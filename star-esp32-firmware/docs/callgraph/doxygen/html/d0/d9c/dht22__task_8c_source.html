50static esp_err_t internal_read_dht22(dht22_task_context_t* const ctx)
51{
52 star_dht22_data_t dht22_data;
53 const esp_err_t read_result = star_bus_dht22_read(ctx->bus_manager, ctx->bus_name, &dht22_data);
54
55 const bool data_valid = (read_result == ESP_OK && dht22_data.checksum_valid);
56
57 const temperature_data_t temp_data =
58 shared_data_create_temperature_data(dht22_data.temperature_c,
59 dht22_data.humidity_percent,
60 read_result,
61 dht22_data.checksum_valid);
62
63 shared_data_update_health(STAR_SYSTEM_CONFIG.num_hcsr04_sensors, data_valid);
64
65 shared_context_t* shared_ctx = shared_data_get_context();
66 if (shared_ctx != NULL && shared_ctx->temperature_data_queue != NULL) {
67 if (xQueueSend(shared_ctx->temperature_data_queue, &temp_data, 0) != pdTRUE) {
68 ESP_LOGW(s_TAG, "Temperature data queue full, dropping measurement");
69 }
70 }
71
72 if (data_valid) {
73 ctx->last_valid_temperature = dht22_data.temperature_c;
74 ctx->consecutive_failures = 0;
75
76 shared_data_set_temperature(dht22_data.temperature_c, true);
77
78 const esp_err_t temp_update_ret = sensor_task_update_temperature(dht22_data.temperature_c);
79 if (temp_update_ret != ESP_OK) {
80 ESP_LOGE(s_TAG,
81 "Failed to update sensor task temperature: %s",
82 esp_err_to_name(temp_update_ret));
83 }
84
85 ESP_LOGI(s_TAG,
86 "DHT22 reading: %.1f°C, %.1f%% RH",
87 dht22_data.temperature_c,
88 dht22_data.humidity_percent);
89 } else {
90 ctx->consecutive_failures++;
91 ESP_LOGW(s_TAG,
92 "DHT22 read failed: %s (consecutive failures: %lu)",
93 esp_err_to_name(read_result),
94 ctx->consecutive_failures);
95
96 if (ctx->consecutive_failures >= STAR_SYSTEM_CONFIG.health.max_sensor_failures) {
97 ESP_LOGE(s_TAG,
98 "DHT22 sensor may be faulty - %lu consecutive failures",
99 ctx->consecutive_failures);
100
101 shared_data_set_temperature(ctx->last_valid_temperature, false);
102 }
103 }
104
105 return read_result;
106}

108static void internal_dht22_task_loop(void* const parameter)
109{
110 dht22_task_context_t* const ctx = (dht22_task_context_t*)parameter;
111
112 ESP_LOGI(s_TAG,
113 "DHT22 monitoring task started (interval: %dms)",
114 STAR_SYSTEM_CONFIG.tasks.dht22.interval_ms);
115
116 TickType_t last_wake_time = xTaskGetTickCount();
117
118 while (ctx->task_running) {
119 if (ctx->sensor_initialized) {
120 internal_read_dht22(ctx);
121 } else {
122 ESP_LOGW(s_TAG, "DHT22 not initialized, skipping read");
123 }
124
125 vTaskDelayUntil(&last_wake_time, pdMS_TO_TICKS(STAR_SYSTEM_CONFIG.tasks.dht22.interval_ms));
126 }
127
128 ESP_LOGI(s_TAG, "DHT22 monitoring task ending");
129 ctx->task_handle = NULL;
130 vTaskDelete(NULL);
131}

135esp_err_t dht22_task_start(star_bus_manager_t* const bus_manager, const char* const bus_name)
136{
137 if (bus_manager == NULL || bus_name == NULL) {
138 ESP_LOGE(s_TAG, "Invalid arguments for DHT22 task start");
139 return ESP_ERR_INVALID_ARG;
140 }
141
142 if (s_g_context_initialized && s_task_context.task_running) {
143 ESP_LOGI(s_TAG, "DHT22 task already running");
144 return ESP_OK;
145 }
146
147 memset(&s_task_context, 0, sizeof(dht22_task_context_t));
148
149 s_task_context.bus_manager = bus_manager;
150 s_task_context.bus_name = bus_name;
151 s_task_context.task_running = false;
152 s_task_context.sensor_initialized = false;
153 s_task_context.last_valid_temperature = STAR_SYSTEM_CONFIG.health.default_temperature_c;
154 s_task_context.consecutive_failures = 0;
155
156 esp_err_t ret = internal_init_dht22(&s_task_context);
157 if (ret != ESP_OK) {
158 ESP_LOGE(s_TAG, "Failed to initialize DHT22 sensor");
159 return ret;
160 }
161
162 s_task_context.task_running = true;
163
164 const BaseType_t task_ret = xTaskCreate(internal_dht22_task_loop,
165 STAR_SYSTEM_CONFIG.tasks.dht22.name,
166 STAR_SYSTEM_CONFIG.tasks.dht22.stack_size,
167 &s_task_context,
168 STAR_SYSTEM_CONFIG.tasks.dht22.priority,
169 &s_task_context.task_handle);
170
171 if (task_ret != pdPASS) {
172 ESP_LOGE(s_TAG, "Failed to create DHT22 task");
173 s_task_context.task_running = false;
174 internal_deinit_dht22(&s_task_context);
175 return ESP_ERR_NO_MEM;
176 }
177
178 s_g_context_initialized = true;
179
180 shared_context_t* shared_ctx = shared_data_get_context();
181 if (shared_ctx != NULL) {
182 shared_ctx->dht22_task_handle = s_task_context.task_handle;
183 }
184
185 ESP_LOGI(s_TAG, "DHT22 task started successfully");
186 return ESP_OK;
187}

189esp_err_t dht22_task_stop(void)
190{
191 if (!s_g_context_initialized || !s_task_context.task_running) {
192 ESP_LOGI(s_TAG, "DHT22 task not running");

198 s_task_context.task_running = false;

205
206 shared_context_t* shared_ctx = shared_data_get_context();
207 if (shared_ctx != NULL) {
208 shared_ctx->dht22_task_handle = NULL;
209 }
210
211 s_g_context_initialized = false;
212 memset(&s_task_context, 0, sizeof(dht22_task_context_t));
213
214 ESP_LOGI(s_TAG, "DHT22 task stopped successfully");
215 return ESP_OK;
