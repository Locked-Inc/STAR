152 static star_pid_handle_t pids[4];

162
163 for (int i = 0; i < 4; i++) {
164 ret = star_pid_init(&pids[i], &pid_cfg);
165 if (ret != ESP_OK) {
166 ESP_LOGE(s_tag, "PID %d init failed: %s", i, esp_err_to_name(ret));
167 return;
168 }
169 }
170 ESP_LOGI(s_tag, "4 PID controllers initialized (Kp=%.3f, Ki=%.3f)", pid_cfg.kp, pid_cfg.ki);
171
172 /* 9. Initialize motor shared state */
173 static motor_shared_state_t shared_state;
174 ret = motor_shared_state_init(&shared_state);
175 if (ret != ESP_OK) {
176 ESP_LOGE(s_tag, "Shared state init failed: %s", esp_err_to_name(ret));
177 return;
178 }
179 ESP_LOGI(s_tag, "Motor shared state initialized");
180
181 /* 10. Initialize ARQ layer */
182 static star_arq_handle_t arq;
183 star_arq_config_t arq_cfg = {
184 .bus_manager = &bus_manager,
185 .spi_bus_name = "rpi_spi",
186 .max_retries = k_arq_max_retries,
187 .retry_timeout_ms = k_arq_retry_timeout_ms,
188 };
189 ret = star_arq_init(&arq, &arq_cfg);
190 if (ret != ESP_OK) {
191 ESP_LOGE(s_tag, "ARQ init failed: %s", esp_err_to_name(ret));
192 return;
193 }
194
195 /* 11. Create FreeRTOS tasks */
196 motor_control_task_create(&shared_state, motors, &encoder_mux, pids);
197 command_handler_task_create(&shared_state, &arq);
198 telemetry_task_create(&temp_sensor, &arq);
199
200 ESP_LOGI(s_tag, "Initialization complete. Motor control system running at 250Hz.");
201
202 /* app_main returns, freeing main task stack */
203}
