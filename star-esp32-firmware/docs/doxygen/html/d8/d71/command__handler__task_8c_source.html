108static void internal_handle_set_velocity_request(motor_shared_state_t* shared_state,
109 star_arq_handle_t* arq,
110 const uint8_t* request_buf,
111 size_t request_len,
112 uint8_t* response_buf,
113 size_t response_buf_size)
114{
115 /* Decode the request */
116 pb_istream_t input_stream = pb_istream_from_buffer(request_buf, request_len);
117 star_v1_SetVelocityRequest request = star_v1_SetVelocityRequest_init_zero;
118
119 bool decode_ret = pb_decode(&input_stream, star_v1_SetVelocityRequest_fields, &request);
120
121 if (!decode_ret) {
122 ESP_LOGE(s_tag, "Failed to decode SetVelocityRequest: %s", PB_GET_ERROR(&input_stream));
123 }
124
125 /* Update shared state with velocity setpoints */
126 esp_err_t ret = ESP_OK;
127
128 if (decode_ret && request.has_command) {
129 /* Skid-steer kinematics: distribute left/right to motors */
130 /* Left side: motors 0, 1 */
131 ret = motor_shared_state_set_velocity(shared_state, 0, request.command.left_velocity_mps);
132 ret |= motor_shared_state_set_velocity(shared_state, 1, request.command.left_velocity_mps);
133
134 /* Right side: motors 2, 3 */
135 ret |= motor_shared_state_set_velocity(shared_state, 2, request.command.right_velocity_mps);
136 ret |= motor_shared_state_set_velocity(shared_state, 3, request.command.right_velocity_mps);
137
138 if (ret != ESP_OK) {
139 ESP_LOGW(s_tag, "Failed to update velocity setpoints");
140 } else {
141 ESP_LOGD(s_tag,
142 "Velocity command: L=%.3f m/s, R=%.3f m/s",
143 request.command.left_velocity_mps,
144 request.command.right_velocity_mps);
145 }
146 }
147
148 /* Build response */
149 star_v1_SetVelocityResponse response = star_v1_SetVelocityResponse_init_zero;
150 response.has_header = true;
151
152 /* Set status */
153 if (!decode_ret) {
154 response.header.status = star_v1_Status_STATUS_INVALID_REQUEST;
155 } else if (ret != ESP_OK) {
156 response.header.status = star_v1_Status_STATUS_INTERNAL_ERROR;
157 } else {
158 response.header.status = star_v1_Status_STATUS_OK;
159 }
160
161 /* Copy request_id for correlation */
162 if (decode_ret && request.has_header) {
163 response.header.request_id = request.header.request_id;
164 }
165
166 /* TODO: Populate motor_status array from shared state */
167 /* This would read motor status from shared state and encode into response */
168
169 /* Encode response */
170 pb_ostream_t output_stream = pb_ostream_from_buffer(response_buf, response_buf_size);
171
172 bool encode_ok = pb_encode(&output_stream, star_v1_SetVelocityResponse_fields, &response);
173
174 if (!encode_ok) {
175 ESP_LOGE(s_tag, "Failed to encode SetVelocityResponse: %s", PB_GET_ERROR(&output_stream));
176 return;
177 }
178
179 size_t response_len = output_stream.bytes_written;
180
181 /* Send response */
182 esp_err_t send_ret =
183 star_arq_send(arq, response_buf, response_len, k_command_arq_send_timeout_ms);
184
185 if (send_ret != ESP_OK) {
186 ESP_LOGW(s_tag, "Failed to send SetVelocityResponse: %s", esp_err_to_name(send_ret));
187 }
188}

204static void internal_handle_emergency_stop_request(motor_shared_state_t* shared_state,

235 response.estop_engaged = false;

237 response.header.status = star_v1_Status_STATUS_INTERNAL_ERROR;
238 response.estop_engaged = false;

240 response.header.status = star_v1_Status_STATUS_OK;

281static void internal_handle_requests(motor_shared_state_t* shared_state, star_arq_handle_t* arq)
282{
283 uint8_t rx_buffer[k_command_rx_buffer_size];
284 uint8_t tx_buffer[k_command_tx_buffer_size];
285
286 while (true) {
287 /* Receive request from RPi5 (blocking) */
288 size_t rx_len;
289 esp_err_t ret =
290 star_arq_receive(arq, rx_buffer, k_command_rx_buffer_size, &rx_len, portMAX_DELAY);
291
292 if (ret != ESP_OK) {
293 ESP_LOGW(s_tag, "Receive failed: %s", esp_err_to_name(ret));
294 vTaskDelay(pdMS_TO_TICKS(k_command_arq_error_backoff_ms));
295 continue;
296 }
297
298 /* TODO: Add message type detection to dispatch to correct handler */
299 /* For now, assume all messages are SetVelocityRequest */
300 /* In production, would parse message type field or use message envelope */
301
302 /* Process as SetVelocityRequest (most common) */
303 internal_handle_set_velocity_request(shared_state,
304 arq,
305 rx_buffer,
306 rx_len,
307 tx_buffer,
308 k_command_tx_buffer_size);
309
310 /* If implementing message type detection:
311 * - Parse message type from envelope or first bytes
312 * - Switch based on type:
313 * - SetVelocityRequest → internal_handle_set_velocity_request()
314 * - EmergencyStopRequest → internal_handle_emergency_stop_request()
315 * - SetMotorPowerRequest → internal_handle_set_motor_power_request()
316 */
317 }
318}
