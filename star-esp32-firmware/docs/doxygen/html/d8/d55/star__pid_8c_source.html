31esp_err_t star_pid_init(star_pid_handle_t* handle, const star_pid_config_t* config)
32{
33 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
34 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_tag, "Config is NULL");
35 ESP_RETURN_ON_FALSE(!handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Already initialized");
36
37 /* Validate configuration */
38 ESP_RETURN_ON_FALSE(config->output_max > config->output_min,
39 ESP_ERR_INVALID_ARG,
40 s_tag,
41 "output_max must be > output_min");
42 ESP_RETURN_ON_FALSE(config->integral_max > config->integral_min,
43 ESP_ERR_INVALID_ARG,
44 s_tag,
45 "integral_max must be > integral_min");
46
47 /* Zero out handle */
48 memset(handle, 0, sizeof(star_pid_handle_t));
49
50 /* Copy configuration */
51 handle->kp = config->kp;
52 handle->ki = config->ki;
53 handle->kd = config->kd;
54 handle->output_min = config->output_min;
55 handle->output_max = config->output_max;
56 handle->integral_min = config->integral_min;
57 handle->integral_max = config->integral_max;
58 handle->integral = 0.0f;
59 handle->prev_error = 0.0f;
60 handle->initialized = true;
61
62 ESP_LOGI(s_tag,
63 "PID initialized: Kp=%.3f, Ki=%.3f, Kd=%.3f, out=[%.1f,%.1f]",
64 config->kp,
65 config->ki,
66 config->kd,
67 config->output_min,
68 config->output_max);
69
70 return ESP_OK;
71}

85esp_err_t
86star_pid_compute(star_pid_handle_t* handle, float setpoint, float measured, float dt, float* output)
87{
88 ESP_RETURN_ON_FALSE(handle && output, ESP_ERR_INVALID_ARG, s_tag, "Handle or output is NULL");
89 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
90 ESP_RETURN_ON_FALSE(dt > 0.0f, ESP_ERR_INVALID_ARG, s_tag, "dt must be > 0");
91
92 /* Calculate error */
93 float error = setpoint - measured;
94
95 /* Proportional term */
96 float p_term = handle->kp * error;
97
98 /* Integral term with anti-windup */
99 handle->integral += error * dt;
100 handle->integral = internal_clamp(handle->integral, handle->integral_min, handle->integral_max);
101 float i_term = handle->ki * handle->integral;
102
103 /* Derivative term */
104 float derivative = (error - handle->prev_error) / dt;
105 float d_term = handle->kd * derivative;
106
107 /* Compute total output */
108 float raw_output = p_term + i_term + d_term;
109
110 /* Clamp output to limits */
111 *output = internal_clamp(raw_output, handle->output_min, handle->output_max);
112
113 /* Store error for next iteration */
114 handle->prev_error = error;
115
116 ESP_LOGD(s_tag,
117 "PID: error=%.2f, P=%.2f, I=%.2f, D=%.2f, out=%.2f",
118 error,
119 p_term,
120 i_term,
121 d_term,
122 *output);
123
124 return ESP_OK;
125}
