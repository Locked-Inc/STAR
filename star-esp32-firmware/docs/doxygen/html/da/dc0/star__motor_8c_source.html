41esp_err_t star_motor_init(star_motor_handle_t* handle, const star_motor_config_t* config)
42{
43 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
44 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_tag, "Config is NULL");
45 ESP_RETURN_ON_FALSE(!handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Already initialized");
46
47 /* Zero out handle */
48 memset(handle, 0, sizeof(star_motor_handle_t));
49
50 esp_err_t ret = ESP_OK;
51
52 /* Calculate period in timer ticks */
53 uint32_t period_ticks = config->timer_resolution_hz / config->pwm_freq_hz;
54
55 /* Create MCPWM timer */
56 mcpwm_timer_config_t timer_config = {
57 .group_id = config->group_id,
58 .clk_src = MCPWM_TIMER_CLK_SRC_DEFAULT,
59 .resolution_hz = config->timer_resolution_hz,
60 .count_mode = MCPWM_TIMER_COUNT_MODE_UP_DOWN, /* Center-aligned PWM */
61 .period_ticks = period_ticks,
62 };
63 ret = mcpwm_new_timer(&timer_config, &handle->timer);
64 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to create MCPWM timer");
65
66 /* Create MCPWM operator */
67 mcpwm_operator_config_t operator_config = {
68 .group_id = config->group_id,
69 };
70 ret = mcpwm_new_operator(&operator_config, &handle->operator);
71 if (ret != ESP_OK) {
72 ESP_LOGE(s_tag, "Failed to create MCPWM operator: %s", esp_err_to_name(ret));
73 mcpwm_del_timer(handle->timer);
74 return ret;
75 }
76
77 /* Connect operator to timer */
78 ret = mcpwm_operator_connect_timer(handle->operator, handle->timer);
79 if (ret != ESP_OK) {
80 ESP_LOGE(s_tag, "Failed to connect operator to timer: %s", esp_err_to_name(ret));
81 mcpwm_del_operator(handle->operator);
82 mcpwm_del_timer(handle->timer);
83 return ret;
84 }
85
86 /* Create comparator */
87 mcpwm_comparator_config_t comparator_config = {
88 .flags.update_cmp_on_tez = true, /* Update on timer equal zero */
89 };
90 ret = mcpwm_new_comparator(handle->operator, &comparator_config, &handle->comparator);
91 if (ret != ESP_OK) {
92 ESP_LOGE(s_tag, "Failed to create comparator: %s", esp_err_to_name(ret));
93 mcpwm_del_operator(handle->operator);
94 mcpwm_del_timer(handle->timer);
95 return ret;
96 }
97
98 /* Create generators for PWM A and B */
99 mcpwm_generator_config_t gen_a_config = {
100 .gen_gpio_num = config->pin_pwm_a,
101 };
102 ret = mcpwm_new_generator(handle->operator, &gen_a_config, &handle->gen_a);
103 if (ret != ESP_OK) {
104 ESP_LOGE(s_tag, "Failed to create generator A: %s", esp_err_to_name(ret));
105 mcpwm_del_comparator(handle->comparator);
106 mcpwm_del_operator(handle->operator);
107 mcpwm_del_timer(handle->timer);
108 return ret;
109 }
110
111 mcpwm_generator_config_t gen_b_config = {
112 .gen_gpio_num = config->pin_pwm_b,
113 };
114 ret = mcpwm_new_generator(handle->operator, &gen_b_config, &handle->gen_b);
115 if (ret != ESP_OK) {
116 ESP_LOGE(s_tag, "Failed to create generator B: %s", esp_err_to_name(ret));
117 mcpwm_del_generator(handle->gen_a);
118 mcpwm_del_comparator(handle->comparator);
119 mcpwm_del_operator(handle->operator);
120 mcpwm_del_timer(handle->timer);
121 return ret;
122 }
123
124 /* Configure dead-time */
125 if (config->dead_time_ns > 0) {
126 /* Convert nanoseconds to timer ticks */
127 uint32_t dead_time_ticks =
128 (uint32_t)((config->dead_time_ns * config->timer_resolution_hz) / s_ns_per_sec);
129
130 mcpwm_dead_time_config_t dt_config = {
131 .posedge_delay_ticks = dead_time_ticks,
132 .negedge_delay_ticks = dead_time_ticks,
133 .flags.invert_output = false,
134 };
135
136 ret = mcpwm_generator_set_dead_time(handle->gen_a, handle->gen_a, &dt_config);
137 if (ret != ESP_OK) {
138 ESP_LOGW(s_tag, "Failed to set dead-time for gen_a: %s", esp_err_to_name(ret));
139 }
140
141 ret = mcpwm_generator_set_dead_time(handle->gen_b, handle->gen_b, &dt_config);
142 if (ret != ESP_OK) {
143 ESP_LOGW(s_tag, "Failed to set dead-time for gen_b: %s", esp_err_to_name(ret));
144 }
145 }
146
147 /* Configure generator actions */
148 ret = internal_configure_generators(handle, config->invert_pwm);
149 if (ret != ESP_OK) {
150 mcpwm_del_generator(handle->gen_b);
151 mcpwm_del_generator(handle->gen_a);
152 mcpwm_del_comparator(handle->comparator);
153 mcpwm_del_operator(handle->operator);
154 mcpwm_del_timer(handle->timer);
155 return ret;
156 }
157
158 /* Configure fault detection if pin provided */
159 if (config->fault_pin >= 0) {
160 mcpwm_gpio_fault_config_t fault_config = {
161 .group_id = config->group_id,
162 .gpio_num = config->fault_pin,
163 .flags.pull_up = true,
164 .flags.active_level = 0, /* Active low */
165 };
166 ret = mcpwm_new_gpio_fault(&fault_config, &handle->fault);
167 if (ret != ESP_OK) {
168 ESP_LOGW(s_tag, "Failed to configure fault detection: %s", esp_err_to_name(ret));
169 handle->fault = NULL;
170 }
171 }
172
173 /* Enable and start timer */
174 ret = mcpwm_timer_enable(handle->timer);
175 if (ret != ESP_OK) {
176 ESP_LOGE(s_tag, "Failed to enable timer: %s", esp_err_to_name(ret));
177 if (handle->fault) {
178 mcpwm_del_fault(handle->fault);
179 }
180 mcpwm_del_generator(handle->gen_b);
181 mcpwm_del_generator(handle->gen_a);
182 mcpwm_del_comparator(handle->comparator);
183 mcpwm_del_operator(handle->operator);
184 mcpwm_del_timer(handle->timer);
185 return ret;
186 }
187
188 ret = mcpwm_timer_start_stop(handle->timer, MCPWM_TIMER_START_NO_STOP);
189 if (ret != ESP_OK) {
190 ESP_LOGE(s_tag, "Failed to start timer: %s", esp_err_to_name(ret));
191 mcpwm_timer_disable(handle->timer);
192 if (handle->fault) {
193 mcpwm_del_fault(handle->fault);
194 }
195 mcpwm_del_generator(handle->gen_b);
196 mcpwm_del_generator(handle->gen_a);
197 mcpwm_del_comparator(handle->comparator);
198 mcpwm_del_operator(handle->operator);
199 mcpwm_del_timer(handle->timer);
200 return ret;
201 }
202
203 /* Store configuration */
204 handle->group_id = config->group_id;
205 handle->pwm_freq_hz = config->pwm_freq_hz;
206 handle->period_ticks = period_ticks;
207 handle->current_duty = (float)k_motor_duty_zero;
208 handle->initialized = true;
209
210 /* Set initial duty to 0 (stopped) */
211 star_motor_set_duty(handle, (float)k_motor_duty_zero);
212
213 ESP_LOGI(s_tag,
214 "Motor initialized: group=%d, freq=%luHz, pins=A%d/B%d, dead_time=%luns",
215 config->group_id,
216 config->pwm_freq_hz,
217 config->pin_pwm_a,
218 config->pin_pwm_b,
219 config->dead_time_ns);
220
221 return ESP_OK;
222}

224esp_err_t star_motor_deinit(star_motor_handle_t* handle)
225{
226 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
227 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
228
229 /* Stop motor first */
230 star_motor_stop(handle, false);
231
232 /* Stop and disable timer */
233 mcpwm_timer_start_stop(handle->timer, MCPWM_TIMER_STOP_EMPTY);
234 mcpwm_timer_disable(handle->timer);
235
236 /* Delete all MCPWM resources */
237 if (handle->fault) {
238 mcpwm_del_fault(handle->fault);
239 }
240 mcpwm_del_generator(handle->gen_b);
241 mcpwm_del_generator(handle->gen_a);
242 mcpwm_del_comparator(handle->comparator);
243 mcpwm_del_operator(handle->operator);
244 mcpwm_del_timer(handle->timer);
245
246 /* Clear handle */
247 memset(handle, 0, sizeof(star_motor_handle_t));
248
249 ESP_LOGI(s_tag, "Motor deinitialized");
250 return ESP_OK;
251}

253esp_err_t star_motor_set_duty(star_motor_handle_t* handle, float duty)
254{
255 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
256 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
257
258 /* Clamp duty cycle */
259 duty = internal_clamp_duty(duty);
260
261 /* Calculate compare value (duty is -100 to +100, so we use absolute value) */
262 float abs_duty = fabsf(duty);
263 uint32_t compare_ticks = (uint32_t)((abs_duty / (float)k_motor_percent_scale) * handle->period_ticks);
264
265 /* Set comparator value */
266 esp_err_t ret = mcpwm_comparator_set_compare_value(handle->comparator, compare_ticks);
267 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to set compare value");
268
269 /* Configure direction (swap generator actions based on duty sign) */
270 if (duty >= (float)k_motor_duty_zero) {
271 /* Forward: A = PWM, B = LOW */
272 mcpwm_generator_set_action_on_compare_event(
273 handle->gen_a,
274 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
275 handle->comparator,
276 MCPWM_GEN_ACTION_HIGH));
277 mcpwm_generator_set_action_on_compare_event(
278 handle->gen_a,
279 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_DOWN,
280 handle->comparator,
281 MCPWM_GEN_ACTION_LOW));
282 mcpwm_generator_set_actions_on_timer_event(
283 handle->gen_b,
284 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
285 MCPWM_TIMER_EVENT_EMPTY,
286 MCPWM_GEN_ACTION_LOW),
287 MCPWM_GEN_TIMER_EVENT_ACTION_END());
288 } else {
289 /* Reverse: A = LOW, B = PWM */
290 mcpwm_generator_set_actions_on_timer_event(
291 handle->gen_a,
292 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
293 MCPWM_TIMER_EVENT_EMPTY,
294 MCPWM_GEN_ACTION_LOW),
295 MCPWM_GEN_TIMER_EVENT_ACTION_END());
296 mcpwm_generator_set_action_on_compare_event(
297 handle->gen_b,
298 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
299 handle->comparator,
300 MCPWM_GEN_ACTION_HIGH));
301 mcpwm_generator_set_action_on_compare_event(
302 handle->gen_b,
303 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_DOWN,
304 handle->comparator,
305 MCPWM_GEN_ACTION_LOW));
306 }
307
308 handle->current_duty = duty;
309
310 ESP_LOGD(s_tag, "Motor duty set: %.1f%% (compare=%lu ticks)", duty, compare_ticks);
311 return ESP_OK;
312}

314esp_err_t star_motor_stop(star_motor_handle_t* handle, bool brake)
315{
316 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
317 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
318
319 if (brake) {
320 /* Brake: both outputs high (short circuit brake) */
321 mcpwm_generator_set_force_level(handle->gen_a, 1, true);
322 mcpwm_generator_set_force_level(handle->gen_b, 1, true);
323 } else {
324 /* Coast: both outputs low */
325 mcpwm_generator_set_force_level(handle->gen_a, 0, true);
326 mcpwm_generator_set_force_level(handle->gen_b, 0, true);
327 }
328
329 handle->current_duty = (float)k_motor_duty_zero;
330
331 ESP_LOGD(s_tag, "Motor stopped (%s)", brake ? "brake" : "coast");
332 return ESP_OK;
333}

346static inline float internal_clamp_duty(float duty)
347{
348 if (duty < (float)k_motor_duty_min) {

357static esp_err_t internal_configure_generators(star_motor_handle_t* handle, bool invert_pwm)
358{
359 /* Configure generator A actions (forward PWM when duty > 0) */
360 esp_err_t ret;
361
362 ret =
363 mcpwm_generator_set_action_on_timer_event(handle->gen_a,
364 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
365 MCPWM_TIMER_EVENT_EMPTY,
366 MCPWM_GEN_ACTION_LOW));
367 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to set gen_a timer action");
368
369 ret = mcpwm_generator_set_action_on_compare_event(
370 handle->gen_a,
371 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
372 handle->comparator,
373 MCPWM_GEN_ACTION_HIGH));
374 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to set gen_a compare up action");
375
376 ret = mcpwm_generator_set_action_on_compare_event(
377 handle->gen_a,
378 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_DOWN,
379 handle->comparator,
380 MCPWM_GEN_ACTION_LOW));
381 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to set gen_a compare down action");
382
383 /* Configure generator B actions (initially low, will be set during duty update) */
384 ret =
385 mcpwm_generator_set_action_on_timer_event(handle->gen_b,
386 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
387 MCPWM_TIMER_EVENT_EMPTY,
388 MCPWM_GEN_ACTION_LOW));
389 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to set gen_b timer action");
390
391 return ESP_OK;
392}
