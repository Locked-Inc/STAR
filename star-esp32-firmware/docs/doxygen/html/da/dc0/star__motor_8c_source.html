32esp_err_t star_motor_init(star_motor_handle_t* handle, const star_motor_config_t* config)
33{
34 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_TAG, "Handle is NULL");
35 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config is NULL");
36 ESP_RETURN_ON_FALSE(!handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Already initialized");
37
38 /* Zero out handle */
39 memset(handle, 0, sizeof(star_motor_handle_t));
40
41 esp_err_t ret = ESP_OK;
42
43 /* Calculate period in timer ticks */
44 uint32_t period_ticks = config->timer_resolution_hz / config->pwm_freq_hz;
45
46 /* Create MCPWM timer */
47 mcpwm_timer_config_t timer_config = {
48 .group_id = config->group_id,
49 .clk_src = MCPWM_TIMER_CLK_SRC_DEFAULT,
50 .resolution_hz = config->timer_resolution_hz,
51 .count_mode = MCPWM_TIMER_COUNT_MODE_UP_DOWN, /* Center-aligned PWM */
52 .period_ticks = period_ticks,
53 };
54 ret = mcpwm_new_timer(&timer_config, &handle->timer);
55 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to create MCPWM timer");
56
57 /* Create MCPWM operator */
58 mcpwm_operator_config_t operator_config = {
59 .group_id = config->group_id,
60 };
61 ret = mcpwm_new_operator(&operator_config, &handle->operator);
62 if (ret != ESP_OK) {
63 ESP_LOGE(s_TAG, "Failed to create MCPWM operator: %s", esp_err_to_name(ret));
64 mcpwm_del_timer(handle->timer);
65 return ret;
66 }
67
68 /* Connect operator to timer */
69 ret = mcpwm_operator_connect_timer(handle->operator, handle->timer);
70 if (ret != ESP_OK) {
71 ESP_LOGE(s_TAG, "Failed to connect operator to timer: %s", esp_err_to_name(ret));
72 mcpwm_del_operator(handle->operator);
73 mcpwm_del_timer(handle->timer);
74 return ret;
75 }
76
77 /* Create comparator */
78 mcpwm_comparator_config_t comparator_config = {
79 .flags.update_cmp_on_tez = true, /* Update on timer equal zero */
80 };
81 ret = mcpwm_new_comparator(handle->operator, &comparator_config, &handle->comparator);
82 if (ret != ESP_OK) {
83 ESP_LOGE(s_TAG, "Failed to create comparator: %s", esp_err_to_name(ret));
84 mcpwm_del_operator(handle->operator);
85 mcpwm_del_timer(handle->timer);
86 return ret;
87 }
88
89 /* Create generators for PWM A and B */
90 mcpwm_generator_config_t gen_a_config = {
91 .gen_gpio_num = config->pin_pwm_a,
92 };
93 ret = mcpwm_new_generator(handle->operator, &gen_a_config, &handle->gen_a);
94 if (ret != ESP_OK) {
95 ESP_LOGE(s_TAG, "Failed to create generator A: %s", esp_err_to_name(ret));
96 mcpwm_del_comparator(handle->comparator);
97 mcpwm_del_operator(handle->operator);
98 mcpwm_del_timer(handle->timer);
99 return ret;
100 }
101
102 mcpwm_generator_config_t gen_b_config = {
103 .gen_gpio_num = config->pin_pwm_b,
104 };
105 ret = mcpwm_new_generator(handle->operator, &gen_b_config, &handle->gen_b);
106 if (ret != ESP_OK) {
107 ESP_LOGE(s_TAG, "Failed to create generator B: %s", esp_err_to_name(ret));
108 mcpwm_del_generator(handle->gen_a);
109 mcpwm_del_comparator(handle->comparator);
110 mcpwm_del_operator(handle->operator);
111 mcpwm_del_timer(handle->timer);
112 return ret;
113 }
114
115 /* Configure dead-time */
116 if (config->dead_time_ns > 0) {
117 /* Convert nanoseconds to timer ticks */
118 uint32_t dead_time_ticks =
119 (uint32_t)((config->dead_time_ns * config->timer_resolution_hz) / 1000000000ULL);
120
121 mcpwm_dead_time_config_t dt_config = {
122 .posedge_delay_ticks = dead_time_ticks,
123 .negedge_delay_ticks = dead_time_ticks,
124 .flags.invert_output = false,
125 };
126
127 ret = mcpwm_generator_set_dead_time(handle->gen_a, handle->gen_a, &dt_config);
128 if (ret != ESP_OK) {
129 ESP_LOGW(s_TAG, "Failed to set dead-time for gen_a: %s", esp_err_to_name(ret));
130 }
131
132 ret = mcpwm_generator_set_dead_time(handle->gen_b, handle->gen_b, &dt_config);
133 if (ret != ESP_OK) {
134 ESP_LOGW(s_TAG, "Failed to set dead-time for gen_b: %s", esp_err_to_name(ret));
135 }
136 }
137
138 /* Configure generator actions */
139 ret = internal_configure_generators(handle, config->invert_pwm);
140 if (ret != ESP_OK) {
141 mcpwm_del_generator(handle->gen_b);
142 mcpwm_del_generator(handle->gen_a);
143 mcpwm_del_comparator(handle->comparator);
144 mcpwm_del_operator(handle->operator);
145 mcpwm_del_timer(handle->timer);
146 return ret;
147 }
148
149 /* Configure fault detection if pin provided */
150 if (config->fault_pin >= 0) {
151 mcpwm_gpio_fault_config_t fault_config = {
152 .group_id = config->group_id,
153 .gpio_num = config->fault_pin,
154 .flags.pull_up = true,
155 .flags.active_level = 0, /* Active low */
156 };
157 ret = mcpwm_new_gpio_fault(&fault_config, &handle->fault);
158 if (ret != ESP_OK) {
159 ESP_LOGW(s_TAG, "Failed to configure fault detection: %s", esp_err_to_name(ret));
160 handle->fault = NULL;
161 }
162 }
163
164 /* Enable and start timer */
165 ret = mcpwm_timer_enable(handle->timer);
166 if (ret != ESP_OK) {
167 ESP_LOGE(s_TAG, "Failed to enable timer: %s", esp_err_to_name(ret));
168 if (handle->fault) {
169 mcpwm_del_fault(handle->fault);
170 }
171 mcpwm_del_generator(handle->gen_b);
172 mcpwm_del_generator(handle->gen_a);
173 mcpwm_del_comparator(handle->comparator);
174 mcpwm_del_operator(handle->operator);
175 mcpwm_del_timer(handle->timer);
176 return ret;
177 }
178
179 ret = mcpwm_timer_start_stop(handle->timer, MCPWM_TIMER_START_NO_STOP);
180 if (ret != ESP_OK) {
181 ESP_LOGE(s_TAG, "Failed to start timer: %s", esp_err_to_name(ret));
182 mcpwm_timer_disable(handle->timer);
183 if (handle->fault) {
184 mcpwm_del_fault(handle->fault);
185 }
186 mcpwm_del_generator(handle->gen_b);
187 mcpwm_del_generator(handle->gen_a);
188 mcpwm_del_comparator(handle->comparator);
189 mcpwm_del_operator(handle->operator);
190 mcpwm_del_timer(handle->timer);
191 return ret;
192 }
193
194 /* Store configuration */
195 handle->group_id = config->group_id;
196 handle->pwm_freq_hz = config->pwm_freq_hz;
197 handle->period_ticks = period_ticks;
198 handle->current_duty = 0.0f;
199 handle->initialized = true;
200
201 /* Set initial duty to 0 (stopped) */
202 star_motor_set_duty(handle, 0.0f);
203
204 ESP_LOGI(s_TAG,
205 "Motor initialized: group=%d, freq=%luHz, pins=A%d/B%d, dead_time=%luns",
206 config->group_id,
207 config->pwm_freq_hz,
208 config->pin_pwm_a,
209 config->pin_pwm_b,
210 config->dead_time_ns);
211
212 return ESP_OK;
213}

215esp_err_t star_motor_deinit(star_motor_handle_t* handle)
216{
217 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_TAG, "Handle is NULL");
218 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Not initialized");
219
220 /* Stop motor first */
221 star_motor_stop(handle, false);
222
223 /* Stop and disable timer */
224 mcpwm_timer_start_stop(handle->timer, MCPWM_TIMER_STOP_EMPTY);
225 mcpwm_timer_disable(handle->timer);
226
227 /* Delete all MCPWM resources */
228 if (handle->fault) {
229 mcpwm_del_fault(handle->fault);
230 }
231 mcpwm_del_generator(handle->gen_b);
232 mcpwm_del_generator(handle->gen_a);
233 mcpwm_del_comparator(handle->comparator);
234 mcpwm_del_operator(handle->operator);
235 mcpwm_del_timer(handle->timer);
236
237 /* Clear handle */
238 memset(handle, 0, sizeof(star_motor_handle_t));
239
240 ESP_LOGI(s_TAG, "Motor deinitialized");
241 return ESP_OK;
242}

244esp_err_t star_motor_set_duty(star_motor_handle_t* handle, float duty)
245{
246 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_TAG, "Handle is NULL");
247 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Not initialized");
248
249 /* Clamp duty cycle */
250 duty = internal_clamp_duty(duty);
251
252 /* Calculate compare value (duty is -100 to +100, so we use absolute value) */
253 float abs_duty = fabsf(duty);
254 uint32_t compare_ticks = (uint32_t)((abs_duty / 100.0f) * handle->period_ticks);
255
256 /* Set comparator value */
257 esp_err_t ret = mcpwm_comparator_set_compare_value(handle->comparator, compare_ticks);
258 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to set compare value");
259
260 /* Configure direction (swap generator actions based on duty sign) */
261 if (duty >= 0.0f) {
262 /* Forward: A = PWM, B = LOW */
263 mcpwm_generator_set_action_on_compare_event(
264 handle->gen_a,
265 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
266 handle->comparator,
267 MCPWM_GEN_ACTION_HIGH));
268 mcpwm_generator_set_action_on_compare_event(
269 handle->gen_a,
270 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_DOWN,
271 handle->comparator,
272 MCPWM_GEN_ACTION_LOW));
273 mcpwm_generator_set_actions_on_timer_event(
274 handle->gen_b,
275 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
276 MCPWM_TIMER_EVENT_EMPTY,
277 MCPWM_GEN_ACTION_LOW),
278 MCPWM_GEN_TIMER_EVENT_ACTION_END());
279 } else {
280 /* Reverse: A = LOW, B = PWM */
281 mcpwm_generator_set_actions_on_timer_event(
282 handle->gen_a,
283 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
284 MCPWM_TIMER_EVENT_EMPTY,
285 MCPWM_GEN_ACTION_LOW),
286 MCPWM_GEN_TIMER_EVENT_ACTION_END());
287 mcpwm_generator_set_action_on_compare_event(
288 handle->gen_b,
289 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
290 handle->comparator,
291 MCPWM_GEN_ACTION_HIGH));
292 mcpwm_generator_set_action_on_compare_event(
293 handle->gen_b,
294 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_DOWN,
295 handle->comparator,
296 MCPWM_GEN_ACTION_LOW));
297 }
298
299 handle->current_duty = duty;
300
301 ESP_LOGD(s_TAG, "Motor duty set: %.1f%% (compare=%lu ticks)", duty, compare_ticks);
302 return ESP_OK;
303}

305esp_err_t star_motor_stop(star_motor_handle_t* handle, bool brake)
306{
307 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_TAG, "Handle is NULL");
308 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Not initialized");
309
310 if (brake) {
311 /* Brake: both outputs high (short circuit brake) */
312 mcpwm_generator_set_force_level(handle->gen_a, 1, true);
313 mcpwm_generator_set_force_level(handle->gen_b, 1, true);
314 } else {
315 /* Coast: both outputs low */
316 mcpwm_generator_set_force_level(handle->gen_a, 0, true);
317 mcpwm_generator_set_force_level(handle->gen_b, 0, true);
318 }
319
320 handle->current_duty = 0.0f;
321
322 ESP_LOGD(s_TAG, "Motor stopped (%s)", brake ? "brake" : "coast");
323 return ESP_OK;
324}

337static inline float internal_clamp_duty(float duty)

348static esp_err_t internal_configure_generators(star_motor_handle_t* handle, bool invert_pwm)
349{
350 /* Configure generator A actions (forward PWM when duty > 0) */
351 esp_err_t ret;
352
353 ret =
354 mcpwm_generator_set_action_on_timer_event(handle->gen_a,
355 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
356 MCPWM_TIMER_EVENT_EMPTY,
357 MCPWM_GEN_ACTION_LOW));
358 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to set gen_a timer action");
359
360 ret = mcpwm_generator_set_action_on_compare_event(
361 handle->gen_a,
362 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
363 handle->comparator,
364 MCPWM_GEN_ACTION_HIGH));
365 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to set gen_a compare up action");
366
367 ret = mcpwm_generator_set_action_on_compare_event(
368 handle->gen_a,
369 MCPWM_GEN_COMPARE_EVENT_ACTION(MCPWM_TIMER_DIRECTION_DOWN,
370 handle->comparator,
371 MCPWM_GEN_ACTION_LOW));
372 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to set gen_a compare down action");
373
374 /* Configure generator B actions (initially low, will be set during duty update) */
375 ret =
376 mcpwm_generator_set_action_on_timer_event(handle->gen_b,
377 MCPWM_GEN_TIMER_EVENT_ACTION(MCPWM_TIMER_DIRECTION_UP,
378 MCPWM_TIMER_EVENT_EMPTY,
379 MCPWM_GEN_ACTION_LOW));
380 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to set gen_b timer action");
381
382 return ESP_OK;
383}
