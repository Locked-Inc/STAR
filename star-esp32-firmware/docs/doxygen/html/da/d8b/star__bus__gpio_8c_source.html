62 ESP_RETURN_ON_FALSE(bus_config->type == k_star_bus_type_gpio,
63 ESP_ERR_INVALID_ARG,
64 s_tag,
65 "Bus '%s' is not a GPIO bus",
66 bus_name);
67 ESP_RETURN_ON_FALSE(bus_config->proto.gpio.ops.write_digital,
68 ESP_ERR_NOT_SUPPORTED,
69 s_tag,
70 "No write_digital op for '%s'",
71 bus_name);

104static esp_err_t internal_star_bus_gpio_write_digital(const star_bus_config_t* config,
105 uint8_t pin_index,
106 uint8_t value)
107{
108 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_tag, "Config is NULL");
109 ESP_RETURN_ON_FALSE(pin_index < config->proto.gpio.pin_count,
110 ESP_ERR_INVALID_ARG,
111 s_tag,
112 "Pin index %d out of range (max %d)",
113 pin_index,
114 config->proto.gpio.pin_count - 1);
115
116 gpio_num_t pin = config->proto.gpio.pins[pin_index];
117
118 esp_err_t ret = gpio_set_level(pin, value ? 1 : 0);
119 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to set GPIO %d", pin);
120
121 ESP_LOGD(s_tag, "GPIO %d set to %d", pin, value ? 1 : 0);
122 return ESP_OK;
123}

125static esp_err_t internal_star_bus_gpio_read_digital(const star_bus_config_t* config,
126 uint8_t pin_index,
127 uint8_t* value)
128{
129 ESP_RETURN_ON_FALSE(config && value, ESP_ERR_INVALID_ARG, s_tag, "Config or value is NULL");
130 ESP_RETURN_ON_FALSE(pin_index < config->proto.gpio.pin_count,
131 ESP_ERR_INVALID_ARG,
132 s_tag,
133 "Pin index %d out of range (max %d)",
134 pin_index,
135 config->proto.gpio.pin_count - 1);
136
137 gpio_num_t pin = config->proto.gpio.pins[pin_index];
138
139 int level = gpio_get_level(pin);
140 *value = (level > 0) ? 1 : 0;
141
142 ESP_LOGD(s_tag, "GPIO %d read as %d", pin, *value);
143 return ESP_OK;
144}
