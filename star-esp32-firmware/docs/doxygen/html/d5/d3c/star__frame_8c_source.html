44esp_err_t star_frame_build(
45 star_frame_handle_t* handle,
46 uint16_t seq,
47 star_frame_type_t type,
48 const uint8_t* payload,
49 uint16_t payload_len,
50 uint8_t** out_frame,
51 size_t* out_frame_len)
52{
53 ESP_RETURN_ON_FALSE(handle && out_frame && out_frame_len,
54 ESP_ERR_INVALID_ARG,
55 s_tag,
56 "NULL pointer argument");
57 ESP_RETURN_ON_FALSE(payload_len <= k_star_frame_max_payload,
58 ESP_ERR_INVALID_SIZE,
59 s_tag,
60 "Payload too large: %u > %d",
61 payload_len,
62 k_star_frame_max_payload);
63 ESP_RETURN_ON_FALSE(!(payload == NULL && payload_len > 0),
64 ESP_ERR_INVALID_ARG,
65 s_tag,
66 "NULL payload with non-zero length");
67
68 /* Build frame header */
69 uint8_t* buf = handle->buffer;
70 size_t offset = 0;
71
72 /* SYNC (2 bytes, big-endian network byte order) */
73 internal_write_u16_be(&buf[offset], k_star_frame_sync_magic);
74 offset += 2;
75
76 /* SEQ (2 bytes, big-endian network byte order) */
77 internal_write_u16_be(&buf[offset], seq);
78 offset += 2;
79
80 /* LEN (2 bytes, big-endian network byte order) */
81 internal_write_u16_be(&buf[offset], payload_len);
82 offset += 2;
83
84 /* TYPE (1 byte) */
85 buf[offset] = (uint8_t)type;
86 offset += 1;
87
88 /* FLAGS (1 byte, reserved for future use) */
89 buf[offset] = 0;
90 offset += 1;
91
92 /* PAYLOAD (0-1024 bytes) */
93 if (payload_len > 0) {
94 memcpy(&buf[offset], payload, payload_len);
95 offset += payload_len;
96 }
97
98 /* Calculate CRC-32 over header + payload */
99 const size_t crc_data_len = k_star_frame_header_size + payload_len;
100 const uint32_t crc = star_frame_crc32(buf, crc_data_len);
101
102 /* CRC-32 (4 bytes, little-endian per IEEE 802.3) */
103 internal_write_u32_le(&buf[offset], crc);
104 offset += k_star_frame_crc_size;
105
106 /* Update handle state */
107 handle->seq = seq;
108 handle->payload_len = payload_len;
109 handle->type = (uint8_t)type;
110 handle->valid = true;
111
112 /* Return frame data (points into handle buffer) */
113 *out_frame = buf;
114 *out_frame_len = offset;
115
116 ESP_LOGD(s_tag,
117 "Built frame: seq=%u, type=%u, payload_len=%u, total_len=%zu",
118 seq,
119 (uint8_t)type,
120 payload_len,
121 offset);
122
123 return ESP_OK;
124}

126esp_err_t star_frame_parse(
127 star_frame_handle_t* handle,
128 const uint8_t* data,
129 size_t data_len,
130 uint16_t* out_seq,
131 star_frame_type_t* out_type,
132 uint8_t** out_payload,
133 uint16_t* out_payload_len)
134{
135 ESP_RETURN_ON_FALSE(handle && data && out_seq && out_type && out_payload && out_payload_len,
136 ESP_ERR_INVALID_ARG,
137 s_tag,
138 "NULL pointer argument");
139
140 const size_t min_frame_size = k_star_frame_header_size + k_star_frame_crc_size;
141 ESP_RETURN_ON_FALSE(data_len >= min_frame_size,
142 ESP_ERR_INVALID_SIZE,
143 s_tag,
144 "Frame too small: %zu < %zu",
145 data_len,
146 min_frame_size);
147
148 size_t offset = 0;
149
150 /* Parse SYNC (2 bytes, big-endian) */
151 const uint16_t sync = internal_read_u16_be(&data[offset]);
152 ESP_RETURN_ON_FALSE(sync == k_star_frame_sync_magic,
153 ESP_ERR_INVALID_STATE,
154 s_tag,
155 "SYNC mismatch: 0x%04X != 0x%04X",
156 sync,
157 k_star_frame_sync_magic);
158 offset += 2;
159
160 /* Parse SEQ (2 bytes, big-endian) */
161 const uint16_t seq = internal_read_u16_be(&data[offset]);
162 offset += 2;
163
164 /* Parse LEN (2 bytes, big-endian) */
165 const uint16_t payload_len = internal_read_u16_be(&data[offset]);
166 offset += 2;
167
168 ESP_RETURN_ON_FALSE(payload_len <= k_star_frame_max_payload,
169 ESP_ERR_INVALID_SIZE,
170 s_tag,
171 "Payload length too large: %u > %d",
172 payload_len,
173 k_star_frame_max_payload);
174
175 /* Parse TYPE (1 byte) */
176 const uint8_t type = data[offset];
177 offset += 1;
178
179 /* Parse FLAGS (1 byte, reserved) */
180 offset += 1;
181
182 /* Verify frame length matches expected */
183 const size_t expected_len = k_star_frame_header_size + payload_len + k_star_frame_crc_size;
184 ESP_RETURN_ON_FALSE(data_len >= expected_len,
185 ESP_ERR_INVALID_SIZE,
186 s_tag,
187 "Frame incomplete: %zu < %zu",
188 data_len,
189 expected_len);
190
191 /* Verify CRC-32 (calculated over header + payload) */
192 const size_t crc_data_len = k_star_frame_header_size + payload_len;
193 const uint32_t calculated_crc = star_frame_crc32(data, crc_data_len);
194 const uint32_t received_crc = internal_read_u32_le(&data[crc_data_len]);
195
196 ESP_RETURN_ON_FALSE(calculated_crc == received_crc,
197 ESP_ERR_INVALID_CRC,
198 s_tag,
199 "CRC mismatch: 0x%08lX != 0x%08lX",
200 calculated_crc,
201 received_crc);
202
203 /* Update handle state */
204 handle->seq = seq;
205 handle->payload_len = payload_len;
206 handle->type = type;
207 handle->valid = true;
208
209 /* Return parsed data (payload points into input data buffer) */
210 *out_seq = seq;
211 *out_type = (star_frame_type_t)type;
212 *out_payload = (uint8_t*)&data[k_star_frame_header_size];
213 *out_payload_len = payload_len;
214
215 ESP_LOGD(s_tag, "Parsed frame: seq=%u, type=%u, payload_len=%u", seq, type, payload_len);
216
217 return ESP_OK;
218}

220uint32_t star_frame_crc32(const uint8_t* data, size_t len) {
221 if (data == NULL || len == 0) {
222 return 0;
223 }
224
225 /*
226 * Use ESP32 ROM-based CRC-32 (IEEE 802.3 polynomial: 0x04C11DB7)
227 * Initial value: 0xFFFFFFFF, final XOR: 0xFFFFFFFF
228 * This implementation has zero SRAM overhead (uses ROM lookup table)
229 */
230 return esp_rom_crc32_le(0xFFFFFFFF, data, len) ^ 0xFFFFFFFF;
231}

269static inline void internal_write_u32_le(uint8_t* dest, uint32_t value) {
270 dest[0] = value & 0xFF;
