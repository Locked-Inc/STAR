92esp_err_t star_sensor_ds18b20_init(star_ds18b20_handle_t* handle,
93 const star_ds18b20_config_t* config)
94{
95 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
96 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_tag, "Config is NULL");
97 ESP_RETURN_ON_FALSE(config->bus_manager, ESP_ERR_INVALID_ARG, s_tag, "Bus manager is NULL");
98 ESP_RETURN_ON_FALSE(config->bus_name, ESP_ERR_INVALID_ARG, s_tag, "Bus name is NULL");
99 ESP_RETURN_ON_FALSE(!handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Already initialized");
100
101 /* Zero out handle */
102 memset(handle, 0, sizeof(star_ds18b20_handle_t));
103
104 /* Store configuration */
105 handle->bus_manager = config->bus_manager;
106 handle->bus_name = config->bus_name;
107 handle->resolution = config->resolution;
108 handle->use_rom = config->use_rom;
109 handle->rom_code = config->rom_code;
110
111 /* If ROM code not provided and use_rom is true, try to read it */
112 if (handle->use_rom && handle->rom_code == 0) {
113 esp_err_t ret = star_sensor_ds18b20_read_rom(handle, &handle->rom_code);
114 if (ret != ESP_OK) {
115 ESP_LOGE(s_tag, "Failed to read ROM code: %s", esp_err_to_name(ret));
116 return ret;
117 }
118 ESP_LOGI(s_tag, "DS18B20 ROM code: 0x%016llX", handle->rom_code);
119 }
120
121 /* Configure resolution */
122 esp_err_t ret = internal_ds18b20_write_scratchpad(handle);
123 if (ret != ESP_OK) {
124 ESP_LOGE(s_tag, "Failed to write scratchpad: %s", esp_err_to_name(ret));
125 return ret;
126 }
127
128 handle->initialized = true;
129
130 ESP_LOGI(s_tag,
131 "DS18B20 initialized: bus=%s, resolution=%d-bit, rom=%s",
132 config->bus_name,
133 9 + config->resolution,
134 config->use_rom ? "yes" : "skip");
135
136 return ESP_OK;
137}

151esp_err_t star_sensor_ds18b20_read_temp(star_ds18b20_handle_t* handle, float* out_temperature)
152{
153 ESP_RETURN_ON_FALSE(handle && out_temperature,
154 ESP_ERR_INVALID_ARG,
155 s_tag,
156 "Handle or out_temperature is NULL");
157 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
158
159 /* Start temperature conversion */
160 esp_err_t ret = internal_ds18b20_start_conversion(handle);
161 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to start conversion");
162
163 /* Wait for conversion to complete */
164 vTaskDelay(pdMS_TO_TICKS(s_conversion_times_ms[handle->resolution]));
165
166 /* Read scratchpad */
167 uint8_t scratchpad[k_ds18b20_scratchpad_size];
168 ret = internal_ds18b20_read_scratchpad(handle, scratchpad);
169 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to read scratchpad");
170
171 /* Verify CRC (CRC is calculated over first 8 bytes, stored in byte 9) */
172 uint8_t crc = internal_ds18b20_crc8(scratchpad, k_ds18b20_scratchpad_size - 1);
173 if (crc != scratchpad[k_ds18b20_scratchpad_size - 1]) {
174 ESP_LOGE(s_tag,
175 "CRC mismatch: calculated=0x%02X, received=0x%02X",
176 crc,
177 scratchpad[k_ds18b20_scratchpad_size - 1]);
178 return ESP_ERR_INVALID_CRC;
179 }
180
181 /* Extract temperature (16-bit value in scratchpad[0:1]) */
182 int16_t raw_temp = (scratchpad[1] << 8) | scratchpad[0];
183
184 /* Convert to Celsius based on resolution */
185 *out_temperature = (float)raw_temp / 16.0f;
186
187 ESP_LOGD(s_tag, "Temperature: %.2f °C (raw=0x%04X)", *out_temperature, raw_temp);
188 return ESP_OK;
189}

211esp_err_t star_sensor_ds18b20_read_rom(star_ds18b20_handle_t* handle, uint64_t* out_rom)
212{
213 ESP_RETURN_ON_FALSE(handle && out_rom, ESP_ERR_INVALID_ARG, s_tag, "Handle or out_rom is NULL");
214
215 /* Send Read ROM command */
216 uint8_t rom_bytes[k_ds18b20_rom_size];
217 esp_err_t ret = star_bus_onewire_read_rom(handle->bus_manager, handle->bus_name, rom_bytes);
218 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to read ROM");
219
220 /* Convert byte array to 64-bit value */
221 *out_rom = 0;
222 for (int32_t i = 0; i < k_ds18b20_rom_size; i++) {
223 *out_rom |= ((uint64_t)rom_bytes[i]) << (i * k_ds18b20_bits_per_byte);
224 }
225
226 return ESP_OK;
227}

231static esp_err_t internal_ds18b20_send_rom_command(star_ds18b20_handle_t* handle)
232{
233 if (handle->use_rom) {
234 /* Match ROM - send ROM code to address specific peripheral */
235 uint8_t rom_bytes[k_ds18b20_rom_size];
236 for (int32_t i = 0; i < k_ds18b20_rom_size; i++) {
237 rom_bytes[i] = (uint8_t)(handle->rom_code >> (i * k_ds18b20_bits_per_byte));
238 }
239 return star_bus_onewire_match_rom(handle->bus_manager, handle->bus_name, rom_bytes);
240 } else {
241 /* Skip ROM - address all peripherals on bus */
242 return star_bus_onewire_skip_rom(handle->bus_manager, handle->bus_name);
243 }
244}

294static esp_err_t internal_ds18b20_write_scratchpad(star_ds18b20_handle_t* handle)
295{
296 /* Reset bus */
297 bool presence;
298 esp_err_t ret = star_bus_onewire_reset(handle->bus_manager, handle->bus_name, &presence);
299 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to reset bus");
300 ESP_RETURN_ON_FALSE(presence, ESP_ERR_NOT_FOUND, s_tag, "No device present on bus");
301
302 /* Send ROM command */
303 ret = internal_ds18b20_send_rom_command(handle);
304 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to send ROM command");
305
306 /* Send Write Scratchpad command */
307 ret = star_bus_onewire_write_byte(handle->bus_manager,
308 handle->bus_name,
309 k_ds18b20_cmd_write_scratchpad);
310 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to send Write Scratchpad command");
311
312 /* Write 3 data bytes: [TH, TL, CONFIG] */
313 uint8_t data[k_ds18b20_scratchpad_bytes];
314 data[0] = k_ds18b20_config_th_default; /* TH (high alarm, not used) */
315 data[1] = k_ds18b20_config_tl_default; /* TL (low alarm, not used) */
316 data[2] = (uint8_t)((handle->resolution << k_ds18b20_config_resolution_shift) |
317 k_ds18b20_config_resolution_mask); /* Configuration register */
318
319 for (int32_t i = 0; i < k_ds18b20_scratchpad_bytes; i++) {
320 ret = star_bus_onewire_write_byte(handle->bus_manager, handle->bus_name, data[i]);
321 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to write scratchpad byte");
322 }
323
324 return ESP_OK;
325}

327static uint8_t internal_ds18b20_crc8(const uint8_t* data, size_t len)
328{
329 uint8_t crc = 0;
330
331 for (size_t i = 0; i < len; i++) {
332 uint8_t byte = data[i];
333 for (int32_t j = 0; j < k_ds18b20_bits_per_byte; j++) {
334 uint8_t mix = (crc ^ byte) & k_ds18b20_crc_lsb_mask;
335 crc >>= 1;
336 if (mix) {
337 crc ^= k_ds18b20_crc_polynomial;
338 }
339 byte >>= 1;
340 }
341 }
342
343 return crc;
344}
