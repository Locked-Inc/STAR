119static void internal_motor_control_loop(motor_task_params_t* params)
120{
121 const TickType_t period_ticks = pdMS_TO_TICKS(k_motor_control_period_ms);
122 TickType_t last_wake_time = xTaskGetTickCount();
123 const float dt_s = k_motor_control_period_ms / 1000.0f;
124
125 int32_t positions[k_motor_count];
126 float velocities_rpm[k_motor_count];
127 float velocities_mps[k_motor_count];
128 float setpoints_mps[k_motor_count];
129 float outputs[k_motor_count];
130
131 ESP_LOGI(s_tag, "Motor control loop started at %dHz", k_motor_control_loop_hz);
132
133 while (true) {
134 /* 1. Read all 4 encoders (~80μs) */
135 esp_err_t ret = star_encoder_mux_read_all(params->encoder_mux,
136 positions,
137 velocities_rpm,
138 k_motor_control_period_ms,
139 k_encoder_ppr);
140
141 if (ret != ESP_OK) {
142 ESP_LOGW(s_tag, "Failed to read encoders: %s", esp_err_to_name(ret));
143 /* Continue with last known values */
144 }
145
146 /* 2. Convert RPM to m/s */
147 internal_convert_rpm_to_mps(velocities_rpm, velocities_mps);
148
149 /* 3. Read velocity setpoints from shared state (~20μs per motor) */
150 for (int i = 0; i < k_motor_count; i++) {
151 ret = motor_shared_state_get_velocity(params->shared_state, i, &setpoints_mps[i]);
152 if (ret != ESP_OK) {
153 /* On timeout, use last known setpoint (already in setpoints_mps) */
154 ESP_LOGD(s_tag, "Timeout reading setpoint for motor %d, using cached value", i);
155 }
156 }
157
158 /* 4. Compute PID for all 4 motors (~20μs per motor) */
159 for (int i = 0; i < k_motor_count; i++) {
160 ret =
161 star_pid_compute(&params->pids[i], setpoints_mps[i], velocities_mps[i], dt_s, &outputs[i]);
162
163 if (ret != ESP_OK) {
164 ESP_LOGW(s_tag, "PID compute failed for motor %d: %s", i, esp_err_to_name(ret));
165 outputs[i] = 0.0f; /* Safe fallback */
166 }
167 }
168
169 /* 5. Apply outputs to motors (~20μs per motor) */
170 for (int i = 0; i < k_motor_count; i++) {
171 ret = star_motor_set_duty(&params->motors[i], outputs[i]);
172
173 if (ret != ESP_OK) {
174 ESP_LOGW(s_tag, "Failed to set motor %d duty: %s", i, esp_err_to_name(ret));
175 }
176 }
177
178 /* 6. Update shared state with status (~25μs per motor) */
179 for (int i = 0; i < k_motor_count; i++) {
180 star_v1_MotorState motor_state;
181
182 /* Determine motor state */
183 if (fabsf(setpoints_mps[i]) < 0.01f && fabsf(outputs[i]) < 1.0f) {
184 motor_state = star_v1_MotorState_MOTOR_STATE_IDLE;
185 } else {
186 motor_state = star_v1_MotorState_MOTOR_STATE_RUNNING;
187 }
188
189 /* Update status (ignore timeout - not critical) */
190 motor_shared_state_set_status(params->shared_state,
191 i,
192 velocities_mps[i],
193 outputs[i],
194 0.0f, /* TODO: Add current sensing */
195 motor_state);
196 }
197
198 /* 7. Check emergency stop (~20μs) */
199 bool estop_active;
200 ret = motor_shared_state_get_estop(params->shared_state, &estop_active);
201
202 if (ret == ESP_ERR_TIMEOUT) {
203 /* Fail-safe: assume emergency stop on mutex timeout */
204 estop_active = true;
205 }
206
207 if (estop_active) {
208 /* Emergency stop: brake all motors */
209 for (int i = 0; i < k_motor_count; i++) {
210 star_motor_stop(&params->motors[i], true); /* Brake */
211
212 /* Update state to ESTOP */
213 motor_shared_state_set_status(params->shared_state,
214 i,
215 0.0f,
216 0.0f,
217 0.0f,
218 star_v1_MotorState_MOTOR_STATE_ESTOP);
219 }
220 }
221
222 /* 8. Wait for next period (maintains precise 250Hz) */
223 vTaskDelayUntil(&last_wake_time, period_ticks);
224 }
225}

259static void internal_convert_rpm_to_mps(const float velocities_rpm[k_motor_count],
260 float velocities_mps[k_motor_count])
261{
262 const float diameter_m = k_wheel_diameter_m / 1000.0f; /* mm to m */
263 const float conversion_factor = (M_PI * diameter_m) / 60.0f;
264
265 for (int i = 0; i < k_motor_count; i++) {
266 velocities_mps[i] = velocities_rpm[i] * conversion_factor;
267 }
