50star_bus_config_t* star_bus_config_create_i2c(const char* name,
51 i2c_port_t port,
52 uint8_t address,
53 gpio_num_t sda_pin,
54 gpio_num_t scl_pin,
55 uint32_t clk_speed)
56{
57 star_bus_config_t* config = internal_star_bus_config_create_common(name, k_star_bus_type_i2c);
58 ESP_RETURN_ON_FALSE(config,
59 NULL,
60 s_TAG,
61 "Failed to create common config for %s",
62 name ? name : "NULL");
63
64 /* Configure I2C-specific parameters */
65 config->proto.i2c.port = port;
66 config->proto.i2c.address = address;
67
68 /* Initialize with default I2C configuration */
69 memset(&config->proto.i2c.config, 0, sizeof(i2c_config_t));
70 config->proto.i2c.config.mode = I2C_MODE_MASTER;
71 config->proto.i2c.config.sda_io_num = sda_pin;
72 config->proto.i2c.config.scl_io_num = scl_pin;
73 config->proto.i2c.config.sda_pullup_en = GPIO_PULLUP_ENABLE; /* Common practice */
74 config->proto.i2c.config.scl_pullup_en = GPIO_PULLUP_ENABLE; /* Common practice */
75 config->proto.i2c.config.master.clk_speed = clk_speed;
76
77 /* Initialize default operations */
78 star_bus_i2c_init_default_ops(&config->proto.i2c.ops);
79
80 ESP_LOGI(s_TAG,
81 "Created I2C config '%s' (Port: %d, Addr: 0x%02X, SDA: %d, SCL: %d, Speed: %" PRIu32
82 " Hz)",
83 name,
84 port,
85 address,
86 sda_pin,
87 scl_pin,
88 clk_speed);
89 return config;
90}

92star_bus_config_t* star_bus_config_create_spi_device(const char* name,
93 spi_host_device_t host,
94 gpio_num_t copi_pin,
95 gpio_num_t cipo_pin,
96 gpio_num_t sclk_pin,
97 int32_t dma_chan,
98 const spi_device_interface_config_t* dev_cfg)
99{
100 ESP_RETURN_ON_FALSE(dev_cfg, NULL, s_TAG, "SPI device config cannot be NULL");
101 ESP_RETURN_ON_FALSE(host >= 0 && host < SPI_HOST_MAX,
102 NULL,
103 s_TAG,
104 "Invalid SPI host number: %d",
105 host);
106
107 star_bus_config_t* config = internal_star_bus_config_create_common(name, k_star_bus_type_spi);
108 ESP_RETURN_ON_FALSE(config,
109 NULL,
110 s_TAG,
111 "Failed to create common config for %s",
112 name ? name : "NULL");
113
114 /* Configure SPI-specific parameters */
115 config->proto.spi.host = host;
116
117 /* Initialize SPI bus configuration structure within the config */
118 /* We copy the relevant pin numbers to our internal structure */
119 memset(&config->proto.spi.bus_cfg, 0, sizeof(spi_bus_config_t));
120 config->proto.spi.copi_io_num = copi_pin;
121 config->proto.spi.cipo_io_num = cipo_pin;
122 config->proto.spi.sclk_io_num = sclk_pin;
123 config->proto.spi.quadwp_io_num = -1; /* Not used for standard SPI, only QuadSPI */
124 config->proto.spi.quadhd_io_num = -1;
125 config->proto.spi.max_transfer_sz =
126 0; /* 0 means use default (usually 4092 bytes if DMA enabled, 64 otherwise) */
127
128 /* Copy the provided device configuration */
129 memcpy(&config->proto.spi.dev_cfg, dev_cfg, sizeof(spi_device_interface_config_t));
130
131 /* Store DMA channel request */
132 config->proto.spi.bus_cfg.flags = (dma_chan > 0) ? SPICOMMON_BUSFLAG_MASTER : 0;
133 /* The actual DMA channel assignment happens in spi_bus_initialize */
134
135 /* Mark as controller mode (not peripheral) */
136 config->proto.spi.is_peripheral = false;
137
138 /* Initialize default operations */
139 star_bus_spi_init_default_ops(&config->proto.spi.ops);
140
141 ESP_LOGI(
142 s_TAG,
143 "Created SPI config '%s' (Host: %d, COPI: %d, CIPO: %d, SCLK: %d, CS: %d, Mode: %d, Speed: %" PRId32
144 " Hz, DMA: %s)",
145 name,
146 host,
147 copi_pin,
148 cipo_pin,
149 sclk_pin,
150 dev_cfg->spics_io_num,
151 dev_cfg->mode,
152 (long)dev_cfg->clock_speed_hz,
153 (dma_chan > 0) ? "Enabled" : "Disabled");
154
155 return config;
156}

313esp_err_t star_bus_config_destroy(star_bus_config_t* config)
314{
315 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config pointer is NULL");
316
317 /* If the bus/device is initialized, deinitialize it first */
318 if (config->initialized) {
319 esp_err_t ret = star_bus_config_deinit(config);
320 if (ret != ESP_OK) {
321 ESP_LOGE(s_TAG,
322 "Failed to deinitialize bus '%s' before destroying: %s. Continuing cleanup.",
323 config->name ? config->name : "UNKNOWN",
324 esp_err_to_name(ret));
325 /* Continue cleanup even if deinit failed to avoid memory leaks */
326 }
327 }
328
329 ESP_LOGI(s_TAG, "Destroying bus config '%s'", config->name ? config->name : "UNKNOWN");
330
331 /* Specific cleanup for SPI is handled by the manager during remove_bus */
332 /* The manager checks if the bus needs to be freed based on device count */
333
334 /* Clean up and free the configuration */
335 internal_star_bus_config_cleanup(config);
336
337 /* IMPORTANT: After this call, the config pointer is invalid. */
338 return ESP_OK;
339}

352esp_err_t star_bus_config_init(star_bus_config_t* config, star_bus_manager_t* manager)
353{
354 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config pointer is NULL");
355 ESP_RETURN_ON_FALSE(!config->initialized,
356 ESP_ERR_INVALID_STATE,
357 s_TAG,
358 "Bus '%s' is already initialized",
359 config->name ? config->name : "UNKNOWN");
360
361 esp_err_t ret = ESP_OK;
362 const char* bus_name = config->name ? config->name : "UNKNOWN"; /* Use for logging */
363
364 switch (config->type) {
365 case k_star_bus_type_i2c:
366 /* Initialize I2C bus */
367 ESP_LOGI(s_TAG, "Initializing I2C bus '%s' (Port %d)", bus_name, config->proto.i2c.port);
368
369 ret = i2c_param_config(config->proto.i2c.port, &config->proto.i2c.config);
370 ESP_GOTO_ON_ERROR(ret,
371 fail,
372 s_TAG,
373 "i2c_param_config failed for '%s': %s",
374 bus_name,
375 esp_err_to_name(ret));
376
377 ret = i2c_driver_install(config->proto.i2c.port,
378 config->proto.i2c.config.mode,
379 0, /* RX buffer size (controller mode) */
380 0, /* TX buffer size (controller mode) */
381 0); /* Flags */
382 ESP_GOTO_ON_ERROR(ret,
383 fail_i2c_driver,
384 s_TAG,
385 "i2c_driver_install failed for '%s': %s",
386 bus_name,
387 esp_err_to_name(ret));
388 break;
389
390 case k_star_bus_type_spi:
391 ESP_RETURN_ON_FALSE(manager,
392 ESP_ERR_INVALID_ARG,
393 s_TAG,
394 "Bus manager pointer is required for SPI initialization");
395
396 spi_host_device_t host = config->proto.spi.host;
397 ESP_RETURN_ON_FALSE(host >= 0 && host < SPI_HOST_MAX,
398 ESP_ERR_INVALID_ARG,
399 s_TAG,
400 "Invalid SPI host number: %d",
401 host);
402
403 /* Check if this is peripheral or controller mode */
404 if (config->proto.spi.is_peripheral) {
405 /* SPI Peripheral Mode */
406 ESP_LOGI(s_TAG, "Initializing SPI peripheral '%s' (Host %d)", bus_name, host);
407
408 spi_slave_interface_config_t slave_cfg = {
409 .spics_io_num = config->proto.spi.cs_io_num,
410 .flags = 0,
411 .queue_size = config->proto.spi.queue_size,
412 .mode = config->proto.spi.mode,
413 .post_setup_cb = NULL,
414 .post_trans_cb = NULL,
415 };
416
417 /* Map COPI/CIPO to ESP-IDF's MOSI/MISO */
418 spi_bus_config_t bus_cfg = {
419 .mosi_io_num = config->proto.spi.copi_io_num,
420 .miso_io_num = config->proto.spi.cipo_io_num,
421 .sclk_io_num = config->proto.spi.sclk_io_num,
422 .quadwp_io_num = -1,
423 .quadhd_io_num = -1,
424 .max_transfer_sz = 0,
425 };
426
427 /* DMA for SPI peripheral is controlled by the DMA channel parameter */
428#if CONFIG_IDF_TARGET_ESP32S3
429 /* ESP32-S3: Enable DMA for better performance (95% CPU reduction) */
430 int dma_chan = SPI_DMA_CH_AUTO;
431#else
432 /* ESP32-WROOM: No DMA support for SPI peripheral */
433 int dma_chan = SPI_DMA_DISABLED;
434#endif
435
436 ret = spi_slave_initialize(host, &bus_cfg, &slave_cfg, dma_chan);
437 ESP_GOTO_ON_ERROR(ret,
438 fail,
439 s_TAG,
440 "spi_slave_initialize failed for '%s': %s",
441 bus_name,
442 esp_err_to_name(ret));
443
444 manager->spi_host_initialized[host] = true;
445 ESP_LOGI(s_TAG, "SPI peripheral '%s' initialized successfully", bus_name);
446 } else {
447 /* SPI Controller Mode */
448 ESP_LOGI(s_TAG, "Initializing SPI device '%s' (Host %d)", bus_name, host);
449
450 /* Initialize the SPI bus *only if* it hasn't been initialized yet for this host */
451 if (!manager->spi_host_initialized[host]) {
452 ESP_LOGI(s_TAG, "Initializing SPI bus driver for host %d...", host);
453 /* Map COPI/CIPO to ESP-IDF's MOSI/MISO */
454 spi_bus_config_t esp_idf_bus_cfg = {
455 .mosi_io_num = config->proto.spi.copi_io_num,
456 .miso_io_num = config->proto.spi.cipo_io_num,
457 .sclk_io_num = config->proto.spi.sclk_io_num,
458 .quadwp_io_num = config->proto.spi.quadwp_io_num,
459 .quadhd_io_num = config->proto.spi.quadhd_io_num,
460 .max_transfer_sz = config->proto.spi.max_transfer_sz,
461 .flags = config->proto.spi.bus_cfg.flags,
462 };
463 int32_t dma_chan = (config->proto.spi.bus_cfg.flags & SPICOMMON_BUSFLAG_MASTER)
464 ? SPI_DMA_CH_AUTO
465 : SPI_DMA_DISABLED;
466 ret = spi_bus_initialize(host, &esp_idf_bus_cfg, dma_chan);
467 ESP_GOTO_ON_ERROR(ret,
468 fail,
469 s_TAG,
470 "spi_bus_initialize failed for host %d: %s",
471 host,
472 esp_err_to_name(ret));
473 manager->spi_host_initialized[host] = true;
474 manager->spi_device_count[host] = 0;
475 ESP_LOGI(s_TAG, "SPI bus driver for host %d initialized.", host);
476 } else {
477 ESP_LOGD(s_TAG, "SPI bus driver for host %d already initialized.", host);
478 }
479
480 /* Add the SPI device to the bus */
481 spi_device_handle_t spi_handle;
482 ret = spi_bus_add_device(host, &config->proto.spi.dev_cfg, &spi_handle);
483 ESP_GOTO_ON_ERROR(ret,
484 fail_spi_add_device,
485 s_TAG,
486 "spi_bus_add_device failed for '%s': %s",
487 bus_name,
488 esp_err_to_name(ret));
489 config->handle = (void*)spi_handle;
490 manager->spi_device_count[host]++;
491 }
492 break;
493
494 case k_star_bus_type_gpio: {
495 /* Initialize GPIO pins */
496 ESP_LOGI(s_TAG,
497 "Initializing GPIO bus '%s' with %d pins",
498 bus_name,
499 config->proto.gpio.pin_count);
500
501 for (uint8_t i = 0; i < config->proto.gpio.pin_count; i++) {
502 gpio_num_t pin = config->proto.gpio.pins[i];
503
504 /* Configure each pin as output with default low level */
505 gpio_config_t io_conf = {
506 .pin_bit_mask = (1ULL << pin),
507 .mode = GPIO_MODE_INPUT_OUTPUT, /* Allow both input and output */
508 .pull_up_en = GPIO_PULLUP_DISABLE,
509 .pull_down_en = GPIO_PULLDOWN_DISABLE,
510 .intr_type = GPIO_INTR_DISABLE,
511 };
512
513 ret = gpio_config(&io_conf);
514 ESP_GOTO_ON_ERROR(ret,
515 fail,
516 s_TAG,
517 "gpio_config failed for pin %d: %s",
518 pin,
519 esp_err_to_name(ret));
520 }
521
522 ESP_LOGI(s_TAG, "GPIO bus '%s' initialized successfully", bus_name);
523 break;
524 }
525
526 case k_star_bus_type_adc: {
527 /* Initialize ADC unit */
528 ESP_RETURN_ON_FALSE(manager,
529 ESP_ERR_INVALID_ARG,
530 s_TAG,
531 "Bus manager pointer is required for ADC initialization");
532
533 adc_unit_t unit = config->proto.adc.unit;
534 ESP_LOGI(s_TAG,
535 "Initializing ADC bus '%s' (Unit: %d, Channel: %d)",
536 bus_name,
537 unit,
538 config->proto.adc.channel);
539
540 /* Check if ADC unit is already initialized (shared across channels) */
541 if (manager->adc_unit_handles[unit] == NULL) {
542 /* First channel on this unit - create the unit handle */
543 ESP_LOGI(s_TAG, "Creating new ADC unit handle for unit %d", unit);
544
545 adc_oneshot_unit_init_cfg_t unit_cfg = {
546 .unit_id = unit,
547 .ulp_mode = ADC_ULP_MODE_DISABLE,
548 };
549
550 ret = adc_oneshot_new_unit(&unit_cfg, &manager->adc_unit_handles[unit]);
551 ESP_GOTO_ON_ERROR(ret,
552 fail,
553 s_TAG,
554 "adc_oneshot_new_unit failed for '%s': %s",
555 bus_name,
556 esp_err_to_name(ret));
557 } else {
558 /* Unit already initialized - reuse the handle */
559 ESP_LOGI(s_TAG, "Reusing existing ADC unit handle for unit %d", unit);
560 }
561
562 /* Store the shared unit handle in this config */
563 config->proto.adc.unit_handle = manager->adc_unit_handles[unit];
564
565 /* Configure this specific ADC channel */
566 adc_oneshot_chan_cfg_t chan_cfg = {
567 .bitwidth = config->proto.adc.bitwidth,
568 .atten = config->proto.adc.atten,
569 };
570
571 ret = adc_oneshot_config_channel(config->proto.adc.unit_handle,
572 config->proto.adc.channel,
573 &chan_cfg);
574 ESP_GOTO_ON_ERROR(ret,
575 fail_adc_unit,
576 s_TAG,
577 "adc_oneshot_config_channel failed for '%s': %s",
578 bus_name,
579 esp_err_to_name(ret));
580
581 /* Increment channel count for this unit */
582 manager->adc_channel_count[unit]++;
583
584 /* Try to initialize calibration (optional, may fail on some chips) */
585#if ADC_CALI_SCHEME_CURVE_FITTING_SUPPORTED
586 adc_cali_curve_fitting_config_t cali_cfg = {
587 .unit_id = unit,
588 .chan = config->proto.adc.channel,
589 .atten = config->proto.adc.atten,
590 .bitwidth = config->proto.adc.bitwidth,
591 };
592
593 ret = adc_cali_create_scheme_curve_fitting(&cali_cfg, &config->proto.adc.cali_handle);
594 if (ret == ESP_OK) {
595 ESP_LOGI(s_TAG, "ADC calibration (curve fitting) initialized for '%s'", bus_name);
596 } else {
597 ESP_LOGW(s_TAG,
598 "ADC calibration initialization failed for '%s': %s (continuing without "
599 "calibration)",
600 bus_name,
601 esp_err_to_name(ret));
602 config->proto.adc.cali_handle = NULL;
603 }
604#elif ADC_CALI_SCHEME_LINE_FITTING_SUPPORTED
605 adc_cali_line_fitting_config_t cali_cfg = {
606 .unit_id = unit,
607 .atten = config->proto.adc.atten,
608 .bitwidth = config->proto.adc.bitwidth,
609 };
610
611 ret = adc_cali_create_scheme_line_fitting(&cali_cfg, &config->proto.adc.cali_handle);
612 if (ret == ESP_OK) {
613 ESP_LOGI(s_TAG, "ADC calibration (line fitting) initialized for '%s'", bus_name);
614 } else {
615 ESP_LOGW(s_TAG,
616 "ADC calibration initialization failed for '%s': %s (continuing without "
617 "calibration)",
618 bus_name,
619 esp_err_to_name(ret));
620 config->proto.adc.cali_handle = NULL;
621 }
622#else
623 ESP_LOGW(s_TAG, "ADC calibration not supported on this chip, continuing without calibration");
624 config->proto.adc.cali_handle = NULL;
625#endif
626
627 ESP_LOGI(s_TAG, "ADC bus '%s' initialized successfully", bus_name);
628 break;
629
630 fail_adc_unit:
631 /* Only delete the unit if we created it (first channel on this unit) */
632 manager->adc_channel_count[unit]--;
633 if (manager->adc_channel_count[unit] == 0 && manager->adc_unit_handles[unit]) {
634 adc_oneshot_del_unit(manager->adc_unit_handles[unit]);
635 manager->adc_unit_handles[unit] = NULL;
636 }
637 config->proto.adc.unit_handle = NULL;
638 goto fail;
639 }
640
641 case k_star_bus_type_onewire: {
642 /* Initialize OneWire bus */
643 ESP_RETURN_ON_FALSE(manager,
644 ESP_ERR_INVALID_ARG,
645 s_TAG,
646 "Bus manager pointer is required for OneWire initialization");
647
648 ESP_LOGI(s_TAG,
649 "Initializing OneWire bus '%s' (Pin: %d)",
650 bus_name,
651 config->proto.onewire.gpio_pin);
652
653 /* Create onewire config and call star_bus_onewire_init() */
654 star_onewire_config_t ow_cfg;
655 ow_cfg.gpio_pin = config->proto.onewire.gpio_pin;
656 ow_cfg.speed = (star_onewire_speed_t)config->proto.onewire.speed;
657 ow_cfg.use_parasitic_power = config->proto.onewire.use_parasitic_power;
658 ow_cfg.use_strong_pullup = config->proto.onewire.use_strong_pullup;
659 ow_cfg.search_timeout_ms = config->proto.onewire.search_timeout_ms;
660
661 ret = star_bus_onewire_init(manager, config->name, &ow_cfg);
662 ESP_GOTO_ON_ERROR(ret,
663 fail,
664 s_TAG,
665 "star_bus_onewire_init failed for '%s': %s",
666 bus_name,
667 esp_err_to_name(ret));
668
669 ESP_LOGI(s_TAG, "OneWire bus '%s' initialized successfully", bus_name);
670 break;
671 }
672
673 default:
674 ESP_LOGE(s_TAG, "Unsupported bus type for initialization: %d", config->type);
675 ret = ESP_ERR_NOT_SUPPORTED;
676 goto fail;
677 }
678
679 /* Mark as initialized */
680 config->initialized = true;
681 ESP_LOGI(s_TAG,
682 "Successfully initialized bus/device '%s' (%s)",
683 bus_name,
684 star_bus_type_to_string(config->type));
685 return ESP_OK;
686
687/* Failure points with specific cleanup needs */
688fail_i2c_driver:
689 /* If driver install failed after param_config, no specific cleanup needed for param_config */
690 goto fail;
691
692fail_spi_add_device:
693 /* If add_device failed, we might need to free the bus if we just initialized it */
694 /* This logic is now handled by the manager during remove_bus */
695 goto fail;
696
697/* Generic failure point */
698fail:
699 ESP_LOGE(s_TAG, "Failed to initialize bus/device '%s'", bus_name);
700 return ret;
701}

703esp_err_t star_bus_config_deinit(star_bus_config_t* config)
704{
705 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config pointer is NULL");
706 ESP_RETURN_ON_FALSE(config->initialized,
707 ESP_ERR_INVALID_STATE,
708 s_TAG,
709 "Bus/device '%s' is not initialized",
710 config->name ? config->name : "UNKNOWN");
711
712 esp_err_t ret = ESP_OK;
713 const char* bus_name = config->name ? config->name : "UNKNOWN";
714
715 switch (config->type) {
716 case k_star_bus_type_i2c:
717 /* Deinitialize I2C bus */
718 ESP_LOGI(s_TAG, "Deinitializing I2C bus '%s' (Port %d)", bus_name, config->proto.i2c.port);
719
720 ret = i2c_driver_delete(config->proto.i2c.port);
721 break;
722
723 case k_star_bus_type_spi:
724 if (config->proto.spi.is_peripheral) {
725 /* SPI Peripheral Mode - Free the peripheral driver */
726 ESP_LOGI(s_TAG,
727 "Deinitializing SPI peripheral '%s' (Host %d)",
728 bus_name,
729 config->proto.spi.host);
730 ret = spi_slave_free(config->proto.spi.host);
731 } else {
732 /* SPI Controller Mode - Remove device */
733 ESP_LOGI(s_TAG,
734 "Deinitializing SPI device '%s' (Host %d)",
735 bus_name,
736 config->proto.spi.host);
737 if (config->handle) {
738 ret = spi_bus_remove_device((spi_device_handle_t)config->handle);
739 config->handle = NULL;
740 } else {
741 ESP_LOGW(s_TAG, "SPI device '%s' handle was already NULL before deinit.", bus_name);
742 ret = ESP_OK;
743 }
744 }
745 break;
746
747 case k_star_bus_type_gpio:
748 /* GPIO pins don't need explicit deinitialization - they can be reconfigured anytime */
749 ESP_LOGI(s_TAG, "Deinitializing GPIO bus '%s'", bus_name);
750 ret = ESP_OK;
751 break;
752
753 case k_star_bus_type_adc:
754 /* Deinitialize ADC channel */
755 ESP_LOGI(s_TAG, "Deinitializing ADC bus '%s'", bus_name);
756
757 /* Delete calibration if it was initialized */
758 if (config->proto.adc.cali_handle) {
759#if ADC_CALI_SCHEME_CURVE_FITTING_SUPPORTED
760 ret = adc_cali_delete_scheme_curve_fitting(config->proto.adc.cali_handle);
761#elif ADC_CALI_SCHEME_LINE_FITTING_SUPPORTED
762 ret = adc_cali_delete_scheme_line_fitting(config->proto.adc.cali_handle);
763#endif
764 if (ret != ESP_OK) {
765 ESP_LOGW(s_TAG,
766 "Failed to delete ADC calibration for '%s': %s",
767 bus_name,
768 esp_err_to_name(ret));
769 }
770 config->proto.adc.cali_handle = NULL;
771 }
772
773 /* Note: ADC unit handle is shared and managed by bus_manager.
774 * It will be deleted when the last channel using this unit is removed. */
775 config->proto.adc.unit_handle = NULL;
776 ret = ESP_OK;
777 break;
778
779 case k_star_bus_type_onewire:
780 /* Deinitialize OneWire bus */
781 ESP_LOGI(s_TAG, "Deinitializing OneWire bus '%s'", bus_name);
782 /* Call with NULL manager - the function doesn't use it, only needs bus_name */
783 ret = star_bus_onewire_deinit(NULL, bus_name);
784 if (ret != ESP_OK) {
785 ESP_LOGW(s_TAG,
786 "Failed to deinitialize OneWire bus '%s': %s",
787 bus_name,
788 esp_err_to_name(ret));
789 }
790 break;
791
792 default:
793 ESP_LOGE(s_TAG, "Unsupported bus type for deinitialization: %d", config->type);
794 ret = ESP_ERR_NOT_SUPPORTED;
795 break;
796 }
797
798 if (ret != ESP_OK) {
799 ESP_LOGE(s_TAG, "Failed to deinitialize bus/device '%s': %s", bus_name, esp_err_to_name(ret));
800 /* Do not return early, mark as uninitialized anyway */
801 }
802
803 /* Mark as not initialized */
804 config->initialized = false;
805 config->handle = NULL; /* Clear handle */
806
807 if (ret == ESP_OK) {
808 ESP_LOGI(s_TAG, "Successfully deinitialized bus/device '%s'", bus_name);
809 }
810
811 return ret; /* Return the ret of the deinit operation */
812}

849static star_bus_config_t* internal_star_bus_config_create_common(const char* name,
850 star_bus_type_t type)
851{
852 /* Name validation */
853 if (name == NULL) {
854 ESP_LOGE(s_TAG, "Bus name cannot be NULL");
855 return NULL;
856 }
857
858 /* Allocate memory for the bus configuration */
859 star_bus_config_t* config = (star_bus_config_t*)malloc(sizeof(star_bus_config_t));
860 if (config == NULL) {
861 ESP_LOGE(s_TAG, "Failed to allocate memory for bus configuration");
862 return NULL;
863 }
864
865 /* Initialize the configuration to zeros */
866 memset(config, 0, sizeof(star_bus_config_t));
867
868 /* Make a copy of the bus name */
869 char* name_copy = strdup(name);
870 if (name_copy == NULL) {
871 ESP_LOGE(s_TAG, "Failed to allocate memory for bus name");
872 free(config);
873 return NULL;
874 }
875 config->name = name_copy; /* Store the copy */
876
877 /* Set common parameters */
878 config->type = type;
879 config->initialized = false;
880 config->handle = NULL;
881 config->user_ctx = NULL;
882 config->next = NULL;
883
884 /* Initialize callback structs to NULL */
885 memset(&config->proto.i2c.callbacks, 0, sizeof(config->proto.i2c.callbacks));
886 memset(&config->proto.spi.callbacks, 0, sizeof(config->proto.spi.callbacks));
887
888 return config;
889}
