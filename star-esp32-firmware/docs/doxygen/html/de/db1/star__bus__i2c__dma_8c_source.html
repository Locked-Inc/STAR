86bool star_i2c_is_dma_capable(const void* buffer, size_t size)
87{
88 if (buffer == NULL || size == 0) {
89 return false;
90 }
91
92 /* Check alignment */
93 if (((uintptr_t)buffer % STAR_I2C_DMA_ALIGNMENT) != 0) {
94 return false;
95 }
96
97 /* Check if buffer is in internal RAM (DMA-capable on ESP32) */
98 if (!esp_ptr_internal(buffer)) {
99 return false;
100 }
101
102 /* Verify the buffer region is in DMA-capable memory */
103 return esp_ptr_dma_capable(buffer);
104}

106esp_err_t star_bus_i2c_configure_dma(star_bus_manager_t* manager,
107 const char* bus_name,
108 const star_i2c_dma_config_t* config)
109{
110 if (manager == NULL || bus_name == NULL || config == NULL) {
111 ESP_LOGE(s_TAG, "Invalid parameters");
112 return ESP_ERR_INVALID_ARG;
113 }
114
115 /* Get I2C port */
116 i2c_port_t port = get_i2c_port_from_bus(manager, bus_name);
117 if (port == I2C_NUM_MAX) {
118 ESP_LOGE(s_TAG, "Bus '%s' not found or not I2C", bus_name);
119 return ESP_ERR_NOT_FOUND;
120 }
121
122 /* Get DMA state */
123 i2c_dma_state_t* state = get_dma_state(port);
124 if (state == NULL) {
125 return ESP_ERR_INVALID_STATE;
126 }
127
128 /* Validate configuration */
129 if (config->dma_channel >= 2) { /* ESP32 has 2 DMA channels */
130 ESP_LOGE(s_TAG, "Invalid DMA channel: %d", config->dma_channel);
131 return ESP_ERR_INVALID_ARG;
132 }
133
134 /* Store configuration */
135 memcpy(&state->config, config, sizeof(star_i2c_dma_config_t));
136
137 ESP_LOGI(s_TAG,
138 "DMA configured for bus '%s': enabled=%d, min_size=%" PRIu32 ", channel=%d",
139 bus_name,
140 config->enabled,
141 config->min_transfer_size,
142 config->dma_channel);
143
144 return ESP_OK;
145}

173esp_err_t star_bus_i2c_write_dma(star_bus_manager_t* manager,
174 const char* bus_name,
175 const uint8_t* data,
176 size_t length,
177 uint8_t command,
178 star_i2c_dma_callback_t callback,
179 void* context)
180{
181 if (manager == NULL || bus_name == NULL || data == NULL || length == 0) {
182 ESP_LOGE(s_TAG, "Invalid parameters");
183 return ESP_ERR_INVALID_ARG;
184 }
185
186 /* Get I2C port */
187 i2c_port_t port = get_i2c_port_from_bus(manager, bus_name);
188 if (port == I2C_NUM_MAX) {
189 ESP_LOGE(s_TAG, "Bus '%s' not found or not I2C", bus_name);
190 return ESP_ERR_NOT_FOUND;
191 }
192
193 /* Get DMA state */
194 i2c_dma_state_t* state = get_dma_state(port);
195 if (state == NULL) {
196 return ESP_ERR_INVALID_STATE;
197 }
198
199 /* Check if DMA should be used */
200 bool use_dma = internal_should_use_dma(state, length);
201
202 if (!use_dma) {
203 /* Fall back to standard I2C write */
204 ESP_LOGD(s_TAG,
205 "Using standard I2C write (length=%zu, threshold=%" PRIu32 ")",
206 length,
207 state->config.min_transfer_size);
208 esp_err_t result = star_bus_i2c_write(manager, bus_name, data, length, command, NULL);
209
210 /* Call callback if provided */
211 if (callback != NULL) {
212 callback(result, context);
213 }
214
215 return result;
216 }
217
218 /* Verify buffer is DMA-capable */
219 if (!star_i2c_is_dma_capable(data, length)) {
220 ESP_LOGW(s_TAG, "Buffer not DMA-capable, falling back to standard I2C");
221 esp_err_t result = star_bus_i2c_write(manager, bus_name, data, length, command, NULL);
222
223 if (callback != NULL) {
224 callback(result, context);
225 }
226
227 return result;
228 }
229
230 ESP_LOGD(s_TAG, "Using DMA for write: %zu bytes", length);
231
232 /* Store callback context */
233 state->transfer_context.callback = callback;
234 state->transfer_context.user_context = context;
235 state->transfer_context.in_progress = true;
236
237 /* Perform DMA-backed I2C write */
238 /* Note: ESP-IDF i2c_master_write_to_device supports DMA automatically
239 * when buffers are in DMA-capable memory */
240 esp_err_t result = star_bus_i2c_write(manager, bus_name, data, length, command, NULL);
241
242 /* Update statistics */
243 if (result == ESP_OK) {
244 state->transfer_count++;
245 state->bytes_transferred += length;
246 }
247
248 /* Mark transfer complete */
249 state->transfer_context.in_progress = false;
250
251 /* Call callback if provided */
252 if (callback != NULL) {
253 callback(result, context);
254 }
255
256 return result;
257}

405static bool internal_should_use_dma(const i2c_dma_state_t* state, size_t length)
406{
407 if (state == NULL || !state->config.enabled) {
408 return false;
409 }
410
411 if (length < state->config.min_transfer_size) {
412 return false;
413 }
414
415 if (length > STAR_I2C_DMA_MAX_SIZE) {
416 return false;
417 }
418
419 return true;
