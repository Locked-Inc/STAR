143static inline uint8_t pca9685_freq_to_prescale_calc(uint16_t freq_hz)
144{
145 // Using oscillator frequency from hardware limits
146 // Formula: prescale = round(oscillator / (4096 * freq)) - 1
147 return (uint8_t)((25000000 / (4096UL * (uint32_t)freq_hz)) - 1);
148}
149
155static inline uint16_t pca9685_prescale_to_freq_calc(uint8_t prescale)
156{
157 // Formula: freq = oscillator / (4096 * (prescale + 1))
158 return (uint16_t)(25000000 / (4096UL * ((uint32_t)prescale + 1)));
159}

166static inline uint8_t pca9685_channel_base_register(uint8_t channel)
167{
168 // Each channel uses 4 registers (ON_L, ON_H, OFF_L, OFF_H)
169 return k_pca9685_reg_led0_on_l + (channel * 4);
170}
