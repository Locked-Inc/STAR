67esp_err_t star_bus_spi_transmit(const star_bus_manager_t* manager,
68 const char* name,
69 const void* tx_buffer,
70 size_t len,
71 uint32_t user_flags)
72{
73 ESP_RETURN_ON_FALSE(manager && name && tx_buffer,
74 ESP_ERR_INVALID_ARG,
75 s_tag,
76 "Manager, name, or tx_buffer is NULL");
77 /* Allow len == 0 for command-only transfers? Maybe not useful for SPI. */
78 ESP_RETURN_ON_FALSE(len > 0, ESP_ERR_INVALID_ARG, s_tag, "Transmit length must be > 0");
79
80 star_bus_config_t* bus_config = star_bus_manager_find_bus(manager, name);
81 ESP_RETURN_ON_FALSE(bus_config, ESP_ERR_NOT_FOUND, s_tag, "Bus '%s' not found", name);
82 ESP_RETURN_ON_FALSE(bus_config->type == k_star_bus_type_spi,
83 ESP_ERR_INVALID_ARG,
84 s_tag,
85 "Bus '%s' is not an SPI bus",
86 name);
87 ESP_RETURN_ON_FALSE(bus_config->proto.spi.ops.transmit,
88 ESP_ERR_NOT_SUPPORTED,
89 s_tag,
90 "No transmit op for '%s'",
91 name);
92
93 return bus_config->proto.spi.ops.transmit(bus_config, tx_buffer, len, user_flags);
94}

155/* Default implementation using queued transactions */
156static esp_err_t internal_star_bus_spi_transmit(const star_bus_config_t* config,
157 const void* tx_buffer,
158 size_t len,
159 uint32_t user_flags)
160{
161 ESP_RETURN_ON_FALSE(config->initialized,
162 ESP_ERR_INVALID_STATE,
163 s_tag,
164 "SPI device '%s' is not initialized",
165 config->name);
166 ESP_RETURN_ON_FALSE(config->handle,
167 ESP_ERR_INVALID_STATE,
168 s_tag,
169 "SPI device '%s' handle is NULL",
170 config->name);
171
172 /* Check for overflow: length in bits must fit in uint32_t */
173 ESP_RETURN_ON_FALSE(len <= (UINT32_MAX / 8),
174 ESP_ERR_INVALID_ARG,
175 s_tag,
176 "SPI transmit length too large: %zu bytes (max: %lu)",
177 len,
178 (unsigned long)(UINT32_MAX / 8));
179
180 spi_transaction_t trans;
181 memset(&trans, 0, sizeof(trans));
182 trans.length = len * 8; /* Length is in bits */
183 trans.tx_buffer = tx_buffer;
184 trans.rx_buffer = NULL;
185 trans.user = (void*)user_flags; /* Pass user flags (e.g., for DC pin) */
186
187 esp_err_t ret = spi_device_polling_transmit((spi_device_handle_t)config->handle, &trans);
188 /* Alternative: Queued transmit */
189 /* esp_err_t ret = spi_device_queue_trans((spi_device_handle_t)config->handle, &trans, portMAX_DELAY); */
190 /* if (ret == ESP_OK) { */
191 /* spi_transaction_t *ret_trans; */
192 /* ret = spi_device_get_trans_result((spi_device_handle_t)config->handle, &ret_trans, portMAX_DELAY); */
193 /* } */
194
195 if (ret != ESP_OK) {
196 ESP_LOGE(s_tag, "SPI transmit failed for '%s': %s", config->name, esp_err_to_name(ret));
197 /* Call error callback */
198 if (config->proto.spi.callbacks.on_error) {
199 star_bus_event_t event = {.bus_type = k_star_bus_type_spi,
200 .bus_name = config->name,
201 .spi = {.host = config->proto.spi.host,
202 .cs_pin = config->proto.spi.dev_cfg.spics_io_num,
203 .is_write = true,
204 .tx_len = 0, /* Indicate failure */
205 .rx_len = 0}};
206 config->proto.spi.callbacks.on_error(&event, ret, config->user_ctx);
207 }
208 } else {
209 ESP_LOGD(s_tag, "SPI transmit success: %zu bytes from '%s'", len, config->name);
210 /* Call success callback */
211 if (config->proto.spi.callbacks.on_transfer_complete) {
212 star_bus_event_t event = {.bus_type = k_star_bus_type_spi,
213 .bus_name = config->name,
214 .spi = {.host = config->proto.spi.host,
215 .cs_pin = config->proto.spi.dev_cfg.spics_io_num,
216 .is_write = true,
217 .tx_len = len,
218 .rx_len = 0}};
219 config->proto.spi.callbacks.on_transfer_complete(&event, config->user_ctx);
220 }
221 }
222 return ret;
223}
