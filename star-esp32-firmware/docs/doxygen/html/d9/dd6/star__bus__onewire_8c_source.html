88static esp_err_t internal_init_onewire_mutex(void)
89{
90 /* Quick check without lock for common case */
91 if (g_onewire_mutex != NULL) {
92 return ESP_OK;
93 }
94
95 /* Use spinlock for thread-safe lazy initialization */
96 portENTER_CRITICAL(&g_onewire_init_spinlock);
97
98 /* Double-check after acquiring spinlock */
99 if (g_onewire_mutex == NULL) {
100 g_onewire_mutex = xSemaphoreCreateMutex();
101 if (g_onewire_mutex == NULL) {
102 portEXIT_CRITICAL(&g_onewire_init_spinlock);
103 ESP_LOGE(s_TAG, "Failed to create 1-Wire global mutex");
104 return ESP_ERR_NO_MEM;
105 }
106 }
107
108 portEXIT_CRITICAL(&g_onewire_init_spinlock);
109 return ESP_OK;
110}

115static onewire_state_t* internal_internal_get_onewire_state_safe(const char* bus_name, bool create)
116{
117 for (uint8_t i = 0; i < g_num_onewire_states; i++) {
118 if (strcmp(g_onewire_states[i].bus_name, bus_name) == 0) {
119 return &g_onewire_states[i];
120 }
121 }
122
123 if (create && g_num_onewire_states < k_max_onewire_buses) {
124 onewire_state_t* state = &g_onewire_states[g_num_onewire_states];
125 memset(state, 0, sizeof(onewire_state_t));
126 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
127 state->bus_name[sizeof(state->bus_name) - 1] = '\0';
128 g_num_onewire_states++;
129 return state;
130 }
131
132 return NULL;
133}

138static onewire_state_t* internal_get_onewire_state_safe(const char* bus_name, bool create)
139{
140 if (internal_init_onewire_mutex() != ESP_OK) {
141 return NULL;
142 }
143
144 if (xSemaphoreTake(g_onewire_mutex, pdMS_TO_TICKS(s_onewire_mutex_timeout_ms)) != pdTRUE) {
145 ESP_LOGE(s_TAG,
146 "Failed to take 1-Wire mutex (timeout after %lu ms)",
147 (unsigned long)s_onewire_mutex_timeout_ms);
148 return NULL;
149 }
150
151 onewire_state_t* state = internal_internal_get_onewire_state_safe(bus_name, create);
152
153 xSemaphoreGive(g_onewire_mutex);
154 return state;
155}

191esp_err_t star_bus_onewire_init(star_bus_manager_t* manager,
192 const char* bus_name,
193 const star_onewire_config_t* config)
194{
195 if (manager == NULL || bus_name == NULL || config == NULL) {
196 ESP_LOGE(s_TAG, "Invalid parameters");
197 return ESP_ERR_INVALID_ARG;
198 }
199
200 if (config->gpio_pin < 0) {
201 ESP_LOGE(s_TAG, "Invalid GPIO pin");
202 return ESP_ERR_INVALID_ARG;
203 }
204
205 onewire_state_t* state = internal_get_onewire_state_safe(bus_name, true);
206 if (state == NULL) {
207 ESP_LOGE(s_TAG, "Failed to create state for bus '%s'", bus_name);
208 return ESP_ERR_NO_MEM;
209 }
210
211 memcpy(&state->config, config, sizeof(star_onewire_config_t));
212 memset(&state->stats, 0, sizeof(star_onewire_stats_t));
213
214 /* Configure GPIO as open-drain output with pull-up */
215 gpio_config_t io_conf = {
216 .pin_bit_mask = (1ULL << config->gpio_pin),
217 .mode = GPIO_MODE_INPUT_OUTPUT_OD,
218 .pull_up_en = GPIO_PULLUP_ENABLE,
219 .pull_down_en = GPIO_PULLDOWN_DISABLE,
220 .intr_type = GPIO_INTR_DISABLE,
221 };
222
223 esp_err_t result = gpio_config(&io_conf);
224 if (result != ESP_OK) {
225 ESP_LOGE(s_TAG, "Failed to configure GPIO: %s", esp_err_to_name(result));
226 return result;
227 }
228
229 /* Release bus (high) */
230 internal_gpio_high(config->gpio_pin);
231
232 /* Reset search state */
233 state->last_discrepancy = 0;
234 state->last_family_discrepancy = 0;
235 state->last_device_flag = false;
236 memset(state->search_rom, 0, sizeof(state->search_rom));
237
238 state->initialized = true;
239
240 ESP_LOGI(s_TAG, "1-Wire bus '%s' initialized on GPIO %d", bus_name, config->gpio_pin);
241
242 return ESP_OK;
243}

266esp_err_t star_bus_onewire_reset(star_bus_manager_t* manager, const char* bus_name, bool* present)
267{
268 if (manager == NULL || bus_name == NULL || present == NULL) {
269 return ESP_ERR_INVALID_ARG;
270 }
271
272 onewire_state_t* state = internal_get_onewire_state_safe(bus_name, false);
273 if (state == NULL || !state->initialized) {
274 return ESP_ERR_NOT_FOUND;
275 }
276
277 gpio_num_t pin = state->config.gpio_pin;
278 uint32_t start_time = esp_timer_get_time();
279
280 state->stats.total_resets++;
281
282 /* Pull bus low for reset pulse */
283 internal_gpio_low(pin);
284 internal_delay_us(k_timing_reset_pulse);
285
286 /* Release bus and wait for presence pulse */
287 internal_gpio_high(pin);
288 internal_delay_us(k_timing_presence_wait);
289
290 /* Sample bus for presence pulse (device pulls low) */
291 int32_t level = internal_gpio_read(pin);
292 *present = (level == 0);
293
294 /* Wait for end of presence pulse */
295 internal_delay_us(k_timing_presence_sample);
296
297 uint32_t elapsed = (uint32_t)(esp_timer_get_time() - start_time);
298 state->stats.last_operation_time_us = elapsed;
299
300 if (*present) {
301 state->stats.successful_resets++;
302 } else {
303 state->stats.failed_resets++;
304 }
305
306 return ESP_OK;
307}

312 return ESP_ERR_INVALID_ARG;
313 }
314
315 onewire_state_t* state = internal_get_onewire_state_safe(bus_name, false);
316 if (state == NULL || !state->initialized) {
317 return ESP_ERR_NOT_FOUND;
318 }
319
320 gpio_num_t pin = state->config.gpio_pin;
321
322 if (bit & 1) {
323 /* Write 1 */
324 internal_gpio_low(pin);
325 internal_delay_us(k_timing_write_1_low);
326 internal_gpio_high(pin);
327 internal_delay_us(k_timing_write_0_low - k_timing_write_1_low + k_timing_write_recovery);
328 } else {
329 /* Write 0 */
330 internal_gpio_low(pin);
331 internal_delay_us(k_timing_write_0_low);
332 internal_gpio_high(pin);
333 internal_delay_us(k_timing_write_recovery);
334 }
335
336 return ESP_OK;

418esp_err_t star_bus_onewire_write_bytes(star_bus_manager_t* manager,
419 const char* bus_name,
420 star_onewire_rom_t rom,
421 const uint8_t* data,
422 size_t length)
423{
424 if (manager == NULL || bus_name == NULL || data == NULL) {
425 return ESP_ERR_INVALID_ARG;
426 }
427
428 onewire_state_t* state = internal_get_onewire_state_safe(bus_name, false);
429 if (state == NULL || !state->initialized) {
430 return ESP_ERR_NOT_FOUND;
431 }
432
433 /* Reset bus */
434 bool present;
435 esp_err_t result = star_bus_onewire_reset(manager, bus_name, &present);
436 if (result != ESP_OK || !present) {
437 return ESP_FAIL;
438 }
439
440 /* Select device or skip ROM */
441 if (rom == 0) {
442 /* Skip ROM - address all devices */
443 result = star_bus_onewire_write_byte(manager, bus_name, (uint8_t)k_onewire_cmd_skip_rom);
444 } else {
445 /* Match ROM - address specific device */
446 result = star_bus_onewire_write_byte(manager, bus_name, (uint8_t)k_onewire_cmd_match_rom);
447 if (result == ESP_OK) {
448 uint8_t rom_bytes[8];
449 star_bus_onewire_rom_to_bytes(rom, rom_bytes);
450 for (uint32_t i = 0; i < 8; i++) {
451 result = star_bus_onewire_write_byte(manager, bus_name, rom_bytes[i]);
452 if (result != ESP_OK) {
453 break;
454 }
455 }
456 }
457 }
458
459 if (result != ESP_OK) {
460 return result;
461 }
462
463 /* Write data */
464 for (size_t i = 0; i < length; i++) {
465 result = star_bus_onewire_write_byte(manager, bus_name, data[i]);
466 if (result != ESP_OK) {
467 return result;
468 }
469 }
470
471 return ESP_OK;
472}

493esp_err_t star_bus_onewire_read_rom(star_bus_manager_t* manager,
494 const char* bus_name,
495 star_onewire_rom_t* rom)
496{
497 if (manager == NULL || bus_name == NULL || rom == NULL) {
498 return ESP_ERR_INVALID_ARG;
499 }
500
501 /* Reset bus */
502 bool present;
503 esp_err_t result = star_bus_onewire_reset(manager, bus_name, &present);
504 if (result != ESP_OK || !present) {
505 return ESP_FAIL;
506 }
507
508 /* Send Read ROM command */
509 result = star_bus_onewire_write_byte(manager, bus_name, (uint8_t)k_onewire_cmd_read_rom);
510 if (result != ESP_OK) {
511 return result;
512 }
513
514 /* Read 8 bytes */
515 uint8_t rom_bytes[8];
516 result = star_bus_onewire_read_bytes(manager, bus_name, rom_bytes, 8);
517 if (result != ESP_OK) {
518 return result;
519 }
520
521 *rom = star_bus_onewire_bytes_to_rom(rom_bytes);
522
523 /* Verify CRC */
524 if (!star_bus_onewire_verify_rom(*rom)) {
525 onewire_state_t* state = internal_get_onewire_state_safe(bus_name, false);
526 if (state != NULL) {
527 state->stats.crc_errors++;
528 }
529 return ESP_ERR_INVALID_CRC;
530 }
531
532 return ESP_OK;
533}

535esp_err_t star_bus_onewire_search(star_bus_manager_t* manager,
536 const char* bus_name,
537 star_onewire_rom_t* rom_codes,
538 size_t* count)
539{
540 if (manager == NULL || bus_name == NULL || rom_codes == NULL || count == NULL) {
541 return ESP_ERR_INVALID_ARG;
542 }
543
544 onewire_state_t* state = internal_get_onewire_state_safe(bus_name, false);
545 if (state == NULL || !state->initialized) {
546 return ESP_ERR_NOT_FOUND;
547 }
548
549 state->stats.search_operations++;
550
551 size_t max_devices = *count;
552 *count = 0;
553
554 /* Reset search state */
555 state->last_discrepancy = 0;
556 state->last_family_discrepancy = 0;
557 state->last_device_flag = false;
558
559 /* Search for devices */
560 while (*count < max_devices) {
561 /* Reset bus */
562 bool present;
563 esp_err_t result = star_bus_onewire_reset(manager, bus_name, &present);
564 if (result != ESP_OK || !present) {
565 break;
566 }
567
568 /* Send search command */
569 result = star_bus_onewire_write_byte(manager, bus_name, (uint8_t)k_onewire_cmd_search_rom);
570 if (result != ESP_OK) {
571 break;
572 }
573
574 /* Perform search algorithm */
575 uint8_t last_zero = 0;
576 uint8_t rom_byte_number = 0;
577 uint8_t rom_byte_mask = 1;
578 bool search_result = false;
579
580 for (uint8_t bit_pos = 1; bit_pos <= 64; bit_pos++) {
581 uint8_t id_bit, cmp_id_bit;
582
583 /* Read bit and its complement */
584 star_bus_onewire_read_bit(manager, bus_name, &id_bit);
585 star_bus_onewire_read_bit(manager, bus_name, &cmp_id_bit);
586
587 if (id_bit && cmp_id_bit) {
588 /* No devices responded */
589 break;
590 }
591
592 uint8_t search_direction;
593
594 if (id_bit != cmp_id_bit) {
595 /* All devices have same bit */
596 search_direction = id_bit;
597 } else {
598 /* Discrepancy - choose direction */
599 if (bit_pos < state->last_discrepancy) {
600 search_direction = (state->search_rom[rom_byte_number] & rom_byte_mask) ? 1 : 0;
601 } else if (bit_pos == state->last_discrepancy) {
602 search_direction = 1;
603 } else {
604 search_direction = 0;
605 }
606
607 if (search_direction == 0) {
608 last_zero = bit_pos;
609 if (last_zero < 9) {
610 state->last_family_discrepancy = last_zero;
611 }
612 }
613 }
614
615 /* Write search direction */
616 star_bus_onewire_write_bit(manager, bus_name, search_direction);
617
618 /* Update ROM buffer */
619 if (search_direction) {
620 state->search_rom[rom_byte_number] |= rom_byte_mask;
621 } else {
622 state->search_rom[rom_byte_number] &= ~rom_byte_mask;
623 }
624
625 /* Advance to next bit */
626 rom_byte_mask <<= 1;
627 if (rom_byte_mask == 0) {
628 rom_byte_number++;
629 rom_byte_mask = 1;
630 }
631 }
632
633 if (rom_byte_number >= 8) {
634 search_result = true;
635 }
636
637 if (search_result) {
638 /* Store found ROM */
639 rom_codes[*count] = star_bus_onewire_bytes_to_rom(state->search_rom);
640
641 /* Verify CRC */
642 if (!star_bus_onewire_verify_rom(rom_codes[*count])) {
643 state->stats.crc_errors++;
644 continue;
645 }
646
647 (*count)++;
648
649 state->last_discrepancy = last_zero;
650 if (state->last_discrepancy == 0) {
651 state->last_device_flag = true;
652 }
653 }
654
655 if (state->last_device_flag || state->last_discrepancy == 0) {
656 break;
657 }
658 }
659
660 state->stats.devices_found = *count;
661
662 ESP_LOGI(s_TAG, "Found %d device(s) on bus '%s'", *count, bus_name);
663
664 return ESP_OK;
665}

677uint8_t star_bus_onewire_crc8(const uint8_t* data, size_t length)
678{

688bool star_bus_onewire_verify_rom(star_onewire_rom_t rom)
689{
690 uint8_t rom_bytes[8];
691 star_bus_onewire_rom_to_bytes(rom, rom_bytes);

697uint16_t star_bus_onewire_crc16(const uint8_t* data, size_t length)
698{
699 uint16_t crc = 0;
700
701 for (size_t i = 0; i < length; i++) {
702 uint8_t x = (crc ^ data[i]) & 0xFF;
703 crc = (crc >> 8) ^ ((x ^ (x << 4)) << 8) ^ (x << 3) ^ (x >> 4);
704 }
705
706 return crc;
707}
