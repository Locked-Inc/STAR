75/* TODO(human): Implement the telemetry request handler
76 *
77 * This function should:
78 * 1. Decode the incoming GetTelemetryRequest using pb_decode()
79 * 2. Read temperature from the DS18B20 sensor
80 * 3. Build a GetTelemetryResponse with the temperature data
81 * 4. Encode the response using pb_encode()
82 * 5. Send the response via star_arq_send()
83 *
84 * Key considerations:
85 * - Error handling: What if decode fails? What if sensor read fails?
86 * - Response status: Should use star_v1_Status_STATUS_OK on success
87 * - Request ID: Must be copied from request to response for correlation
88 * - Timestamp: Optional but useful for debugging (can set to 0 for now)
89 */
90static void internal_handle_telemetry_request(star_ds18b20_handle_t* sensor,
91 star_arq_handle_t* arq,
92 const uint8_t* request_buf,
93 size_t request_len,
94 uint8_t* response_buf,
95 size_t response_buf_size)
96{
97 /* Decode the request */
98 pb_istream_t input_stream = pb_istream_from_buffer(request_buf, request_len);
99 star_v1_GetTelemetryRequest request = star_v1_GetTelemetryRequest_init_zero;
100
101 bool decode_ret = pb_decode(&input_stream, star_v1_GetTelemetryRequest_fields, &request);
102
103 if (!decode_ret) {
104 ESP_LOGE(s_tag, "Failed to decode telemetry request: %s", PB_GET_ERROR(&input_stream));
105 }
106
107 /* Read sensor */
108 float temp_c = 0.0f;
109 esp_err_t sensor_ret = ESP_FAIL;
110
111 if (decode_ret) {
112 sensor_ret = star_sensor_ds18b20_read_temp(sensor, &temp_c);
113
114 if (sensor_ret != ESP_OK) {
115 ESP_LOGW(s_tag, "Failed to read temperature from sensor: %s", esp_err_to_name(sensor_ret));
116 }
117 }
118
119 /* Build response */
120 star_v1_GetTelemetryResponse response = star_v1_GetTelemetryResponse_init_zero;
121 response.has_header = true;
122
123 /* Set status */
124 if (!decode_ret) {
125 response.header.status = star_v1_Status_STATUS_INVALID_REQUEST;
126 } else if (sensor_ret != ESP_OK) {
127 response.header.status = star_v1_Status_STATUS_INTERNAL_ERROR;
128 } else {
129 response.header.status = star_v1_Status_STATUS_OK;
130 }
131
132 /* Copy request_id for correlation */
133 if (decode_ret && request.has_header) {
134 response.header.request_id = request.header.request_id;
135 }
136
137 /* Set telemetry data */
138 if (sensor_ret == ESP_OK) {
139 response.has_telemetry = true;
140 response.telemetry.temperature_celsius = temp_c;
141 }
142
143 /* Encode response */
144 pb_ostream_t output_stream = pb_ostream_from_buffer(response_buf, response_buf_size);
145
146 bool encode_ok = pb_encode(&output_stream, star_v1_GetTelemetryResponse_fields, &response);
147
148 if (!encode_ok) {
149 ESP_LOGE(s_tag, "Failed to encode response: %s", PB_GET_ERROR(&output_stream));
150 return;
151 }
152
153 size_t response_len = output_stream.bytes_written;
154
155 /* Send response */
156 esp_err_t send_ret = star_arq_send(arq, response_buf, response_len, k_arq_send_timeout_ms);
157
158 if (send_ret != ESP_OK) {
159 ESP_LOGW(s_tag, "Failed to send telemetry response: %s", esp_err_to_name(send_ret));
160 }
161}

163static void internal_handle_requests(star_ds18b20_handle_t* sensor, star_arq_handle_t* arq)
164{
165 uint8_t rx_buffer[k_rx_buffer_size];
166 uint8_t tx_buffer[k_tx_buffer_size];
167
168 while (true) {
169 /* Receive request from RPi5 */
170 size_t rx_len;
171 esp_err_t ret = star_arq_receive(arq, rx_buffer, k_rx_buffer_size, &rx_len, portMAX_DELAY);
172
173 if (ret != ESP_OK) {
174 ESP_LOGW(s_tag, "Receive failed: %s", esp_err_to_name(ret));
175 vTaskDelay(pdMS_TO_TICKS(k_arq_error_backoff_ms));
176 continue;
177 }
178
179 /* Process telemetry request */
180 internal_handle_telemetry_request(sensor, arq, rx_buffer, rx_len, tx_buffer, k_tx_buffer_size);
181 }
182}
