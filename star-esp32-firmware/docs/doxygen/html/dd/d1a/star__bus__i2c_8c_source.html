88 const uint8_t* data,
89 size_t len,
90 uint8_t reg_addr,
91 size_t* bytes_written)
92{
93 ESP_RETURN_ON_FALSE(manager && name && data,
94 ESP_ERR_INVALID_ARG,
95 s_tag,
96 "Manager, name, or data is NULL");

98
99 star_bus_config_t* bus_config = star_bus_manager_find_bus(manager, name);
100 ESP_RETURN_ON_FALSE(bus_config, ESP_ERR_NOT_FOUND, s_tag, "Bus '%s' not found", name);
101 ESP_RETURN_ON_FALSE(bus_config->type == k_star_bus_type_i2c,
102 ESP_ERR_INVALID_ARG,
103 s_tag,
104 "Bus '%s' is not an I2C bus",
105 name);
106 ESP_RETURN_ON_FALSE(bus_config->proto.i2c.ops.write,
107 ESP_ERR_NOT_SUPPORTED,
108 s_tag,
109 "No write op for '%s'",
110 name);
111
112 /* Clear bytes_written initially */
113 if (bytes_written) {
114 *bytes_written = 0;
115 }
116
117 return bus_config->proto.i2c.ops.write(bus_config, data, len, reg_addr, bytes_written);

230static esp_err_t internal_star_bus_i2c_write(const star_bus_config_t* config,
231 const uint8_t* data,
232 size_t len,
233 uint8_t reg_addr,
234 size_t* bytes_written)
235{
236 /* Basic validation (config, data assumed non-NULL, len > 0 by caller) */
237 ESP_RETURN_ON_FALSE(config->initialized,
238 ESP_ERR_INVALID_STATE,
239 s_tag,
240 "Bus '%s' is not initialized",
241 config->name);
242
243 i2c_port_t port = config->proto.i2c.port;
244 uint8_t address = config->proto.i2c.address;
245 esp_err_t ret = ESP_OK;
246 i2c_cmd_handle_t cmd = NULL;
247
248 cmd = i2c_cmd_link_create();
249 ESP_RETURN_ON_FALSE(cmd,
250 ESP_ERR_NO_MEM,
251 s_tag,
252 "Failed to create I2C command link for write '%s'",
253 config->name);
254
255 ret = i2c_master_start(cmd);
256 ESP_GOTO_ON_ERROR(ret,
257 write_fail,
258 s_tag,
259 "Write '%s': START failed: %s",
260 config->name,
261 esp_err_to_name(ret));
262
263 ret = i2c_master_write_byte(cmd, (address << 1) | I2C_MASTER_WRITE, true);
264 ESP_GOTO_ON_ERROR(ret,
265 write_fail,
266 s_tag,
267 "Write '%s': Address (W) failed: %s",
268 config->name,
269 esp_err_to_name(ret));
270
271 /* Write register address */
272 ret = i2c_master_write_byte(cmd, reg_addr, true);
273 ESP_GOTO_ON_ERROR(ret,
274 write_fail,
275 s_tag,
276 "Write '%s': Reg Addr failed: %s",
277 config->name,
278 esp_err_to_name(ret));
279
280 /* Write data */
281 ret = i2c_master_write(cmd, data, len, true);
282 ESP_GOTO_ON_ERROR(ret,
283 write_fail,
284 s_tag,
285 "Write '%s': Data write failed: %s",
286 config->name,
287 esp_err_to_name(ret));
288
289 ret = i2c_master_stop(cmd);
290 ESP_GOTO_ON_ERROR(ret,
291 write_fail,
292 s_tag,
293 "Write '%s': STOP failed: %s",
294 config->name,
295 esp_err_to_name(ret));
296
297 /* Execute with retry logic */
298 ret = internal_i2c_execute_with_retry(config, port, cmd, "Write");
299 ESP_GOTO_ON_ERROR(ret, write_fail, s_tag, "Write '%s': CMD failed after retries", config->name);
300
301 /* Success path */
302 i2c_cmd_link_delete(cmd);
303 cmd = NULL; /* Prevent double delete in fail path */
304 ESP_LOGD(s_tag,
305 "Write Success: %zu bytes to '%s' (Addr 0x%02X, Reg 0x%02X)",
306 len,
307 config->name,
308 address,
309 reg_addr);
310 if (bytes_written) {
311 *bytes_written = len;
312 }
313
314 /* Call success callback */
315 if (config->proto.i2c.callbacks.on_transfer_complete) {
316 star_bus_event_t event = {
317 .bus_type = k_star_bus_type_i2c,
318 .bus_name = config->name,
319 .i2c = {.port = port, .address = address, .is_write = true, .len = len}};
320 config->proto.i2c.callbacks.on_transfer_complete(&event, config->user_ctx);
321 }
322 return ESP_OK;
323
324write_fail:
325 if (cmd) {
326 i2c_cmd_link_delete(cmd);
327 }
328 ESP_LOGE(s_tag, "Write failed for '%s': %s", config->name, esp_err_to_name(ret));
329 /* Call error callback */
330 if (config->proto.i2c.callbacks.on_error) {
331 star_bus_event_t event = {.bus_type = k_star_bus_type_i2c,
332 .bus_name = config->name,
333 .i2c = {.port = port,
334 .address = address,
335 .is_write = true,
336 .len = 0}}; /* Len is 0 on error */
337 config->proto.i2c.callbacks.on_error(&event, ret, config->user_ctx);
338 }
339 return ret;
340}

343 uint8_t* data,
344 size_t len,
345 uint8_t reg_addr,
346 size_t* bytes_read)
347{
348 /* Basic validation (config, data assumed non-NULL, len > 0 by caller) */

365 config->name);
366
367 /* --- Write Phase (Device Address + Register Address) --- */
368 ret = i2c_master_start(cmd);

385 ESP_GOTO_ON_ERROR(ret,
386 read_fail,
387 s_tag,
388 "Read '%s': Reg Addr failed: %s",
389 config->name,
390 esp_err_to_name(ret));
391
392 /* --- Read Phase (Repeated Start + Device Address + Data) --- */
393 ret = i2c_master_start(cmd); /* Repeated start */
394 ESP_GOTO_ON_ERROR(ret,
395 read_fail,
396 s_tag,
397 "Read '%s': Repeated START failed: %s",
398 config->name,
399 esp_err_to_name(ret));
400
401 ret = i2c_master_write_byte(cmd, (address << 1) | I2C_MASTER_READ, true);
402 ESP_GOTO_ON_ERROR(ret,
403 read_fail,
404 s_tag,
405 "Read '%s': Address (R) failed: %s",
406 config->name,
407 esp_err_to_name(ret));
408
409 if (len > 1) {
410 ret = i2c_master_read(cmd, data, len - 1, I2C_MASTER_ACK);
411 ESP_GOTO_ON_ERROR(ret,
412 read_fail,

416 esp_err_to_name(ret));
417 }
418

420 ret = i2c_master_read_byte(cmd, data + len - 1, I2C_MASTER_NACK);
421 ESP_GOTO_ON_ERROR(ret,
422 read_fail,
423 s_tag,
424 "Read '%s': Data read (NACK) failed: %s",
425 config->name,
426 esp_err_to_name(ret));
427
428 ret = i2c_master_stop(cmd);
429 ESP_GOTO_ON_ERROR(ret,
430 read_fail,
431 s_tag,
432 "Read '%s': STOP failed: %s",
433 config->name,
434 esp_err_to_name(ret));
435
436 /* Execute with retry logic */
437 ret = internal_i2c_execute_with_retry(config, port, cmd, "Read");
438 ESP_GOTO_ON_ERROR(ret, read_fail, s_tag, "Read '%s': CMD failed after retries", config->name);
439
440 /* Success path */
441 i2c_cmd_link_delete(cmd);
442 cmd = NULL; /* Prevent double delete in fail path */
443 ESP_LOGD(s_tag,
444 "Read Success: %zu bytes from '%s' (Addr 0x%02X, Reg 0x%02X)",
445 len,
446 config->name,
447 address,
448 reg_addr);
449 if (bytes_read) {
450 *bytes_read = len;
451 }
452
453 /* Call success callback */
454 if (config->proto.i2c.callbacks.on_transfer_complete) {
455 star_bus_event_t event = {
456 .bus_type = k_star_bus_type_i2c,
457 .bus_name = config->name,
458 .i2c = {.port = port, .address = address, .is_write = false, .len = len}};
459 config->proto.i2c.callbacks.on_transfer_complete(&event, config->user_ctx);
460 }
461 return ESP_OK;
462
463read_fail:
464 if (cmd) {
465 i2c_cmd_link_delete(cmd);
466 }
467 ESP_LOGE(s_tag, "Read failed for '%s': %s", config->name, esp_err_to_name(ret));
468 /* Call error callback */
469 if (config->proto.i2c.callbacks.on_error) {
470 star_bus_event_t event = {
471 .bus_type = k_star_bus_type_i2c,
472 .bus_name = config->name,
473 .i2c = {.port = port, .address = address, .is_write = false, .len = 0}};
474 config->proto.i2c.callbacks.on_error(&event, ret, config->user_ctx);
475 }
476 return ret;
477}
