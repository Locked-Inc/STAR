28esp_err_t motor_shared_state_init(motor_shared_state_t* state)
29{
30 ESP_RETURN_ON_FALSE(state, ESP_ERR_INVALID_ARG, s_tag, "State is NULL");
31
32 /* Zero out structure */
33 memset(state, 0, sizeof(motor_shared_state_t));
34
35 /* Initialize all motors to IDLE state */
36 for (int i = 0; i < k_motor_count; i++) {
37 state->state[i] = star_v1_MotorState_MOTOR_STATE_IDLE;
38 }
39
40 /* Create mutex for thread safety */
41 state->mutex = xSemaphoreCreateMutex();
42 if (!state->mutex) {
43 ESP_LOGE(s_tag, "Failed to create mutex");
44 return ESP_ERR_NO_MEM;
45 }
46
47 ESP_LOGI(s_tag, "Motor shared state initialized");
48 return ESP_OK;
49}

52motor_shared_state_set_velocity(motor_shared_state_t* state, uint8_t motor_idx, float velocity_mps)
53{
54 ESP_RETURN_ON_FALSE(state, ESP_ERR_INVALID_ARG, s_tag, "State is NULL");
55 ESP_RETURN_ON_FALSE(motor_idx < k_motor_count,
56 ESP_ERR_INVALID_ARG,
57 s_tag,
58 "motor_idx must be < %d",
59 k_motor_count);
60
61 /* Take mutex with write timeout */
62 if (xSemaphoreTake(state->mutex, pdMS_TO_TICKS(k_mutex_timeout_write_ms)) != pdTRUE) {
63 ESP_LOGW(s_tag, "Timeout setting velocity for motor %d", motor_idx);
64 return ESP_ERR_TIMEOUT;
65 }
66
67 /* Update setpoint */
68 state->velocity_setpoint_mps[motor_idx] = velocity_mps;
69
70 /* Release mutex */
71 xSemaphoreGive(state->mutex);
72
73 return ESP_OK;
74}
