97static esp_err_t internal_star_bus_adc_read_raw(const star_bus_config_t* config, int32_t* out_raw)
98{
99 ESP_RETURN_ON_FALSE(config && out_raw, ESP_ERR_INVALID_ARG, s_tag, "Config or out_raw is NULL");
100 ESP_RETURN_ON_FALSE(config->proto.adc.unit_handle,
101 ESP_ERR_INVALID_STATE,
102 s_tag,
103 "ADC unit handle is NULL");
104
105 /* Use int for ESP-IDF function call */
106 int esp_raw_value;
107 esp_err_t ret =
108 adc_oneshot_read(config->proto.adc.unit_handle, config->proto.adc.channel, &esp_raw_value);
109 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to read ADC channel %d", config->proto.adc.channel);
110
111 /* Explicit conversion from int to int32_t */
112 *out_raw = (int32_t)esp_raw_value;
113
114 ESP_LOGD(s_tag, "ADC channel %d raw value: %" PRId32, config->proto.adc.channel, *out_raw);
115 return ESP_OK;
116}

118static esp_err_t internal_star_bus_adc_read_voltage(const star_bus_config_t* config,
119 int32_t* out_voltage_mv)
120{
121 ESP_RETURN_ON_FALSE(config && out_voltage_mv,
122 ESP_ERR_INVALID_ARG,
123 s_tag,
124 "Config or out_voltage_mv is NULL");
125 ESP_RETURN_ON_FALSE(config->proto.adc.unit_handle,
126 ESP_ERR_INVALID_STATE,
127 s_tag,
128 "ADC unit handle is NULL");
129
130 /* Use int for ESP-IDF function calls */
131 int esp_raw_value;
132 esp_err_t ret =
133 adc_oneshot_read(config->proto.adc.unit_handle, config->proto.adc.channel, &esp_raw_value);
134 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to read ADC channel %d", config->proto.adc.channel);
135
136 /* Convert to voltage if calibration is available */
137 if (config->proto.adc.cali_handle) {
138 int esp_voltage_mv;
139 ret = adc_cali_raw_to_voltage(config->proto.adc.cali_handle, esp_raw_value, &esp_voltage_mv);
140 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to convert ADC raw to voltage");
141
142 /* Explicit conversion from int to int32_t */
143 *out_voltage_mv = (int32_t)esp_voltage_mv;
144
145 ESP_LOGD(s_tag,
146 "ADC channel %d: raw=%d, voltage=%" PRId32 " mV",
147 config->proto.adc.channel,
148 esp_raw_value,
149 *out_voltage_mv);
150 } else {
151 /* No calibration available, estimate voltage from raw value */
152 /* For 12-bit ADC with 3.3V reference: voltage = (raw / 4095) * 3300 */
153 *out_voltage_mv = (int32_t)((esp_raw_value * 3300) / 4095);
154 ESP_LOGW(s_tag,
155 "ADC channel %d: No calibration, estimated voltage=%" PRId32 " mV (raw=%d)",
156 config->proto.adc.channel,
157 *out_voltage_mv,
158 esp_raw_value);
159 }
160
161 return ESP_OK;
162}
