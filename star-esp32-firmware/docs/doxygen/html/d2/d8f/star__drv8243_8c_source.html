39esp_err_t star_drv8243_init(star_drv8243_handle_t* handle, const star_drv8243_config_t* config)
40{
41 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_TAG, "Handle is NULL");
42 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config is NULL");
43 ESP_RETURN_ON_FALSE(config->bus_manager, ESP_ERR_INVALID_ARG, s_TAG, "Bus manager is NULL");
44 ESP_RETURN_ON_FALSE(config->gpio_bus_name, ESP_ERR_INVALID_ARG, s_TAG, "GPIO bus name is NULL");
45 ESP_RETURN_ON_FALSE(config->adc_bus_name, ESP_ERR_INVALID_ARG, s_TAG, "ADC bus name is NULL");
46 ESP_RETURN_ON_FALSE(!handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Already initialized");
47
48 /* Validate PWM frequency */
49 ESP_RETURN_ON_FALSE(config->pwm_freq_hz <= DRV8243_MAX_PWM_FREQ_HZ,
50 ESP_ERR_INVALID_ARG,
51 s_TAG,
52 "PWM frequency exceeds 25 kHz limit");
53
54 /* Zero out handle */
55 memset(handle, 0, sizeof(star_drv8243_handle_t));
56
57 /* Store configuration */
58 handle->bus_manager = config->bus_manager;
59 handle->gpio_bus_name = config->gpio_bus_name;
60 handle->adc_bus_name = config->adc_bus_name;
61 handle->pin_ipropi = config->pin_ipropi;
62 handle->pin_nfault = config->pin_nfault;
63 handle->current_limit_ma = config->current_limit_ma;
64 handle->ki_propi = config->ki_propi > 0 ? config->ki_propi : DRV8243_DEFAULT_KI_PROPI;
65
66 esp_err_t ret = ESP_OK;
67
68 /* Initialize motor control (MCPWM for H-bridge) */
69 star_motor_config_t motor_config = {
70 .group_id = config->group_id,
71 .pin_pwm_a = config->pin_pwm_ph,
72 .pin_pwm_b = config->pin_pwm_en,
73 .pwm_freq_hz = config->pwm_freq_hz,
74 .timer_resolution_hz =
75 config->timer_resolution_hz > 0 ? config->timer_resolution_hz : DRV8243_DEFAULT_TIMER_RES_HZ,
76 .dead_time_ns = config->dead_time_ns,
77 .invert_pwm = false,
78 .fault_pin = -1, /* We handle fault separately via GPIO bus */
79 };
80
81 ret = star_motor_init(&handle->motor, &motor_config);
82 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to initialize motor controller");
83
84 /* Configure nFAULT pin as input with pull-up (active low) */
85 ret = internal_drv8243_configure_fault_pin(handle);
86 if (ret != ESP_OK) {
87 ESP_LOGE(s_TAG, "Failed to configure fault pin: %s", esp_err_to_name(ret));
88 star_motor_deinit(&handle->motor);
89 return ret;
90 }
91
92 handle->current_speed = 0.0f;
93 handle->fault_active = false;
94 handle->initialized = true;
95
96 ESP_LOGI(s_TAG,
97 "DRV8243 initialized: PH=%d, EN=%d, IPROPI=%d, nFAULT=%d, freq=%luHz, limit=%umA",
98 config->pin_pwm_ph,
99 config->pin_pwm_en,
100 config->pin_ipropi,
101 config->pin_nfault,
102 config->pwm_freq_hz,
103 config->current_limit_ma);
104
105 return ESP_OK;
106}

126esp_err_t star_drv8243_set_speed(star_drv8243_handle_t* handle, float speed)
127{
128 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_TAG, "Handle is NULL");
129 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Not initialized");
130
131 /* Check for fault condition */
132 bool fault_active;
133 esp_err_t ret = star_drv8243_get_fault_status(handle, &fault_active);
134 if (ret == ESP_OK && fault_active) {
135 ESP_LOGW(s_TAG, "Cannot set speed: fault condition active");
136 return ESP_ERR_INVALID_STATE;
137 }
138
139 /* Check current limit if enabled */
140 if (handle->current_limit_ma > 0) {
141 ret = internal_drv8243_check_current_limit(handle);
142 if (ret != ESP_OK) {
143 ESP_LOGW(s_TAG, "Current limit exceeded, reducing speed");
144 /* Reduce speed by 10% when current limit hit */
145 speed *= 0.9f;
146 }
147 }
148
149 /* Set motor duty cycle */
150 ret = star_motor_set_duty(&handle->motor, speed);
151 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to set motor duty");
152
153 handle->current_speed = speed;
154
155 ESP_LOGD(s_TAG, "Motor speed set: %.1f%%", speed);
156 return ESP_OK;
157}

173esp_err_t star_drv8243_read_current(star_drv8243_handle_t* handle, float* out_current)
174{
175 ESP_RETURN_ON_FALSE(handle && out_current,
176 ESP_ERR_INVALID_ARG,
177 s_TAG,
178 "Handle or out_current is NULL");
179 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Not initialized");
180
181 /* Read ADC voltage from IPROPI pin */
182 uint32_t voltage_mv;
183 esp_err_t ret = star_bus_adc_read_voltage(handle->bus_manager,
184 handle->adc_bus_name,
185 handle->pin_ipropi,
186 &voltage_mv);
187 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to read IPROPI ADC");
188
189 /* Convert voltage to current using ki_propi ratio
190 * Current (mA) = Voltage (mV) * Ki_PROPI (A/V) / 1000
191 * Example: 1000mV * 525 A/V / 1000 = 525mA
192 */
193 *out_current = (float)(voltage_mv * handle->ki_propi) / 1000.0f;
194
195 ESP_LOGD(s_TAG, "Motor current: %.1f mA (IPROPI=%lu mV)", *out_current, voltage_mv);
196 return ESP_OK;
197}

199esp_err_t star_drv8243_get_fault_status(star_drv8243_handle_t* handle, bool* out_fault)
200{
201 ESP_RETURN_ON_FALSE(handle && out_fault,
202 ESP_ERR_INVALID_ARG,
203 s_TAG,
204 "Handle or out_fault is NULL");
205 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_TAG, "Not initialized");
206
207 /* Read nFAULT pin (active low) directly using ESP-IDF GPIO */
208 int level = gpio_get_level(handle->pin_nfault);
209
210 /* Fault is active when pin is LOW */
211 *out_fault = (level == 0);
212 handle->fault_active = *out_fault;
213
214 if (*out_fault) {
215 ESP_LOGW(s_TAG, "DRV8243 fault detected on nFAULT pin");
216 }
217
218 return ESP_OK;
219}

246static esp_err_t internal_drv8243_check_current_limit(star_drv8243_handle_t* handle)
247{
248 float current_ma;
249 esp_err_t ret = star_drv8243_read_current(handle, &current_ma);
250 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to read current for limit check");
251
252 if (current_ma > (float)handle->current_limit_ma) {
253 ESP_LOGW(s_TAG,
254 "Current limit exceeded: %.1f mA > %u mA",
255 current_ma,
256 handle->current_limit_ma);
257 return ESP_ERR_INVALID_STATE;
258 }
259
260 return ESP_OK;
261}

263static esp_err_t internal_drv8243_configure_fault_pin(star_drv8243_handle_t* handle)
264{
265 /* Configure nFAULT pin as input with pull-up (active low) using ESP-IDF GPIO */
266 gpio_config_t fault_cfg = {
267 .pin_bit_mask = (1ULL << handle->pin_nfault),
268 .mode = GPIO_MODE_INPUT,
269 .pull_up_en = GPIO_PULLUP_ENABLE,
270 .pull_down_en = GPIO_PULLDOWN_DISABLE,
271 .intr_type = GPIO_INTR_DISABLE,
272 };
273
274 esp_err_t ret = gpio_config(&fault_cfg);
275 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to configure nFAULT pin");
276
277 return ESP_OK;
