68esp_err_t star_drv8243_init(star_drv8243_handle_t* handle, const star_drv8243_config_t* config)
69{
70 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
71 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_tag, "Config is NULL");
72 ESP_RETURN_ON_FALSE(config->bus_manager, ESP_ERR_INVALID_ARG, s_tag, "Bus manager is NULL");
73 ESP_RETURN_ON_FALSE(config->gpio_bus_name, ESP_ERR_INVALID_ARG, s_tag, "GPIO bus name is NULL");
74 ESP_RETURN_ON_FALSE(config->adc_bus_name, ESP_ERR_INVALID_ARG, s_tag, "ADC bus name is NULL");
75 ESP_RETURN_ON_FALSE(!handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Already initialized");
76
77 /* Validate PWM frequency */
78 ESP_RETURN_ON_FALSE(config->pwm_freq_hz <= k_drv8243_max_pwm_freq_hz,
79 ESP_ERR_INVALID_ARG,
80 s_tag,
81 "PWM frequency exceeds 25 kHz limit");
82
83 /* Zero out handle */
84 memset(handle, 0, sizeof(star_drv8243_handle_t));
85
86 /* Store configuration */
87 handle->bus_manager = config->bus_manager;
88 handle->gpio_bus_name = config->gpio_bus_name;
89 handle->adc_bus_name = config->adc_bus_name;
90 handle->pin_ipropi = config->pin_ipropi;
91 handle->pin_nfault = config->pin_nfault;
92 handle->current_limit_ma = config->current_limit_ma;
93 handle->ki_propi = config->ki_propi > 0 ? config->ki_propi : k_drv8243_default_ki_propi;
94
95 esp_err_t ret = ESP_OK;
96
97 /* Initialize motor control (MCPWM for H-bridge) */
98 star_motor_config_t motor_config = {
99 .group_id = config->group_id,
100 .pin_pwm_a = config->pin_pwm_ph,
101 .pin_pwm_b = config->pin_pwm_en,
102 .pwm_freq_hz = config->pwm_freq_hz,
103 .timer_resolution_hz = config->timer_resolution_hz > 0 ? config->timer_resolution_hz
104 : k_drv8243_default_timer_res_hz,
105 .dead_time_ns = config->dead_time_ns,
106 .invert_pwm = false,
107 .fault_pin = -1, /* We handle fault separately via GPIO bus */
108 };
109
110 ret = star_motor_init(&handle->motor, &motor_config);
111 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to initialize motor controller");
112
113 /* Configure nFAULT pin as input with pull-up (active low) */
114 ret = internal_drv8243_configure_fault_pin(handle);
115 if (ret != ESP_OK) {
116 ESP_LOGE(s_tag, "Failed to configure fault pin: %s", esp_err_to_name(ret));
117 star_motor_deinit(&handle->motor);
118 return ret;
119 }
120
121 handle->current_speed = 0.0f;
122 handle->fault_active = false;
123 handle->initialized = true;
124
125 ESP_LOGI(s_tag,
126 "DRV8243 initialized: PH=%d, EN=%d, IPROPI=%d, nFAULT=%d, freq=%luHz, limit=%umA",
127 config->pin_pwm_ph,
128 config->pin_pwm_en,
129 config->pin_ipropi,
130 config->pin_nfault,
131 config->pwm_freq_hz,
132 config->current_limit_ma);
133
134 return ESP_OK;
135}

155esp_err_t star_drv8243_set_speed(star_drv8243_handle_t* handle, float speed)
156{
157 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
158 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
159
160 /* Check for fault condition */
161 bool fault_active;
162 esp_err_t ret = star_drv8243_get_fault_status(handle, &fault_active);
163 if (ret == ESP_OK && fault_active) {
164 ESP_LOGW(s_tag, "Cannot set speed: fault condition active");
165 return ESP_ERR_INVALID_STATE;
166 }
167
168 /* Check current limit if enabled */
169 if (handle->current_limit_ma > 0) {
170 ret = internal_drv8243_check_current_limit(handle);
171 if (ret != ESP_OK) {
172 ESP_LOGW(s_tag, "Current limit exceeded, reducing speed");
173 speed *= s_current_limit_reduction_factor;
174 }
175 }
176
177 /* Set motor duty cycle */
178 ret = star_motor_set_duty(&handle->motor, speed);
179 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to set motor duty");
180
181 handle->current_speed = speed;
182
183 ESP_LOGD(s_tag, "Motor speed set: %.1f%%", speed);
184 return ESP_OK;
185}

201esp_err_t star_drv8243_read_current(star_drv8243_handle_t* handle, float* out_current)
202{
203 ESP_RETURN_ON_FALSE(handle && out_current,
204 ESP_ERR_INVALID_ARG,
205 s_tag,
206 "Handle or out_current is NULL");
207 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
208
209 /*
210 * Read ADC voltage from IPROPI pin.
211 * The ADC bus was configured with the IPROPI channel during init.
212 * The voltage is returned in millivolts after ESP-IDF calibration.
213 */
214 int32_t voltage_mv;
215 esp_err_t ret = star_bus_adc_read_voltage(handle->bus_manager, handle->adc_bus_name, &voltage_mv);
216 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to read IPROPI ADC");
217
218 /*
219 * Convert IPROPI voltage to motor current using the Ki_PROPI ratio.
220 *
221 * Formula derivation:
222 * - IPROPI outputs voltage V = I_motor / Ki_PROPI (where Ki_PROPI is in A/V)
223 * - Therefore: I_motor (A) = V (V) * Ki_PROPI (A/V)
224 * - Converting units: I_motor (mA) = V (mV) * Ki_PROPI / 1000
225 *
226 * Example with typical values:
227 * - voltage_mv = 1000 mV (1V from IPROPI)
228 * - ki_propi = 525 A/V (datasheet typical at 25C)
229 * - Result: 1000 * 525 / 1000 = 525 mA motor current
230 *
231 * Note: Ki_PROPI varies with temperature; for high-precision applications,
232 * consider temperature compensation using the DS18B20 sensor.
233 */
234 *out_current = (float)(voltage_mv * handle->ki_propi) / s_mv_to_v_divisor;
235
236 ESP_LOGD(s_tag, "Motor current: %.1f mA (IPROPI=%" PRId32 " mV)", *out_current, voltage_mv);
237 return ESP_OK;
238}

240esp_err_t star_drv8243_get_fault_status(star_drv8243_handle_t* handle, bool* out_fault)
241{
242 ESP_RETURN_ON_FALSE(handle && out_fault,
243 ESP_ERR_INVALID_ARG,
244 s_tag,
245 "Handle or out_fault is NULL");
246 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
247
248 /* Read nFAULT pin (active low) directly using ESP-IDF GPIO */
249 int level = gpio_get_level(handle->pin_nfault);
250
251 /* Fault is active when pin is LOW */
252 *out_fault = (level == 0);
253 handle->fault_active = *out_fault;
254
255 if (*out_fault) {
256 ESP_LOGW(s_tag, "DRV8243 fault detected on nFAULT pin");
257 }
258
259 return ESP_OK;
260}

314
