32static esp_err_t internal_bq7850_manufacturer_access(star_bus_manager_t* manager,
33 const char* bus_name,
34 uint8_t addr,
35 uint16_t subcmd,
36 uint16_t* result)
37{
38 esp_err_t ret;
39
40 /* Write subcommand to Manufacturer Access register */
41 ret = star_smbus_write_word(manager, bus_name, addr, k_bq7850_cmd_manufacturer_access, subcmd);
42 if (ret != ESP_OK) {
43 ESP_LOGE(s_tag,
44 "Failed to write manufacturer access subcommand 0x%04X: %s",
45 subcmd,
46 esp_err_to_name(ret));
47 return ret;
48 }
49
50 /* Small delay for command processing (required by BQ7850 datasheet) */
51 vTaskDelay(pdMS_TO_TICKS(k_bq7850_mfr_access_delay_ms));
52
53 /* Read result if requested */
54 if (result != NULL) {
55 ret = star_smbus_read_word(manager, bus_name, addr, k_bq7850_cmd_manufacturer_access, result);
56 if (ret != ESP_OK) {
57 ESP_LOGE(s_tag, "Failed to read manufacturer access result: %s", esp_err_to_name(ret));
58 return ret;
59 }
60 }
61
62 return ESP_OK;
63}

84esp_err_t star_bms_bq7850_init(bq7850_handle_t* handle,
85 star_bus_manager_t* manager,
86 const char* bus_name,
87 const bq7850_config_t* config)
88{
89 if (handle == NULL || manager == NULL || bus_name == NULL || config == NULL) {
90 ESP_LOGE(s_tag, "NULL parameter in init");
91 return ESP_ERR_INVALID_ARG;
92 }
93
94 if (config->num_cells == 0 || config->num_cells > k_bq7850_max_cells) {
95 ESP_LOGE(s_tag,
96 "Invalid number of cells: %d (must be 1-%d)",
97 config->num_cells,
98 k_bq7850_max_cells);
99 return ESP_ERR_INVALID_ARG;
100 }
101
102 /* Initialize handle */
103 memset(handle, 0, sizeof(bq7850_handle_t));
104 handle->manager = manager;
105 handle->bus_name = bus_name;
106 handle->smbus_addr = config->smbus_addr;
107 memcpy(&handle->config, config, sizeof(bq7850_config_t));
108
109 /* Create mutex */
110 handle->mutex = xSemaphoreCreateMutex();
111 if (handle->mutex == NULL) {
112 ESP_LOGE(s_tag, "Failed to create mutex");
113 return ESP_ERR_NO_MEM;
114 }
115
116 ESP_LOGI(s_tag,
117 "Initializing BQ7850 BMS on bus '%s' at address 0x%02X",
118 bus_name,
119 config->smbus_addr);
120
121 /* Verify device communication by reading device type */
122 uint16_t device_type = 0;
123 esp_err_t ret = internal_bq7850_manufacturer_access(manager,
124 bus_name,
125 config->smbus_addr,
126 k_bq7850_subcmd_device_type,
127 &device_type);
128
129 if (ret != ESP_OK) {
130 ESP_LOGE(s_tag, "Failed to communicate with BQ7850: %s", esp_err_to_name(ret));
131 vSemaphoreDelete(handle->mutex);
132 handle->mutex = NULL;
133 return ret;
134 }
135
136 ESP_LOGI(s_tag, "BQ7850 detected, device type: 0x%04X", device_type);
137
138 /* Read firmware version */
139 uint16_t fw_version = 0;
140 ret = internal_bq7850_manufacturer_access(manager,
141 bus_name,
142 config->smbus_addr,
143 k_bq7850_subcmd_fw_version,
144 &fw_version);
145 if (ret == ESP_OK) {
146 ESP_LOGI(s_tag, "Firmware version: 0x%04X", fw_version);
147 }
148
149 /* Verify we can read basic parameters */
150 uint16_t voltage = 0;
151 ret = star_smbus_read_word(manager, bus_name, config->smbus_addr, k_bq7850_cmd_voltage, &voltage);
152 if (ret != ESP_OK) {
153 ESP_LOGE(s_tag, "Failed to read pack voltage: %s", esp_err_to_name(ret));
154 vSemaphoreDelete(handle->mutex);
155 handle->mutex = NULL;
156 return ret;
157 }
158
159 handle->initialized = true;
160 ESP_LOGI(s_tag,
161 "BQ7850 initialization successful (thread-safe with mutex), pack voltage: %d mV",
162 voltage);
163
164 return ESP_OK;
165}

167esp_err_t star_bms_bq7850_deinit(bq7850_handle_t* handle)
168{
169 if (handle == NULL || !handle->initialized) {
170 return ESP_ERR_INVALID_ARG;
171 }
172
173 /* Acquire mutex before cleanup */
174 SemaphoreHandle_t mutex = handle->mutex;
175 if (mutex != NULL && xSemaphoreTake(mutex, pdMS_TO_TICKS(s_bq7850_mutex_timeout_ms)) != pdTRUE) {
176 ESP_LOGW(s_tag, "Failed to acquire mutex for deinit, continuing anyway");
177 }
178
179 /* Clear handle fields */
180 handle->initialized = false;
181 handle->mutex = NULL;
182
183 /* Release and delete mutex */
184 if (mutex != NULL) {
185 xSemaphoreGive(mutex);
186 vSemaphoreDelete(mutex);
187 }
188
189 ESP_LOGI(s_tag, "BQ7850 BMS deinitialized");
190 return ESP_OK;
191}

282 bus_name,

284 k_bq7850_cmd_device_chemistry,
285 chem_data,
286 k_bq7850_device_info_string_len,

288 if (ret == ESP_OK && chem_len > 0) {

322esp_err_t star_bms_bq7850_read_cells(const bq7850_handle_t* handle, bq7850_cell_data_t* cell_data)
323{
324 if (handle == NULL || cell_data == NULL) {
325 return ESP_ERR_INVALID_ARG;
326 }
327
328 star_bus_manager_t* manager = handle->manager;
329 const char* bus_name = handle->bus_name;
330 uint8_t addr = handle->smbus_addr;
331 esp_err_t ret = ESP_OK;
332
333 memset(cell_data, 0, sizeof(bq7850_cell_data_t));
334
335 /* Read individual cell voltages */
336 for (uint8_t i = 0; i < k_bq7850_max_cells; i++) {
337 ret = internal_bq7850_read_cell_voltage(manager, bus_name, addr, i, &cell_data->cell_mv[i]);
338
339 if (ret == ESP_OK && cell_data->cell_mv[i] > 0) {
340 cell_data->valid_cells++;
341 } else if (ret != ESP_OK) {
342 ESP_LOGW(s_tag, "Failed to read cell %d voltage: %s", i + 1, esp_err_to_name(ret));
343 /* Continue reading other cells */
344 }
345 }
346
347 /* Read total pack voltage */
348 ret = star_smbus_read_word(manager, bus_name, addr, k_bq7850_cmd_voltage, &cell_data->pack_mv);
349 if (ret != ESP_OK) {
350 ESP_LOGE(s_tag, "Failed to read pack voltage: %s", esp_err_to_name(ret));
351 }
352
353 return (cell_data->valid_cells > 0) ? ESP_OK : ESP_FAIL;
354}

388/* --- Temperature Functions --- */
389
390esp_err_t star_bms_bq7850_read_temperatures(const bq7850_handle_t* handle,
391 bq7850_temp_data_t* temp_data)
392{
393 if (handle == NULL || temp_data == NULL) {
394 return ESP_ERR_INVALID_ARG;
395 }
396
397 star_bus_manager_t* manager = handle->manager;
398 const char* bus_name = handle->bus_name;
399 uint8_t addr = handle->smbus_addr;
400 esp_err_t ret;
401
402 memset(temp_data, 0, sizeof(bq7850_temp_data_t));
403
404 /* Read temperature sensors via manufacturer access */
405 int32_t temp_sum = 0;
406
407 for (uint8_t i = 0; i < k_bq7850_max_temp_sensors; i++) {
408 uint16_t temp_raw = 0;
409 uint16_t subcmd = k_bq7850_subcmd_temp_sensor_base + i;
410
411 ret = internal_bq7850_manufacturer_access(manager, bus_name, addr, subcmd, &temp_raw);
412
413 if (ret == ESP_OK && temp_raw > 0) {
414 /* BQ7850 returns temperature in 0.1K units */
415 /* Convert to 0.1C: T(C) = T(K) - 273.15 */
416 temp_data->temp_c[i] = temp_raw - k_bq7850_kelvin_offset_deci_c;
417 temp_data->valid_sensors++;
418 temp_sum += temp_data->temp_c[i];
419 }
420 }
421
422 /* Calculate average temperature */
423 if (temp_data->valid_sensors > 0) {
424 temp_data->avg_temp_c = temp_sum / temp_data->valid_sensors;
425 }
426
427 return (temp_data->valid_sensors > 0) ? ESP_OK : ESP_FAIL;
428}

453esp_err_t star_bms_bq7850_read_current(const bq7850_handle_t* handle,
454 bq7850_current_data_t* current_data)
455{
456 if (handle == NULL || current_data == NULL) {
457 return ESP_ERR_INVALID_ARG;
458 }
459
460 star_bus_manager_t* manager = handle->manager;
461 const char* bus_name = handle->bus_name;
462 uint8_t addr = handle->smbus_addr;
463 esp_err_t ret;
464
465 memset(current_data, 0, sizeof(bq7850_current_data_t));
466
467 /* Read instantaneous current (signed) - use intermediate variable to avoid aliasing */
468 uint16_t current_raw = 0;
469 ret = star_smbus_read_word(manager, bus_name, addr, k_bq7850_cmd_current, &current_raw);
470 if (ret != ESP_OK) {
471 ESP_LOGE(s_tag, "Failed to read current: %s", esp_err_to_name(ret));
472 return ret;
473 }
474 current_data->current_ma = (int16_t)current_raw;
475
476 /* Read average current - use intermediate variable to avoid aliasing */
477 uint16_t avg_current_raw = 0;
478 ret =
479 star_smbus_read_word(manager, bus_name, addr, k_bq7850_cmd_average_current, &avg_current_raw);
480 if (ret != ESP_OK) {
481 ESP_LOGW(s_tag, "Failed to read average current: %s", esp_err_to_name(ret));
482 } else {
483 current_data->avg_current_ma = (int16_t)avg_current_raw;
484 }
485
486 /* Read voltage */
487 ret =
488 star_smbus_read_word(manager, bus_name, addr, k_bq7850_cmd_voltage, &current_data->voltage_mv);
489 if (ret != ESP_OK) {
490 ESP_LOGE(s_tag, "Failed to read voltage: %s", esp_err_to_name(ret));
491 return ret;
492 }
493
494 /* Calculate power (mW) = voltage (mV) * current (mA) / 1000 */
495 current_data->power_mw = ((int32_t)current_data->voltage_mv * current_data->current_ma) / 1000;
496
497 return ESP_OK;
498}

761esp_err_t star_bms_bq7850_write_protection(const bq7850_handle_t* handle,
762 const bq7850_protection_t* protection)
763{
764 if (handle == NULL || protection == NULL) {
765 return ESP_ERR_INVALID_ARG;
766 }
767
768 star_bus_manager_t* manager = handle->manager;
769 const char* bus_name = handle->bus_name;
770 uint8_t addr = handle->smbus_addr;
771 esp_err_t ret;
772
773 ESP_LOGW(s_tag, "Writing protection thresholds (device must be unsealed)");
774
775 /* Write protection thresholds via manufacturer access */
776 ret = internal_bq7850_manufacturer_access(manager,
777 bus_name,
778 addr,
779 k_bq7850_subcmd_protection_ov_thresh,
780 NULL);
781 if (ret == ESP_OK) {
782 ret = star_smbus_write_word(manager,
783 bus_name,
784 addr,
785 k_bq7850_cmd_manufacturer_access,
786 protection->overvoltage_mv);
787 }
788
789 if (ret != ESP_OK) {
790 ESP_LOGE(s_tag, "Failed to write OV threshold: %s", esp_err_to_name(ret));
791 return ret;
792 }
793
794 /* Add delay between writes (required for BQ7850 EEPROM write settling) */
795 vTaskDelay(pdMS_TO_TICKS(k_bq7850_cmd_write_delay_ms));
796
797 /* Write undervoltage threshold */
798 ret = internal_bq7850_manufacturer_access(manager,
799 bus_name,
800 addr,
801 k_bq7850_subcmd_protection_uv_thresh,
802 NULL);
803 if (ret == ESP_OK) {
804 ret = star_smbus_write_word(manager,
805 bus_name,
806 addr,
807 k_bq7850_cmd_manufacturer_access,
808 protection->undervoltage_mv);
809 }
810
811 if (ret != ESP_OK) {
812 ESP_LOGE(s_tag, "Failed to write UV threshold: %s", esp_err_to_name(ret));
813 return ret;
814 }
815
816 vTaskDelay(pdMS_TO_TICKS(k_bq7850_cmd_write_delay_ms));
817
818 /* Write overcurrent threshold */
819 ret = internal_bq7850_manufacturer_access(manager,
820 bus_name,
821 addr,
822 k_bq7850_subcmd_protection_oc_thresh,
823 NULL);
824 if (ret == ESP_OK) {
825 ret = star_smbus_write_word(manager,
826 bus_name,
827 addr,
828 k_bq7850_cmd_manufacturer_access,
829 protection->overcharge_ma);
830 }
831
832 if (ret != ESP_OK) {
833 ESP_LOGE(s_tag, "Failed to write OC threshold: %s", esp_err_to_name(ret));
834 return ret;
835 }
836
837 vTaskDelay(pdMS_TO_TICKS(k_bq7850_cmd_write_delay_ms));
838
839 /* Write overtemperature threshold */
840 ret = internal_bq7850_manufacturer_access(manager,
841 bus_name,
842 addr,
843 k_bq7850_subcmd_protection_ot_thresh,
844 NULL);
845 if (ret == ESP_OK) {
846 ret = star_smbus_write_word(manager,
847 bus_name,
848 addr,
849 k_bq7850_cmd_manufacturer_access,
850 (uint16_t)protection->overtemp_c);
851 }
852
853 if (ret != ESP_OK) {
854 ESP_LOGE(s_tag, "Failed to write OT threshold: %s", esp_err_to_name(ret));
855 return ret;
856 }
857
858 vTaskDelay(pdMS_TO_TICKS(k_bq7850_cmd_write_delay_ms));
859
860 /* Write undertemperature threshold */
861 ret = internal_bq7850_manufacturer_access(manager,
862 bus_name,
863 addr,
864 k_bq7850_subcmd_protection_ut_thresh,
865 NULL);
866 if (ret == ESP_OK) {
867 ret = star_smbus_write_word(manager,
868 bus_name,
869 addr,
870 k_bq7850_cmd_manufacturer_access,
871 (uint16_t)protection->undertemp_c);
872 }
873
874 if (ret != ESP_OK) {
875 ESP_LOGE(s_tag, "Failed to write UT threshold: %s", esp_err_to_name(ret));
876 return ret;
877 }
878
879 ESP_LOGI(s_tag, "All protection thresholds updated (OV, UV, OC, OT, UT)");
880
881 return ESP_OK;
882}

930size_t star_bms_bq7850_status_to_string(const bq7850_status_t* status, char* buffer, size_t size)
931{

933 return 0;
934 }
935
936 size_t offset = 0;
937
938 /* Helper macro to safely append to buffer with overflow protection */
939#define SAFE_APPEND(fmt, ...) \
940 do { \
941 if (offset < size) { \
942 int written = snprintf(buffer + offset, size - offset, fmt, ##__VA_ARGS__); \
943 if (written > 0) { \
944 offset += (size_t)written; \
945 if (offset > size) { \
946 offset = size; \
947 } \
948 } \
949 } \
950 } while (0)
951
952 SAFE_APPEND("Battery Status: ");
953
954 if (status->charging) {
955 SAFE_APPEND("CHARGING ");
956 }
957 if (status->discharging) {
958 SAFE_APPEND("DISCHARGING ");
959 }
960 if (status->fully_charged) {
961 SAFE_APPEND("FULL ");
962 }
963
964 if (status->safety_status) {
965 SAFE_APPEND("\nSafety: ");
966 if (status->safety_status & k_bq7850_safety_status_cuv) {
967 SAFE_APPEND("CUV ");
968 }
969 if (status->safety_status & k_bq7850_safety_status_cov) {
970 SAFE_APPEND("COV ");
971 }
972 if (status->safety_status & k_bq7850_safety_status_occ) {
973 SAFE_APPEND("OCC ");
974 }
975 if (status->safety_status & k_bq7850_safety_status_ocd) {
976 SAFE_APPEND("OCD ");
977 }
978 }
979
980#undef SAFE_APPEND
981
982 return offset;
983}
