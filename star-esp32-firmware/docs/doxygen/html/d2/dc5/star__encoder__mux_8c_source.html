50esp_err_t star_encoder_mux_init(star_encoder_mux_handle_t* handle,
51 const star_encoder_mux_config_t* config)
52{
53 ESP_RETURN_ON_FALSE(handle, ESP_ERR_INVALID_ARG, s_tag, "Handle is NULL");
54 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_tag, "Config is NULL");
55 ESP_RETURN_ON_FALSE(!handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Already initialized");
56
57 /* Zero out handle */
58 memset(handle, 0, sizeof(star_encoder_mux_handle_t));
59
60 esp_err_t ret = ESP_OK;
61
62 /* Configure multiplexer select lines as outputs */
63 gpio_config_t gpio_conf = {
64 .pin_bit_mask = (1ULL << config->pin_sel0) | (1ULL << config->pin_sel1),
65 .mode = GPIO_MODE_OUTPUT,
66 .pull_up_en = GPIO_PULLUP_DISABLE,
67 .pull_down_en = GPIO_PULLDOWN_DISABLE,
68 .intr_type = GPIO_INTR_DISABLE,
69 };
70
71 ret = gpio_config(&gpio_conf);
72 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to configure select line GPIOs");
73
74 /* Set select lines to encoder 0 initially */
75 gpio_set_level(config->pin_sel0, 0);
76 gpio_set_level(config->pin_sel1, 0);
77
78 /* Initialize shared encoder instance */
79 star_encoder_config_t enc_config = {
80 .pin_a = config->pin_out_a,
81 .pin_b = config->pin_out_b,
82 .filter_value = config->filter_value,
83 .high_limit = config->high_limit,
84 .low_limit = config->low_limit,
85 };
86
87 ret = star_encoder_init(&handle->encoder, &enc_config);
88 if (ret != ESP_OK) {
89 ESP_LOGE(s_tag, "Failed to initialize encoder: %s", esp_err_to_name(ret));
90 return ret;
91 }
92
93 /* Store configuration */
94 handle->pin_sel0 = config->pin_sel0;
95 handle->pin_sel1 = config->pin_sel1;
96 handle->settling_time_us = config->settling_time_us;
97 handle->last_time_us = esp_timer_get_time();
98
99 /* Initialize position tracking */
100 for (int i = 0; i < k_encoder_count; i++) {
101 handle->positions[i] = 0;
102 handle->last_counts[i] = 0;
103 }
104
105 handle->initialized = true;
106
107 ESP_LOGI(s_tag,
108 "Encoder mux initialized: sel=%d/%d, out=%d/%d, settling=%luμs",
109 config->pin_sel0,
110 config->pin_sel1,
111 config->pin_out_a,
112 config->pin_out_b,
113 config->settling_time_us);
114
115 return ESP_OK;
116}

137esp_err_t star_encoder_mux_read_all(star_encoder_mux_handle_t* handle,
138 int32_t out_positions[k_encoder_count],
139 float out_velocities_rpm[k_encoder_count],
140 float dt_ms,
141 uint32_t counts_per_rev)
142{
143 ESP_RETURN_ON_FALSE(handle && out_positions && out_velocities_rpm,
144 ESP_ERR_INVALID_ARG,
145 s_tag,
146 "Handle or output arrays are NULL");
147 ESP_RETURN_ON_FALSE(handle->initialized, ESP_ERR_INVALID_STATE, s_tag, "Not initialized");
148 ESP_RETURN_ON_FALSE(dt_ms > 0.0f, ESP_ERR_INVALID_ARG, s_tag, "dt_ms must be > 0");
149 ESP_RETURN_ON_FALSE(counts_per_rev > 0, ESP_ERR_INVALID_ARG, s_tag, "counts_per_rev must be > 0");
150
151 esp_err_t ret = ESP_OK;
152
153 /* Sequentially read all 4 encoders */
154 for (uint8_t i = 0; i < k_encoder_count; i++) {
155 ret = internal_read_encoder_count(handle,
156 i,
157 &out_positions[i],
158 &out_velocities_rpm[i],
159 dt_ms,
160 counts_per_rev);
161
162 if (ret != ESP_OK) {
163 ESP_LOGW(s_tag, "Failed to read encoder %d: %s", i, esp_err_to_name(ret));
164 /* Continue reading other encoders even if one fails */
165 out_positions[i] = handle->positions[i]; /* Use last known position */
166 out_velocities_rpm[i] = 0.0f; /* Zero velocity on error */
167 }
168 }
169
170 /* Update timestamp */
171 handle->last_time_us = esp_timer_get_time();
172
173 return ESP_OK;
174}

235static esp_err_t internal_select_encoder(star_encoder_mux_handle_t* handle, uint8_t encoder_idx)
236{
237 if (encoder_idx >= k_encoder_count) {
238 return ESP_ERR_INVALID_ARG;
239 }
240
241 /* Set select lines based on encoder index (binary encoding) */
242 gpio_set_level(handle->pin_sel0, encoder_idx & 0x01); /* Bit 0 */
243 gpio_set_level(handle->pin_sel1, (encoder_idx >> 1) & 0x01); /* Bit 1 */
244
245 /* Wait for multiplexer settling time */
246 ets_delay_us(handle->settling_time_us);
247
248 return ESP_OK;
249}

269static esp_err_t internal_read_encoder_count(star_encoder_mux_handle_t* handle,
270 uint8_t encoder_idx,
271 int32_t* out_position,
272 float* out_velocity_rpm,
273 float dt_ms,
274 uint32_t counts_per_rev)
275{
276 /* Select encoder via multiplexer */
277 esp_err_t ret = internal_select_encoder(handle, encoder_idx);
278 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to select encoder %d", encoder_idx);
279
280 /* Read current PCNT count */
281 int32_t current_count;
282 ret = star_encoder_get_count(&handle->encoder, &current_count);
283 ESP_RETURN_ON_ERROR(ret, s_tag, "Failed to read encoder %d count", encoder_idx);
284
285 /* Calculate delta count from last read */
286 int32_t delta_count = current_count - handle->last_counts[encoder_idx];
287
288 /* Update accumulated position */
289 handle->positions[encoder_idx] += delta_count;
290 *out_position = handle->positions[encoder_idx];
291
292 /* Calculate velocity: (counts/dt_ms) * (ms/min) / (counts/rev) = rev/min */
293 *out_velocity_rpm = ((float)delta_count * s_ms_per_minute) / (dt_ms * (float)counts_per_rev);
294
295 /* Store current count for next delta calculation */
296 handle->last_counts[encoder_idx] = current_count;
297
298 return ESP_OK;
299}
