\section{Executive Summary}

STAR is a distributed robotics platform with a custom PCB RX72N motor controller, Raspberry Pi 5 control system, ROS2 Jazzy middleware, and a Go gateway service. The system communicates over 10~Mbps SPI using Protocol Buffers with HARQ/FEC error correction.

\subsubsection{WireMessage Command Dispatcher (confirmed complete)}

\begin{okbox}
  \filepath{src/tasks/comm\_task.c:1959} -- \texttt{internal\_handle\_command\_frame()} handles all incoming protobuf commands.
\end{okbox}

\begin{lstlisting}[language=C, caption={Command dispatcher in comm\_task.c (lines 1959--2060)}]
static void internal_handle_command_frame(
    rx_comm_channel_t channel, const rx_frame_t* frame)
{
    /* Try SetVelocityRequest (4-motor differential drive) */
    err = rx_nanopb_decode_velocity_request(..., &velocity_req);
    if (err == k_rx_ok && velocity_req.has_command) {
        /* Map proto -> motor_command_t -> shared_data */
        cmd.target_velocity_mps[k_motor_idx_front_left]  = ...;
        cmd.target_velocity_mps[k_motor_idx_front_right] = ...;
        cmd.target_velocity_mps[k_motor_idx_back_left]   = ...;
        cmd.target_velocity_mps[k_motor_idx_back_right]  = ...;
        shared_data_set_motor_command(&cmd);   // -> wakes motor_control_task
        return;
    }

    /* Try EmergencyStopRequest */
    err = rx_nanopb_decode_estop_request(..., &estop_req);
    if (err == k_rx_ok) {
        shared_data_trigger_estop(k_estop_reason_manual); // -> POEG HW stop
        return;
    }

    /* Try SetPIDGainsRequest */
    err = rx_nanopb_decode_pid_gains_request(..., &pid_req);
    if (err == k_rx_ok && pid_req.has_pid_config) {
        gains.kp = (float)pid_req.pid_config.kp;  // Kp, Ki, Kd
        shared_data_set_pid_gains(&gains);
        return;
    }

    /* Try SetRetransmitConfigRequest */
    err = rx_nanopb_decode_retransmit_config_request(..., &retransmit_req);
    if (err == k_rx_ok && retransmit_req.has_retransmit_config) {
        rx_comm_manager_set_auto_retransmit(...);
        return;
    }

    rx_log_warn(s_tag, "Could not decode command payload");
}
\end{lstlisting}

\textbf{OTA state machine:}

\begin{lstlisting}[language=C, caption={OTA state machine types (rx\_ota.h)}]
typedef enum : uint8_t {
    k_ota_state_idle        = 0,  /**< No update in progress */
    k_ota_state_starting    = 1,  /**< BeginUpdate received */
    k_ota_state_receiving   = 2,  /**< WriteChunk frames arriving */
    k_ota_state_validating  = 3,  /**< Checking CRC-32 + size */
    k_ota_state_ready       = 4,  /**< Ready to reboot */
    k_ota_state_error       = 5,  /**< Error state */
} ota_state_t;

typedef struct {
    ota_state_t state;         /**< Current FSM state */
    uint32_t    total_size;    /**< Expected firmware size (bytes) */
    uint32_t    bytes_written; /**< Progress counter */
    uint32_t    expected_crc;  /**< Expected CRC-32 from BeginUpdate */
    uint32_t    running_crc;   /**< Running CRC accumulator */
    uint32_t    session_id;    /**< From BeginUpdateRequest.header */
} ota_context_t;

rx_err_t rx_ota_begin(const star_v1_BeginUpdateRequest* req);
rx_err_t rx_ota_write_chunk(const star_v1_FirmwareChunk* chunk);
rx_err_t rx_ota_finalize(void);
rx_err_t rx_ota_abort(void);
rx_err_t rx_ota_get_progress(uint8_t* percent_out);
rx_err_t rx_ota_reboot(void);
\end{lstlisting}

\textbf{Estimated effort:} 2--3 days (firmware state machine + flash write + gateway integration)

\subsubsection{Gap F-2: NVS Configuration Persistence (HIGH)}

\begin{warnbox}
  \textbf{Impact:} PID gains received via \texttt{SetPIDGainsRequest} are applied to the running system via \texttt{shared\_data\_set\_pid\_gains()} but are NOT saved to flash. Every reboot resets to compile-time defaults.
\end{warnbox}

The RX72N has 64 KB Data Flash (separate from code flash) ideal for NVS. No library currently uses it.

\textbf{Files to create:}

\begin{lstlisting}[caption={NVS library structure}]
e2-studio-star-rx72n-firmware/

  tests/
    test_rx_nvs.c   -- Unity tests (mock flash)
\end{lstlisting}

\begin{lstlisting}[language=C, caption={NVS API (rx\_nvs.h)}]
/**
 * @brief Write named config blob to Data Flash.
 * @param[in] key   ASCII key (max 32 chars)
 * @param[in] data  Buffer to save
 * @param[in] len   Length (max 4096 bytes)
 */
rx_err_t rx_nvs_write(const char* key,
                       const void* data, uint16_t len);

/**
 * @brief Read named config blob from Data Flash.
 */
rx_err_t rx_nvs_read(const char* key,
                      void* data, uint16_t* len);

/** @brief Factory reset (erase all Data Flash). */
rx_err_t rx_nvs_erase_all(void);
\end{lstlisting}

\textbf{Integration points after creating NVS:}
\begin{itemize}
  \item \filepath{src/tasks/comm\_task.c} -- After \texttt{shared\_data\_set\_pid\_gains()}, call \texttt{rx\_nvs\_write("pid\_config", ...)}
  \item \filepath{src/main.c} -- On startup, call \texttt{rx\_nvs\_read("pid\_config", ...)} before \texttt{tx\_kernel\_enter()}
\end{itemize}

\textbf{Estimated effort:} 4--6 hours

\textbf{Motor dynamics model} (to implement in \filepath{cmd/virtual\_rx72n/motor\_sim.go}):

\begin{lstlisting}[language=Go, caption={Motor dynamics (discrete-time, 100 Hz)}]
// First-order model: G(s) = K/(tau*s + 1)
// K = 3.665 rad/s/V, tau = 0.075 s
// Discrete backward Euler at T=0.010 s:
//   v[k] = (1 - exp(-T/tau)) * K * u[k] + exp(-T/tau) * v[k-1]
//         = 0.1248 * K * u[k] + 0.8752 * v[k-1]
alpha := math.Exp(-motorSamplePeriod / motorTimeConstant)  // 0.8752
m.velocityRadps = (1-alpha)*motorGain*inputVoltage +
                  alpha*m.velocityRadps
m.positionRad += m.velocityRadps * motorSamplePeriod
ticks := int64(m.positionRad / (2*math.Pi) * ticksPerRev)
m.encoderTicks.Store(ticks)
\end{lstlisting}
