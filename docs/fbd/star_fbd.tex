% ============================================================================
% SPACING CONSTANTS - All spacing derived from these values
% ============================================================================
% Vertical spacing: 0.4cm everywhere
%   - Top of block to bottom of title: 0.4cm (title centered at top)
%   - Bottom of title to first label: 0.4cm
%   - Between labels: 0.4cm
%   - Last label to bottom of block: 0.4cm
%
% Horizontal spacing:
%   - Labels inset from block edge: 0.4cm
%   - Gap between left and right labels: 0.8cm (minimum)
%
% Block Height Formula (CORRECTED):
%   H = 0.4 (title area) + 0.4 (title-to-labels gap)
%       + (max_labels * 0.4) + 0.4 (bottom padding)
%   H = 1.2 + (max_labels * 0.4) cm
%
% Label Y-Position Formula (from block center):
%   For label i (1-indexed) with max_labels N and height H:
%   y_offset = H/2 - 1.2 - (i-1)*0.4
%   This places first label at H/2 - 1.2, then 0.4cm apart
% ============================================================================

\newcommand{\vertspace}{0.4}        % Vertical spacing in cm (between elements)
\newcommand{\horzinset}{0.4}        % Horizontal inset from block edge in cm
\newcommand{\labelgap}{0.8}         % Minimum gap between left/right labels in cm

% Height macros: H = 1.2 + N*0.4 where N = max(left_labels, right_labels)
