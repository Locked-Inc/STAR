\subsubsection { Legend }

\paragraph { Shapes }
\begin{itemize}[nosep]
    \item \textbf{Ellipse} : Start / End nodes
    \item \textbf{Rectangle} : Process steps
    \item \textbf{Diamond} : Decision points
    \item \textbf{Parallelogram} : Data input / output
\end {
  itemize
}

\paragraph { Colors }
\begin{itemize}[nosep]
    \item \textbf{Yellow} : Start / End / Decision nodes
    \item \textbf{Blue} : Initialization steps
    \item \textbf{Green} : Processing steps
    \item \textbf{Orange} : Optimization / Safety operations
    \item \textbf{Red} : Emergency handling
    \item \textbf{Purple} : Output / Communication
    \item \textbf{Gray} : Data flow
\end {
  itemize
}

\paragraph { Line Types }
\begin{itemize}[nosep]
    \item \textbf{Solid} : Normal control flow
    \item \textbf{Dashed} : Data flow or feedback
    \item \textbf{Bold Red} : Emergency paths
\end {
  itemize
}
