\begin{itemize}[nosep]
    \item \textbf{ES} = Early Start
    : Earliest day the task can begin
    \item \textbf{EF} =
          Early Finish : Earliest day the task can complete(ES + Duration)
    \item \textbf{LS} = Late Start
    : Latest day the task can begin without delaying the project
    \item \textbf{LF} = Late Finish
    : Latest day the task can complete without delaying the project
    \item \textbf{Float} = Total Float / Slack : LS - ES(days of flexibility)
    \item \textbf{CP} = Critical Path : Tasks with zero float(bold rows)
