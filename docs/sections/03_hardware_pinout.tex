\subsubsection{Pin Function Categories}

\textbf{SPI3 (Peripheral Mode - Communication with RPi5):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item IO10: Chip Select (CS)
    \item IO11: Controller Out Peripheral In (COPI)
    \item IO12: Serial Clock (SCLK)
    \item IO13: Controller In Peripheral Out (CIPO)
    \item TXD0 (IO1): Data Ready 1 interrupt to Pi
    \item RXD0 (IO3): Data Ready 2 interrupt to Pi
\end{itemize}

\textbf{SPI2 (Controller Mode - Communication with Motor Drivers):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item IO35: Controller Out Peripheral In (COPI)
    \item IO36: Serial Clock (SCLK)
    \item IO37: Controller In Peripheral Out (CIPO)
    \item Chip selects generated via 74HC238 decoder (see below)
\end{itemize}

\textbf{MCPWM (Motor Control):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Motor 1: IO6 (EN), IO7 (PH)
    \item Motor 2: IO14 (EN), IO15 (PH)
    \item Motor 3: IO16 (EN), IO17 (PH)
    \item Motor 4: IO18 (EN), IO21 (PH)
\end{itemize}

\textbf{ADC (Current Sensing - IPROPI):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item IO1: Motor 1 current
    \item IO2: Motor 2 current
    \item IO4: Motor 3 current
    \item IO5: Motor 4 current
\end{itemize}

\textbf{Encoder Selection and Reading:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item IO40, IO41: Multiplexer select lines
    \item IO42, IO47: Encoder output channels
\end{itemize}

\subsection{Additional Peripheral Connections}

\subsubsection{IMU (Inertial Measurement Unit)}

\textbf{Hardware:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item BNO055: 9-axis absolute orientation sensor
    \item BMP280: Barometric pressure sensor
\end{itemize}

\textbf{Protocol:} I2C communication

\textbf{Connection:} Connects to RPi5 via Header J10:
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item PI\_SCL (Clock) routes to RPi5 Pin 5
    \item PI\_SDA (Data) routes to RPi5 Pin 3
    \item IMU\_RST routes to RPi5 Pin 16 (IO23)
    \item IMU\_INT routes to RPi5 Pin 18 (IO24)
\end{itemize}

\subsubsection{Stereo Cameras}

\textbf{Hardware:} External stereo camera modules

\textbf{Protocol:} I2C communication

\textbf{Connection:} Connects via Header J12 to the same I2C bus (PI\_SCL and PI\_SDA), allowing the Pi to address them alongside the IMU using different I2C addresses.

\subsubsection{Temperature Sensor}

\textbf{Hardware:} DS18B20+ 1-Wire digital temperature sensor

\textbf{Protocol:} 1-Wire communication

\textbf{Connection:} Connected to ESP32-S3 IO48
