\section{Hardware Pin Connections}

\subsection{RX72N Microcontroller Pinout (R5F572NNHGFP\#30)}

\noindent\colorbox{keyinsightcolor}{\parbox{\dimexpr\linewidth-2\fboxsep}{%
\textbf{Hardware Advantage:} The RX72N provides \textbf{182 direct GPIO pins} with dedicated peripheral functions. Unlike the ESP32 design, \textbf{no multiplexers or decoders are required} - each motor gets its own dedicated MTU channels for both PWM and encoder, with direct GPIO access to fault lines and current sense ADC channels.
}}

\vspace{0.3cm}

\subsubsection{Package Information}

\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item \textbf{MCU:} Renesas RX72N (R5F572NNHGFP\#30)
    \item \textbf{Package:} 100-pin LFQFP (Low-profile Quad Flat Package)
    \item \textbf{GPIO Pins:} 182 (multiplexed across package pins)
    \item \textbf{Flash:} 4 MB
    \item \textbf{RAM:} 1 MB
    \item \textbf{CPU:} RXv3 core at 240 MHz
\end{itemize}

\textbf{PWM Configuration:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Frequency: 20 kHz (center-aligned PWM mode 1)
    \item Dead-time: 1 $\mu$s for H-bridge safety
    \item Resolution: $\sim$12-bit (3000 counts at 120 MHz PCLKA)
    \item MTIOCA: PH (Phase) signal to DRV8243S
    \item MTIOCB: EN (Enable) signal to DRV8243S
\end{itemize}

\textbf{Encoder Configuration:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Mode: MTU Phase Counting Mode 1 (4x decoding)
    \item Inputs: MTCLKA (Phase A), MTCLKB (Phase B)
    \item Resolution: 341 PPR (Pulses Per Revolution)
    \item Counter: 16-bit hardware counter with software wraparound tracking
    \item Update Rate: Read at 250 Hz control loop
\end{itemize}

\textbf{Fault Detection:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Active-low nFAULT signals from DRV8243S
    \item Direct GPIO input (no external hardware needed)
    \item Polled at 250 Hz control loop
\end{itemize}

\textbf{Current Sensing:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item IPROPI analog output from DRV8243S
    \item 12-bit S12ADFa ADC with PWM synchronization
    \item Range: 0-3.3V (0-2000 mA motor current)
    \item Sample time: 24 states for low-noise acquisition
\end{itemize}

\subsubsection{Communication Interfaces}

\textbf{Debug UART (SCI1):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item PORT2.6: TXD1 (Transmit)
    \item PORT2.5: RXD1 (Receive)
    \item Baud Rate: 115200 bps
    \item Purpose: Debug output, logging, firmware update
\end{itemize}

\textbf{SPI to Raspberry Pi 5 (RSPI0):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item PORT7.4: SSLA0 (Chip Select)
    \item PORT7.5: RSPCKA (Serial Clock)
    \item PORT7.6: MISOA (Controller In Peripheral Out - CIPO)
    \item PORT7.7: MOSIA (Controller Out Peripheral In - COPI)
    \item Mode: Peripheral (RX72N receives commands from RPi5)
    \item Speed: 10 Mbps
    \item Protocol: nanopb + CRC-32
\end{itemize}

\textbf{I2C Bus 0 (RIIC0) - BQ25798 Charger:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item SDA0, SCL0: I2C data and clock
    \item PORT5.0: BQ25798 INT (interrupt)
    \item Purpose: Battery charging control
\end{itemize}

\textbf{I2C Bus 1 (RIIC1) - Fuel Gauge (SMBUS):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item SDA1, SCL1: SMBUS-compatible I2C
    \item PORT5.1: Fuel Gauge ALERT (interrupt)
    \item Purpose: Battery state-of-charge monitoring
    \item Features: Hardware CRC-8 PEC support
\end{itemize}

\subsubsection{Status and Safety}

\textbf{Status LEDs:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item PORT0.3: Status LED (heartbeat)
    \item PORT0.5: Error LED (fault indication)
    \item PORT0.7: Power LED (system active)
\end{itemize}

\textbf{Emergency Stop:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item PORT1.0: E-STOP button input
    \item Pull-up resistor, active-low
    \item Polled at 250 Hz, immediately stops all motors
\end{itemize}

\subsection{Comparison: RX72N vs ESP32-S3 Architecture}

\begin{table}[H]
\centering
\small
\begin{tabular}{|l|c|c|}
\hline
\textbf{Feature} & \textbf{ESP32-S3} & \textbf{RX72N} \\
\hline
GPIO Pins & 35 usable & \textbf{182 direct} \\
\hline
Motor PWM & MCPWM (shared) & MTU3a (dedicated) \\
\hline
Encoders & 4x 4:1 mux + PCNT & \textbf{4x dedicated MTU} \\
\hline
Motor CS Lines & 74HC238 decoder & \textbf{Direct GPIO} \\
\hline
Fault Detection & Via decoder outputs & \textbf{Direct GPIO} \\
\hline
Control Loop & 250 Hz (mux overhead) & \textbf{250 Hz (no mux)} \\
\hline
Encoder Latency & $\sim$40 $\mu$s (mux switching) & \textbf{$\sim$10 $\mu$s (direct)} \\
\hline
\end{tabular}
\caption{Hardware architecture comparison}
\label{tab:architecture-comparison}
\end{table}

\noindent\colorbox{keyinsightcolor}{\parbox{\dimexpr\linewidth-2\fboxsep}{%
\textbf{Key Advantage:} The RX72N eliminates all multiplexing overhead from the ESP32 design. Each motor has dedicated hardware resources with direct GPIO access, reducing control loop latency by 75\% and simplifying the PCB design (no external decoder/mux ICs required).
}}

\subsubsection{RPi5 Pin Function Categories}

\textbf{SPI (Controller Mode - Communication with RX72N):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Pin 24 (CE0): Chip Select
    \item Pin 19 (COPI): Controller Out Peripheral In
    \item Pin 23 (SCLK): Serial Clock
    \item Pin 21 (CIPO): Controller In Peripheral Out
    \item Pin 38 (IO20): Data Ready 1 interrupt from RX72N
    \item Pin 40 (IO21): Data Ready 2 interrupt from RX72N
\end{itemize}

\textbf{I2C (IMU and Stereo Cameras):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Pin 3 (SDA): I2C Data
    \item Pin 5 (SCL): I2C Clock
    \item Pin 16 (IO23): IMU Reset
    \item Pin 18 (IO24): IMU Interrupt
\end{itemize}

\textbf{UART (RPLiDAR C1):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Pin 8 (TXD): LiDAR RX (Pi TX)
    \item Pin 10 (RXD): LiDAR TX (Pi RX)
\end{itemize}

\textbf{Motor Driver Fault Lines (from RX72N):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Pin 33 (IO13): Motor 0 nFAULT
    \item Pin 31 (IO6): Motor 1 nFAULT
    \item Pin 29 (IO5): Motor 2 nFAULT
    \item Pin 27 (ID\_SD): Motor 3 nFAULT
\end{itemize}

\textbf{HC-SR04 Ultrasonic Sensors (6 sensors):}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item Sensor 1: Pin 7 (TRIG), Pin 11 (ECHO)
    \item Sensor 2: Pin 13 (TRIG), Pin 15 (ECHO)
    \item Sensor 3: Pin 35 (TRIG), Pin 37 (ECHO)
    \item Sensor 4: Pin 12 (TRIG), Pin 22 (ECHO)
    \item Sensor 5: Pin 26 (TRIG), Pin 28 (ECHO)
    \item Sensor 6: Pin 32 (TRIG), Pin 36 (ECHO)
\end{itemize}

\subsection{Additional Peripheral Connections}

\subsubsection{IMU (Inertial Measurement Unit)}

\textbf{Hardware:}
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item BNO055: 9-axis absolute orientation sensor
    \item BMP280: Barometric pressure sensor
\end{itemize}

\textbf{Protocol:} I2C communication

\textbf{Connection:} Connects to RPi5 via Header J10:
\begin{itemize}[leftmargin=1.5em,topsep=4pt,itemsep=2pt]
    \item PI\_SCL (Clock) routes to RPi5 Pin 5
    \item PI\_SDA (Data) routes to RPi5 Pin 3
    \item IMU\_RST routes to RPi5 Pin 16 (IO23)
    \item IMU\_INT routes to RPi5 Pin 18 (IO24)
\end{itemize}

\subsubsection{Stereo Cameras}

\textbf{Hardware:} External stereo camera modules

\textbf{Protocol:} I2C communication

\textbf{Connection:} Connects via Header J12 to the same I2C bus (PI\_SCL and PI\_SDA), allowing the Pi to address them alongside the IMU using different I2C addresses.

\subsubsection{Temperature Sensor}

\textbf{Hardware:} DS18B20+ 1-Wire digital temperature sensor

\textbf{Protocol:} 1-Wire communication

\textbf{Connection:} Connected to RX72N GPIO (specific pin TBD in final hardware design)
