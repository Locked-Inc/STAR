┌─ PROTOCOL TIMING ─────────────────────────────────────────────────────────┐
│                                                                            │
│  MCU Initialization:                                                       │
│    Pull LOW:     1-10ms minimum (use 1ms)                                 │
│    Release HIGH: 20-40μs                                                   │
│                                                                            │
│  Sensor Response:                                                          │
│    Pull LOW:     80μs                                                      │
│    Release HIGH: 80μs                                                      │
│                                                                            │
│  Data Bits (40 total):                                                     │
│    Start:        50μs LOW (both 0 and 1)                                  │
│    Bit 0:        26-28μs HIGH                                             │
│    Bit 1:        ~70μs HIGH                                               │
│                                                                            │
│  Sampling:       2000ms minimum between reads (HARD LIMIT)                │
│  Duration:       ~5-6 milliseconds total                                  │
│                                                                            │
└─────────────────────────────────────────────────────────────────────────────┘

┌─ 40-BIT DATA FORMAT ──────────────────────────────────────────────────────┐
│                                                                            │
│  ┌─────────┬─────────┬──────────┬──────────┬──────────┐                  │
│  │ Byte 0  │ Byte 1  │ Byte 2   │ Byte 3   │ Byte 4   │                  │
│  │ RH High │ RH Low  │ Tmp High │ Tmp Low  │ Checksum │                  │
│  └─────────┴─────────┴──────────┴──────────┴──────────┘                  │
│                                                                            │
│  RH  = ((B0 << 8) | B1) / 10.0          → 0.0 to 100.0%                │
│  Tmp = (int16_t)((B2 << 8) | B3) / 10.0 → -40 to +80°C                 │
│  CHK = (B0 + B1 + B2 + B3) & 0xFF       → Validate                      │
│                                                                            │
│  CRITICAL: Use int16_t for temperature (two's complement)                │
│                                                                            │
└─────────────────────────────────────────────────────────────────────────────┘

│    // Timing code here (pull low, wait, read bits)                       │

┌─ CYCLE COUNTER TIMING (CPU @ 240MHz) ─────────────────────────────────────┐
│                                                                            │
│  #include "xtensa/hal.h"                                                  │
│  #define CPU_FREQ_MHZ 240                                                │
│  #define CYCLES_PER_US (CPU_FREQ_MHZ)                                   │
│                                                                            │
│  uint32_t start = xthal_get_ccount();  // Get cycle counter              │
│  // ... wait for something ...                                            │
│  uint32_t elapsed_cycles = xthal_get_ccount() - start;                  │
│  uint32_t elapsed_us = elapsed_cycles / CYCLES_PER_US;  // Convert      │
│                                                                            │
│  Note: 1 cycle = 4.17ns, xthal_get_ccount() overhead ≈ 50 cycles        │
│                                                                            │
└─────────────────────────────────────────────────────────────────────────────┘

┌─ BIT VALUE DETERMINATION ─────────────────────────────────────────────────┐
│                                                                            │
│  Conservative thresholds (account for WiFi jitter):                       │
│                                                                            │
│    Bit 0 range:  18-36μs  (nominal: 22-30μs)                            │
│    Bit 1 range:  60-85μs  (nominal: 68-75μs)                            │
│    Separation:   20+μs                                                    │
│                                                                            │
│  // Determine bit value                                                    │
│  if (pulse_us >= 60) {                                                    │
│    bit = 1;  // HIGH pulse = 1                                           │
│  } else if (pulse_us <= 36) {                                             │
│    bit = 0;  // LOW pulse = 0                                            │
│  } else {                                                                  │
│    return ERROR;  // Timing error                                        │
│  }                                                                         │
│                                                                            │
└─────────────────────────────────────────────────────────────────────────────┘

┌─ CHECKSUM VALIDATION ─────────────────────────────────────────────────────┐
│                                                                            │
│  // Calculate expected checksum                                            │
│  uint8_t expected = (data[0] + data[1] + data[2] + data[3]) & 0xFF;     │
│                                                                            │
│  // Validate                                                               │
│  if (expected != data[4]) {                                               │
│    // Error: data corrupted                                              │
│    return ESP_ERR_INVALID_CRC;                                            │
│  }                                                                         │
│                                                                            │
│  Note: Checksum uses only last 8 bits (overflow ignored)                 │
│                                                                            │
└─────────────────────────────────────────────────────────────────────────────┘

┌─ SAMPLING INTERVAL ENFORCEMENT ───────────────────────────────────────────┐
│                                                                            │
│  #define MIN_INTERVAL_MS 2000                                            │
│                                                                            │
│  static uint32_t last_read_ms = 0;                                        │
│                                                                            │
│  uint32_t now = esp_timer_get_time() / 1000;  // Convert to ms          │
│  if ((now - last_read_ms) < MIN_INTERVAL_MS) {                           │
│    return ESP_ERR_INVALID_STATE;  // Too fast!                          │
│  }                                                                         │
│  last_read_ms = now;                                                      │
│                                                                            │
│  // Then call dht22_read()                                                │
│                                                                            │
└─────────────────────────────────────────────────────────────────────────────┘

┌─ TIMEOUT PROTECTION ──────────────────────────────────────────────────────┐
│                                                                            │
│  #define TIMEOUT_US 250000  // 250ms maximum                             │
│  #define CYCLES_PER_US 240                                               │
│                                                                            │
│  uint32_t start = xthal_get_ccount();                                    │
│  uint32_t max_cycles = TIMEOUT_US * CYCLES_PER_US;                      │
│                                                                            │
│  while (gpio_get_level(pin) == EXPECTED) {                               │
│    if ((xthal_get_ccount() - start) > max_cycles) {                     │
│      return ESP_ERR_TIMEOUT;  // Safety exit                            │
│    }                                                                       │
│  }                                                                         │
│                                                                            │
│  Note: Never use iteration count for timeout (platform dependent)        │
│        Always use cycle counter for reliable timing                      │
│                                                                            │
└─────────────────────────────────────────────────────────────────────────────┘
