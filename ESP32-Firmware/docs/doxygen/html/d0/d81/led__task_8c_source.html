71static void internal_update_leds_from_latest_readings(led_task_context_t* ctx)
72{
73 for (uint8_t i = 0; i < NUM_HCSR04; i++) {
74 sensor_data_t sensor_data;
75 if (shared_data_get_sensor_reading(i, &sensor_data)) {
76 uint32_t age_ms = shared_data_get_uptime_ms() - sensor_data.timestamp_ms;
77
78 if (age_ms > (TASK_INTERVAL_SENSORS * 5)) {
79 ESP_LOGW(s_TAG, "Sensor %d data too old (%lums), turning LED off", i, age_ms);
80 led_controller_turn_off(&ctx->led_controller, (led_index_t)i);
81 } else {
82 internal_update_led_from_sensor_data(ctx, &sensor_data);
83 }
84 } else {
85 led_controller_turn_off(&ctx->led_controller, (led_index_t)i);

90static void internal_led_task_loop(void* parameter)
91{
92 led_task_context_t* ctx = (led_task_context_t*)parameter;
93
94 ESP_LOGI(s_TAG, "LED display task started (interval: %dms)", TASK_INTERVAL_LEDS);
95
96 TickType_t last_wake_time = xTaskGetTickCount();
97
98 while (ctx->task_running) {
99 if (ctx->leds_initialized) {
100 internal_process_sensor_data_queue(ctx);
101
102 if (uxQueueMessagesWaiting(shared_data_get_context()->sensor_data_queue) == 0) {
103 internal_update_leds_from_latest_readings(ctx);
104 }
105 } else {
106 ESP_LOGW(s_TAG, "LED controller not initialized, skipping update");
107 }
108
109 vTaskDelayUntil(&last_wake_time, pdMS_TO_TICKS(TASK_INTERVAL_LEDS));
110 }
111
112 ESP_LOGI(s_TAG, "LED display task ending");
113 ctx->task_handle = NULL;
114 vTaskDelete(NULL);
115}

144
145 s_task_context.leds_initialized = true;

147
148 BaseType_t task_ret = xTaskCreate(internal_led_task_loop,
149 "led_task",
150 TASK_STACK_SIZE_LEDS,
151 &s_task_context,
152 TASK_PRIORITY_LEDS,
153 &s_task_context.task_handle);
154

157 s_task_context.task_running = false;
158 led_controller_deinit(&s_task_context.led_controller);

163
164 shared_context_t* shared_ctx = shared_data_get_context();
165 if (shared_ctx != NULL) {
166 shared_ctx->leds_task_handle = s_task_context.task_handle;
167 }

169 ESP_LOGI(s_TAG, "LED task started successfully");
170 return ESP_OK;

173esp_err_t led_task_stop(void)
174{
175 if (!s_g_context_initialized || !s_task_context.task_running) {
176 ESP_LOGI(s_TAG, "LED task not running");

182 s_task_context.task_running = false;

187 ESP_LOGW(s_TAG, "Task did not stop gracefully within timeout");
188 }
189 }
190
191 if (s_task_context.leds_initialized) {
192 led_controller_deinit(&s_task_context.led_controller);
193 s_task_context.leds_initialized = false;
194 }
195
196 shared_context_t* shared_ctx = shared_data_get_context();
197 if (shared_ctx != NULL) {
198 shared_ctx->leds_task_handle = NULL;
199 }
200
201 s_g_context_initialized = false;
202
203 ESP_LOGI(s_TAG, "LED task stopped successfully (failures: %lu, successes: %lu)",
204 s_task_context.update_failures, s_task_context.successful_updates);
205
206 memset(&s_task_context, 0, sizeof(led_task_context_t));
