76static float internal_scale_pwm_for_color(float original_percent, float min_pwm)
77{
78 if (original_percent <= 0.1f) {
79 return 0.0f;
80 }
81 return min_pwm + (original_percent / 100.0f) * (MAX_PWM_ALL - min_pwm);

100static rgb_color_t internal_distance_to_rgb(float distance_cm)
101{
102 rgb_color_t color = {0.0f, 0.0f, 0.0f};
103
104 if (distance_cm > DISTANCE_VERY_FAR) {
105 return color;
106 }
107
108 float brightness_factor = 1.0f;
109
110 if (distance_cm < DISTANCE_VERY_CLOSE) {
111 color.red = 100.0f;
112 brightness_factor = 1.0f;
113 } else if (distance_cm < DISTANCE_CLOSE) {
114 float ratio = (distance_cm - DISTANCE_VERY_CLOSE) / (DISTANCE_CLOSE - DISTANCE_VERY_CLOSE);
115 color.red = 100.0f;
116 color.green = ratio * 100.0f;
117 brightness_factor = 1.0f - (ratio * 0.3f);
118 } else if (distance_cm < DISTANCE_MEDIUM) {
119 float ratio = (distance_cm - DISTANCE_CLOSE) / (DISTANCE_MEDIUM - DISTANCE_CLOSE);
120 color.red = (1.0f - ratio) * 100.0f;
121 color.green = 100.0f;
122 brightness_factor = 0.7f - (ratio * 0.2f);
123 } else if (distance_cm < DISTANCE_FAR) {
124 float ratio = (distance_cm - DISTANCE_MEDIUM) / (DISTANCE_FAR - DISTANCE_MEDIUM);
125 color.green = (1.0f - ratio) * 100.0f;
126 color.blue = ratio * 100.0f;
127 brightness_factor = 0.5f - (ratio * 0.2f);
128 } else if (distance_cm <= DISTANCE_VERY_FAR) {
129 color.blue = 100.0f;
130 float ratio = (distance_cm - DISTANCE_FAR) / (DISTANCE_VERY_FAR - DISTANCE_FAR);
131 brightness_factor = 0.3f - (ratio * 0.2f);
132 }
133
134 color.red *= brightness_factor;
135 color.green *= brightness_factor;
136 color.blue *= brightness_factor;
137
138 return color;
139}
