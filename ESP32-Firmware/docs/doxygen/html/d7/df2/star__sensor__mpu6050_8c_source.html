77esp_err_t star_sensor_mpu6050_init(mpu6050_handle_t* handle,
78 star_bus_manager_t* manager,
79 const char* bus_name,
80 star_error_interface_t* error_iface,
81 const mpu6050_config_t* config)
82{
83 if (handle == NULL || manager == NULL || bus_name == NULL || config == NULL) {
84 ESP_LOGE(s_TAG, "NULL parameter in init");
85 return ESP_ERR_INVALID_ARG;
86 }
87
88 memset(handle, 0, sizeof(mpu6050_handle_t));
89 handle->manager = manager;
90 handle->bus_name = bus_name;
91 handle->i2c_addr = config->i2c_addr;
92 memcpy(&handle->config, config, sizeof(mpu6050_config_t));
93
94 internal_get_accel_sensitivity(config->accel_range, &handle->accel_sensitivity);
95 internal_get_gyro_sensitivity(config->gyro_range, &handle->gyro_sensitivity);
96
97 /* Create mutex first */
98 handle->mutex = xSemaphoreCreateMutex();
99 if (handle->mutex == NULL) {
100 ESP_LOGE(s_TAG, "Failed to create mutex");
101 return ESP_ERR_NO_MEM;
102 }
103
104 // Handle error interface injection
105 if (error_iface == NULL) {
106 handle->error_iface = star_error_interface_create_default();
107 if (handle->error_iface == NULL) {
108 ESP_LOGE(s_TAG, "Failed to create default error interface");
109 vSemaphoreDelete(handle->mutex);
110 handle->mutex = NULL;
111 return ESP_ERR_NO_MEM;
112 }
113 handle->owns_error_handler = true;
114 ESP_LOGI(s_TAG, "Created default error handler internally");
115 } else {
116 handle->error_iface = error_iface;
117 handle->owns_error_handler = false;
118 ESP_LOGI(s_TAG, "Using injected error interface");
119 }
120
121 ESP_LOGI(s_TAG, "Initializing MPU6050 on bus '%s' at address 0x%02X", bus_name, config->i2c_addr);
122
123 uint8_t who_am_i;
124 const esp_err_t ret =
125 internal_mpu6050_read_reg(manager, bus_name, k_mpu6050_reg_who_am_i, &who_am_i);
126 if (ret != ESP_OK || who_am_i != MPU6050_WHO_AM_I_VAL) {
127 ESP_LOGE(s_TAG, "Failed to detect MPU6050 (who_am_i=0x%02X)", who_am_i);
128 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
129 handle->error_iface = NULL;
130 vSemaphoreDelete(handle->mutex);
131 handle->mutex = NULL;
132 return ESP_ERR_NOT_FOUND;
133 }
134
135 ret = star_sensor_mpu6050_reset(handle);
136 if (ret != ESP_OK) {
137 ESP_LOGE(s_TAG, "Failed to reset: %s", esp_err_to_name(ret));
138 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
139 handle->error_iface = NULL;
140 vSemaphoreDelete(handle->mutex);
141 handle->mutex = NULL;
142 return ret;
143 }
144
145 vTaskDelay(pdMS_TO_TICKS(100));
146
147 ret = internal_mpu6050_write_reg(manager, bus_name, k_mpu6050_reg_pwr_mgmt_1, 0x01);
148 if (ret != ESP_OK) {
149 ESP_LOGE(s_TAG, "Failed to set clock source: %s", esp_err_to_name(ret));
150 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
151 handle->error_iface = NULL;
152 vSemaphoreDelete(handle->mutex);
153 handle->mutex = NULL;
154 return ret;
155 }
156
157 const uint8_t gyro_config = (config->gyro_range & 0x03) << 3;
158 ret = internal_mpu6050_write_reg(manager, bus_name, k_mpu6050_reg_gyro_config, gyro_config);
159 if (ret != ESP_OK) {
160 ESP_LOGE(s_TAG, "Failed to configure gyro: %s", esp_err_to_name(ret));
161 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
162 handle->error_iface = NULL;
163 vSemaphoreDelete(handle->mutex);
164 handle->mutex = NULL;
165 return ret;
166 }
167
168 const uint8_t accel_config = (config->accel_range & 0x03) << 3;
169 ret = internal_mpu6050_write_reg(manager, bus_name, k_mpu6050_reg_accel_config, accel_config);
170 if (ret != ESP_OK) {
171 ESP_LOGE(s_TAG, "Failed to configure accel: %s", esp_err_to_name(ret));
172 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
173 handle->error_iface = NULL;
174 vSemaphoreDelete(handle->mutex);
175 handle->mutex = NULL;
176 return ret;
177 }
178
179 ret = internal_mpu6050_write_reg(manager, bus_name, k_mpu6050_reg_config, config->dlpf & 0x07);
180 if (ret != ESP_OK) {
181 ESP_LOGE(s_TAG, "Failed to configure DLPF: %s", esp_err_to_name(ret));
182 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
183 handle->error_iface = NULL;
184 vSemaphoreDelete(handle->mutex);
185 handle->mutex = NULL;
186 return ret;
187 }
188
189 ret = internal_mpu6050_write_reg(manager,
190 bus_name,
191 k_mpu6050_reg_smplrt_div,
192 config->sample_rate_div);
193 if (ret != ESP_OK) {
194 ESP_LOGE(s_TAG, "Failed to configure sample rate: %s", esp_err_to_name(ret));
195 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
196 handle->error_iface = NULL;
197 vSemaphoreDelete(handle->mutex);
198 handle->mutex = NULL;
199 return ret;
200 }
201
202 if (config->enable_fifo) {
203 ret = star_sensor_mpu6050_fifo_enable(handle, true);
204 if (ret != ESP_OK) {
205 ESP_LOGW(s_TAG, "Failed to enable FIFO: %s", esp_err_to_name(ret));
206 }
207 }
208
209 handle->initialized = true;
210 ESP_LOGI(s_TAG, "MPU6050 initialization successful (thread-safe with mutex)");
211 return ESP_OK;
212}

214esp_err_t star_sensor_mpu6050_deinit(mpu6050_handle_t* handle)
215{
216 if (handle == NULL || !handle->initialized) {
217 return ESP_ERR_INVALID_ARG;
218 }
219
220 // Acquire mutex before cleanup
221 const SemaphoreHandle_t mutex = handle->mutex;
222 if (mutex != NULL && xSemaphoreTake(mutex, pdMS_TO_TICKS(s_mpu6050_mutex_timeout_ms)) != pdTRUE) {
223 ESP_LOGW(s_TAG, "Failed to acquire mutex for deinit, continuing anyway");
224 }
225
226 // Put device to sleep (no separate mutex needed, we already hold it)
227 uint8_t pwr_mgmt;
228 if (internal_mpu6050_read_reg(handle->manager,
229 handle->bus_name,
230 k_mpu6050_reg_pwr_mgmt_1,
231 &pwr_mgmt) == ESP_OK) {
232 pwr_mgmt |= MPU6050_PWR1_SLEEP;
233 internal_mpu6050_write_reg(handle->manager,
234 handle->bus_name,
235 k_mpu6050_reg_pwr_mgmt_1,
236 pwr_mgmt);
237 }
238
239 // Clean up error handler if we own it
240 star_error_interface_cleanup(handle->error_iface, handle->owns_error_handler);
241 handle->error_iface = NULL;
242 if (handle->owns_error_handler) {
243 ESP_LOGI(s_TAG, "Destroyed internally-owned error handler");
244 }
245
246 // Clear handle fields
247 handle->initialized = false;
248 handle->mutex = NULL;
249
250 // Release and delete mutex
251 if (mutex != NULL) {
252 xSemaphoreGive(mutex);
253 vSemaphoreDelete(mutex);
254 }
255
256 return ESP_OK;
257}

363 accel->y_g = (float)raw.y / handle->accel_sensitivity;
364 accel->z_g = (float)raw.z / handle->accel_sensitivity;
365
366 return ESP_OK;

428esp_err_t star_sensor_mpu6050_read_all(const mpu6050_handle_t* handle,
429 mpu6050_accel_t* const accel,
430 mpu6050_gyro_t* const gyro,

432{

437 /* Guard against division by zero from corrupted sensitivity */
438 if ((accel != NULL && handle->accel_sensitivity <= 0.0f) ||
439 (gyro != NULL && handle->gyro_sensitivity <= 0.0f)) {
440 ESP_LOGE(s_TAG, "Invalid sensitivity values");
441 return ESP_ERR_INVALID_STATE;
442 }
443
444 // Acquire mutex for thread-safe read
445 const SemaphoreHandle_t mutex = handle->mutex;
446 if (mutex == NULL || xSemaphoreTake(mutex, pdMS_TO_TICKS(s_mpu6050_mutex_timeout_ms)) != pdTRUE) {
447 ESP_LOGE(s_TAG, "Failed to acquire mutex for read_all");
448 return ESP_ERR_TIMEOUT;
449 }

453 xSemaphoreGive(mutex);
454 return ESP_ERR_INVALID_STATE;

456

458 uint8_t data[14];
459 esp_err_t ret = star_bus_i2c_read(handle->manager, handle->bus_name, data, 14, reg, NULL);
460 if (ret != ESP_OK) {
461 xSemaphoreGive(mutex);
462 return ret;
463 }
464
465 if (accel != NULL) {
466 const int16_t ax = (int16_t)((data[0] << 8) | data[1]);

468 const int16_t az = (int16_t)((data[4] << 8) | data[5]);
469 accel->x_g = (float)ax / handle->accel_sensitivity;
470 accel->y_g = (float)ay / handle->accel_sensitivity;
471 accel->z_g = (float)az / handle->accel_sensitivity;
472 }
473
474 if (temp_c != NULL) {
475 const int16_t temp_raw = (int16_t)((data[6] << 8) | data[7]);
476 *temp_c = (float)temp_raw / MPU6050_TEMP_SENSITIVITY + MPU6050_TEMP_OFFSET;
477 }
478
479 if (gyro != NULL) {
480 const int16_t gx = (int16_t)((data[8] << 8) | data[9]);
481 const int16_t gy = (int16_t)((data[10] << 8) | data[11]);

484 gyro->y_dps = (float)gy / handle->gyro_sensitivity;
485 gyro->z_dps = (float)gz / handle->gyro_sensitivity;
486 }

488 xSemaphoreGive(mutex);

490}

603 xSemaphoreGive(mutex);
604 return ESP_ERR_INVALID_STATE;
605 }
606
607 const uint8_t reg = k_mpu6050_reg_fifo_count_h;
608 uint8_t data[2];
609 esp_err_t ret = star_bus_i2c_read(handle->manager, handle->bus_name, data, 2, reg, NULL);
610 if (ret != ESP_OK) {
611 xSemaphoreGive(mutex);
612 return ret;
613 }
614
615 *count = ((uint16_t)data[0] << 8) | data[1];
616
617 if (*count > MPU6050_FIFO_SIZE) {
618 ESP_LOGW(s_TAG, "FIFO overflow detected: count=%d, auto-resetting FIFO", *count);
619 /* Auto-reset FIFO on overflow to recover from corrupted state */
620 uint8_t user_ctrl;
621 ret = internal_mpu6050_read_reg(handle->manager,
622 handle->bus_name,
623 k_mpu6050_reg_user_ctrl,
624 &user_ctrl);
625 if (ret == ESP_OK) {
626 user_ctrl |= MPU6050_USERCTRL_FIFO_RESET;
627 internal_mpu6050_write_reg(handle->manager,
628 handle->bus_name,
629 k_mpu6050_reg_user_ctrl,
630 user_ctrl);
631 }
632 *count = 0;
633 xSemaphoreGive(mutex);
634 return ESP_ERR_INVALID_STATE;
635 }
636
637 xSemaphoreGive(mutex);
638 return ESP_OK;
639}

641esp_err_t star_sensor_mpu6050_fifo_read(const mpu6050_handle_t* handle,
642 uint8_t* const data,
643 const uint16_t len)
644{
645 if (handle == NULL || data == NULL || !handle->initialized) {
646 return ESP_ERR_INVALID_ARG;
647 }
648
649 if (len > MPU6050_FIFO_SIZE) {
650 ESP_LOGE(s_TAG, "Read length %d exceeds FIFO size %d", len, MPU6050_FIFO_SIZE);
651 return ESP_ERR_INVALID_ARG;
652 }
653
654 // Acquire mutex for thread-safe FIFO read
655 const SemaphoreHandle_t mutex = handle->mutex;
656 if (mutex == NULL || xSemaphoreTake(mutex, pdMS_TO_TICKS(s_mpu6050_mutex_timeout_ms)) != pdTRUE) {
657 ESP_LOGE(s_TAG, "Failed to acquire mutex for fifo_read");
658 return ESP_ERR_TIMEOUT;
659 }
660
661 // Re-check after mutex acquisition
662 if (!handle->initialized || handle->mutex == NULL) {
663 xSemaphoreGive(mutex);
664 return ESP_ERR_INVALID_STATE;
665 }
666
667 const uint8_t reg = k_mpu6050_reg_fifo_r_w;
668 const esp_err_t ret = star_bus_i2c_read(handle->manager, handle->bus_name, data, len, reg, NULL);
669
670 xSemaphoreGive(mutex);
671 return ret;
672}
