20esp_err_t error_handler_init(error_handler_t* handler,
21 uint32_t max_retries,
22 uint32_t base_retry_delay,
23 uint32_t max_retry_delay,
24 esp_err_t (*reset_fn)(void* context),
25 void* reset_context)
26/* clang-format on */
27{
28 if (handler == NULL) {
29 ESP_LOGE(TAG, "Init Error: Handler pointer is NULL");
30 return ESP_ERR_INVALID_ARG;
31 }
32
33 /* Check if already initialized to prevent mutex leak */
34 if (handler->mutex != NULL) {
35 ESP_LOGE(TAG, "Init Error: Handler already initialized. Call deinit first.");
36 return ESP_ERR_INVALID_STATE;
37 }
38
39 /* Initialize struct members first */
40 handler->max_retries = max_retries;
41 handler->base_retry_delay = base_retry_delay > 0 ? base_retry_delay : 100;
42 handler->max_retry_delay =
43 max_retry_delay > handler->base_retry_delay ? max_retry_delay : handler->base_retry_delay;
44 handler->current_retry_delay = handler->base_retry_delay;
45 handler->reset_fn = reset_fn;
46 handler->reset_context = reset_context;
47 handler->last_error = ESP_OK;
48 handler->in_error_state = false;
49 handler->error_count = 0;
50 handler->current_retry = 0;
51
52 /* Create mutex last after other fields are set */
53 handler->mutex = xSemaphoreCreateMutex();
54 if (handler->mutex == NULL) {
55 ESP_LOGE(TAG, "Mutex Error: Failed to create mutex during initialization");
56 return ESP_ERR_NO_MEM;
57 }
58
59 return ESP_OK;
60}

62esp_err_t error_handler_deinit(error_handler_t* handler)
63{
64 if (handler == NULL) {
65 return ESP_ERR_INVALID_ARG;
66 }
67
68 SemaphoreHandle_t mutex_to_delete = handler->mutex;
69 if (mutex_to_delete == NULL) {
70 return ESP_ERR_INVALID_STATE;
71 }
72
73 /* Acquire mutex to ensure no other operation is in progress */
74 if (xSemaphoreTake(mutex_to_delete, pdMS_TO_TICKS(s_error_handler_mutex_timeout_ms)) != pdTRUE) {
75 /* If we can't acquire mutex within timeout, do NOT proceed - another thread is using it */
76 ESP_LOGE(TAG, "Deinit: Could not acquire mutex within timeout, aborting deinit");
77 return ESP_ERR_TIMEOUT;
78 }
79
80 /* Set to NULL while holding mutex to prevent new operations from starting */
81 handler->mutex = NULL;
82
83 /* Clear reset function pointers to prevent use-after-free */
84 handler->reset_fn = NULL;
85 handler->reset_context = NULL;
86
87 /* Give back the mutex before deleting (required by FreeRTOS) */
88 xSemaphoreGive(mutex_to_delete);
89
90 /* Now safe to delete */
91 vSemaphoreDelete(mutex_to_delete);
92
93 return ESP_OK;
94}

97esp_err_t error_handler_record_error(error_handler_t* handler,
98 esp_err_t error,
99 const char* description,
100 const char* file,
101 int32_t line,
102 const char* func)
103/* clang-format on */
104{
105 if (handler == NULL) {
106 return error;
107 }
108
109 const char* desc = description ? description : "No description";
110 const char* filename = file ? file : "Unknown file";
111 const char* funcname = func ? func : "Unknown function";
112
113 /* Atomically capture mutex pointer to prevent TOCTOU race */
114 SemaphoreHandle_t mutex = handler->mutex;
115 if (mutex == NULL) {
116 ESP_LOGE(TAG,
117 "Mutex Error: Mutex not initialized for error handler used in %s:%d",
118 filename,
119 line);
120 return ESP_ERR_INVALID_STATE;
121 }
122
123 if (xSemaphoreTake(mutex, pdMS_TO_TICKS(s_error_handler_mutex_timeout_ms)) != pdTRUE) {
124 ESP_LOGE(TAG, "Mutex Timeout: Failed to acquire mutex in %s:%d", filename, line);
125 return ESP_ERR_TIMEOUT;
126 }
127
128 /* Re-check mutex validity after acquisition (handler could have been deinit'd) */
129 if (handler->mutex == NULL) {
130 xSemaphoreGive(mutex);
131 ESP_LOGE(TAG, "Handler Error: Handler was deinitialized while waiting for mutex");
132 return ESP_ERR_INVALID_STATE;
133 }
134
135 /* Only increment retry count if we haven't reached the max yet */
136 uint32_t retry_count_to_log = handler->current_retry;
137 if (!handler->in_error_state) {
138 handler->in_error_state = true;
139 handler->current_retry = 0;
140 handler->current_retry_delay = handler->base_retry_delay;
141 retry_count_to_log = 0;
142 } else if (handler->current_retry < handler->max_retries) {
143 handler->current_retry++;
144 retry_count_to_log = handler->current_retry;
145 uint64_t new_delay_64 = (uint64_t)handler->current_retry_delay * 2;
146 if (new_delay_64 > handler->max_retry_delay) {
147 handler->current_retry_delay = handler->max_retry_delay;
148 } else if (new_delay_64 < handler->base_retry_delay) {
149 handler->current_retry_delay = handler->base_retry_delay;
150 } else {
151 handler->current_retry_delay = (uint32_t)new_delay_64;
152 }
153 ESP_LOGI(TAG, "Backoff: Next retry delay: %" PRIu32 " ms", handler->current_retry_delay);
154 } else {
155 retry_count_to_log = handler->max_retries;
156 handler->current_retry_delay = handler->max_retry_delay;
157 ESP_LOGI(TAG, "Backoff: Next retry delay: %" PRIu32 " ms", handler->current_retry_delay);
158 }
159
160 char log_buffer[256];
161 snprintf(log_buffer,
162 sizeof(log_buffer),
163 "Desc: %s | Code: %d (%s) | Loc: %s:%" PRId32 " (%s) | Retry: %" PRIu32 "/%" PRIu32 "",
164 desc,
165 error,
166 esp_err_to_name(error),
167 filename,
168 line,
169 funcname,
170 retry_count_to_log,
171 handler->max_retries);
172 log_buffer[sizeof(log_buffer) - 1] = '\0';
173
174 ESP_LOGE(TAG, "Error Recorded: %s", log_buffer);
175
176 handler->error_count++;
177 handler->last_error = error;
178
179 esp_err_t recovery_result = ESP_FAIL;
180
181 if (handler->current_retry >= handler->max_retries) {
182 ESP_LOGE(TAG,
183 "Max Retries: Max retries (%" PRIu32 ") exceeded for error %d (%s).",
184 handler->max_retries,
185 error,
186 esp_err_to_name(error));
187
188 /* Capture reset function and context while holding mutex */
189 esp_err_t (*reset_fn_local)(void* context) = handler->reset_fn;
190 void* reset_context_local = handler->reset_context;
191
192 if (reset_fn_local != NULL) {
193 ESP_LOGI(TAG, "Reset Attempt: Attempting reset function after max retries...");
194
195 /* Release mutex before calling potentially blocking reset function */
196 xSemaphoreGive(mutex);
197
198 /* Call reset with captured local copies */
199 recovery_result = reset_fn_local(reset_context_local);
200
201 /* Re-acquire mutex - use the original captured mutex pointer */
202 if (xSemaphoreTake(mutex, pdMS_TO_TICKS(s_error_handler_mutex_timeout_ms)) != pdTRUE) {
203 ESP_LOGE(TAG, "Mutex Error: Failed to re-acquire mutex after reset attempt!");
204 return ESP_ERR_TIMEOUT;
205 }
206
207 /* Verify handler wasn't deinitialized during reset */
208 if (handler->mutex == NULL) {
209 xSemaphoreGive(mutex);
210 ESP_LOGE(TAG, "Handler Error: Handler was deinitialized during reset");
211 return ESP_ERR_INVALID_STATE;
212 }
213
214 if (recovery_result == ESP_OK) {
215 ESP_LOGI(TAG, "Reset Success: Reset function succeeded. Resetting error state.");
216 handler->error_count = 0;
217 handler->current_retry = 0;
218 handler->current_retry_delay = handler->base_retry_delay;
219 handler->last_error = ESP_OK;
220 handler->in_error_state = false;
221 xSemaphoreGive(mutex);
222 return ESP_OK;
223 } else {
224 ESP_LOGE(TAG,
225 "Reset Failed: Reset function failed with code: %d (%s)",
226 recovery_result,
227 esp_err_to_name(recovery_result));
228 }
229 } else {
230 ESP_LOGW(TAG, "No Reset: Max retries exceeded and no reset function provided.");
231 }
232 xSemaphoreGive(mutex);
233 return error;
234 } else {
235 xSemaphoreGive(mutex);
236 return error;
237 }

240bool error_handler_can_retry(error_handler_t* handler)
241{
242 bool can_retry = false;
243 if (handler == NULL) {
244 return false;
245 }
246
247 /* Atomically capture mutex pointer */
248 SemaphoreHandle_t mutex = handler->mutex;
249 if (mutex == NULL) {
250 ESP_LOGE(TAG, "Mutex Error: Mutex not initialized for error handler (can_retry)");
251 return false;
252 }
253
254 if (xSemaphoreTake(mutex, pdMS_TO_TICKS(s_error_handler_mutex_timeout_ms)) == pdTRUE) {
255 /* Re-check mutex validity after acquisition */
256 if (handler->mutex == NULL) {
257 xSemaphoreGive(mutex);
258 return false;
259 }
260 can_retry = (handler->in_error_state && (handler->current_retry < handler->max_retries));
261 xSemaphoreGive(mutex);
262 } else {
263 ESP_LOGE(TAG, "Mutex Timeout: Failed to acquire mutex for can_retry check");
264 can_retry = false;
265 }
266 return can_retry;
267}

269esp_err_t error_handler_reset_state(error_handler_t* handler)
270{
271 if (handler == NULL) {
272 return ESP_ERR_INVALID_ARG;
273 }
274
275 /* Atomically capture mutex pointer */
276 SemaphoreHandle_t mutex = handler->mutex;
277 if (mutex == NULL) {
278 ESP_LOGE(TAG, "Mutex Error: Mutex not initialized for error handler (reset_state)");
279 return ESP_ERR_INVALID_STATE;
280 }
281
282 if (xSemaphoreTake(mutex, pdMS_TO_TICKS(s_error_handler_mutex_timeout_ms)) == pdTRUE) {
283 /* Re-check mutex validity after acquisition */
284 if (handler->mutex == NULL) {
285 xSemaphoreGive(mutex);
286 return ESP_ERR_INVALID_STATE;
287 }
288 if (handler->in_error_state) {
289 ESP_LOGI(TAG, "State Reset: Resetting error handler state.");
290 }
291 handler->error_count = 0;
292 handler->current_retry = 0;
293 handler->current_retry_delay = handler->base_retry_delay;
294 handler->last_error = ESP_OK;
295 handler->in_error_state = false;
296 xSemaphoreGive(mutex);
297 return ESP_OK;
298 } else {
299 ESP_LOGE(TAG, "Mutex Timeout: Failed to acquire mutex for state reset");
300 return ESP_ERR_TIMEOUT;
301 }
302}

359error_handler_t* error_handler_create_custom(uint32_t max_retries,
360 uint32_t base_retry_delay,
361 uint32_t max_retry_delay,
362 esp_err_t (*reset_fn)(void* context),
363 void* reset_context)
364{
365 /* Allocate memory for error handler */
366 error_handler_t* handler = (error_handler_t*)malloc(sizeof(error_handler_t));
367 if (handler == NULL) {
368 ESP_LOGE(TAG, "Factory Error: Failed to allocate memory for error handler");
369 return NULL;
370 }
371
372 /* Zero-initialize to ensure clean state */
373 memset(handler, 0, sizeof(error_handler_t));
374
375 /* Initialize the error handler */
376 esp_err_t ret = error_handler_init(handler,
377 max_retries,
378 base_retry_delay,
379 max_retry_delay,
380 reset_fn,
381 reset_context);
382 if (ret != ESP_OK) {
383 ESP_LOGE(TAG, "Factory Error: Failed to initialize error handler: %s", esp_err_to_name(ret));
384 free(handler);
385 return NULL;
386 }
387
388 ESP_LOGI(
389 TAG,
390 "Factory Success: Created error handler (retries=%lu, base_delay=%lums, max_delay=%lums)",
391 (unsigned long)max_retries,
392 (unsigned long)base_retry_delay,
393 (unsigned long)max_retry_delay);
394
395 return handler;
396}
