369 acc_data,
370 6,
371 BNO055_REG_ACC_DATA_X_LSB,
372 NULL);
373 if (ret == ESP_OK) {
374 data->accelerometer.x = (int16_t)((acc_data[1] << 8) | acc_data[0]);
375 data->accelerometer.y = (int16_t)((acc_data[3] << 8) | acc_data[2]);
376 data->accelerometer.z = (int16_t)((acc_data[5] << 8) | acc_data[4]);
377 }

379 // Read BMP280 pressure and temperature
380 uint8_t bmp_data[6];
381 ret =
382 star_bus_i2c_read(handle->manager, handle->bus_name, bmp_data, 6, BMP280_REG_PRESS_MSB, NULL);
383 if (ret == ESP_OK) {
384 int32_t adc_P = (bmp_data[0] << 12) | (bmp_data[1] << 4) | (bmp_data[2] >> 4);
385 int32_t adc_T = (bmp_data[3] << 12) | (bmp_data[4] << 4) | (bmp_data[5] >> 4);
386
387 // Temperature compensation
388 int32_t var1 =
389 ((((adc_T >> 3) - ((int32_t)handle->dig_T1 << 1))) * ((int32_t)handle->dig_T2)) >> 11;
390 int32_t var2 =
391 (((((adc_T >> 4) - ((int32_t)handle->dig_T1)) * ((adc_T >> 4) - ((int32_t)handle->dig_T1))) >>
392 12) *
393 ((int32_t)handle->dig_T3)) >>
394 14;
395 int32_t t_fine = var1 + var2;
396 data->bmp_temp_c = (t_fine * 5 + 128) >> 8;
397 data->bmp_temp_c /= 100.0f;
398
399 // Pressure compensation
400 int64_t var1_64 = ((int64_t)t_fine) - 128000;
401 int64_t var2_64 = var1_64 * var1_64 * (int64_t)handle->dig_P6;
402 var2_64 = var2_64 + ((var1_64 * (int64_t)handle->dig_P5) << 17);
403 var2_64 = var2_64 + (((int64_t)handle->dig_P4) << 35);
404 var1_64 = ((var1_64 * var1_64 * (int64_t)handle->dig_P3) >> 8) +

406 var1_64 = (((((int64_t)1) << 47) + var1_64)) * ((int64_t)handle->dig_P1) >> 33;
407
408 if (var1_64 != 0) {
