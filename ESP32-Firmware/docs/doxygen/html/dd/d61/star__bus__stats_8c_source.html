44static esp_err_t internal_init_stats_mutex(void)
45{
46 /* Quick check without lock for common case */
47 if (g_stats_mutex != NULL) {
48 return ESP_OK;
49 }
50
51 /* Use spinlock for thread-safe lazy initialization */
52 portENTER_CRITICAL(&g_stats_init_spinlock);
53
54 /* Double-check after acquiring spinlock */
55 if (g_stats_mutex == NULL) {
56 g_stats_mutex = xSemaphoreCreateMutex();
57 if (g_stats_mutex == NULL) {
58 portEXIT_CRITICAL(&g_stats_init_spinlock);
59 ESP_LOGE(s_TAG, "Failed to create stats global mutex");
60 return ESP_ERR_NO_MEM;
61 }
62 }
63
64 portEXIT_CRITICAL(&g_stats_init_spinlock);
65 return ESP_OK;
66}

71static stats_state_t* internal_get_stats_state(const char* bus_name, bool create)
72{
73 /* Search for existing state */
74 for (uint8_t i = 0; i < g_num_stats_states; i++) {
75 if (strcmp(g_stats_states[i].bus_name, bus_name) == 0) {
76 return &g_stats_states[i];
77 }
78 }
79
80 /* Create new state if requested and space available */
81 if (create && g_num_stats_states < MAX_STATS_BUSES) {
82 stats_state_t* state = &g_stats_states[g_num_stats_states];
83 memset(state, 0, sizeof(stats_state_t));
84 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
85 state->bus_name[sizeof(state->bus_name) - 1] = '\0';
86 state->stats.min_operation_time_us = UINT32_MAX;
87 state->stats.stats_start_time_ms = pdTICKS_TO_MS(xTaskGetTickCount());
88 g_num_stats_states++;
89 return state;
90 }
91
92 return NULL;
93}

98static stats_state_t* get_stats_state(const char* bus_name, bool create)
99{
100 if (internal_init_stats_mutex() != ESP_OK) {
101 return NULL;
102 }
103
104 if (xSemaphoreTake(g_stats_mutex, portMAX_DELAY) != pdTRUE) {
105 ESP_LOGE(s_TAG, "Failed to take stats mutex");
106 return NULL;
107 }
108
109 stats_state_t* state = internal_get_stats_state(bus_name, create);
110
111 xSemaphoreGive(g_stats_mutex);
112 return state;
113}

117esp_err_t star_bus_stats_enable(star_bus_manager_t* manager,
118 const char* bus_name,
119 const star_stats_config_t* config)
120{
121 if (manager == NULL || bus_name == NULL) {
122 ESP_LOGE(s_TAG, "Invalid parameters");
123 return ESP_ERR_INVALID_ARG;
124 }
125
126 stats_state_t* state = get_stats_state(bus_name, true);
127 if (state == NULL) {
128 ESP_LOGE(s_TAG, "Failed to create statistics state for bus '%s'", bus_name);
129 return ESP_ERR_NO_MEM;
130 }
131
132 /* Apply configuration */
133 if (config != NULL) {
134 memcpy(&state->config, config, sizeof(star_stats_config_t));
135 } else {
136 /* Use defaults */
137 star_stats_config_t default_config = STAR_STATS_CONFIG_DEFAULT();
138 memcpy(&state->config, &default_config, sizeof(star_stats_config_t));
139 }
140
141 state->enabled = true;
142 state->stats.stats_start_time_ms = pdTICKS_TO_MS(xTaskGetTickCount());
143
144 ESP_LOGI(s_TAG, "Statistics enabled for bus '%s'", bus_name);
145
146 return ESP_OK;

267esp_err_t star_bus_stats_diff(const star_bus_manager_t* manager,
268 const char* bus_name,
269 const star_stats_snapshot_t* snapshot,
270 star_bus_stats_t* diff)
271{
272 if (manager == NULL || bus_name == NULL || snapshot == NULL || diff == NULL) {
273 return ESP_ERR_INVALID_ARG;
274 }
275
276 star_bus_stats_t current;
277 esp_err_t result = star_bus_stats_get(manager, bus_name, &current);
278 if (result != ESP_OK) {
279 return result;
280 }
281
282 /* Calculate differences */
283 diff->total_operations = current.total_operations - snapshot->stats.total_operations;
284 diff->read_operations = current.read_operations - snapshot->stats.read_operations;
285 diff->write_operations = current.write_operations - snapshot->stats.write_operations;
286 diff->failed_operations = current.failed_operations - snapshot->stats.failed_operations;
287 diff->bytes_read = current.bytes_read - snapshot->stats.bytes_read;
288 diff->bytes_written = current.bytes_written - snapshot->stats.bytes_written;
289
290 /* Timing stats - use current values (not differences) */
291 diff->min_operation_time_us = current.min_operation_time_us;
292 diff->max_operation_time_us = current.max_operation_time_us;
293 diff->avg_operation_time_us = current.avg_operation_time_us;
294
295 /* Error stats */
296 diff->timeout_errors = current.timeout_errors - snapshot->stats.timeout_errors;
297 diff->nack_errors = current.nack_errors - snapshot->stats.nack_errors;
298 diff->bus_errors = current.bus_errors - snapshot->stats.bus_errors;
299 diff->other_errors = current.other_errors - snapshot->stats.other_errors;
300
301 /* Utilization */
302 diff->total_bus_time_ms = current.total_bus_time_ms - snapshot->stats.total_bus_time_ms;
303 diff->idle_time_ms = current.idle_time_ms - snapshot->stats.idle_time_ms;
304
305 /* Last operation info */
306 diff->last_error = current.last_error;
307 diff->last_operation_time_us = current.last_operation_time_us;
308
309 /* Time period */
310 uint32_t current_time = pdTICKS_TO_MS(xTaskGetTickCount());
311 diff->stats_start_time_ms = current_time - snapshot->timestamp_ms;
312
313 return ESP_OK;
314}

316float star_bus_stats_error_rate(const star_bus_stats_t* stats)
317{

339uint32_t star_bus_stats_throughput(const star_bus_stats_t* stats)
340{
341 if (stats == NULL || stats->total_bus_time_ms == 0) {
342 return 0;
343 }
344
345 uint64_t total_bytes = stats->bytes_read + stats->bytes_written;
346 uint32_t time_seconds = stats->total_bus_time_ms / 1000;
347
348 if (time_seconds == 0) {
349 return 0;
350 }
351
352 return (uint32_t)(total_bytes / time_seconds);
353}
