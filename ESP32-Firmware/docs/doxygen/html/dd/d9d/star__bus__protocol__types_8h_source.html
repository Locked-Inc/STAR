86 /* --- Renamed members within spi_bus_config_t for internal use --- */
87 /* These correspond to the fields in the ESP-IDF spi_bus_config_t */
88 /* SPI pin naming follows OSHWA (Open Source Hardware Association) standard:
89 * - COPI (Controller Out, Peripheral In) - replaces MOSI (Master Out, Slave In)
90 * - CIPO (Controller In, Peripheral Out) - replaces MISO (Master In, Slave Out)
91 * This modern terminology removes master/slave language while maintaining clarity.
92 * The ESP-IDF still uses mosi_io_num/miso_io_num internally, which we map here.
93 */
94 gpio_num_t copi_io_num;
95 gpio_num_t cipo_io_num;
96 gpio_num_t sclk_io_num;
