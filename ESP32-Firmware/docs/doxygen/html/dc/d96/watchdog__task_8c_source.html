20static watchdog_health_level_t internal_assess_sensor_health(void)
21{
22 shared_context_t* shared_ctx = shared_data_get_context();
23 if (shared_ctx == NULL) {
24 return WATCHDOG_HEALTH_FAULT;
25 }
26
27 uint32_t total_sensor_failures = 0;
28 uint32_t total_sensor_reads = 0;
29
30 if (xSemaphoreTake(shared_ctx->health_mutex, pdMS_TO_TICKS(MUTEX_TIMEOUT_MS)) == pdTRUE) {
31 for (uint8_t i = 0; i < NUM_HCSR04; i++) {
32 total_sensor_failures += shared_ctx->health_data.sensor_read_failures[i];
33 }
34 total_sensor_failures += shared_ctx->health_data.temperature_read_failures;
35 total_sensor_reads = shared_ctx->health_data.total_sensor_reads +
36 shared_ctx->health_data.total_temperature_reads;
37
38 xSemaphoreGive(shared_ctx->health_mutex);
39 } else {
40 ESP_LOGE(s_TAG, "Failed to acquire health mutex for sensor assessment");
41 return WATCHDOG_HEALTH_FAULT;
42 }
43
44 if (total_sensor_reads == 0) {
45 ESP_LOGW(s_TAG, "No sensor readings available for health assessment");
46 return WATCHDOG_HEALTH_DEGRADED;
47 }
48
49 float failure_rate = (float)total_sensor_failures / total_sensor_reads;
50
51 if (failure_rate > SENSOR_FAILURE_RATE_CRITICAL) {
52 ESP_LOGE(s_TAG, "Critical sensor failure rate: %.1f%%", failure_rate * 100);
53 return WATCHDOG_HEALTH_CRITICAL;
54 } else if (failure_rate > SENSOR_FAILURE_RATE_DEGRADED) {
55 ESP_LOGW(s_TAG, "Elevated sensor failure rate: %.1f%%", failure_rate * 100);
56 return WATCHDOG_HEALTH_DEGRADED;
57 }
58
59 ESP_LOGI(s_TAG, "Sensor health good: %.1f%% failure rate", failure_rate * 100);
60 return WATCHDOG_HEALTH_GOOD;
61}

63static watchdog_health_level_t internal_assess_task_health(void)
64{
65 bool all_tasks_running = true;
66 uint32_t unresponsive_tasks = 0;
67
68 if (!sensor_task_is_running()) {
69 ESP_LOGW(s_TAG, "Sensor task not running");
70 all_tasks_running = false;
71 }
72
73 if (!dht22_task_is_running()) {
74 ESP_LOGW(s_TAG, "DHT22 task not running");
75 unresponsive_tasks++;
76 }
77
78 if (!led_task_is_running()) {
79 ESP_LOGW(s_TAG, "LED task not running");
80 unresponsive_tasks++;
81 }
82
83 if (!all_tasks_running) {
84 return WATCHDOG_HEALTH_CRITICAL;
85 } else if (unresponsive_tasks > 0) {
86 return WATCHDOG_HEALTH_DEGRADED;
87 }
88
89 return WATCHDOG_HEALTH_GOOD;
90}

92static watchdog_health_level_t internal_assess_system_state(void)
93{
94 system_state_t current_state = shared_data_get_system_state();
95
96 switch (current_state) {

98 return WATCHDOG_HEALTH_GOOD;
99 case SYSTEM_STATE_ERROR_RECOVERY:
100 return WATCHDOG_HEALTH_DEGRADED;
101 case SYSTEM_STATE_INITIALIZING:
102 return WATCHDOG_HEALTH_DEGRADED;

109static watchdog_health_level_t internal_assess_overall_health(void)
110{
111 watchdog_health_level_t sensor_health = internal_assess_sensor_health();
112 watchdog_health_level_t task_health = internal_assess_task_health();
113 watchdog_health_level_t state_health = internal_assess_system_state();
114
115 watchdog_health_level_t worst_health = WATCHDOG_HEALTH_GOOD;
116
117 if (sensor_health > worst_health) worst_health = sensor_health;
118 if (task_health > worst_health) worst_health = task_health;
119 if (state_health > worst_health) worst_health = state_health;
120
121 return worst_health;
122}

124static esp_err_t internal_take_recovery_action(watchdog_health_level_t health_level)
125{
126 esp_err_t ret = ESP_OK;
127 s_task_context.recovery_actions_taken++;
128
129 switch (health_level) {
130 case WATCHDOG_HEALTH_DEGRADED:
131 ESP_LOGW(s_TAG, "System degraded - logging detailed health status");
132 shared_data_set_system_state(SYSTEM_STATE_ERROR_RECOVERY);
133 break;
134
135 case WATCHDOG_HEALTH_CRITICAL:
136 ESP_LOGE(s_TAG, "System critical - attempting emergency LED shutdown");
137 shared_data_set_system_state(SYSTEM_STATE_ERROR_RECOVERY);
138
139 if (led_task_is_running()) {
140 led_task_emergency_off();
141 }
142 break;
143
144 case WATCHDOG_HEALTH_FAULT:
145 ESP_LOGE(s_TAG, "System fault - initiating system restart");
146 shared_data_set_system_state(SYSTEM_STATE_FAULT);
147
148 ESP_LOGE(s_TAG, "System restart in %d ms...", SYSTEM_RESTART_DELAY_MS);
149 vTaskDelay(pdMS_TO_TICKS(SYSTEM_RESTART_DELAY_MS));
150 esp_restart();
151 break;
152
153 case WATCHDOG_HEALTH_GOOD:
154 if (shared_data_get_system_state() == SYSTEM_STATE_ERROR_RECOVERY) {
155 ESP_LOGI(s_TAG, "System recovered - returning to normal operation");
156 shared_data_set_system_state(SYSTEM_STATE_RUNNING);
157 }
158 break;
159
160 default:
161 ret = ESP_ERR_INVALID_ARG;
162 break;
163 }
164
165 return ret;
166}

168static void internal_watchdog_task_loop(void* parameter)
169{
170 watchdog_task_context_t* ctx = (watchdog_task_context_t*)parameter;
171
172 ESP_LOGI(s_TAG, "System watchdog task started (interval: %dms)", TASK_INTERVAL_WATCHDOG);
173
174 TickType_t last_wake_time = xTaskGetTickCount();
175
176 while (ctx->task_running) {
177 watchdog_health_level_t current_health = internal_assess_overall_health();
178 ctx->last_health_level = current_health;
179 ctx->health_checks_performed++;
180
181 if (current_health != WATCHDOG_HEALTH_GOOD) {
182 ESP_LOGW(s_TAG, "Health check %lu: Level %d",
183 ctx->health_checks_performed, current_health);
184 internal_take_recovery_action(current_health);
185 } else {
186 ESP_LOGI(s_TAG, "Health check %lu: System healthy", ctx->health_checks_performed);
187 }
188
189 if (ctx->health_checks_performed % HEALTH_CHECK_LOG_INTERVAL == 0) {
190 uint32_t uptime = shared_data_get_uptime_ms();
191 ESP_LOGI(s_TAG, "Watchdog status - uptime: %lu ms, checks: %lu, recoveries: %lu",
192 uptime, ctx->health_checks_performed, ctx->recovery_actions_taken);
193 }
194
195 vTaskDelayUntil(&last_wake_time, pdMS_TO_TICKS(TASK_INTERVAL_WATCHDOG));
196 }
197
198 ESP_LOGI(s_TAG, "System watchdog task ending");
199 ctx->task_handle = NULL;
200 vTaskDelete(NULL);
201}

205esp_err_t watchdog_task_start(void)
206{
207 if (s_g_context_initialized && s_task_context.task_running) {
208 ESP_LOGI(s_TAG, "Watchdog task already running");
209 return ESP_OK;
210 }
211
212 memset(&s_task_context, 0, sizeof(watchdog_task_context_t));
213
214 s_task_context.task_running = true;
215 s_task_context.health_checks_performed = 0;
216 s_task_context.recovery_actions_taken = 0;
217 s_task_context.last_health_level = WATCHDOG_HEALTH_GOOD;
218
219 BaseType_t task_ret = xTaskCreate(internal_watchdog_task_loop,
220 "watchdog_task",
221 TASK_STACK_SIZE_WATCHDOG,
222 &s_task_context,
223 TASK_PRIORITY_WATCHDOG,
224 &s_task_context.task_handle);
225
226 if (task_ret != pdPASS) {
227 ESP_LOGE(s_TAG, "Failed to create watchdog task");
228 s_task_context.task_running = false;
229 return ESP_ERR_NO_MEM;
230 }
231
232 s_g_context_initialized = true;
233
234 shared_context_t* shared_ctx = shared_data_get_context();
235 if (shared_ctx != NULL) {
236 shared_ctx->watchdog_task_handle = s_task_context.task_handle;
237 }
238
239 ESP_LOGI(s_TAG, "System watchdog task started successfully");
240 return ESP_OK;
241}

243esp_err_t watchdog_task_stop(void)
244{
245 if (!s_g_context_initialized || !s_task_context.task_running) {
246 ESP_LOGI(s_TAG, "Watchdog task not running");
247 return ESP_OK;
248 }
249
250 ESP_LOGI(s_TAG, "Stopping watchdog task...");
251
252 s_task_context.task_running = false;
253
254 if (s_task_context.task_handle != NULL) {
255 uint32_t notification_value = 0;
256 if (xTaskNotifyWait(0, 0, &notification_value, pdMS_TO_TICKS(NOTIFICATION_TIMEOUT_MS)) != pdTRUE) {
257 ESP_LOGW(s_TAG, "Task did not stop gracefully within timeout");
258 }
259 }
260
261 shared_context_t* shared_ctx = shared_data_get_context();
262 if (shared_ctx != NULL) {
263 shared_ctx->watchdog_task_handle = NULL;
264 }
265
266 s_g_context_initialized = false;
267
268 ESP_LOGI(s_TAG, "Watchdog task stopped (checks: %lu, recoveries: %lu)",
269 s_task_context.health_checks_performed, s_task_context.recovery_actions_taken);
270
271 memset(&s_task_context, 0, sizeof(watchdog_task_context_t));
272
273 return ESP_OK;
274}
