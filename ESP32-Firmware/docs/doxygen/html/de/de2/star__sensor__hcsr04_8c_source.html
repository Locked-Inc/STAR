19static void IRAM_ATTR hcsr04_echo_isr_handler(void* arg)
20{
21 hcsr04_handle_t* handle = (hcsr04_handle_t*)arg;
22
23 if (gpio_get_level(handle->echo_pin)) {
24 // Rising edge - start timing
25 handle->echo_start_time = (uint32_t)esp_timer_get_time();
26 handle->measurement_complete = false;
27 } else {
28 // Falling edge - end timing
29 handle->echo_end_time = (uint32_t)esp_timer_get_time();
30 handle->measurement_complete = true;
31 }
32}

34esp_err_t star_sensor_hcsr04_init(hcsr04_handle_t* handle,
35 star_bus_manager_t* manager,
36 const char* bus_name,
37 star_error_interface_t* error_iface,
38 const hcsr04_config_t* config)
39{
40 if (handle == NULL || manager == NULL || bus_name == NULL || config == NULL) {
41 ESP_LOGE(s_TAG, "NULL parameter in init");
42 return ESP_ERR_INVALID_ARG;
43 }
44
45 memset(handle, 0, sizeof(hcsr04_handle_t));
46 handle->manager = manager;
47 handle->bus_name = bus_name;
48 handle->trigger_pin = config->trigger_pin;
49 handle->echo_pin = config->echo_pin;
50 handle->temperature_c = config->temperature_c;
51
52 // Create mutex first
53 handle->mutex = xSemaphoreCreateMutex();
54 if (handle->mutex == NULL) {
55 ESP_LOGE(s_TAG, "Failed to create mutex");
56 return ESP_ERR_NO_MEM;
57 }
58
59 // Handle error interface injection
60 if (error_iface == NULL) {
61 // Create default error handler internally
62 error_handler_t* default_handler = error_handler_create_default();
63 if (default_handler == NULL) {
64 ESP_LOGE(s_TAG, "Failed to create default error handler");
65 vSemaphoreDelete(handle->mutex);
66 handle->mutex = NULL;
67 return ESP_ERR_NO_MEM;
68 }
69
70 // Allocate and store the interface
71 handle->error_iface = (star_error_interface_t*)malloc(sizeof(star_error_interface_t));
72 if (handle->error_iface == NULL) {
73 ESP_LOGE(s_TAG, "Failed to allocate error interface");
74 error_handler_destroy_default(default_handler);
75 vSemaphoreDelete(handle->mutex);
76 handle->mutex = NULL;
77 return ESP_ERR_NO_MEM;
78 }
79
80 error_handler_get_interface(handle->error_iface, default_handler);
81 handle->owns_error_handler = true;
82 ESP_LOGI(s_TAG, "Created default error handler internally");
83 } else {
84 handle->error_iface = error_iface;
85 handle->owns_error_handler = false;
86 ESP_LOGI(s_TAG, "Using injected error interface");
87 }
88
89 ESP_LOGI(s_TAG,
90 "Initializing HC-SR04 (trigger=%d, echo=%d)",
91 config->trigger_pin,
92 config->echo_pin);
93
94 // Configure trigger pin as output through GPIO bus
95 esp_err_t ret = star_bus_gpio_configure(manager,
96 bus_name,
97 config->trigger_pin,
98 GPIO_MODE_OUTPUT,
99 GPIO_FLOATING);
100 if (ret != ESP_OK) {
101 ESP_LOGE(s_TAG, "Failed to configure trigger pin: %s", esp_err_to_name(ret));
102 if (handle->owns_error_handler && handle->error_iface != NULL) {
103 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
104 error_handler_destroy_default(handler_ptr);
105 free(handle->error_iface);
106 }
107 vSemaphoreDelete(handle->mutex);
108 handle->mutex = NULL;
109 return ret;
110 }
111
112 // Set trigger pin low initially
113 ret = star_bus_gpio_write(manager, bus_name, config->trigger_pin, 0);
114 if (ret != ESP_OK) {
115 ESP_LOGE(s_TAG, "Failed to set trigger pin low: %s", esp_err_to_name(ret));
116 if (handle->owns_error_handler && handle->error_iface != NULL) {
117 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
118 error_handler_destroy_default(handler_ptr);
119 free(handle->error_iface);
120 }
121 vSemaphoreDelete(handle->mutex);
122 handle->mutex = NULL;
123 return ret;
124 }
125
126 // Configure echo pin as input with interrupt through GPIO bus
127 ret =
128 star_bus_gpio_configure(manager, bus_name, config->echo_pin, GPIO_MODE_INPUT, GPIO_FLOATING);
129 if (ret != ESP_OK) {
130 ESP_LOGE(s_TAG, "Failed to configure echo pin: %s", esp_err_to_name(ret));
131 if (handle->owns_error_handler && handle->error_iface != NULL) {
132 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
133 error_handler_destroy_default(handler_ptr);
134 free(handle->error_iface);
135 }
136 vSemaphoreDelete(handle->mutex);
137 handle->mutex = NULL;
138 return ret;
139 }
140
141 // Set up interrupt handler through GPIO bus
142 ret = star_bus_gpio_set_interrupt(manager,
143 bus_name,
144 config->echo_pin,
145 GPIO_INTR_ANYEDGE,
146 hcsr04_echo_isr_handler,
147 (void*)handle);
148 if (ret != ESP_OK) {
149 ESP_LOGE(s_TAG, "Failed to set interrupt: %s", esp_err_to_name(ret));
150 if (handle->owns_error_handler && handle->error_iface != NULL) {
151 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
152 error_handler_destroy_default(handler_ptr);
153 free(handle->error_iface);
154 }
155 vSemaphoreDelete(handle->mutex);
156 handle->mutex = NULL;
157 return ret;
158 }
159
160 handle->initialized = true;
161 ESP_LOGI(s_TAG, "HC-SR04 initialization successful (thread-safe with mutex)");
162 return ESP_OK;
163}

165esp_err_t star_sensor_hcsr04_deinit(hcsr04_handle_t* handle)
166{
167 if (handle == NULL || !handle->initialized) {
168 return ESP_ERR_INVALID_ARG;
169 }
170
171 // Acquire mutex before cleanup
172 SemaphoreHandle_t mutex = handle->mutex;
173 if (mutex != NULL && xSemaphoreTake(mutex, pdMS_TO_TICKS(HCSR04_MUTEX_TIMEOUT_MS)) != pdTRUE) {
174 ESP_LOGW(s_TAG, "Failed to acquire mutex for deinit, continuing anyway");
175 }
176
177 // Remove ISR handler (direct GPIO call, not through bus, for cleanup)
178 gpio_isr_handler_remove(handle->echo_pin);
179
180 // Clean up error handler if we own it
181 if (handle->owns_error_handler && handle->error_iface != NULL) {
182 error_handler_t* error_handler = (error_handler_t*)handle->error_iface->ctx;
183 error_handler_destroy_default(error_handler);
184 free(handle->error_iface);
185 handle->error_iface = NULL;
186 ESP_LOGI(s_TAG, "Destroyed internally-owned error handler");
187 }
188
189 // Clear handle fields
190 handle->initialized = false;
191 handle->mutex = NULL;
192
193 // Release and delete mutex
194 if (mutex != NULL) {
195 xSemaphoreGive(mutex);
196 vSemaphoreDelete(mutex);
197 }
198
199 ESP_LOGI(s_TAG, "HC-SR04 deinitialized");
200 return ESP_OK;
201}

203esp_err_t star_sensor_hcsr04_trigger(hcsr04_handle_t* handle)
204{
205 if (handle == NULL || !handle->initialized) {
206 return ESP_ERR_INVALID_STATE;
207 }
208
209 // Acquire mutex
210 SemaphoreHandle_t mutex = handle->mutex;
211 if (mutex == NULL || xSemaphoreTake(mutex, pdMS_TO_TICKS(HCSR04_MUTEX_TIMEOUT_MS)) != pdTRUE) {
212 ESP_LOGE(s_TAG, "Failed to acquire mutex for trigger");
213 return ESP_ERR_TIMEOUT;
214 }
215
216 // Re-check after mutex acquisition
217 if (!handle->initialized || handle->mutex == NULL) {
218 xSemaphoreGive(mutex);
219 return ESP_ERR_INVALID_STATE;
220 }
221
222 handle->measurement_complete = false;
223 handle->timeout_occurred = false;
224 handle->echo_start_time = 0;
225 handle->echo_end_time = 0;
226
227 // Send 10us trigger pulse (use GPIO bus)
228 esp_err_t ret = star_bus_gpio_write(handle->manager, handle->bus_name, handle->trigger_pin, 0);
229 if (ret == ESP_OK) {
230 esp_rom_delay_us(2);
231 ret = star_bus_gpio_write(handle->manager, handle->bus_name, handle->trigger_pin, 1);
232 }
233 if (ret == ESP_OK) {
234 esp_rom_delay_us(10);
235 ret = star_bus_gpio_write(handle->manager, handle->bus_name, handle->trigger_pin, 0);
236 }
237
238 xSemaphoreGive(mutex);
239 return ret;
240}

252esp_err_t star_sensor_hcsr04_calculate_distance(uint32_t echo_time_us,
253 float temperature_c,
254 float* distance_cm)
255{
256 if (distance_cm == NULL) {
257 return ESP_ERR_INVALID_ARG;
258 }
259
260 if (echo_time_us < 116 || echo_time_us > HCSR04_TIMEOUT_US) {
261 return HCSR04_ERR_INVALID_PULSE;
262 }
263
264 // Speed of sound = 331.4 + 0.606 * T (m/s)
265 float speed_cm_us = (331.4f + 0.606f * temperature_c) / 10000.0f;
266
267 // Distance = (time * speed) / 2 (divide by 2 for round trip)
268 *distance_cm = (echo_time_us * speed_cm_us) / 2.0f;
269
270 if (*distance_cm < HCSR04_MIN_DISTANCE_CM) {
271 return HCSR04_ERR_OUT_OF_RANGE_MIN;
272 }
273 if (*distance_cm > HCSR04_MAX_DISTANCE_CM) {
274 return HCSR04_ERR_OUT_OF_RANGE_MAX;
275 }
276
277 return ESP_OK;
278}

280esp_err_t star_sensor_hcsr04_get_result(hcsr04_handle_t* handle, float* distance_cm)
281{
282 if (handle == NULL || distance_cm == NULL || !handle->initialized) {
283 return ESP_ERR_INVALID_ARG;
284 }
285
286 if (!handle->measurement_complete) {
287 return HCSR04_ERR_NOT_READY;
288 }
289
290 if (handle->echo_end_time <= handle->echo_start_time) {
291 ESP_LOGE(s_TAG,
292 "Invalid timing: start=%lu, end=%lu",
293 handle->echo_start_time,
294 handle->echo_end_time);
295 return HCSR04_ERR_INVALID_PULSE;
296 }
297
298 uint32_t echo_time = handle->echo_end_time - handle->echo_start_time;
299
300 esp_err_t ret =
301 star_sensor_hcsr04_calculate_distance(echo_time, handle->temperature_c, distance_cm);
302
303 if (ret == ESP_OK) {
304 ESP_LOGD(s_TAG, "Distance: %.2f cm (echo: %lu us)", *distance_cm, echo_time);
305 } else {
306 ESP_LOGW(s_TAG, "Measurement error: %d (echo: %lu us)", ret, echo_time);
307 }
308
309 return ret;
310}

312esp_err_t star_sensor_hcsr04_read_distance(hcsr04_handle_t* handle, float* distance_cm)
313{
314 if (handle == NULL || distance_cm == NULL || !handle->initialized) {
315 return ESP_ERR_INVALID_ARG;
316 }
317
318 esp_err_t ret = star_sensor_hcsr04_trigger(handle);
319 if (ret != ESP_OK) {
320 return ret;
321 }
322
323 // Wait for measurement with timeout
324 uint32_t start_time = (uint32_t)esp_timer_get_time();
325 uint32_t timeout_ms = 60; // Max time for 400cm measurement + margin
326
327 while (!handle->measurement_complete) {
328 uint32_t elapsed = ((uint32_t)esp_timer_get_time() - start_time) / 1000;
329 if (elapsed > timeout_ms) {
330 ESP_LOGW(s_TAG, "Measurement timeout");
331 return HCSR04_ERR_TIMEOUT;
332 }
333 vTaskDelay(pdMS_TO_TICKS(1));
334 }
335
336 return star_sensor_hcsr04_get_result(handle, distance_cm);
337}
