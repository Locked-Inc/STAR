95static void internal_sensor_task_loop(void* parameter)
96{
97 sensor_task_context_t* ctx = (sensor_task_context_t*)parameter;
98
99 ESP_LOGI(s_TAG, "Sensor monitoring task started");
100
101 while (ctx->task_running) {
102 TickType_t start_time = xTaskGetTickCount();
103
104 for (uint8_t i = 0; i < NUM_HCSR04; i++) {
105 if (!ctx->task_running) {
106 break;
107 }
108
109 internal_read_single_sensor(ctx, i);
110
111 if (i < NUM_HCSR04 - 1) {
112 vTaskDelay(pdMS_TO_TICKS(20));
113 }
114 }
115
116 TickType_t elapsed = xTaskGetTickCount() - start_time;
117 TickType_t target_period = pdMS_TO_TICKS(TASK_INTERVAL_SENSORS);
118
119 if (elapsed < target_period) {
120 vTaskDelay(target_period - elapsed);
121 }
122 }
123
124 ESP_LOGI(s_TAG, "Sensor monitoring task ending");
125 ctx->task_handle = NULL;
126 vTaskDelete(NULL);
127}

175 s_g_context_initialized = true;

177 shared_context_t* shared_ctx = shared_data_get_context();
178 if (shared_ctx != NULL) {
179 shared_ctx->sensors_task_handle = s_task_context.task_handle;
180 }

182 ESP_LOGI(s_TAG, "Sensor task started successfully");
