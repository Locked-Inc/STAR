25static esp_err_t internal_init_sensors(sensor_task_context_t* const ctx)
26{
27 const hcsr04_config_t sensor_configs[STAR_SYSTEM_HCSR04_SENSOR_COUNT] = {
28 /* HC-SR04 sensor configurations */
29 {STAR_SYSTEM_CONFIG.gpio.left_trig,
30 STAR_SYSTEM_CONFIG.gpio.left_echo,
31 STAR_SYSTEM_CONFIG.health.default_temperature_c},
32 {STAR_SYSTEM_CONFIG.gpio.right_trig,
33 STAR_SYSTEM_CONFIG.gpio.right_echo,
34 STAR_SYSTEM_CONFIG.health.default_temperature_c}};
35
36 esp_err_t ret = ESP_OK;
37
38 for (uint8_t i = 0; i < STAR_SYSTEM_CONFIG.num_hcsr04_sensors; i++) {
39 esp_err_t init_ret = star_sensor_hcsr04_init(&ctx->sensor_handles[i],
40 ctx->bus_manager,
41 ctx->bus_name,
42 ctx->error_iface,
43 &sensor_configs[i]);
44 if (init_ret != ESP_OK) {
45 ESP_LOGE(s_TAG, "Failed to init HC-SR04 sensor %d: %s", i, esp_err_to_name(init_ret));
46
47 for (uint8_t j = 0; j < i; j++) {
48 star_sensor_hcsr04_deinit(&ctx->sensor_handles[j]);
49 }
50 ret = init_ret;
51 break;
52 }
53 }
54
55 if (ret == ESP_OK) {
56 ctx->sensors_initialized = true;
57 ESP_LOGI(s_TAG,
58 "All %d HC-SR04 sensors initialized successfully",
59 STAR_SYSTEM_CONFIG.num_hcsr04_sensors);
60 }
61
62 return ret;
63}

79static esp_err_t internal_read_single_sensor(sensor_task_context_t* const ctx,
80 const uint8_t sensor_index)
81{
82 if (!star_validate_sensor_index(sensor_index)) {
83 return ESP_ERR_INVALID_ARG;
84 }
85
86 float distance_cm;
87 esp_err_t read_result =
88 star_sensor_hcsr04_read_distance(&ctx->sensor_handles[sensor_index], &distance_cm);
89
90 sensor_data_t sensor_data =
91 shared_data_create_sensor_data(sensor_index, distance_cm, read_result);
92
93 shared_data_set_sensor_reading(sensor_index, &sensor_data);
94 shared_data_update_health(sensor_index, (read_result == ESP_OK));
95
96 shared_context_t* shared_ctx = shared_data_get_context();
97 if (shared_ctx != NULL && shared_ctx->sensor_data_queue != NULL) {
98 if (xQueueSend(shared_ctx->sensor_data_queue, &sensor_data, 0) != pdTRUE) {
99 ESP_LOGW(s_TAG, "Sensor data queue full, dropping measurement");
100 }
101 }
102
103 if (read_result == ESP_OK) {
104 const char* const sensor_name = (sensor_index == k_star_system_hcsr04_left) ? "Left" : "Right";
105 ESP_LOGI(s_TAG, "%s sensor: %.1f cm", sensor_name, distance_cm);
106 } else {
107 ESP_LOGE(s_TAG, "Failed to read sensor %d: %s", sensor_index, esp_err_to_name(read_result));
108 }
109
110 return read_result;

113static void internal_sensor_task_loop(void* const parameter)
114{
115 sensor_task_context_t* const ctx = (sensor_task_context_t*)parameter;
116
117 ESP_LOGI(s_TAG, "Sensor monitoring task started");
118
119 while (ctx->task_running) {
120 const TickType_t start_time = xTaskGetTickCount();
121
122 for (uint8_t i = 0; i < STAR_SYSTEM_CONFIG.num_hcsr04_sensors; i++) {
123 if (!ctx->task_running) {
124 break;
125 }
126
127 internal_read_single_sensor(ctx, i);
128
129 if (i < STAR_SYSTEM_CONFIG.num_hcsr04_sensors - 1) {
130 vTaskDelay(pdMS_TO_TICKS(sensor_task_inter_sensor_delay_ms));
131 }
132 }
133
134 const TickType_t elapsed = xTaskGetTickCount() - start_time;
135 const TickType_t target_period = pdMS_TO_TICKS(STAR_SYSTEM_CONFIG.tasks.sensors.interval_ms);
136
137 if (elapsed < target_period) {
138 vTaskDelay(target_period - elapsed);
139 }
140 }
141
142 ESP_LOGI(s_TAG, "Sensor monitoring task ending");
143 ctx->task_handle = NULL;
144 vTaskDelete(NULL);
145}

149esp_err_t sensor_task_start(star_bus_manager_t* const bus_manager,
150 star_error_interface_t* const error_iface,
151 const char* const bus_name)
152{
153 if (bus_manager == NULL || bus_name == NULL) {
154 ESP_LOGE(s_TAG, "Invalid arguments for sensor task start");
155 return ESP_ERR_INVALID_ARG;
156 }
157
158 if (s_g_context_initialized && s_task_context.task_running) {
159 ESP_LOGI(s_TAG, "Sensor task already running");
160 return ESP_OK;
161 }
162
163 memset(&s_task_context, 0, sizeof(sensor_task_context_t));
164
165 s_task_context.bus_manager = bus_manager;
166 s_task_context.error_iface = error_iface;
167 s_task_context.bus_name = bus_name;
168 s_task_context.task_running = false;
169 s_task_context.sensors_initialized = false;
170
171 esp_err_t ret = internal_init_sensors(&s_task_context);
172 if (ret != ESP_OK) {
173 ESP_LOGE(s_TAG, "Failed to initialize sensors");
174 return ret;
175 }
176
177 s_task_context.task_running = true;
178
179 const BaseType_t task_ret = xTaskCreate(internal_sensor_task_loop,
180 STAR_SYSTEM_CONFIG.tasks.sensors.name,
181 STAR_SYSTEM_CONFIG.tasks.sensors.stack_size,
182 &s_task_context,
183 STAR_SYSTEM_CONFIG.tasks.sensors.priority,
184 &s_task_context.task_handle);
185
186 if (task_ret != pdPASS) {
187 ESP_LOGE(s_TAG, "Failed to create sensor task");
188 s_task_context.task_running = false;
189 internal_deinit_sensors(&s_task_context);
190 return ESP_ERR_NO_MEM;
191 }
192
193 s_g_context_initialized = true;
194
195 shared_context_t* shared_ctx = shared_data_get_context();
196 if (shared_ctx != NULL) {
197 shared_ctx->sensors_task_handle = s_task_context.task_handle;
198 }
199
200 ESP_LOGI(s_TAG, "Sensor task started successfully");
201 return ESP_OK;
202}
