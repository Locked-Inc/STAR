27static esp_err_t internal_bq7850_manufacturer_access(star_bus_manager_t* manager,
28 const char* bus_name,
29 uint8_t addr,
30 uint16_t subcmd,
31 uint16_t* result)
32{
33 esp_err_t ret;
34
35 /* Write subcommand to Manufacturer Access register */
36 ret = star_smbus_write_word(manager, bus_name, addr, BQ7850_CMD_MANUFACTURER_ACCESS, subcmd);
37 if (ret != ESP_OK) {
38 ESP_LOGE(s_TAG,
39 "Failed to write manufacturer access subcommand 0x%04X: %s",
40 subcmd,
41 esp_err_to_name(ret));
42 return ret;
43 }
44
45 /* Small delay for command processing */
46 vTaskDelay(pdMS_TO_TICKS(10));
47
48 /* Read result if requested */
49 if (result != NULL) {
50 ret = star_smbus_read_word(manager, bus_name, addr, BQ7850_CMD_MANUFACTURER_ACCESS, result);
51 if (ret != ESP_OK) {
52 ESP_LOGE(s_TAG, "Failed to read manufacturer access result: %s", esp_err_to_name(ret));
53 return ret;
54 }
55 }
56
57 return ESP_OK;
58}

65 uint8_t addr,

79esp_err_t star_bms_bq7850_init(bq7850_handle_t* handle,
80 star_bus_manager_t* manager,
81 const char* bus_name,
82 star_error_interface_t* error_iface,
83 const bq7850_config_t* config)
84{
85 if (handle == NULL || manager == NULL || bus_name == NULL || config == NULL) {
86 ESP_LOGE(s_TAG, "NULL parameter in init");
87 return ESP_ERR_INVALID_ARG;
88 }
89
90 if (config->num_cells == 0 || config->num_cells > BQ7850_MAX_CELLS) {
91 ESP_LOGE(s_TAG,
92 "Invalid number of cells: %d (must be 1-%d)",
93 config->num_cells,
94 BQ7850_MAX_CELLS);
95 return ESP_ERR_INVALID_ARG;
96 }
97
98 /* Initialize handle */
99 memset(handle, 0, sizeof(bq7850_handle_t));
100 handle->manager = manager;
101 handle->bus_name = bus_name;
102 handle->smbus_addr = config->smbus_addr;
103 memcpy(&handle->config, config, sizeof(bq7850_config_t));
104
105 /* Create mutex first */
106 handle->mutex = xSemaphoreCreateMutex();
107 if (handle->mutex == NULL) {
108 ESP_LOGE(s_TAG, "Failed to create mutex");
109 return ESP_ERR_NO_MEM;
110 }
111
112 /* Handle error interface injection */
113 if (error_iface == NULL) {
114 /* Create default error handler internally */
115 error_handler_t* default_handler = error_handler_create_default();
116 if (default_handler == NULL) {
117 ESP_LOGE(s_TAG, "Failed to create default error handler");
118 vSemaphoreDelete(handle->mutex);
119 handle->mutex = NULL;
120 return ESP_ERR_NO_MEM;
121 }
122
123 /* Allocate and store the interface */
124 handle->error_iface = (star_error_interface_t*)malloc(sizeof(star_error_interface_t));
125 if (handle->error_iface == NULL) {
126 ESP_LOGE(s_TAG, "Failed to allocate error interface");
127 error_handler_destroy_default(default_handler);
128 vSemaphoreDelete(handle->mutex);
129 handle->mutex = NULL;
130 return ESP_ERR_NO_MEM;
131 }
132
133 error_handler_get_interface(handle->error_iface, default_handler);
134 handle->owns_error_handler = true;
135 ESP_LOGI(s_TAG, "Created default error handler internally");
136 } else {
137 handle->error_iface = error_iface;
138 handle->owns_error_handler = false;
139 ESP_LOGI(s_TAG, "Using injected error interface");
140 }
141
142 ESP_LOGI(s_TAG,
143 "Initializing BQ7850 BMS on bus '%s' at address 0x%02X",
144 bus_name,
145 config->smbus_addr);
146
147 /* Verify device communication by reading device type */
148 uint16_t device_type = 0;
149 esp_err_t ret = internal_bq7850_manufacturer_access(manager,
150 bus_name,
151 config->smbus_addr,
152 BQ7850_SUBCMD_DEVICE_TYPE,
153 &device_type);
154
155 if (ret != ESP_OK) {
156 ESP_LOGE(s_TAG, "Failed to communicate with BQ7850: %s", esp_err_to_name(ret));
157 if (handle->owns_error_handler && handle->error_iface != NULL) {
158 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
159 error_handler_destroy_default(handler_ptr);
160 free(handle->error_iface);
161 }
162 vSemaphoreDelete(handle->mutex);
163 handle->mutex = NULL;
164 return ret;
165 }
166
167 ESP_LOGI(s_TAG, "BQ7850 detected, device type: 0x%04X", device_type);
168
169 /* Read firmware version */
170 uint16_t fw_version = 0;
171 ret = internal_bq7850_manufacturer_access(manager,
172 bus_name,
173 config->smbus_addr,
174 BQ7850_SUBCMD_FW_VERSION,
175 &fw_version);
176 if (ret == ESP_OK) {
177 ESP_LOGI(s_TAG, "Firmware version: 0x%04X", fw_version);
178 }
179
180 /* Verify we can read basic parameters */
181 uint16_t voltage = 0;
182 ret = star_smbus_read_word(manager, bus_name, config->smbus_addr, BQ7850_CMD_VOLTAGE, &voltage);
183 if (ret != ESP_OK) {
184 ESP_LOGE(s_TAG, "Failed to read pack voltage: %s", esp_err_to_name(ret));
185 if (handle->owns_error_handler && handle->error_iface != NULL) {
186 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
187 error_handler_destroy_default(handler_ptr);
188 free(handle->error_iface);
189 }
190 vSemaphoreDelete(handle->mutex);
191 handle->mutex = NULL;
192 return ret;
193 }
194
195 handle->initialized = true;
196 ESP_LOGI(s_TAG,
197 "BQ7850 initialization successful (thread-safe with mutex), pack voltage: %d mV",
198 voltage);
199
200 return ESP_OK;
201}

203esp_err_t star_bms_bq7850_deinit(bq7850_handle_t* handle)
204{
205 if (handle == NULL || !handle->initialized) {
206 return ESP_ERR_INVALID_ARG;
207 }
208
209 /* Acquire mutex before cleanup */
210 SemaphoreHandle_t mutex = handle->mutex;
211 if (mutex != NULL && xSemaphoreTake(mutex, pdMS_TO_TICKS(BQ7850_MUTEX_TIMEOUT_MS)) != pdTRUE) {
212 ESP_LOGW(s_TAG, "Failed to acquire mutex for deinit, continuing anyway");
213 }
214
215 /* Clean up error handler if we own it */
216 if (handle->owns_error_handler && handle->error_iface != NULL) {
217 error_handler_t* error_handler = (error_handler_t*)handle->error_iface->ctx;
218 error_handler_destroy_default(error_handler);
219 free(handle->error_iface);
220 handle->error_iface = NULL;
221 ESP_LOGI(s_TAG, "Destroyed internally-owned error handler");
222 }
223
224 /* Clear handle fields */
225 handle->initialized = false;
226 handle->mutex = NULL;
227
228 /* Release and delete mutex */
229 if (mutex != NULL) {
230 xSemaphoreGive(mutex);
231 vSemaphoreDelete(mutex);
232 }
233
234 ESP_LOGI(s_TAG, "BQ7850 BMS deinitialized");
235 return ESP_OK;
236}

359esp_err_t star_bms_bq7850_read_cells(const bq7850_handle_t* handle, bq7850_cell_data_t* cell_data)
360{
361 if (handle == NULL || cell_data == NULL) {
362 return ESP_ERR_INVALID_ARG;
363 }
364
365 star_bus_manager_t* manager = handle->manager;
366 const char* bus_name = handle->bus_name;
367 uint8_t addr = handle->smbus_addr;
368 esp_err_t ret = ESP_OK;
369
370 memset(cell_data, 0, sizeof(bq7850_cell_data_t));
371
372 /* Read individual cell voltages */
373 for (uint8_t i = 0; i < BQ7850_MAX_CELLS; i++) {
374 ret = internal_bq7850_read_cell_voltage(manager, bus_name, addr, i, &cell_data->cell_mv[i]);
375
376 if (ret == ESP_OK && cell_data->cell_mv[i] > 0) {
377 cell_data->valid_cells++;
378 } else if (ret != ESP_OK) {
379 ESP_LOGW(s_TAG, "Failed to read cell %d voltage: %s", i + 1, esp_err_to_name(ret));
380 /* Continue reading other cells */
381 }
382 }
383
384 /* Read total pack voltage */
385 ret = star_smbus_read_word(manager, bus_name, addr, BQ7850_CMD_VOLTAGE, &cell_data->pack_mv);
386 if (ret != ESP_OK) {
387 ESP_LOGE(s_TAG, "Failed to read pack voltage: %s", esp_err_to_name(ret));
388 }
389
390 return (cell_data->valid_cells > 0) ? ESP_OK : ESP_FAIL;
391}

425/* --- Temperature Functions --- */
426
427esp_err_t star_bms_bq7850_read_temperatures(const bq7850_handle_t* handle,
428 bq7850_temp_data_t* temp_data)
429{
430 if (handle == NULL || temp_data == NULL) {
431 return ESP_ERR_INVALID_ARG;
432 }
433
434 star_bus_manager_t* manager = handle->manager;
435 const char* bus_name = handle->bus_name;
436 uint8_t addr = handle->smbus_addr;
437 esp_err_t ret;
438
439 memset(temp_data, 0, sizeof(bq7850_temp_data_t));
440
441 /* Read temperature sensors via manufacturer access */
442 int32_t temp_sum = 0;
443
444 for (uint8_t i = 0; i < BQ7850_MAX_TEMP_SENSORS; i++) {
445 uint16_t temp_raw = 0;
446 uint16_t subcmd = BQ7850_SUBCMD_TEMP_SENSOR_BASE + i;
447
448 ret = internal_bq7850_manufacturer_access(manager, bus_name, addr, subcmd, &temp_raw);
449
450 if (ret == ESP_OK && temp_raw > 0) {
451 /* BQ7850 returns temperature in 0.1K units */
452 /* Convert to 0.1C: T(C) = T(K) - 273.15 */
453 temp_data->temp_c[i] = temp_raw - 2731; /* 273.1K = 2731 in 0.1K units */
454 temp_data->valid_sensors++;
455 temp_sum += temp_data->temp_c[i];
456 }
457 }
458
459 /* Calculate average temperature */
460 if (temp_data->valid_sensors > 0) {
461 temp_data->avg_temp_c = temp_sum / temp_data->valid_sensors;
462 }
463
464 return (temp_data->valid_sensors > 0) ? ESP_OK : ESP_FAIL;
465}

519 (uint16_t*)&current_data->avg_current_ma);

654}

788esp_err_t star_bms_bq7850_write_protection(const bq7850_handle_t* handle,
789 const bq7850_protection_t* protection)
790{
791 if (handle == NULL || protection == NULL) {
792 return ESP_ERR_INVALID_ARG;
793 }
794
795 star_bus_manager_t* manager = handle->manager;
796 const char* bus_name = handle->bus_name;
797 uint8_t addr = handle->smbus_addr;
798 esp_err_t ret;
799
800 ESP_LOGW(s_TAG, "Writing protection thresholds (device must be unsealed)");
801
802 /* Write protection thresholds via manufacturer access */
803 ret = internal_bq7850_manufacturer_access(manager,
804 bus_name,
805 addr,
806 BQ7850_SUBCMD_PROTECTION_OV_THRESH,
807 NULL);
808 if (ret == ESP_OK) {
809 ret = star_smbus_write_word(manager,
810 bus_name,
811 addr,
812 BQ7850_CMD_MANUFACTURER_ACCESS,
813 protection->overvoltage_mv);
814 }
815
816 if (ret != ESP_OK) {
817 ESP_LOGE(s_TAG, "Failed to write OV threshold: %s", esp_err_to_name(ret));
818 return ret;
819 }
820
821 /* Add delay between writes */
822 vTaskDelay(pdMS_TO_TICKS(50));
823
824 /* Similar process for other thresholds... */
825 ESP_LOGI(s_TAG, "Protection thresholds updated");
826
827 return ESP_OK;
828}
