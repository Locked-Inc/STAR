13static float internal_scale_pwm_for_color(const float original_percent, const float min_pwm)
14{
15 /* Ultra-smooth PWM scaling with no discrete jumps */
16
17 /* Use an extremely low threshold for near-seamless transitions */
18 if (original_percent <= STAR_SYSTEM_CONFIG.pwm.off_threshold) {
19 return 0.0f; /* True off state */
20 }
21
22 /* Smooth exponential scaling to avoid linear "steps" */
23 /* This creates a more natural brightness perception curve */
24 const float normalized = original_percent / s_led_normalize_percent;
25
26 /* Apply gamma correction for perceptually linear brightness */
27 /* Human eye perceives brightness non-linearly */
28 const float gamma_corrected = powf(normalized, s_led_gamma_correction_factor);
29
30 /* Map to safe PWM range with smooth interpolation */
31 const float pwm_range = STAR_SYSTEM_CONFIG.pwm.max_pwm_all - min_pwm;
32 float scaled = min_pwm + (gamma_corrected * pwm_range);
33
34 /* Ensure we stay within safe bounds */
35 scaled = fmaxf(min_pwm, fminf(STAR_SYSTEM_CONFIG.pwm.max_pwm_all, scaled));
36
37 /* Enhanced debug logging for PWM analysis */
38 ESP_LOGI(s_TAG,
39 "PWM: %.3f%% -> γ:%.3f -> %.2f%% (range: %.1f-%.1f)",
40 original_percent,
41 gamma_corrected,
42 scaled,
43 min_pwm,
44 STAR_SYSTEM_CONFIG.pwm.max_pwm_all);
45
46 return scaled;
47}

90rgb_color_t led_controller_hsv_to_rgb(const hsv_color_t* hsv)
91{
92 rgb_color_t rgb = {0.0f, 0.0f, 0.0f};
93
94 if (hsv == NULL || hsv->saturation < 0.0f || hsv->saturation > s_led_hsv_saturation_max ||
95 hsv->value < 0.0f || hsv->value > s_led_hsv_value_max) {
96 return rgb;
97 }
98
99 float hue = fmodf(hsv->hue, s_led_hue_max_degrees);
100 if (hue < 0.0f)
101 hue += s_led_hue_max_degrees;
102
103 const float c = hsv->value * hsv->saturation;
104 const float x =
105 c *
106 (s_led_hsv_saturation_max -
107 fabsf(fmodf(hue / s_led_hue_sector_size, s_led_hue_sector_double) - s_led_hsv_saturation_max));
108 const float m = hsv->value - c;
109
110 float r_prime, g_prime, b_prime;
111
112 if (hue < s_led_hue_sector_1) {
113 r_prime = c;
114 g_prime = x;
115 b_prime = 0.0f;
116 } else if (hue < s_led_hue_sector_2) {
117 r_prime = x;
118 g_prime = c;
119 b_prime = 0.0f;
120 } else if (hue < s_led_hue_sector_3) {
121 r_prime = 0.0f;
122 g_prime = c;
123 b_prime = x;
124 } else if (hue < s_led_hue_sector_4) {
125 r_prime = 0.0f;
126 g_prime = x;
127 b_prime = c;
128 } else if (hue < s_led_hue_sector_5) {
129 r_prime = x;
130 g_prime = 0.0f;
131 b_prime = c;
132 } else {
133 r_prime = c;
134 g_prime = 0.0f;
135 b_prime = x;
136 }
137
138 rgb.red = (r_prime + m) * s_led_rgb_max_percent;
139 rgb.green = (g_prime + m) * s_led_rgb_max_percent;
140 rgb.blue = (b_prime + m) * s_led_rgb_max_percent;
141
142 return rgb;
143}

147esp_err_t led_controller_init(led_controller_t* const controller,
148 const pca9685_handle_t* const pca_handle)
149{
150 if (controller == NULL || pca_handle == NULL) {
151 ESP_LOGE(s_TAG, "Invalid arguments for LED controller init");
152 return ESP_ERR_INVALID_ARG;
153 }
154
155 memset(controller, 0, sizeof(led_controller_t));
156
157 controller->mutex = xSemaphoreCreateMutex();
158 if (controller->mutex == NULL) {
159 ESP_LOGE(s_TAG, "Failed to create mutex for LED controller");
160 return ESP_ERR_NO_MEM;
161 }
162
163 controller->pca_handle = pca_handle;
164 controller->initialized = true;
165
166 ESP_LOGI(s_TAG, "LED controller initialized successfully");
167
168 const esp_err_t ret = led_controller_turn_off_all(controller);
169 if (ret != ESP_OK) {
170 ESP_LOGW(s_TAG, "Failed to turn off all LEDs during init");
171 }
172
173 return ESP_OK;
174}

176void led_controller_deinit(led_controller_t* const controller)
177{

184 led_controller_turn_off_all(controller);
185 controller->initialized = false;
186 controller->pca_handle = NULL;
187 xSemaphoreGive(controller->mutex);

189
190 if (controller->mutex != NULL) {
191 vSemaphoreDelete(controller->mutex);
192 controller->mutex = NULL;

198rgb_color_t led_controller_distance_to_color(const float distance_cm)
199{
200 rgb_color_t color = {0.0f, 0.0f, 0.0f};
201
202 /* Handle invalid distances gracefully */
203 if (!star_validate_distance(distance_cm)) {
204 return color; /* Return off for invalid readings */
205 }
206
207 /* Turn off LEDs for distances beyond maximum range */
208 if (distance_cm > STAR_SYSTEM_CONFIG.distance.max_distance_cm) {
209 return color; /* Return off color */
210 }
211
212 /* CONFIGURABLE DISTANCE RANGE MAPPING */
213 /* Ultra-smooth mapping with configurable min/max distances */
214 const float min_distance =
215 STAR_SYSTEM_CONFIG.distance.min_distance_cm * STAR_SYSTEM_CONFIG.distance.extension_factor;
216 float max_distance = STAR_SYSTEM_CONFIG.distance.max_distance_cm;
217
218 /* Handle edge case where min might equal max */
219 if (max_distance <= min_distance) {
220 max_distance = min_distance + s_led_distance_range_min_offset; /* Ensure at least 1cm range */
221 }
222
223 /* Smooth clamping with exponential approach to boundaries */
224 float effective_distance = distance_cm;
225 if (effective_distance < min_distance) {
226 /* Exponential approach to minimum for ultra-close distances */
227 float ratio = effective_distance / min_distance;
228 effective_distance =
229 min_distance * (s_led_hsv_saturation_max - expf(-s_led_distance_smooth_factor * ratio));
230 }
231
232 /* Map to normalized value with configurable range */
233 const float range = max_distance - min_distance;
234 float t = (effective_distance - min_distance) / range;
235
236 /* Ensure t is in valid range with smooth clamping */
237 t = fmaxf(0.0f, fminf(s_led_hsv_saturation_max, t));
238
239 /* Ultra-smooth HSV gradient: Red (0°) -> Blue (240°) via rainbow spectrum */
240 hsv_color_t hsv;
241
242 /* Smooth hue transition across extended color spectrum */
243 /* 0° = Red, 60° = Yellow, 120° = Green, 180° = Cyan, 240° = Blue */
244 hsv.hue = t * s_led_gradient_max_hue;
245
246 /* Full saturation for vibrant, rich colors */
247 hsv.saturation = s_led_hsv_saturation_max;
248
249 /* Smooth brightness curve with emphasis on close-range visibility */
250 /* Use smooth polynomial instead of linear for more natural transitions */
251 const float t_squared = t * t;
252 const float t_cubed = t_squared * t;
253
254 /* Hermite interpolation for ultra-smooth brightness transitions */
255 const float brightness_base =
256 s_led_hsv_saturation_max -
257 (0.5f * (s_led_brightness_curve_factor * t_squared - s_led_brightness_power_factor * t_cubed));
258
259 /* Configurable brightness boost for safety warning at close range */
260 float close_boost = 0.0f;
261 if (t < STAR_SYSTEM_CONFIG.distance.close_boost_threshold) {
262 /* Smooth exponential boost for close distances */
263 float close_factor =
264 s_led_hsv_saturation_max - (t / STAR_SYSTEM_CONFIG.distance.close_boost_threshold);
265 close_boost = s_led_close_boost_factor * close_factor * close_factor; /* Quadratic falloff */
266 }
267
268 hsv.value =
269 fmaxf(s_led_brightness_min, fminf(s_led_brightness_max, brightness_base + close_boost));
270
271 /* Convert HSV to RGB */
272 color = led_controller_hsv_to_rgb(&hsv);
273
274 /* Enhanced debug logging showing configurable range */
275 ESP_LOGI(
276 s_TAG,
277 "Range[%.1f-%.1f]cm: %.2fcm -> Eff: %.2fcm, t=%.4f, HSV(%.2f,%.2f,%.2f) -> RGB(%.2f,%.2f,%.2f)",
278 STAR_SYSTEM_CONFIG.distance.min_distance_cm,
279 STAR_SYSTEM_CONFIG.distance.max_distance_cm,
280 distance_cm,
281 effective_distance,
282 t,
283 hsv.hue,
284 hsv.saturation,
285 hsv.value,
286 color.red,
287 color.green,
288 color.blue);
289
290 return color;
291}

296{

301
302 if (led_index >= k_led_index_max) {
303 ESP_LOGE(s_TAG, "Invalid LED index: %d", led_index);

306
307 if (xSemaphoreTake(controller->mutex,
308 pdMS_TO_TICKS(STAR_SYSTEM_CONFIG.freertos.mutex_timeout_ms)) != pdTRUE) {

310 return ESP_ERR_TIMEOUT;

312
313 esp_err_t ret = ESP_OK;
314 if (controller->initialized && controller->pca_handle != NULL) {
315 ret = internal_set_rgb_channels(controller->pca_handle, led_index, color);
316 } else {
317 ret = ESP_ERR_INVALID_STATE;
318 }
319
320 xSemaphoreGive(controller->mutex);
321 return ret;
