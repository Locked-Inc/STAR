48static peripheral_state_t* get_peripheral_state(const char* bus_name, bool create)
49{
50 for (uint8_t i = 0; i < g_num_peripheral_states; i++) {
51 if (strcmp(g_peripheral_states[i].bus_name, bus_name) == 0) {
52 return &g_peripheral_states[i];
53 }
54 }
55
56 if (create && g_num_peripheral_states < MAX_PERIPHERAL_BUSES) {
57 peripheral_state_t* state = &g_peripheral_states[g_num_peripheral_states];
58 memset(state, 0, sizeof(peripheral_state_t));
59 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
60 g_num_peripheral_states++;
61 return state;
62 }
63
64 return NULL;
65}

70static star_onewire_rom_t calculate_rom_crc(star_onewire_rom_t rom)
71{
72 uint8_t rom_bytes[8];
73 star_bus_onewire_rom_to_bytes(rom, rom_bytes);
74
75 /* Calculate CRC for first 7 bytes */
76 uint8_t crc = star_bus_onewire_crc8(rom_bytes, 7);
77
78 /* Set CRC as last byte */
79 rom_bytes[7] = crc;
80
81 return star_bus_onewire_bytes_to_rom(rom_bytes);
82}

86esp_err_t star_bus_onewire_peripheral_enable(star_bus_manager_t* manager,
87 const char* bus_name,
88 const star_onewire_peripheral_config_t* config)
89{
90 if (manager == NULL || bus_name == NULL || config == NULL) {
91 ESP_LOGE(s_TAG, "Invalid parameters");
92 return ESP_ERR_INVALID_ARG;
93 }
94
95 if (config->on_read == NULL || config->on_write == NULL) {
96 ESP_LOGE(s_TAG, "Read and write callbacks are required");
97 return ESP_ERR_INVALID_ARG;
98 }
99
100 peripheral_state_t* state = get_peripheral_state(bus_name, true);
101 if (state == NULL) {
102 ESP_LOGE(s_TAG, "Failed to create peripheral state for bus '%s'", bus_name);
103 return ESP_ERR_NO_MEM;
104 }
105
106 /* Store configuration */
107 memcpy(&state->config, config, sizeof(star_onewire_peripheral_config_t));
108 memset(&state->stats, 0, sizeof(star_onewire_peripheral_stats_t));
109
110 /* Calculate CRC if requested */
111 if (config->auto_calculate_crc) {
112 state->config.rom_code = calculate_rom_crc(config->rom_code);
113 }
114
115 /* Verify ROM CRC */
116 if (!star_bus_onewire_verify_rom(state->config.rom_code)) {
117 ESP_LOGW(s_TAG, "ROM code has invalid CRC: 0x%016llX", state->config.rom_code);
118 }
119
120 state->enabled = true;
121 state->alarm_active = false;
122
123 ESP_LOGI(s_TAG,
124 "1-Wire peripheral enabled on bus '%s' with ROM 0x%016llX",
125 bus_name,
126 state->config.rom_code);
127
128 ESP_LOGW(s_TAG,
129 "Note: 1-Wire peripheral mode requires precise timing and "
130 "is best suited for slower bus speeds. Hardware limitations "
131 "may affect reliability at standard 1-Wire speeds.");
132
133 return ESP_OK;
134}

194 return ESP_OK;
