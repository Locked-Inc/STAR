22static smbus_peripheral_state_t* get_peripheral_state(const char* bus_name, bool create)
23{
24 for (uint8_t i = 0; i < g_num_peripheral_states; i++) {
25 if (strcmp(g_peripheral_states[i].bus_name, bus_name) == 0) {
26 return &g_peripheral_states[i];
27 }
28 }
29
30 if (create && g_num_peripheral_states < MAX_PERIPHERAL_BUSES) {
31 smbus_peripheral_state_t* state = &g_peripheral_states[g_num_peripheral_states];
32 memset(state, 0, sizeof(smbus_peripheral_state_t));
33 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
34 g_num_peripheral_states++;
35 return state;
36 }
37
38 return NULL;
39}

41esp_err_t star_bus_smbus_peripheral_enable(star_bus_manager_t* manager,
42 const char* bus_name,
43 const star_smbus_peripheral_config_t* config)
44{
45 if (manager == NULL || bus_name == NULL || config == NULL) {
46 return ESP_ERR_INVALID_ARG;
47 }
48
49 smbus_peripheral_state_t* state = get_peripheral_state(bus_name, true);
50 if (state == NULL) {
51 return ESP_ERR_NO_MEM;
52 }
53
54 memcpy(&state->config, config, sizeof(star_smbus_peripheral_config_t));
55 memset(&state->stats, 0, sizeof(star_smbus_peripheral_stats_t));
56 state->enabled = true;
57
58 ESP_LOGI(s_TAG,
59 "SMBus peripheral enabled on bus '%s' at address 0x%02X",
60 bus_name,
61 config->address);
62
63 return ESP_OK;
64}
