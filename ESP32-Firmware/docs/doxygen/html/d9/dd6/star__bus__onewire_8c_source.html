80static esp_err_t internal_init_onewire_mutex(void)
81{
82 /* Quick check without lock for common case */
83 if (g_onewire_mutex != NULL) {
84 return ESP_OK;
85 }
86
87 /* Use spinlock for thread-safe lazy initialization */
88 portENTER_CRITICAL(&g_onewire_init_spinlock);
89
90 /* Double-check after acquiring spinlock */
91 if (g_onewire_mutex == NULL) {
92 g_onewire_mutex = xSemaphoreCreateMutex();
93 if (g_onewire_mutex == NULL) {
94 portEXIT_CRITICAL(&g_onewire_init_spinlock);
95 ESP_LOGE(s_TAG, "Failed to create 1-Wire global mutex");
96 return ESP_ERR_NO_MEM;
97 }
98 }
99
100 portEXIT_CRITICAL(&g_onewire_init_spinlock);
101 return ESP_OK;
102}

107static onewire_state_t* internal_get_onewire_state(const char* bus_name, bool create)
108{
109 for (uint8_t i = 0; i < g_num_onewire_states; i++) {
110 if (strcmp(g_onewire_states[i].bus_name, bus_name) == 0) {
111 return &g_onewire_states[i];
112 }
113 }
114
115 if (create && g_num_onewire_states < MAX_ONEWIRE_BUSES) {
116 onewire_state_t* state = &g_onewire_states[g_num_onewire_states];
117 memset(state, 0, sizeof(onewire_state_t));
118 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
119 state->bus_name[sizeof(state->bus_name) - 1] = '\0';
120 g_num_onewire_states++;
121 return state;
122 }
123
124 return NULL;
125}

130static onewire_state_t* get_onewire_state(const char* bus_name, bool create)
131{
132 if (internal_init_onewire_mutex() != ESP_OK) {
133 return NULL;
134 }
135
136 if (xSemaphoreTake(g_onewire_mutex, portMAX_DELAY) != pdTRUE) {
137 ESP_LOGE(s_TAG, "Failed to take 1-Wire mutex");
138 return NULL;

141 onewire_state_t* state = internal_get_onewire_state(bus_name, create);
142
143 xSemaphoreGive(g_onewire_mutex);
144 return state;

181esp_err_t star_bus_onewire_init(star_bus_manager_t* manager,
182 const char* bus_name,
183 const star_onewire_config_t* config)
184{
185 if (manager == NULL || bus_name == NULL || config == NULL) {
186 ESP_LOGE(s_TAG, "Invalid parameters");
187 return ESP_ERR_INVALID_ARG;
188 }
189
190 if (config->gpio_pin < 0) {
191 ESP_LOGE(s_TAG, "Invalid GPIO pin");
192 return ESP_ERR_INVALID_ARG;
193 }
194
195 onewire_state_t* state = get_onewire_state(bus_name, true);
196 if (state == NULL) {
197 ESP_LOGE(s_TAG, "Failed to create state for bus '%s'", bus_name);
198 return ESP_ERR_NO_MEM;
199 }
200
201 memcpy(&state->config, config, sizeof(star_onewire_config_t));
202 memset(&state->stats, 0, sizeof(star_onewire_stats_t));
203
204 /* Configure GPIO as open-drain output with pull-up */
205 gpio_config_t io_conf = {
206 .pin_bit_mask = (1ULL << config->gpio_pin),
207 .mode = GPIO_MODE_INPUT_OUTPUT_OD,
208 .pull_up_en = GPIO_PULLUP_ENABLE,
209 .pull_down_en = GPIO_PULLDOWN_DISABLE,
210 .intr_type = GPIO_INTR_DISABLE,
211 };
212
213 esp_err_t result = gpio_config(&io_conf);
214 if (result != ESP_OK) {
215 ESP_LOGE(s_TAG, "Failed to configure GPIO: %s", esp_err_to_name(result));
216 return result;
217 }
218
219 /* Release bus (high) */
220 gpio_high(config->gpio_pin);
221
222 /* Reset search state */
223 state->last_discrepancy = 0;
224 state->last_family_discrepancy = 0;
225 state->last_device_flag = false;
226 memset(state->search_rom, 0, sizeof(state->search_rom));
227
228 state->initialized = true;
229
230 ESP_LOGI(s_TAG, "1-Wire bus '%s' initialized on GPIO %d", bus_name, config->gpio_pin);
231
232 return ESP_OK;
233}

256esp_err_t star_bus_onewire_reset(star_bus_manager_t* manager, const char* bus_name, bool* present)
257{
258 if (manager == NULL || bus_name == NULL || present == NULL) {
259 return ESP_ERR_INVALID_ARG;
260 }
261
262 onewire_state_t* state = get_onewire_state(bus_name, false);
263 if (state == NULL || !state->initialized) {
264 return ESP_ERR_NOT_FOUND;
265 }
266
267 gpio_num_t pin = state->config.gpio_pin;
268 uint32_t start_time = esp_timer_get_time();
269
270 state->stats.total_resets++;
271
272 /* Pull bus low for reset pulse */
273 gpio_low(pin);
274 delay_us(TIMING_RESET_PULSE);
275
276 /* Release bus and wait for presence pulse */
277 gpio_high(pin);
278 delay_us(TIMING_PRESENCE_WAIT);
279
280 /* Sample bus for presence pulse (device pulls low) */
281 int32_t level = gpio_read(pin);
282 *present = (level == 0);
283
284 /* Wait for end of presence pulse */
285 delay_us(TIMING_PRESENCE_SAMPLE);
286
287 uint32_t elapsed = (uint32_t)(esp_timer_get_time() - start_time);
288 state->stats.last_operation_time_us = elapsed;
289
290 if (*present) {
291 state->stats.successful_resets++;
292 } else {
293 state->stats.failed_resets++;
294 }
295
296 return ESP_OK;
297}

408esp_err_t star_bus_onewire_write_bytes(star_bus_manager_t* manager,
409 const char* bus_name,
410 star_onewire_rom_t rom,
411 const uint8_t* data,
412 size_t length)
413{
414 if (manager == NULL || bus_name == NULL || data == NULL) {
415 return ESP_ERR_INVALID_ARG;
416 }
417
418 onewire_state_t* state = get_onewire_state(bus_name, false);
419 if (state == NULL || !state->initialized) {
420 return ESP_ERR_NOT_FOUND;
421 }
422
423 /* Reset bus */
424 bool present;
425 esp_err_t result = star_bus_onewire_reset(manager, bus_name, &present);
426 if (result != ESP_OK || !present) {
427 return ESP_FAIL;
428 }
429
430 /* Select device or skip ROM */
431 if (rom == 0) {
432 /* Skip ROM - address all devices */
433 result = star_bus_onewire_write_byte(manager, bus_name, STAR_ONEWIRE_CMD_SKIP_ROM);
434 } else {
435 /* Match ROM - address specific device */
436 result = star_bus_onewire_write_byte(manager, bus_name, STAR_ONEWIRE_CMD_MATCH_ROM);
437 if (result == ESP_OK) {
438 uint8_t rom_bytes[8];
439 star_bus_onewire_rom_to_bytes(rom, rom_bytes);
440 for (uint32_t i = 0; i < 8; i++) {
441 result = star_bus_onewire_write_byte(manager, bus_name, rom_bytes[i]);
442 if (result != ESP_OK) {
443 break;
444 }
445 }
446 }
447 }
448
449 if (result != ESP_OK) {
450 return result;
451 }
452
453 /* Write data */
454 for (size_t i = 0; i < length; i++) {
455 result = star_bus_onewire_write_byte(manager, bus_name, data[i]);
456 if (result != ESP_OK) {
457 return result;
458 }
459 }
460
461 return ESP_OK;
462}

483esp_err_t star_bus_onewire_read_rom(star_bus_manager_t* manager,
484 const char* bus_name,
485 star_onewire_rom_t* rom)
486{
487 if (manager == NULL || bus_name == NULL || rom == NULL) {
488 return ESP_ERR_INVALID_ARG;
489 }
490
491 /* Reset bus */
492 bool present;
493 esp_err_t result = star_bus_onewire_reset(manager, bus_name, &present);
494 if (result != ESP_OK || !present) {
495 return ESP_FAIL;
496 }
497
498 /* Send Read ROM command */
499 result = star_bus_onewire_write_byte(manager, bus_name, STAR_ONEWIRE_CMD_READ_ROM);
500 if (result != ESP_OK) {
501 return result;
502 }
503
504 /* Read 8 bytes */
505 uint8_t rom_bytes[8];
506 result = star_bus_onewire_read_bytes(manager, bus_name, rom_bytes, 8);
507 if (result != ESP_OK) {

509 }

511 *rom = star_bus_onewire_bytes_to_rom(rom_bytes);
512
513 /* Verify CRC */
514 if (!star_bus_onewire_verify_rom(*rom)) {
515 onewire_state_t* state = get_onewire_state(bus_name, false);
516 if (state != NULL) {
517 state->stats.crc_errors++;
518 }
519 return ESP_ERR_INVALID_CRC;
520 }
521
522 return ESP_OK;
523}

525esp_err_t star_bus_onewire_search(star_bus_manager_t* manager,
526 const char* bus_name,
527 star_onewire_rom_t* rom_codes,
528 size_t* count)
529{
530 if (manager == NULL || bus_name == NULL || rom_codes == NULL || count == NULL) {
531 return ESP_ERR_INVALID_ARG;
532 }
533
534 onewire_state_t* state = get_onewire_state(bus_name, false);
535 if (state == NULL || !state->initialized) {
536 return ESP_ERR_NOT_FOUND;
537 }
538
539 state->stats.search_operations++;
540
541 size_t max_devices = *count;
542 *count = 0;
543
544 /* Reset search state */
545 state->last_discrepancy = 0;
546 state->last_family_discrepancy = 0;
547 state->last_device_flag = false;
548
549 /* Search for devices */
550 while (*count < max_devices) {
551 /* Reset bus */
552 bool present;
553 esp_err_t result = star_bus_onewire_reset(manager, bus_name, &present);
554 if (result != ESP_OK || !present) {
555 break;
556 }
557
558 /* Send search command */
559 result = star_bus_onewire_write_byte(manager, bus_name, STAR_ONEWIRE_CMD_SEARCH_ROM);
560 if (result != ESP_OK) {
561 break;
562 }
563
564 /* Perform search algorithm */
565 uint8_t last_zero = 0;
566 uint8_t rom_byte_number = 0;
567 uint8_t rom_byte_mask = 1;
568 bool search_result = false;
569
570 for (uint8_t bit_pos = 1; bit_pos <= 64; bit_pos++) {
571 uint8_t id_bit, cmp_id_bit;
572
573 /* Read bit and its complement */
574 star_bus_onewire_read_bit(manager, bus_name, &id_bit);
575 star_bus_onewire_read_bit(manager, bus_name, &cmp_id_bit);
576
577 if (id_bit && cmp_id_bit) {
578 /* No devices responded */
579 break;
580 }
581
582 uint8_t search_direction;
583
584 if (id_bit != cmp_id_bit) {
585 /* All devices have same bit */
586 search_direction = id_bit;
587 } else {
588 /* Discrepancy - choose direction */
589 if (bit_pos < state->last_discrepancy) {
590 search_direction = (state->search_rom[rom_byte_number] & rom_byte_mask) ? 1 : 0;
591 } else if (bit_pos == state->last_discrepancy) {
592 search_direction = 1;
593 } else {
594 search_direction = 0;
595 }
596
597 if (search_direction == 0) {
598 last_zero = bit_pos;
599 if (last_zero < 9) {
600 state->last_family_discrepancy = last_zero;
601 }
602 }
603 }
604
605 /* Write search direction */
606 star_bus_onewire_write_bit(manager, bus_name, search_direction);
607
608 /* Update ROM buffer */
609 if (search_direction) {
610 state->search_rom[rom_byte_number] |= rom_byte_mask;
611 } else {
612 state->search_rom[rom_byte_number] &= ~rom_byte_mask;
613 }
614
615 /* Advance to next bit */
616 rom_byte_mask <<= 1;
617 if (rom_byte_mask == 0) {
618 rom_byte_number++;
619 rom_byte_mask = 1;
620 }
621 }
622
623 if (rom_byte_number >= 8) {
624 search_result = true;
625 }
626
627 if (search_result) {
628 /* Store found ROM */
629 rom_codes[*count] = star_bus_onewire_bytes_to_rom(state->search_rom);
630
631 /* Verify CRC */
632 if (!star_bus_onewire_verify_rom(rom_codes[*count])) {
633 state->stats.crc_errors++;
634 continue;
635 }
636
637 (*count)++;
638
639 state->last_discrepancy = last_zero;
640 if (state->last_discrepancy == 0) {
641 state->last_device_flag = true;
642 }
643 }
644
645 if (state->last_device_flag || state->last_discrepancy == 0) {
646 break;
647 }
648 }
649
650 state->stats.devices_found = *count;
651
652 ESP_LOGI(s_TAG, "Found %d device(s) on bus '%s'", *count, bus_name);
653
654 return ESP_OK;
655}

687uint16_t star_bus_onewire_crc16(const uint8_t* data, size_t length)
688{
689 uint16_t crc = 0;
690
691 for (size_t i = 0; i < length; i++) {
692 uint8_t x = (crc ^ data[i]) & 0xFF;
693 crc = (crc >> 8) ^ ((x ^ (x << 4)) << 8) ^ (x << 3) ^ (x >> 4);
694 }
695
696 return crc;
697}

717esp_err_t star_bus_onewire_get_stats(const star_bus_manager_t* manager,
