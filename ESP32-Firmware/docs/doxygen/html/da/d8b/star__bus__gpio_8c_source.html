21static esp_err_t gpio_configure_default(const star_bus_config_t* config,
22 gpio_num_t pin,
23 gpio_mode_t mode,
24 gpio_pull_mode_t pull)
25{
26 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config is NULL");
27 ESP_RETURN_ON_FALSE(config->type == k_star_bus_type_gpio,
28 ESP_ERR_INVALID_ARG,
29 s_TAG,
30 "Config is not GPIO type");
31 ESP_RETURN_ON_FALSE(config->initialized,
32 ESP_ERR_INVALID_STATE,
33 s_TAG,
34 "GPIO bus '%s' not initialized",
35 config->name);
36
37 /* Verify pin is managed by this bus */
38 const star_gpio_bus_config_t* gpio_cfg = &config->proto.gpio;
39 bool pin_found = false;
40 for (uint8_t i = 0; i < gpio_cfg->pin_count; ++i) {
41 if (gpio_cfg->pins[i] == pin) {
42 pin_found = true;
43 break;
44 }
45 }
46
47 ESP_RETURN_ON_FALSE(pin_found,
48 ESP_ERR_INVALID_ARG,
49 s_TAG,
50 "Pin %d not managed by GPIO bus '%s'",
51 pin,
52 config->name);
53
54 /* Configure GPIO */
55 gpio_config_t io_conf = {
56 .pin_bit_mask = (1ULL << pin),
57 .mode = mode,
58 .pull_up_en = (pull == GPIO_PULLUP_ONLY || pull == GPIO_PULLUP_PULLDOWN) ? GPIO_PULLUP_ENABLE
59 : GPIO_PULLUP_DISABLE,
60 .pull_down_en = (pull == GPIO_PULLDOWN_ONLY || pull == GPIO_PULLUP_PULLDOWN)
61 ? GPIO_PULLDOWN_ENABLE
62 : GPIO_PULLDOWN_DISABLE,
63 .intr_type = GPIO_INTR_DISABLE, /* Set via set_interrupt function */
64 };
65
66 esp_err_t ret = gpio_config(&io_conf);
67 ESP_RETURN_ON_ERROR(ret, s_TAG, "Failed to configure GPIO pin %d on bus '%s'", pin, config->name);
68
69 ESP_LOGD(s_TAG,
70 "Configured GPIO pin %d on bus '%s' (mode=%d, pull=%d)",
71 pin,
72 config->name,
73 mode,
74 pull);
75 return ESP_OK;
76}

175static esp_err_t gpio_set_interrupt_default(const star_bus_config_t* config,
176 gpio_num_t pin,
177 gpio_int_type_t intr_type,
178 gpio_isr_t isr_handler,
179 void* args)
180{
181 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config is NULL");
182 ESP_RETURN_ON_FALSE(config->type == k_star_bus_type_gpio,
183 ESP_ERR_INVALID_ARG,
184 s_TAG,
185 "Config is not GPIO type");
186 ESP_RETURN_ON_FALSE(config->initialized,
187 ESP_ERR_INVALID_STATE,
188 s_TAG,
189 "GPIO bus '%s' not initialized",
190 config->name);
191
192 /* Verify pin is managed by this bus */
193 star_gpio_bus_config_t* gpio_cfg = (star_gpio_bus_config_t*)&config->proto.gpio;
194 bool pin_found = false;
195 for (uint8_t i = 0; i < gpio_cfg->pin_count; ++i) {
196 if (gpio_cfg->pins[i] == pin) {
197 pin_found = true;
198 break;
199 }
200 }
201
202 ESP_RETURN_ON_FALSE(pin_found,
203 ESP_ERR_INVALID_ARG,
204 s_TAG,
205 "Pin %d not managed by GPIO bus '%s'",
206 pin,
207 config->name);
208
209 esp_err_t ret = ESP_OK;
210
211 /* Install ISR service if not already installed */
212 if (!gpio_cfg->isr_installed) {
213 ret = gpio_install_isr_service(0);
214 if (ret == ESP_OK || ret == ESP_ERR_INVALID_STATE) {
215 /* ESP_ERR_INVALID_STATE means service already installed (by another bus or component) */
216 gpio_cfg->isr_installed = true;
217 ret = ESP_OK;
218 ESP_LOGD(s_TAG, "GPIO ISR service installed for bus '%s'", config->name);
219 } else {
220 ESP_RETURN_ON_ERROR(ret,
221 s_TAG,
222 "Failed to install GPIO ISR service for bus '%s'",
223 config->name);
224 }
225 }
226
227 /* Set interrupt type */
228 ret = gpio_set_intr_type(pin, intr_type);
229 ESP_RETURN_ON_ERROR(ret,
230 s_TAG,
231 "Failed to set interrupt type for GPIO pin %d on bus '%s'",
232 pin,
233 config->name);
234
235 /* Remove existing ISR handler if setting to disable */
236 if (intr_type == GPIO_INTR_DISABLE || isr_handler == NULL) {
237 ret = gpio_isr_handler_remove(pin);
238 /* Ignore error if handler doesn't exist */
239 if (ret != ESP_OK && ret != ESP_ERR_INVALID_STATE) {
240 ESP_LOGW(s_TAG, "Failed to remove ISR handler for pin %d: %s", pin, esp_err_to_name(ret));
241 }
242 ESP_LOGD(s_TAG, "Disabled interrupt for GPIO pin %d on bus '%s'", pin, config->name);
243 return ESP_OK;
244 }
245
246 /* Add ISR handler */
247 ret = gpio_isr_handler_add(pin, isr_handler, args);
248 ESP_RETURN_ON_ERROR(ret,
249 s_TAG,
250 "Failed to add ISR handler for GPIO pin %d on bus '%s'",
251 pin,
252 config->name);
253
254 ESP_LOGD(s_TAG,
255 "Set interrupt (type=%d) for GPIO pin %d on bus '%s'",
256 intr_type,
257 pin,
258 config->name);
259 return ESP_OK;
260}
