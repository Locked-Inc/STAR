55static peripheral_state_t* get_peripheral_state(const char* bus_name, bool create)
56{
57 /* Search for existing state */
58 for (uint8_t i = 0; i < g_num_peripheral_states; i++) {
59 if (strcmp(g_peripheral_states[i].bus_name, bus_name) == 0) {
60 return &g_peripheral_states[i];
61 }
62 }
63
64 /* Create new state if requested and space available */
65 if (create && g_num_peripheral_states < MAX_PERIPHERAL_BUSES) {
66 peripheral_state_t* state = &g_peripheral_states[g_num_peripheral_states];
67 memset(state, 0, sizeof(peripheral_state_t));
68 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
69 g_num_peripheral_states++;
70 return state;
71 }
72
73 return NULL;
74}

100static void internal_peripheral_task(void* param)
101{
102 peripheral_state_t* state = (peripheral_state_t*)param;
103
104 ESP_LOGI(s_TAG, "Peripheral task started for bus '%s'", state->bus_name);
105
106 while (state->task_running) {
107 /* Read from I2C slave buffer */
108 int32_t len = i2c_slave_read_buffer(state->port,
109 state->rx_buffer,
110 state->config.rx_buffer_size,
111 pdMS_TO_TICKS(100));
112
113 if (len > 0) {
114 /* Master wrote data to us */
115 state->stats.total_write_requests++;
116 state->stats.total_bytes_received += len;
117
118 /* Store last write data */
119 if (state->last_write_data != NULL && len <= state->config.rx_buffer_size) {
120 memcpy(state->last_write_data, state->rx_buffer, len);
121 state->last_write_length = len;
122 }
123
124 /* Call write callback */
125 if (state->config.on_write != NULL) {
126 uint32_t start_time = esp_timer_get_time();
127 esp_err_t result = state->config.on_write(state->rx_buffer, len, state->config.context);
128 uint32_t elapsed = (uint32_t)(esp_timer_get_time() - start_time);
129 state->stats.last_request_time_us = elapsed;
130
131 if (result != ESP_OK) {
132 state->stats.failed_writes++;
133 state->stats.callback_errors++;
134 state->stats.last_error = result;
135
136 if (state->config.on_error != NULL) {
137 state->config.on_error(result, state->config.context);
138 }
139 }
140 }
141 }
142
143 /* Check if master is reading from us */
144 /* Note: For ESP32 I2C slave, we pre-load data into the tx buffer
145 * The driver will send this data when master requests it */
146 if (state->tx_data_length > 0) {
147 /* Data is already loaded, just track stats when it gets sent */
148 /* ESP32 I2C slave driver doesn't provide easy way to detect when
149 * data was actually sent, so we estimate based on buffer state */
150 }
151
152 /* Small delay to prevent task from hogging CPU */
153 vTaskDelay(pdMS_TO_TICKS(10));
154 }
155
156 ESP_LOGI(s_TAG, "Peripheral task stopped for bus '%s'", state->bus_name);
157 vTaskDelete(NULL);
158}

163static esp_err_t internal_start_peripheral_task(peripheral_state_t* state)
164{
165 if (state->task_running) {
166 return ESP_OK; /* Already running */
167 }
168
169 state->task_running = true;
170
171 BaseType_t result =
172 xTaskCreate(internal_peripheral_task, "i2c_periph", 4096, state, 5, &state->task_handle);
173
174 if (result != pdPASS) {
175 state->task_running = false;
176 return ESP_ERR_NO_MEM;
177 }
178
179 return ESP_OK;
180}

185static void internal_stop_peripheral_task(peripheral_state_t* state)
186{
187 if (!state->task_running) {
188 return;
189 }
190
191 state->task_running = false;
192
193 /* Wait for task to exit */
194 if (state->task_handle != NULL) {
195 /* Task will delete itself */
196 vTaskDelay(pdMS_TO_TICKS(200)); /* Give time to exit */
197 state->task_handle = NULL;
198 }
199}

203esp_err_t star_bus_i2c_peripheral_enable(star_bus_manager_t* manager,
204 const char* bus_name,
205 const star_i2c_peripheral_config_t* config)
206{
207 if (manager == NULL || bus_name == NULL || config == NULL) {
208 ESP_LOGE(s_TAG, "Invalid parameters");
209 return ESP_ERR_INVALID_ARG;
210 }
211
212 if (config->on_read == NULL || config->on_write == NULL) {
213 ESP_LOGE(s_TAG, "Read and write callbacks are required");
214 return ESP_ERR_INVALID_ARG;
215 }
216
217 /* Validate address */
218 if (config->addr_mode == STAR_I2C_ADDR_7BIT) {
219 if (!STAR_I2C_ADDR_7BIT_VALID(config->address)) {
220 ESP_LOGE(s_TAG, "Invalid 7-bit address: 0x%02x", config->address);
221 return ESP_ERR_INVALID_ARG;
222 }
223 } else {
224 if (!STAR_I2C_ADDR_10BIT_VALID(config->address)) {
225 ESP_LOGE(s_TAG, "Invalid 10-bit address: 0x%03x", config->address);
226 return ESP_ERR_INVALID_ARG;
227 }
228 }
229
230 /* Get or create peripheral state */
231 peripheral_state_t* state = get_peripheral_state(bus_name, true);
232 if (state == NULL) {
233 ESP_LOGE(s_TAG, "Failed to create peripheral state for bus '%s'", bus_name);
234 return ESP_ERR_NO_MEM;
235 }
236
237 /* Get I2C port number */
238 esp_err_t result = internal_get_i2c_port(bus_name, &state->port);
239 if (result != ESP_OK) {
240 ESP_LOGE(s_TAG, "Failed to get I2C port for bus '%s'", bus_name);
241 return result;
242 }
243
244 /* Store configuration */
245 memcpy(&state->config, config, sizeof(star_i2c_peripheral_config_t));
246
247 /* Allocate buffers */
248 state->rx_buffer = malloc(config->rx_buffer_size);
249 state->tx_buffer = malloc(config->tx_buffer_size);
250 state->last_write_data = malloc(config->rx_buffer_size);
251
252 if (state->rx_buffer == NULL || state->tx_buffer == NULL || state->last_write_data == NULL) {
253 ESP_LOGE(s_TAG, "Failed to allocate buffers");
254 free(state->rx_buffer);
255 free(state->tx_buffer);
256 free(state->last_write_data);
257 return ESP_ERR_NO_MEM;
258 }
259
260 memset(state->rx_buffer, 0, config->rx_buffer_size);
261 memset(state->tx_buffer, 0, config->tx_buffer_size);
262 memset(state->last_write_data, 0, config->rx_buffer_size);
263 state->tx_data_length = 0;
264 state->last_write_length = 0;
265
266 /* Configure I2C peripheral mode */
267 i2c_config_t i2c_config = {
268 .mode = I2C_MODE_SLAVE,
269 .sda_pullup_en = GPIO_PULLUP_ENABLE,
270 .scl_pullup_en = GPIO_PULLUP_ENABLE,
271 .slave.addr_10bit_en = config->enable_10bit_addr ? 1 : 0,
272 .slave.slave_addr = config->address,
273 .slave.maximum_speed = 100000, /* 100 kHz default */
274 };
275
276 /* Note: SDA and SCL pins must already be configured in master mode setup */
277 /* This function only changes the mode to slave */
278
279 result = i2c_param_config(state->port, &i2c_config);
280 if (result != ESP_OK) {
281 ESP_LOGE(s_TAG, "Failed to configure I2C peripheral: %s", esp_err_to_name(result));
282 free(state->rx_buffer);
283 free(state->tx_buffer);
284 free(state->last_write_data);
285 return result;
286 }
287
288 /* Install I2C driver in slave mode */
289 result = i2c_driver_install(state->port,
290 I2C_MODE_SLAVE,
291 config->rx_buffer_size,
292 config->tx_buffer_size,
293 0);
294
295 if (result != ESP_OK) {
296 ESP_LOGE(s_TAG, "Failed to install I2C peripheral driver: %s", esp_err_to_name(result));
297 free(state->rx_buffer);
298 free(state->tx_buffer);
299 free(state->last_write_data);
300 return result;
301 }
302
303 /* Start peripheral task */
304 result = internal_start_peripheral_task(state);
305 if (result != ESP_OK) {
306 ESP_LOGE(s_TAG, "Failed to start peripheral task");
307 i2c_driver_delete(state->port);
308 free(state->rx_buffer);
309 free(state->tx_buffer);
310 free(state->last_write_data);
311 return result;
312 }
313
314 state->enabled = true;
315 memset(&state->stats, 0, sizeof(star_i2c_peripheral_stats_t));
316
317 ESP_LOGI(s_TAG,
318 "I2C peripheral enabled on bus '%s' at address 0x%02x",
319 bus_name,
320 config->address);
321
322 return ESP_OK;
323}

325esp_err_t star_bus_i2c_peripheral_disable(star_bus_manager_t* manager, const char* bus_name)
326{
327 if (manager == NULL || bus_name == NULL) {
328 return ESP_ERR_INVALID_ARG;
329 }
330
331 peripheral_state_t* state = get_peripheral_state(bus_name, false);
332 if (state == NULL || !state->enabled) {
333 return ESP_ERR_NOT_FOUND;
334 }
335
336 /* Stop peripheral task */
337 internal_stop_peripheral_task(state);
338
339 /* Delete I2C driver */
340 i2c_driver_delete(state->port);
341
342 /* Free buffers */
343 free(state->rx_buffer);
344 free(state->tx_buffer);
345 free(state->last_write_data);
346
347 state->rx_buffer = NULL;
348 state->tx_buffer = NULL;
349 state->last_write_data = NULL;
350 state->enabled = false;
351
352 ESP_LOGI(s_TAG, "I2C peripheral disabled on bus '%s'", bus_name);
353
354 return ESP_OK;
355}

375esp_err_t star_bus_i2c_peripheral_set_response(star_bus_manager_t* manager,
376 const char* bus_name,
377 const uint8_t* data,
378 size_t length)
379{
380 if (manager == NULL || bus_name == NULL || data == NULL) {
381 return ESP_ERR_INVALID_ARG;
382 }
383
384 peripheral_state_t* state = get_peripheral_state(bus_name, false);
385 if (state == NULL || !state->enabled) {
386 return ESP_ERR_NOT_FOUND;
387 }
388
389 if (length > state->config.tx_buffer_size) {
390 ESP_LOGE(s_TAG, "Response data too large: %d > %d", length, state->config.tx_buffer_size);
391 return ESP_ERR_INVALID_SIZE;
392 }
393
394 /* Copy data to tx buffer */
395 memcpy(state->tx_buffer, data, length);
396 state->tx_data_length = length;
397
398 /* Write to I2C slave tx buffer */
399 int32_t written =
400 i2c_slave_write_buffer(state->port, data, length, pdMS_TO_TICKS(state->config.timeout_ms));
401
402 if (written < 0) {
403 ESP_LOGE(s_TAG, "Failed to write to slave tx buffer");
404 return ESP_FAIL;
405 }
406
407 state->stats.total_read_requests++;
408 state->stats.total_bytes_sent += written;
409
410 return ESP_OK;
411}

413esp_err_t star_bus_i2c_peripheral_get_last_write(const star_bus_manager_t* manager,
414 const char* bus_name,
415 uint8_t* data,
416 size_t* length)
417{
418 if (manager == NULL || bus_name == NULL || data == NULL || length == NULL) {
419 return ESP_ERR_INVALID_ARG;
420 }
421
422 peripheral_state_t* state = get_peripheral_state(bus_name, false);
423 if (state == NULL || !state->enabled) {
424 return ESP_ERR_NOT_FOUND;
425 }
426
427 size_t copy_len = state->last_write_length < *length ? state->last_write_length : *length;
428
429 if (copy_len > 0 && state->last_write_data != NULL) {
430 memcpy(data, state->last_write_data, copy_len);
431 }
432
433 *length = copy_len;
434
435 return ESP_OK;
436}
