39static esp_err_t internal_execute_operation(batch_context_t* ctx, uint32_t index)
40{
41 if (ctx == NULL || index >= ctx->operation_count) {
42 return ESP_ERR_INVALID_ARG;
43 }
44
45 star_batch_operation_t* op = &ctx->operations[index];
46 esp_err_t result = ESP_FAIL;
47
48 switch (op->type) {
49 case STAR_BATCH_OP_I2C_WRITE:
50 result = star_bus_i2c_write(ctx->manager,
51 op->bus_name,
52 op->params.i2c.data,
53 op->params.i2c.length,
54 op->params.i2c.command,
55 NULL);
56 break;
57
58 case STAR_BATCH_OP_I2C_READ:
59 result = star_bus_i2c_read(ctx->manager,
60 op->bus_name,
61 op->params.i2c.data,
62 op->params.i2c.length,
63 op->params.i2c.command,
64 NULL);
65 break;
66
67 case STAR_BATCH_OP_SPI_TRANSMIT:
68 result = star_bus_spi_transmit(ctx->manager,
69 op->bus_name,
70 op->params.spi.tx_data,
71 op->params.spi.length,
72 op->params.spi.flags);
73 break;
74
75 case STAR_BATCH_OP_SPI_RECEIVE:
76 result = star_bus_spi_receive(ctx->manager,
77 op->bus_name,
78 op->params.spi.rx_data,
79 op->params.spi.length,
80 op->params.spi.flags);
81 break;
82
83 case STAR_BATCH_OP_SPI_TRANSFER:
84 result = star_bus_spi_transfer(ctx->manager,
85 op->bus_name,
86 op->params.spi.tx_data,
87 op->params.spi.rx_data,
88 op->params.spi.length,
89 op->params.spi.flags);
90 break;
91
92 case STAR_BATCH_OP_SMBUS_READ_BYTE:
93 result = star_smbus_read_byte(ctx->manager,
94 op->bus_name,
95 op->params.smbus.addr,
96 op->params.smbus.command,
97 op->params.smbus.data);
98 break;
99
100 case STAR_BATCH_OP_SMBUS_WRITE_BYTE:
101 result = star_smbus_write_byte(ctx->manager,
102 op->bus_name,
103 op->params.smbus.addr,
104 op->params.smbus.command,
105 *op->params.smbus.data);
106 break;
107
108 case STAR_BATCH_OP_DELAY:
109 vTaskDelay(pdMS_TO_TICKS(op->params.delay.delay_ms));
110 result = ESP_OK;
111 break;
112
113 default:
114 ESP_LOGE(s_TAG, "Unknown operation type: %d", op->type);
115 result = ESP_ERR_NOT_SUPPORTED;
116 break;
117 }
118
119 op->result = result;
120 return result;
121}

125esp_err_t star_batch_create(star_bus_manager_t* manager,
126 const star_batch_config_t* config,
127 star_batch_handle_t* handle)
128{
129 if (manager == NULL || config == NULL || handle == NULL) {
130 ESP_LOGE(s_TAG, "Invalid parameters");
131 return ESP_ERR_INVALID_ARG;
132 }
133
134 batch_context_t* ctx = (batch_context_t*)malloc(sizeof(batch_context_t));
135 if (ctx == NULL) {
136 ESP_LOGE(s_TAG, "Failed to allocate batch context");
137 return ESP_ERR_NO_MEM;
138 }
139
140 memset(ctx, 0, sizeof(batch_context_t));
141 ctx->manager = manager;
142 memcpy(&ctx->config, config, sizeof(star_batch_config_t));
143
144 *handle = (star_batch_handle_t)ctx;
145
146 ESP_LOGD(s_TAG,
147 "Batch created: mode=%d, timeout=%" PRIu32 " ms",
148 config->mode,
149 config->timeout_ms);
150
151 return ESP_OK;
152}

154void star_batch_free(star_batch_handle_t handle)
155{
156 if (handle != NULL) {
157 batch_context_t* ctx = (batch_context_t*)handle;
158
159 /* Free any allocated SMBus write data */
160 for (uint32_t i = 0; i < ctx->operation_count; i++) {
161 star_batch_operation_t* op = &ctx->operations[i];
162 if (op->type == STAR_BATCH_OP_SMBUS_WRITE_BYTE && op->params.smbus.data != NULL) {
163 free(op->params.smbus.data);
164 op->params.smbus.data = NULL;
165 }
166 }
167
168 free(ctx);
169 }

421esp_err_t star_batch_execute(star_batch_handle_t handle, star_batch_stats_t* stats)
422{
423 if (handle == NULL) {
424 return ESP_ERR_INVALID_ARG;
425 }
426
427 batch_context_t* ctx = (batch_context_t*)handle;
428
429 if (ctx->operation_count == 0) {
430 ESP_LOGW(s_TAG, "Batch is empty");
431 return ESP_ERR_INVALID_STATE;
432 }
433
434 if (ctx->executed) {
435 ESP_LOGW(s_TAG, "Batch already executed");
436 return ESP_ERR_INVALID_STATE;
437 }
438
439 ESP_LOGI(s_TAG, "Executing batch with %" PRIu32 " operations", ctx->operation_count);
440
441 ctx->start_tick = xTaskGetTickCount();
442 esp_err_t first_error = ESP_OK;
443 uint32_t succeeded = 0;
444 uint32_t failed = 0;
445
446 /* Execute operations sequentially */
447 for (uint32_t i = 0; i < ctx->operation_count; i++) {
448 esp_err_t result = internal_execute_operation(ctx, i);
449
450 if (result == ESP_OK) {
451 succeeded++;
452 } else {
453 failed++;
454 ESP_LOGE(s_TAG, "Operation %" PRIu32 " failed: %s", i, esp_err_to_name(result));
455
456 if (first_error == ESP_OK) {
457 first_error = result;
458 }
459
460 if (ctx->config.stop_on_error) {
461 ESP_LOGW(s_TAG, "Stopping batch execution on error");
462 break;
463 }
464 }
465
466 /* Check timeout */
467 if (ctx->config.timeout_ms > 0) {
468 TickType_t elapsed = xTaskGetTickCount() - ctx->start_tick;
469 if (pdTICKS_TO_MS(elapsed) >= ctx->config.timeout_ms) {
470 ESP_LOGE(s_TAG, "Batch execution timeout");
471 first_error = ESP_ERR_TIMEOUT;
472 break;
473 }
474 }
475 }
476
477 ctx->end_tick = xTaskGetTickCount();
478 ctx->executed = true;
479
480 /* Fill in statistics if requested */
481 if (stats != NULL) {
482 stats->operations_executed = succeeded + failed;
483 stats->operations_succeeded = succeeded;
484 stats->operations_failed = failed;
485 stats->total_time_ms = pdTICKS_TO_MS(ctx->end_tick - ctx->start_tick);
486 }
487
488 ESP_LOGI(s_TAG,
489 "Batch execution complete: %" PRIu32 " succeeded, %" PRIu32 " failed",
490 succeeded,
491 failed);
492
493 return first_error;
494}
