39esp_err_t shared_data_init(void)
40{
41 if (s_g_context_initialized) {
42 ESP_LOGI(s_TAG, "Shared data already initialized");
43 return ESP_OK;
44 }
45
46 init_shared_context(&s_g_global_context);
47
48 s_g_global_context.sensor_data_queue = xQueueCreate(SENSOR_DATA_QUEUE_SIZE, sizeof(sensor_data_t));
49 if (s_g_global_context.sensor_data_queue == NULL) {
50 ESP_LOGE(s_TAG, "Failed to create sensor data queue");
51 return ESP_ERR_NO_MEM;
52 }
53
54 s_g_global_context.temperature_data_queue = xQueueCreate(TEMPERATURE_QUEUE_SIZE, sizeof(temperature_data_t));
55 if (s_g_global_context.temperature_data_queue == NULL) {
56 ESP_LOGE(s_TAG, "Failed to create temperature data queue");
57 vQueueDelete(s_g_global_context.sensor_data_queue);
58 return ESP_ERR_NO_MEM;
59 }
60
61 s_g_global_context.health_mutex = xSemaphoreCreateMutex();
62 if (s_g_global_context.health_mutex == NULL) {
63 ESP_LOGE(s_TAG, "Failed to create health mutex");
64 vQueueDelete(s_g_global_context.sensor_data_queue);
65 vQueueDelete(s_g_global_context.temperature_data_queue);
66 return ESP_ERR_NO_MEM;
67 }
68
69 s_g_global_context.state_mutex = xSemaphoreCreateMutex();
70 if (s_g_global_context.state_mutex == NULL) {
71 ESP_LOGE(s_TAG, "Failed to create state mutex");
72 vQueueDelete(s_g_global_context.sensor_data_queue);
73 vQueueDelete(s_g_global_context.temperature_data_queue);
74 vSemaphoreDelete(s_g_global_context.health_mutex);
75 return ESP_ERR_NO_MEM;
76 }
77
78 s_g_init_timestamp = esp_timer_get_time();
79 s_g_context_initialized = true;
80
81 ESP_LOGI(s_TAG, "Shared data context initialized successfully");
82 return ESP_OK;
83}

123void shared_data_update_health(uint8_t sensor_index, bool success)
124{
125 if (!s_g_context_initialized) {
126 return;
127 }
128
129 if (xSemaphoreTake(s_g_global_context.health_mutex, pdMS_TO_TICKS(MUTEX_TIMEOUT_MS)) == pdTRUE) {
130 if (sensor_index < NUM_HCSR04) {
131 s_g_global_context.health_data.total_sensor_reads++;
132 if (!success) {
133 s_g_global_context.health_data.sensor_read_failures[sensor_index]++;
134 }
135 } else if (sensor_index == NUM_HCSR04) {
136 s_g_global_context.health_data.total_temperature_reads++;
137 if (!success) {
138 s_g_global_context.health_data.temperature_read_failures++;
139 }
140 }
141
142 s_g_global_context.health_data.uptime_ms = shared_data_get_uptime_ms();
143
144 xSemaphoreGive(s_g_global_context.health_mutex);
145 }
