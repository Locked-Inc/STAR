40esp_err_t shared_data_init(void)
41{
42 if (s_g_context_initialized) {
43 ESP_LOGI(s_TAG, "Shared data already initialized");
44 return ESP_OK;
45 }
46
47 init_shared_context(&s_g_global_context);
48
49 s_g_global_context.sensor_data_queue =
50 xQueueCreate(STAR_SYSTEM_CONFIG.freertos.sensor_data_queue_size, sizeof(sensor_data_t));
51 if (s_g_global_context.sensor_data_queue == NULL) {
52 ESP_LOGE(s_TAG, "Failed to create sensor data queue");
53 return ESP_ERR_NO_MEM;
54 }
55
56 s_g_global_context.temperature_data_queue =
57 xQueueCreate(STAR_SYSTEM_CONFIG.freertos.temperature_queue_size, sizeof(temperature_data_t));
58 if (s_g_global_context.temperature_data_queue == NULL) {
59 ESP_LOGE(s_TAG, "Failed to create temperature data queue");
60 vQueueDelete(s_g_global_context.sensor_data_queue);
61 return ESP_ERR_NO_MEM;
62 }
63
64 s_g_global_context.health_mutex = xSemaphoreCreateMutex();
65 if (s_g_global_context.health_mutex == NULL) {
66 ESP_LOGE(s_TAG, "Failed to create health mutex");
67 vQueueDelete(s_g_global_context.sensor_data_queue);
68 vQueueDelete(s_g_global_context.temperature_data_queue);
69 return ESP_ERR_NO_MEM;
70 }
71
72 s_g_global_context.state_mutex = xSemaphoreCreateMutex();
73 if (s_g_global_context.state_mutex == NULL) {
74 ESP_LOGE(s_TAG, "Failed to create state mutex");
75 vQueueDelete(s_g_global_context.sensor_data_queue);
76 vQueueDelete(s_g_global_context.temperature_data_queue);
77 vSemaphoreDelete(s_g_global_context.health_mutex);
78 return ESP_ERR_NO_MEM;
79 }
80
81 s_g_init_timestamp = esp_timer_get_time();
82 s_g_context_initialized = true;
83
84 ESP_LOGI(s_TAG, "Shared data context initialized successfully");
85 return ESP_OK;
86}

138void shared_data_update_health(const uint8_t sensor_index, const bool success)
139{
140 if (!s_g_context_initialized) {
141 return;
142 }
143
144 if (xSemaphoreTake(s_g_global_context.health_mutex, pdMS_TO_TICKS(STAR_SYSTEM_CONFIG.freertos.mutex_timeout_ms)) == pdTRUE) {
145 if (sensor_index < STAR_SYSTEM_CONFIG.num_hcsr04_sensors) {
146 s_g_global_context.health_data.total_sensor_reads++;
147 if (!success) {
148 s_g_global_context.health_data.sensor_read_failures[sensor_index]++;
149 }
150 } else if (sensor_index == STAR_SYSTEM_CONFIG.num_hcsr04_sensors) {
151 s_g_global_context.health_data.total_temperature_reads++;
152 if (!success) {
153 s_g_global_context.health_data.temperature_read_failures++;
154 }
155 }
156
157 s_g_global_context.health_data.uptime_ms = shared_data_get_uptime_ms();
158
159 xSemaphoreGive(s_g_global_context.health_mutex);
160 }
161}
