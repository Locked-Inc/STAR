36static void get_accel_sensitivity(mpu6050_accel_range_t range, float* sensitivity)
37{
38 switch (range) {

76esp_err_t star_sensor_mpu6050_init(mpu6050_handle_t* handle,
77 star_bus_manager_t* manager,
78 const char* bus_name,
79 star_error_interface_t* error_iface,
80 const mpu6050_config_t* config)
81{
82 if (handle == NULL || manager == NULL || bus_name == NULL || config == NULL) {
83 ESP_LOGE(s_TAG, "NULL parameter in init");
84 return ESP_ERR_INVALID_ARG;
85 }
86
87 memset(handle, 0, sizeof(mpu6050_handle_t));
88 handle->manager = manager;
89 handle->bus_name = bus_name;
90 handle->i2c_addr = config->i2c_addr;
91 memcpy(&handle->config, config, sizeof(mpu6050_config_t));
92
93 get_accel_sensitivity(config->accel_range, &handle->accel_sensitivity);
94 get_gyro_sensitivity(config->gyro_range, &handle->gyro_sensitivity);
95
96 // Create mutex first
97 handle->mutex = xSemaphoreCreateMutex();
98 if (handle->mutex == NULL) {
99 ESP_LOGE(s_TAG, "Failed to create mutex");
100 return ESP_ERR_NO_MEM;
101 }
102
103 // Handle error interface injection
104 if (error_iface == NULL) {
105 handle->error_iface = star_error_interface_create_default();
106 if (handle->error_iface == NULL) {
107 ESP_LOGE(s_TAG, "Failed to create default error interface");
108 vSemaphoreDelete(handle->mutex);
109 handle->mutex = NULL;
110 return ESP_ERR_NO_MEM;
111 }
112 handle->owns_error_handler = true;
113 ESP_LOGI(s_TAG, "Created default error handler internally");
114 } else {
115 handle->error_iface = error_iface;
116 handle->owns_error_handler = false;
117 ESP_LOGI(s_TAG, "Using injected error interface");
118 }
119
120 ESP_LOGI(s_TAG, "Initializing MPU6050 on bus '%s' at address 0x%02X", bus_name, config->i2c_addr);
121
122 uint8_t who_am_i;
123 esp_err_t ret =
124 mpu6050_read_reg(manager, bus_name, config->i2c_addr, MPU6050_REG_WHO_AM_I, &who_am_i);
125 if (ret != ESP_OK || who_am_i != MPU6050_WHO_AM_I_VAL) {
126 ESP_LOGE(s_TAG, "Failed to detect MPU6050 (who_am_i=0x%02X)", who_am_i);
127 if (handle->owns_error_handler) {
128 star_error_interface_destroy(handle->error_iface);
129 }
130 vSemaphoreDelete(handle->mutex);
131 handle->mutex = NULL;
132 return ESP_ERR_NOT_FOUND;
133 }
134
135 ret = star_sensor_mpu6050_reset(handle);
136 if (ret != ESP_OK) {
137 ESP_LOGE(s_TAG, "Failed to reset: %s", esp_err_to_name(ret));
138 if (handle->owns_error_handler) {
139 star_error_interface_destroy(handle->error_iface);
140 }
141 vSemaphoreDelete(handle->mutex);
142 handle->mutex = NULL;
143 return ret;
144 }
145
146 vTaskDelay(pdMS_TO_TICKS(100));
147
148 ret = mpu6050_write_reg(manager, bus_name, config->i2c_addr, MPU6050_REG_PWR_MGMT_1, 0x01);
149 if (ret != ESP_OK) {
150 ESP_LOGE(s_TAG, "Failed to set clock source: %s", esp_err_to_name(ret));
151 if (handle->owns_error_handler) {
152 star_error_interface_destroy(handle->error_iface);
153 }
154 vSemaphoreDelete(handle->mutex);
155 handle->mutex = NULL;
156 return ret;
157 }
158
159 uint8_t gyro_config = (config->gyro_range & 0x03) << 3;
160 ret =
161 mpu6050_write_reg(manager, bus_name, config->i2c_addr, MPU6050_REG_GYRO_CONFIG, gyro_config);
162 if (ret != ESP_OK) {
163 ESP_LOGE(s_TAG, "Failed to configure gyro: %s", esp_err_to_name(ret));
164 if (handle->owns_error_handler) {
165 star_error_interface_destroy(handle->error_iface);
166 }
167 vSemaphoreDelete(handle->mutex);
168 handle->mutex = NULL;
169 return ret;
170 }
171
172 uint8_t accel_config = (config->accel_range & 0x03) << 3;
173 ret =
174 mpu6050_write_reg(manager, bus_name, config->i2c_addr, MPU6050_REG_ACCEL_CONFIG, accel_config);
175 if (ret != ESP_OK) {
176 ESP_LOGE(s_TAG, "Failed to configure accel: %s", esp_err_to_name(ret));
177 if (handle->owns_error_handler) {
178 star_error_interface_destroy(handle->error_iface);
179 }
180 vSemaphoreDelete(handle->mutex);
181 handle->mutex = NULL;
182 return ret;
183 }
184
185 ret =
186 mpu6050_write_reg(manager, bus_name, config->i2c_addr, MPU6050_REG_CONFIG, config->dlpf & 0x07);
187 if (ret != ESP_OK) {
188 ESP_LOGE(s_TAG, "Failed to configure DLPF: %s", esp_err_to_name(ret));
189 if (handle->owns_error_handler && handle->error_iface != NULL) {
190 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
191 error_handler_destroy_default(handler_ptr);
192 free(handle->error_iface);
193 }
194 vSemaphoreDelete(handle->mutex);
195 handle->mutex = NULL;
196 return ret;
197 }
198
199 ret = mpu6050_write_reg(manager,
200 bus_name,
201 config->i2c_addr,
202 MPU6050_REG_SMPLRT_DIV,
203 config->sample_rate_div);
204 if (ret != ESP_OK) {
205 ESP_LOGE(s_TAG, "Failed to configure sample rate: %s", esp_err_to_name(ret));
206 if (handle->owns_error_handler && handle->error_iface != NULL) {
207 error_handler_t* handler_ptr = (error_handler_t*)handle->error_iface->ctx;
208 error_handler_destroy_default(handler_ptr);
209 free(handle->error_iface);
210 }
211 vSemaphoreDelete(handle->mutex);
212 handle->mutex = NULL;
213 return ret;
214 }
215
216 if (config->enable_fifo) {
217 ret = star_sensor_mpu6050_fifo_enable(handle, true);
218 if (ret != ESP_OK) {
219 ESP_LOGW(s_TAG, "Failed to enable FIFO: %s", esp_err_to_name(ret));
220 }
221 }
222
223 handle->initialized = true;
224 ESP_LOGI(s_TAG, "MPU6050 initialization successful (thread-safe with mutex)");
225 return ESP_OK;
226}

228esp_err_t star_sensor_mpu6050_deinit(mpu6050_handle_t* handle)
229{
230 if (handle == NULL || !handle->initialized) {
231 return ESP_ERR_INVALID_ARG;
232 }
233
234 // Acquire mutex before cleanup
235 SemaphoreHandle_t mutex = handle->mutex;
236 if (mutex != NULL && xSemaphoreTake(mutex, pdMS_TO_TICKS(MPU6050_MUTEX_TIMEOUT_MS)) != pdTRUE) {
237 ESP_LOGW(s_TAG, "Failed to acquire mutex for deinit, continuing anyway");
238 }
239
240 // Put device to sleep (no separate mutex needed, we already hold it)
241 uint8_t pwr_mgmt;
242 if (mpu6050_read_reg(handle->manager,
243 handle->bus_name,
244 handle->i2c_addr,
245 MPU6050_REG_PWR_MGMT_1,
246 &pwr_mgmt) == ESP_OK) {
247 pwr_mgmt |= MPU6050_PWR1_SLEEP;
248 mpu6050_write_reg(handle->manager,
249 handle->bus_name,
250 handle->i2c_addr,
251 MPU6050_REG_PWR_MGMT_1,
252 pwr_mgmt);
253 }
254
255 // Clean up error handler if we own it
256 if (handle->owns_error_handler) {
257 star_error_interface_destroy(handle->error_iface);
258 handle->error_iface = NULL;
259 ESP_LOGI(s_TAG, "Destroyed internally-owned error handler");
260 }
261
262 // Clear handle fields
263 handle->initialized = false;
264 handle->mutex = NULL;
265
266 // Release and delete mutex
267 if (mutex != NULL) {
268 xSemaphoreGive(mutex);
269 vSemaphoreDelete(mutex);
270 }
271
272 return ESP_OK;
273}

599esp_err_t star_sensor_mpu6050_fifo_get_count(const mpu6050_handle_t* handle, uint16_t* count)
600{
601 if (handle == NULL || count == NULL || !handle->initialized) {
602 return ESP_ERR_INVALID_ARG;
603 }
604
605 // Acquire mutex for thread-safe FIFO count read
606 SemaphoreHandle_t mutex = handle->mutex;
607 if (mutex == NULL || xSemaphoreTake(mutex, pdMS_TO_TICKS(MPU6050_MUTEX_TIMEOUT_MS)) != pdTRUE) {
608 ESP_LOGE(s_TAG, "Failed to acquire mutex for fifo_get_count");
609 return ESP_ERR_TIMEOUT;
610 }
611
612 // Re-check after mutex acquisition
613 if (!handle->initialized || handle->mutex == NULL) {
614 xSemaphoreGive(mutex);
615 return ESP_ERR_INVALID_STATE;
616 }
617
618 uint8_t reg = MPU6050_REG_FIFO_COUNT_H;
619 uint8_t data[2];
620 esp_err_t ret = star_bus_i2c_read(handle->manager, handle->bus_name, data, 2, reg, NULL);
621 if (ret != ESP_OK) {
622 xSemaphoreGive(mutex);
623 return ret;
624 }
625
626 *count = ((uint16_t)data[0] << 8) | data[1];
627
628 if (*count > MPU6050_FIFO_SIZE) {
629 ESP_LOGW(s_TAG, "FIFO overflow detected: count=%d", *count);
630 xSemaphoreGive(mutex);
631 return ESP_ERR_INVALID_STATE;

638esp_err_t star_sensor_mpu6050_fifo_read(const mpu6050_handle_t* handle, uint8_t* data, uint16_t len)
639{
640 if (handle == NULL || data == NULL || !handle->initialized) {
641 return ESP_ERR_INVALID_ARG;
642 }
643
644 if (len > MPU6050_FIFO_SIZE) {
645 ESP_LOGE(s_TAG, "Read length %d exceeds FIFO size %d", len, MPU6050_FIFO_SIZE);
646 return ESP_ERR_INVALID_ARG;
647 }
648
649 // Acquire mutex for thread-safe FIFO read
650 SemaphoreHandle_t mutex = handle->mutex;
651 if (mutex == NULL || xSemaphoreTake(mutex, pdMS_TO_TICKS(MPU6050_MUTEX_TIMEOUT_MS)) != pdTRUE) {
652 ESP_LOGE(s_TAG, "Failed to acquire mutex for fifo_read");
653 return ESP_ERR_TIMEOUT;
654 }
655
656 // Re-check after mutex acquisition
657 if (!handle->initialized || handle->mutex == NULL) {
658 xSemaphoreGive(mutex);
659 return ESP_ERR_INVALID_STATE;
660 }
661
662 uint8_t reg = MPU6050_REG_FIFO_R_W;
663 esp_err_t ret = star_bus_i2c_read(handle->manager, handle->bus_name, data, len, reg, NULL);
664
665 xSemaphoreGive(mutex);
666 return ret;
667}

669esp_err_t star_sensor_mpu6050_set_sleep(mpu6050_handle_t* handle, bool sleep)
670{
671 if (handle == NULL || !handle->initialized) {
672 return ESP_ERR_INVALID_STATE;
673 }
674
675 // Acquire mutex
676 SemaphoreHandle_t mutex = handle->mutex;
677 if (mutex == NULL || xSemaphoreTake(mutex, pdMS_TO_TICKS(MPU6050_MUTEX_TIMEOUT_MS)) != pdTRUE) {
678 ESP_LOGE(s_TAG, "Failed to acquire mutex for set_sleep");
679 return ESP_ERR_TIMEOUT;
680 }
681
682 // Re-check after mutex acquisition
683 if (!handle->initialized || handle->mutex == NULL) {
684 xSemaphoreGive(mutex);
685 return ESP_ERR_INVALID_STATE;
686 }
687
688 uint8_t pwr_mgmt;
689 esp_err_t ret = mpu6050_read_reg(handle->manager,
690 handle->bus_name,
691 handle->i2c_addr,
692 MPU6050_REG_PWR_MGMT_1,
693 &pwr_mgmt);
694 if (ret != ESP_OK) {
695 xSemaphoreGive(mutex);
696 return ret;
697 }
698
699 if (sleep) {
700 pwr_mgmt |= MPU6050_PWR1_SLEEP;
701 } else {
702 pwr_mgmt &= ~MPU6050_PWR1_SLEEP;
703 }
704
705 ret = mpu6050_write_reg(handle->manager,
706 handle->bus_name,
707 handle->i2c_addr,
708 MPU6050_REG_PWR_MGMT_1,
709 pwr_mgmt);
710
711 xSemaphoreGive(mutex);
712 return ret;
