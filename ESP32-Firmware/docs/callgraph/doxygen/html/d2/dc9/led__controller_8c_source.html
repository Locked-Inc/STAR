104rgb_color_t led_controller_distance_to_color(float distance_cm)
105{
106 rgb_color_t color = {0.0f, 0.0f, 0.0f};
107
108 if (!VALIDATE_DISTANCE(distance_cm) || distance_cm > DISTANCE_VERY_FAR) {
109 return color;
110 }
111
112 float brightness_factor = 1.0f;
113
114 if (distance_cm < DISTANCE_VERY_CLOSE) {
115 color.red = 100.0f;
116 brightness_factor = 1.0f;
117 } else if (distance_cm < DISTANCE_CLOSE) {
118 float ratio = (distance_cm - DISTANCE_VERY_CLOSE) / (DISTANCE_CLOSE - DISTANCE_VERY_CLOSE);
119 color.red = 100.0f;
120 color.green = ratio * 100.0f;
121 brightness_factor = 1.0f - (ratio * 0.3f);
122 } else if (distance_cm < DISTANCE_MEDIUM) {
123 float ratio = (distance_cm - DISTANCE_CLOSE) / (DISTANCE_MEDIUM - DISTANCE_CLOSE);
124 color.red = (1.0f - ratio) * 100.0f;
125 color.green = 100.0f;
126 brightness_factor = 0.7f - (ratio * 0.2f);
127 } else if (distance_cm < DISTANCE_FAR) {
128 float ratio = (distance_cm - DISTANCE_MEDIUM) / (DISTANCE_FAR - DISTANCE_MEDIUM);
129 color.green = (1.0f - ratio) * 100.0f;
130 color.blue = ratio * 100.0f;
131 brightness_factor = 0.5f - (ratio * 0.2f);
132 } else if (distance_cm <= DISTANCE_VERY_FAR) {
133 color.blue = 100.0f;
134 float ratio = (distance_cm - DISTANCE_FAR) / (DISTANCE_VERY_FAR - DISTANCE_FAR);
135 brightness_factor = 0.3f - (ratio * 0.2f);
136 }
137
138 color.red *= brightness_factor;
139 color.green *= brightness_factor;
140 color.blue *= brightness_factor;
141
142 return color;
143}

164 esp_err_t ret = ESP_OK;
165 if (controller->initialized && controller->pca_handle != NULL) {
166 ret = internal_set_rgb_channels(controller->pca_handle, led_index, color);
167 } else {
168 ret = ESP_ERR_INVALID_STATE;
169 }
170
171 xSemaphoreGive(controller->mutex);
