79bool star_bus_validate_gpio_i2c(gpio_num_t pin)
80{
81 /* ESP32: Most GPIO pins support I2C except strapping pins */
82 if (pin < 0 || pin >= GPIO_NUM_MAX) {
83 return false;
84 }
85 /* Avoid GPIO 6-11 (connected to flash) and GPIO 0, 2 (strapping) */
86 if ((pin >= 6 && pin <= 11) || pin == 0 || pin == 2) {
87 return false;
88 }
89 return true;
90}

92bool star_bus_validate_gpio_spi(gpio_num_t pin)
93{
94 /* Same restrictions as I2C */
95 return star_bus_validate_gpio_i2c(pin);
96}

100esp_err_t star_bus_quick_setup_i2c(star_bus_manager_t* manager,
101 const char* bus_name,
102 uint8_t address,
103 gpio_num_t sda_pin,
104 gpio_num_t scl_pin)
105{
106 if (manager == NULL || bus_name == NULL) {
107 return ESP_ERR_INVALID_ARG;
108 }
109
110 star_bus_config_t* config = star_bus_config_i2c_fast(bus_name, address, sda_pin, scl_pin);
111 if (config == NULL) {
112 return ESP_ERR_NO_MEM;
113 }
114
115 esp_err_t result = star_bus_manager_add_bus(manager, config);
116
117 if (result == ESP_OK) {
118 ESP_LOGI(s_TAG,
119 "Quick I2C setup: '%s' addr=0x%02X on SDA=%d, SCL=%d",
120 bus_name,
121 address,
122 sda_pin,
123 scl_pin);
124 } else {
125 star_bus_config_destroy(config);
126 }
127
128 return result;
129}
