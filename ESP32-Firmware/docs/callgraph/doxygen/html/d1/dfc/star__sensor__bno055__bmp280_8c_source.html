264{

278 error_handler_destroy_default(error_handler);

298esp_err_t star_sensor_imu_10dof_read(const imu_10dof_handle_t* handle, imu_10dof_data_t* data)
299{
300 if (handle == NULL || data == NULL || !handle->initialized) {
301 return ESP_ERR_INVALID_ARG;
302 }
303
304 // Acquire mutex for thread-safe read
305 SemaphoreHandle_t mutex = handle->mutex;
306 if (mutex == NULL || xSemaphoreTake(mutex, pdMS_TO_TICKS(IMU_10DOF_MUTEX_TIMEOUT_MS)) != pdTRUE) {
307 ESP_LOGE(s_TAG, "Failed to acquire mutex for read");
308 return ESP_ERR_TIMEOUT;
309 }
310
311 // Re-check after mutex acquisition
312 if (!handle->initialized || handle->mutex == NULL) {
313 xSemaphoreGive(mutex);
314 return ESP_ERR_INVALID_STATE;
315 }
316
317 memset(data, 0, sizeof(imu_10dof_data_t));
318 esp_err_t ret;
319
320 // Read quaternion (8 bytes)
321 uint8_t quat_data[8];
322 ret = star_bus_i2c_read(handle->manager,
323 handle->bus_name,
324 quat_data,
325 8,
326 (uint8_t)k_bno055_reg_qua_data_w_lsb,
327 NULL);
328 if (ret == ESP_OK) {
329 data->quaternion.w = (int16_t)((quat_data[1] << 8) | quat_data[0]);
330 data->quaternion.x = (int16_t)((quat_data[3] << 8) | quat_data[2]);
331 data->quaternion.y = (int16_t)((quat_data[5] << 8) | quat_data[4]);
332 data->quaternion.z = (int16_t)((quat_data[7] << 8) | quat_data[6]);
333 }
334
335 // Read Euler angles (6 bytes)
336 uint8_t euler_data[6];
337 ret = star_bus_i2c_read(handle->manager,
338 handle->bus_name,
339 euler_data,
340 6,
341 (uint8_t)k_bno055_reg_eul_heading_lsb,
342 NULL);
343 if (ret == ESP_OK) {
344 data->euler.heading = ((int16_t)((euler_data[1] << 8) | euler_data[0])) / 16.0f;
345 data->euler.roll = ((int16_t)((euler_data[3] << 8) | euler_data[2])) / 16.0f;
346 data->euler.pitch = ((int16_t)((euler_data[5] << 8) | euler_data[4])) / 16.0f;
347 }
348
349 // Read accelerometer (6 bytes)
350 uint8_t acc_data[6];
351 ret = star_bus_i2c_read(handle->manager,
352 handle->bus_name,
353 acc_data,
354 6,
355 (uint8_t)k_bno055_reg_acc_data_x_lsb,
356 NULL);
357 if (ret == ESP_OK) {
358 data->accelerometer.x = (int16_t)((acc_data[1] << 8) | acc_data[0]);
359 data->accelerometer.y = (int16_t)((acc_data[3] << 8) | acc_data[2]);
360 data->accelerometer.z = (int16_t)((acc_data[5] << 8) | acc_data[4]);
361 }
362
363 // Read BMP280 pressure and temperature
364 uint8_t bmp_data[6];
365 ret =
366 star_bus_i2c_read(handle->manager, handle->bus_name, bmp_data, 6, k_bmp280_reg_press_msb, NULL);
367 if (ret == ESP_OK) {
368 int32_t adc_P = (bmp_data[0] << 12) | (bmp_data[1] << 4) | (bmp_data[2] >> 4);
369 int32_t adc_T = (bmp_data[3] << 12) | (bmp_data[4] << 4) | (bmp_data[5] >> 4);
370
371 // Temperature compensation
372 int32_t var1 =
373 ((((adc_T >> 3) - ((int32_t)handle->dig_T1 << 1))) * ((int32_t)handle->dig_T2)) >> 11;
374 int32_t var2 =
375 (((((adc_T >> 4) - ((int32_t)handle->dig_T1)) * ((adc_T >> 4) - ((int32_t)handle->dig_T1))) >>
376 12) *
377 ((int32_t)handle->dig_T3)) >>
378 14;
379 int32_t t_fine = var1 + var2;
380 data->bmp_temp_c = (t_fine * 5 + 128) >> 8;
381 data->bmp_temp_c /= 100.0f;
382
383 // Pressure compensation
384 int64_t var1_64 = ((int64_t)t_fine) - 128000;
385 int64_t var2_64 = var1_64 * var1_64 * (int64_t)handle->dig_P6;
386 var2_64 = var2_64 + ((var1_64 * (int64_t)handle->dig_P5) << 17);
387 var2_64 = var2_64 + (((int64_t)handle->dig_P4) << 35);
388 var1_64 = ((var1_64 * var1_64 * (int64_t)handle->dig_P3) >> 8) +
389 ((var1_64 * (int64_t)handle->dig_P2) << 12);
390 var1_64 = (((((int64_t)1) << 47) + var1_64)) * ((int64_t)handle->dig_P1) >> 33;
391
392 if (var1_64 != 0) {
393 int64_t p = 1048576 - adc_P;
394 p = (((p << 31) - var2_64) * 3125) / var1_64;
395 var1_64 = (((int64_t)handle->dig_P9) * (p >> 13) * (p >> 13)) >> 25;
396 var2_64 = (((int64_t)handle->dig_P8) * p) >> 19;
397 p = ((p + var1_64 + var2_64) >> 8) + (((int64_t)handle->dig_P7) << 4);
398 data->pressure_pa = (float)p / 256.0f;
399
400 // Calculate altitude
401 data->altitude_m =
402 44330.0f * (1.0f - powf(data->pressure_pa / SEA_LEVEL_PRESSURE_PA, 0.1903f));
403 }
404 }
405
406 // Read calibration status
407 star_sensor_imu_10dof_get_calibration(handle, &data->calibration);
408
409 xSemaphoreGive(mutex);
410 return ESP_OK;
411}
