36star_bus_config_t* star_bus_config_create_i2c(const char* name,
37 i2c_port_t port,
38 uint8_t address,
39 gpio_num_t sda_pin,
40 gpio_num_t scl_pin,
41 uint32_t clk_speed)
42{
43 star_bus_config_t* config = internal_star_bus_config_create_common(name, k_star_bus_type_i2c);
44 ESP_RETURN_ON_FALSE(config,
45 NULL,
46 s_TAG,
47 "Failed to create common config for %s",
48 name ? name : "NULL");
49
50 /* Configure I2C-specific parameters */
51 config->proto.i2c.port = port;
52 config->proto.i2c.address = address;
53
54 /* Initialize with default I2C configuration */
55 memset(&config->proto.i2c.config, 0, sizeof(i2c_config_t));
56 config->proto.i2c.config.mode = I2C_MODE_MASTER;
57 config->proto.i2c.config.sda_io_num = sda_pin;
58 config->proto.i2c.config.scl_io_num = scl_pin;
59 config->proto.i2c.config.sda_pullup_en = GPIO_PULLUP_ENABLE; /* Common practice */
60 config->proto.i2c.config.scl_pullup_en = GPIO_PULLUP_ENABLE; /* Common practice */
61 config->proto.i2c.config.master.clk_speed = clk_speed;
62
63 /* Initialize default operations */
64 star_bus_i2c_init_default_ops(&config->proto.i2c.ops);
65
66 ESP_LOGI(s_TAG,
67 "Created I2C config '%s' (Port: %d, Addr: 0x%02X, SDA: %d, SCL: %d, Speed: %" PRIu32
68 " Hz)",
69 name,
70 port,
71 address,
72 sda_pin,
73 scl_pin,
74 clk_speed);
75 return config;
76}

78star_bus_config_t* star_bus_config_create_spi_device(const char* name,
79 spi_host_device_t host,
80 gpio_num_t copi_pin,
81 gpio_num_t cipo_pin,
82 gpio_num_t sclk_pin,
83 int32_t dma_chan,
84 const spi_device_interface_config_t* dev_cfg)
85{
86 ESP_RETURN_ON_FALSE(dev_cfg, NULL, s_TAG, "SPI device config cannot be NULL");
87 ESP_RETURN_ON_FALSE(host >= 0 && host < SPI_HOST_MAX,
88 NULL,
89 s_TAG,
90 "Invalid SPI host number: %d",
91 host);
92
93 star_bus_config_t* config = internal_star_bus_config_create_common(name, k_star_bus_type_spi);
94 ESP_RETURN_ON_FALSE(config,
95 NULL,
96 s_TAG,
97 "Failed to create common config for %s",
98 name ? name : "NULL");
99
100 /* Configure SPI-specific parameters */
101 config->proto.spi.host = host;
102
103 /* Initialize SPI bus configuration structure within the config */
104 /* We copy the relevant pin numbers to our internal structure */
105 memset(&config->proto.spi.bus_cfg, 0, sizeof(spi_bus_config_t));
106 config->proto.spi.copi_io_num = copi_pin;
107 config->proto.spi.cipo_io_num = cipo_pin;
108 config->proto.spi.sclk_io_num = sclk_pin;
109 config->proto.spi.quadwp_io_num = -1; /* Not used for standard SPI, only QuadSPI */
110 config->proto.spi.quadhd_io_num = -1;
111 config->proto.spi.max_transfer_sz =
112 0; /* 0 means use default (usually 4092 bytes if DMA enabled, 64 otherwise) */
113
114 /* Copy the provided device configuration */
115 memcpy(&config->proto.spi.dev_cfg, dev_cfg, sizeof(spi_device_interface_config_t));
116
117 /* Store DMA channel request */
118 config->proto.spi.bus_cfg.flags = (dma_chan > 0) ? SPICOMMON_BUSFLAG_MASTER : 0;
119 /* The actual DMA channel assignment happens in spi_bus_initialize */
120
121 /* Mark as controller mode (not peripheral) */
122 config->proto.spi.is_peripheral = false;
123
124 /* Initialize default operations */
125 star_bus_spi_init_default_ops(&config->proto.spi.ops);
126
127 ESP_LOGI(
128 s_TAG,
129 "Created SPI config '%s' (Host: %d, COPI: %d, CIPO: %d, SCLK: %d, CS: %d, Mode: %d, Speed: %" PRId32
130 " Hz, DMA: %s)",
131 name,
132 host,
133 copi_pin,
134 cipo_pin,
135 sclk_pin,
136 dev_cfg->spics_io_num,
137 dev_cfg->mode,
138 (long)dev_cfg->clock_speed_hz,
139 (dma_chan > 0) ? "Enabled" : "Disabled");
140
141 return config;
142}

207star_bus_config_create_gpio(const char* name, const gpio_num_t pins[], uint8_t pin_count)
208{
209 ESP_RETURN_ON_FALSE(pins, NULL, s_TAG, "Pins array is NULL");
210 ESP_RETURN_ON_FALSE(pin_count > 0, NULL, s_TAG, "Pin count must be at least 1");
211
212 star_bus_config_t* config = internal_star_bus_config_create_common(name, k_star_bus_type_gpio);
213 ESP_RETURN_ON_FALSE(config,
214 NULL,
215 s_TAG,
216 "Failed to create common config for %s",
217 name ? name : "NULL");
218
219 /* Allocate memory for pins array */
220 gpio_num_t* pins_copy = (gpio_num_t*)malloc(pin_count * sizeof(gpio_num_t));
221 if (pins_copy == NULL) {
222 ESP_LOGE(s_TAG, "Failed to allocate memory for GPIO pins array");
223 internal_star_bus_config_cleanup(config);
224 return NULL;
225 }
226
227 /* Copy pin numbers */
228 memcpy(pins_copy, pins, pin_count * sizeof(gpio_num_t));
229
230 /* Configure GPIO-specific parameters */
231 config->proto.gpio.pins = pins_copy;
232 config->proto.gpio.pin_count = pin_count;
233 config->proto.gpio.isr_installed = false;
234
235 /* Initialize callbacks to NULL */
236 memset(&config->proto.gpio.callbacks, 0, sizeof(star_gpio_callbacks_t));
237
238 /* Initialize default operations */
239 star_bus_gpio_init_default_ops(&config->proto.gpio.ops);
240
241 ESP_LOGI(s_TAG, "Created GPIO bus config '%s' with %d pins", name, pin_count);
242 for (uint8_t i = 0; i < pin_count; ++i) {
243 ESP_LOGI(s_TAG, " Pin[%d]: GPIO%d", i, pins[i]);
244 }
245
246 return config;

291esp_err_t star_bus_config_destroy(star_bus_config_t* config)
292{
293 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config pointer is NULL");
294
295 /* If the bus/device is initialized, deinitialize it first */
296 if (config->initialized) {
297 esp_err_t ret = star_bus_config_deinit(config);
298 if (ret != ESP_OK) {
299 ESP_LOGE(s_TAG,
300 "Failed to deinitialize bus '%s' before destroying: %s. Continuing cleanup.",
301 config->name ? config->name : "UNKNOWN",
302 esp_err_to_name(ret));
303 /* Continue cleanup even if deinit failed to avoid memory leaks */
304 }
305 }
306
307 ESP_LOGI(s_TAG, "Destroying bus config '%s'", config->name ? config->name : "UNKNOWN");
308
309 /* Specific cleanup for SPI is handled by the manager during remove_bus */
310 /* The manager checks if the bus needs to be freed based on device count */
311
312 /* Clean up and free the configuration */
313 internal_star_bus_config_cleanup(config);
314
315 /* IMPORTANT: After this call, the config pointer is invalid. */
316 return ESP_OK;
317}

330esp_err_t star_bus_config_init(star_bus_config_t* config, star_bus_manager_t* manager)
331{
332 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config pointer is NULL");
333 ESP_RETURN_ON_FALSE(!config->initialized,
334 ESP_ERR_INVALID_STATE,
335 s_TAG,
336 "Bus '%s' is already initialized",
337 config->name ? config->name : "UNKNOWN");
338
339 esp_err_t ret = ESP_OK;
340 const char* bus_name = config->name ? config->name : "UNKNOWN"; /* Use for logging */
341
342 switch (config->type) {
343 case k_star_bus_type_i2c:
344 /* Initialize I2C bus */
345 ESP_LOGI(s_TAG, "Initializing I2C bus '%s' (Port %d)", bus_name, config->proto.i2c.port);
346
347 ret = i2c_param_config(config->proto.i2c.port, &config->proto.i2c.config);
348 ESP_GOTO_ON_ERROR(ret,
349 fail,
350 s_TAG,
351 "i2c_param_config failed for '%s': %s",
352 bus_name,
353 esp_err_to_name(ret));
354
355 ret = i2c_driver_install(config->proto.i2c.port,
356 config->proto.i2c.config.mode,
357 0, /* RX buffer size (master mode) */
358 0, /* TX buffer size (master mode) */
359 0); /* Flags */
360 ESP_GOTO_ON_ERROR(ret,
361 fail_i2c_driver,
362 s_TAG,
363 "i2c_driver_install failed for '%s': %s",
364 bus_name,
365 esp_err_to_name(ret));
366 break;
367
368 case k_star_bus_type_spi:
369 ESP_RETURN_ON_FALSE(manager,
370 ESP_ERR_INVALID_ARG,
371 s_TAG,
372 "Bus manager pointer is required for SPI initialization");
373
374 spi_host_device_t host = config->proto.spi.host;
375 ESP_RETURN_ON_FALSE(host >= 0 && host < SPI_HOST_MAX,
376 ESP_ERR_INVALID_ARG,
377 s_TAG,
378 "Invalid SPI host number: %d",
379 host);
380
381 /* Check if this is peripheral or controller mode */
382 if (config->proto.spi.is_peripheral) {
383 /* SPI Peripheral (Slave) Mode */
384 ESP_LOGI(s_TAG, "Initializing SPI peripheral '%s' (Host %d)", bus_name, host);
385
386 spi_slave_interface_config_t slave_cfg = {
387 .spics_io_num = config->proto.spi.cs_io_num,
388 .flags = 0,
389 .queue_size = config->proto.spi.queue_size,
390 .mode = config->proto.spi.mode,
391 .post_setup_cb = NULL,
392 .post_trans_cb = NULL,
393 };
394
395 /* Map COPI/CIPO to ESP-IDF's MOSI/MISO */
396 spi_bus_config_t bus_cfg = {
397 .mosi_io_num = config->proto.spi.copi_io_num,
398 .miso_io_num = config->proto.spi.cipo_io_num,
399 .sclk_io_num = config->proto.spi.sclk_io_num,
400 .quadwp_io_num = -1,
401 .quadhd_io_num = -1,
402 .max_transfer_sz = 0,
403 };
404
405 ret = spi_slave_initialize(host, &bus_cfg, &slave_cfg, SPI_DMA_CH_AUTO);
406 ESP_GOTO_ON_ERROR(ret,
407 fail,
408 s_TAG,
409 "spi_slave_initialize failed for '%s': %s",
410 bus_name,
411 esp_err_to_name(ret));
412
413 manager->spi_host_initialized[host] = true;
414 ESP_LOGI(s_TAG, "SPI peripheral '%s' initialized successfully", bus_name);
415 } else {
416 /* SPI Controller (Master) Mode */
417 ESP_LOGI(s_TAG, "Initializing SPI device '%s' (Host %d)", bus_name, host);
418
419 /* Initialize the SPI bus *only if* it hasn't been initialized yet for this host */
420 if (!manager->spi_host_initialized[host]) {
421 ESP_LOGI(s_TAG, "Initializing SPI bus driver for host %d...", host);
422 /* Map COPI/CIPO to ESP-IDF's MOSI/MISO */
423 spi_bus_config_t esp_idf_bus_cfg = {
424 .mosi_io_num = config->proto.spi.copi_io_num,
425 .miso_io_num = config->proto.spi.cipo_io_num,
426 .sclk_io_num = config->proto.spi.sclk_io_num,
427 .quadwp_io_num = config->proto.spi.quadwp_io_num,
428 .quadhd_io_num = config->proto.spi.quadhd_io_num,
429 .max_transfer_sz = config->proto.spi.max_transfer_sz,
430 .flags = config->proto.spi.bus_cfg.flags,
431 };
432 int32_t dma_chan = (config->proto.spi.bus_cfg.flags & SPICOMMON_BUSFLAG_MASTER)
433 ? SPI_DMA_CH_AUTO
434 : SPI_DMA_DISABLED;
435 ret = spi_bus_initialize(host, &esp_idf_bus_cfg, dma_chan);
436 ESP_GOTO_ON_ERROR(ret,
437 fail,
438 s_TAG,
439 "spi_bus_initialize failed for host %d: %s",
440 host,
441 esp_err_to_name(ret));
442 manager->spi_host_initialized[host] = true;
443 manager->spi_device_count[host] = 0;
444 ESP_LOGI(s_TAG, "SPI bus driver for host %d initialized.", host);
445 } else {
446 ESP_LOGD(s_TAG, "SPI bus driver for host %d already initialized.", host);
447 }
448
449 /* Add the SPI device to the bus */
450 spi_device_handle_t spi_handle;
451 ret = spi_bus_add_device(host, &config->proto.spi.dev_cfg, &spi_handle);
452 ESP_GOTO_ON_ERROR(ret,
453 fail_spi_add_device,
454 s_TAG,
455 "spi_bus_add_device failed for '%s': %s",
456 bus_name,
457 esp_err_to_name(ret));
458 config->handle = (void*)spi_handle;
459 manager->spi_device_count[host]++;
460 }
461 break;
462
463 case k_star_bus_type_gpio:
464 /* Initialize GPIO bus - pins are configured on-demand via gpio_configure */
465 ESP_LOGI(s_TAG,
466 "Initializing GPIO bus '%s' with %d pins",
467 bus_name,
468 config->proto.gpio.pin_count);
469
470 /* GPIO initialization is minimal - actual pin configuration happens on first use */
471 /* No driver installation needed for GPIO unlike I2C/SPI */
472
473 ESP_LOGI(s_TAG, "GPIO bus '%s' ready (pins will be configured on-demand)", bus_name);
474 break;
475
476 case k_star_bus_type_dht22:
477 /* Initialize DHT22 bus */
478 ESP_LOGI(s_TAG,
479 "Initializing DHT22 bus '%s' on GPIO %d",
480 bus_name,
481 config->proto.dht22.gpio_pin);
482
483 /* Configure GPIO as open-drain with pull-up for DHT22 protocol */
484 {
485 gpio_config_t io_conf = {
486 .pin_bit_mask = (1ULL << config->proto.dht22.gpio_pin),
487 .mode = GPIO_MODE_INPUT_OUTPUT_OD,
488 .pull_up_en = GPIO_PULLUP_ENABLE,
489 .pull_down_en = GPIO_PULLDOWN_DISABLE,
490 .intr_type = GPIO_INTR_DISABLE,
491 };
492
493 ret = gpio_config(&io_conf);
494 ESP_GOTO_ON_ERROR(ret,
495 fail,
496 s_TAG,
497 "Failed to configure GPIO for DHT22 bus '%s': %s",
498 bus_name,
499 esp_err_to_name(ret));
500
501 /* Release bus (high) */
502 gpio_set_level(config->proto.dht22.gpio_pin, 1);
503 }
504
505 ESP_LOGI(s_TAG,
506 "DHT22 bus '%s' initialized on GPIO %d",
507 bus_name,
508 config->proto.dht22.gpio_pin);
509 break;
510
511 default:
512 ESP_LOGE(s_TAG, "Unsupported bus type for initialization: %d", config->type);
513 ret = ESP_ERR_NOT_SUPPORTED;
514 goto fail;
515 }
516
517 /* Mark as initialized */
518 config->initialized = true;
519 ESP_LOGI(s_TAG,
520 "Successfully initialized bus/device '%s' (%s)",
521 bus_name,
522 star_bus_type_to_string(config->type));
523 return ESP_OK;
524
525/* Failure points with specific cleanup needs */
526fail_i2c_driver:
527 /* If driver install failed after param_config, no specific cleanup needed for param_config */
528 goto fail;
529
530fail_spi_add_device:
531 /* If add_device failed, we might need to free the bus if we just initialized it */
532 /* This logic is now handled by the manager during remove_bus */
533 goto fail;
534
535/* Generic failure point */
536fail:
537 ESP_LOGE(s_TAG, "Failed to initialize bus/device '%s'", bus_name);
538 return ret;
539}

541esp_err_t star_bus_config_deinit(star_bus_config_t* config)
542{
543 ESP_RETURN_ON_FALSE(config, ESP_ERR_INVALID_ARG, s_TAG, "Config pointer is NULL");
544 ESP_RETURN_ON_FALSE(config->initialized,
545 ESP_ERR_INVALID_STATE,
546 s_TAG,
547 "Bus/device '%s' is not initialized",
548 config->name ? config->name : "UNKNOWN");
549
550 esp_err_t ret = ESP_OK;
551 const char* bus_name = config->name ? config->name : "UNKNOWN";
552
553 switch (config->type) {
554 case k_star_bus_type_i2c:
555 /* Deinitialize I2C bus */
556 ESP_LOGI(s_TAG, "Deinitializing I2C bus '%s' (Port %d)", bus_name, config->proto.i2c.port);
557
558 ret = i2c_driver_delete(config->proto.i2c.port);
559 break;
560
561 case k_star_bus_type_spi:
562 if (config->proto.spi.is_peripheral) {
563 /* SPI Peripheral Mode - Free the slave driver */
564 ESP_LOGI(s_TAG,
565 "Deinitializing SPI peripheral '%s' (Host %d)",
566 bus_name,
567 config->proto.spi.host);
568 ret = spi_slave_free(config->proto.spi.host);
569 } else {
570 /* SPI Controller Mode - Remove device */
571 ESP_LOGI(s_TAG,
572 "Deinitializing SPI device '%s' (Host %d)",
573 bus_name,
574 config->proto.spi.host);
575 if (config->handle) {
576 ret = spi_bus_remove_device((spi_device_handle_t)config->handle);
577 config->handle = NULL;
578 } else {
579 ESP_LOGW(s_TAG, "SPI device '%s' handle was already NULL before deinit.", bus_name);
580 ret = ESP_OK;
581 }
582 }
583 break;
584
585 case k_star_bus_type_gpio:
586 /* Deinitialize GPIO bus */
587 ESP_LOGI(s_TAG, "Deinitializing GPIO bus '%s'", bus_name);
588
589 /* Remove ISR handlers for all pins if ISR service was installed */
590 if (config->proto.gpio.isr_installed) {
591 for (uint8_t i = 0; i < config->proto.gpio.pin_count; ++i) {
592 gpio_num_t pin = config->proto.gpio.pins[i];
593 if (pin != GPIO_NUM_NC) {
594 /* Try to remove ISR handler, ignore errors if not set */
595 gpio_isr_handler_remove(pin);
596 }
597 }
598 /* Note: We don't uninstall ISR service as it might be shared with other GPIO buses */
599 }
600
601 /* Reset all pins to input with pull-down (safe default state) */
602 for (uint8_t i = 0; i < config->proto.gpio.pin_count; ++i) {
603 gpio_num_t pin = config->proto.gpio.pins[i];
604 if (pin != GPIO_NUM_NC) {
605 gpio_reset_pin(pin);
606 }
607 }
608
609 ret = ESP_OK;
610 break;
611
612 case k_star_bus_type_dht22:
613 /* Deinitialize DHT22 bus */
614 ESP_LOGI(s_TAG,
615 "Deinitializing DHT22 bus '%s' (GPIO %d)",
616 bus_name,
617 config->proto.dht22.gpio_pin);
618
619 /* Reset GPIO to default state */
620 if (config->proto.dht22.gpio_pin != GPIO_NUM_NC) {
621 gpio_reset_pin(config->proto.dht22.gpio_pin);
622 }
623
624 ret = ESP_OK;
625 break;
626
627 default:
628 ESP_LOGE(s_TAG, "Unsupported bus type for deinitialization: %d", config->type);
629 ret = ESP_ERR_NOT_SUPPORTED;
630 break;
631 }
632
633 if (ret != ESP_OK) {

657{
658 if (config == NULL) {
659 return;
660 }
661
662 /* Free the name if it was allocated (strdup'd in create_common) */
663 if (config->name) {

691static star_bus_config_t* internal_star_bus_config_create_common(const char* name,
692 star_bus_type_t type)
693{
694 /* Name validation */
695 if (name == NULL) {
696 ESP_LOGE(s_TAG, "Bus name cannot be NULL");
697 return NULL;
698 }
699
700 /* Allocate memory for the bus configuration */
701 star_bus_config_t* config = (star_bus_config_t*)malloc(sizeof(star_bus_config_t));
702 if (config == NULL) {
703 ESP_LOGE(s_TAG, "Failed to allocate memory for bus configuration");
704 return NULL;
705 }
706
707 /* Initialize the configuration to zeros */
708 memset(config, 0, sizeof(star_bus_config_t));
709
710 /* Make a copy of the bus name */
711 char* name_copy = strdup(name);
712 if (name_copy == NULL) {
713 ESP_LOGE(s_TAG, "Failed to allocate memory for bus name");
714 free(config);
715 return NULL;
716 }
717 config->name = name_copy; /* Store the copy */
718
719 /* Set common parameters */
720 config->type = type;
721 config->initialized = false;
722 config->handle = NULL;
723 config->user_ctx = NULL;
724 config->next = NULL;
725
726 /* Initialize callback structs to NULL */
727 memset(&config->proto.i2c.callbacks, 0, sizeof(config->proto.i2c.callbacks));
728 memset(&config->proto.spi.callbacks, 0, sizeof(config->proto.spi.callbacks));
729
730 return config;
