31uint16_t star_servo_angle_to_pulse(uint8_t angle)
32{
33 angle = clamp_angle(angle);
34
35 /* Linear interpolation: pulse = 1000 + (angle * 1000 / 180) */
36 uint16_t pulse_us =
37 SERVO_PULSE_MIN_US +
38 ((uint32_t)angle * (SERVO_PULSE_MAX_US - SERVO_PULSE_MIN_US)) / SERVO_ANGLE_MAX;
39
40 return pulse_us;
41}

43uint8_t star_servo_pulse_to_angle(uint16_t pulse_us)

54uint16_t star_servo_pulse_to_count(uint16_t pulse_us)
55{
56 pulse_us = clamp_pulse(pulse_us);
57
58 /* For PCA9685 at 50Hz: count = (pulse_us * 4096) / 20000 */
59 uint32_t count = ((uint32_t)pulse_us * PCA9685_MAX_COUNT) / PCA9685_PERIOD_US;
60
61 /* Clamp to valid PCA9685 range */
62 if (count >= PCA9685_MAX_COUNT) {
63 count = PCA9685_MAX_COUNT - 1;
64 }
65
66 return (uint16_t)count;
67}
