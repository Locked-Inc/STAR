34uint16_t star_servo_angle_to_pulse(uint8_t angle)
35{
36 angle = clamp_angle(angle);
37
38 const star_servo_constants_t* constants = star_servo_get_constants();
39 const star_servo_pulse_t* pulse = &constants->pulse;
40 const star_servo_angle_t* angles = &constants->angle;
41
42 /* Linear interpolation: pulse = min + (angle * range / max_angle) */
43 uint16_t pulse_us =
44 pulse->min_us +
45 ((uint32_t)angle * (pulse->max_us - pulse->min_us)) / angles->max;
46
47 return pulse_us;
48}

50uint8_t star_servo_pulse_to_angle(uint16_t pulse_us)
51{
52 pulse_us = clamp_pulse(pulse_us);
53
54 const star_servo_constants_t* constants = star_servo_get_constants();
55 const star_servo_pulse_t* pulse = &constants->pulse;
56 const star_servo_angle_t* angles = &constants->angle;
57
58 /* Linear interpolation: angle = (pulse - min) * max_angle / range */
59 uint32_t angle = ((uint32_t)(pulse_us - pulse->min_us) * angles->max) /

65uint16_t star_servo_pulse_to_count(uint16_t pulse_us)
66{
67 pulse_us = clamp_pulse(pulse_us);
68
69 const star_servo_pca9685_t* pca = star_servo_get_pca9685_constants();
70
71 /* For PCA9685 at 50Hz: count = (pulse_us * max_count) / period_us */
72 uint32_t count = ((uint32_t)pulse_us * pca->max_count) / pca->period_us;
73
74 /* Clamp to valid PCA9685 range */
75 if (count >= pca->max_count) {
76 count = pca->max_count - 1;
77 }
78
79 return (uint16_t)count;
80}
