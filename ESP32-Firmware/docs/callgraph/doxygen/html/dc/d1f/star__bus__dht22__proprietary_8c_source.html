85static dht22_state_t* internal_get_dht22_state(const char* bus_name, bool create)
86{
87 for (uint8_t i = 0; i < g_num_dht22_states; i++) {
88 if (strcmp(g_dht22_states[i].bus_name, bus_name) == 0) {
89 return &g_dht22_states[i];
90 }
91 }
92
93 if (create && g_num_dht22_states < STAR_DHT22_MAX_BUSES) {
94 dht22_state_t* state = &g_dht22_states[g_num_dht22_states];
95 memset(state, 0, sizeof(dht22_state_t));
96 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
97 state->bus_name[sizeof(state->bus_name) - 1] = '\0';
98 g_num_dht22_states++;
99 return state;
100 }
101
102 return NULL;
103}

108static dht22_state_t* get_dht22_state(const char* bus_name, bool create)
109{
110 if (internal_init_dht22_mutex() != ESP_OK) {
111 return NULL;
112 }
113
114 if (xSemaphoreTake(g_dht22_mutex, portMAX_DELAY) != pdTRUE) {
115 ESP_LOGE(s_TAG, "Failed to take DHT22 mutex");
116 return NULL;
117 }
118
119 dht22_state_t* state = internal_get_dht22_state(bus_name, create);
120
121 xSemaphoreGive(g_dht22_mutex);
122 return state;
123}

182static int32_t wait_for_level(gpio_num_t pin, int expected_level, uint32_t timeout_us)
183{
184 int64_t start = esp_timer_get_time();
185 int64_t elapsed;
186
187 while (gpio_read(pin) != expected_level) {
188 elapsed = esp_timer_get_time() - start;
189 if (elapsed > timeout_us) {
190 return -1;
191 }
192 }
193
194 return (int32_t)(esp_timer_get_time() - start);

229static esp_err_t read_raw_data(dht22_state_t* state, uint8_t data[DHT22_DATA_BYTES])
230{
231 gpio_num_t pin = state->config.gpio_pin;
232 uint32_t start_low_us = state->config.start_low_ms * 1000;
233 uint32_t timeout_us = state->config.read_timeout_ms * 1000;
234
235 /* Disable interrupts for timing-critical section */
236 portMUX_TYPE mux = portMUX_INITIALIZER_UNLOCKED;
237 portENTER_CRITICAL(&mux);
238
239 /* Step 1: Send start signal */
240 /* Pull low for at least 1ms (we use 18ms by default) */
241 gpio_set_output(pin);
242 gpio_low(pin);
243 delay_us(start_low_us);
244
245 /* Pull high for 20-40us */
246 gpio_high(pin);
247 delay_us(TIMING_START_HIGH_US);
248
249 /* Step 2: Switch to input and wait for sensor response */
250 gpio_set_input(pin);
251
252 /* Wait for sensor to pull low (response start) */
253 if (wait_for_level(pin, 0, timeout_us) < 0) {
254 portEXIT_CRITICAL(&mux);
255 ESP_LOGD(s_TAG, "No response from sensor (wait for low)");
256 return ESP_ERR_TIMEOUT;
257 }
258
259 /* Wait for sensor to pull high (response end of low pulse ~80us) */
260 if (wait_for_level(pin, 1, TIMING_RESPONSE_LOW_US + 50) < 0) {
261 portEXIT_CRITICAL(&mux);
262 ESP_LOGD(s_TAG, "No response from sensor (wait for high)");
263 return ESP_ERR_TIMEOUT;
264 }
265
266 /* Wait for sensor to pull low (start of first data bit) */
267 if (wait_for_level(pin, 0, TIMING_RESPONSE_HIGH_US + 50) < 0) {
268 portEXIT_CRITICAL(&mux);
269 ESP_LOGD(s_TAG, "No response from sensor (wait for data start)");
270 return ESP_ERR_TIMEOUT;
271 }
272
273 /* Step 3: Read 40 bits of data */
274 memset(data, 0, DHT22_DATA_BYTES);
275
276 for (int i = 0; i < DHT22_DATA_BITS; i++) {
277 /* Wait for high (end of bit start pulse ~50us) */
278 if (wait_for_level(pin, 1, TIMING_BIT_START_US + 30) < 0) {
279 portEXIT_CRITICAL(&mux);
280 ESP_LOGD(s_TAG, "Timeout waiting for bit %d high", i);
281 return ESP_ERR_TIMEOUT;
282 }
283
284 /* Measure high pulse duration to determine bit value */
285 int32_t high_duration = measure_pulse(pin, 1, TIMING_BIT_1_HIGH_US + 30);
286 if (high_duration < 0) {
287 portEXIT_CRITICAL(&mux);
288 ESP_LOGD(s_TAG, "Timeout measuring bit %d", i);
289 return ESP_ERR_TIMEOUT;
290 }
291
292 /* Bit is 1 if high duration > threshold, 0 otherwise */
293 if (high_duration > TIMING_BIT_THRESHOLD_US) {
294 data[i / 8] |= (1 << (7 - (i % 8)));
295 }
296 }
297
298 portEXIT_CRITICAL(&mux);
299
300 return ESP_OK;
301}

306static esp_err_t dht22_read_default(const star_bus_config_t* config, star_dht22_data_t* data)
307{
308 if (config == NULL || data == NULL) {
309 return ESP_ERR_INVALID_ARG;
310 }
311
312 /* This is called through the bus manager, so we need to find the state */
313 dht22_state_t* state = get_dht22_state(config->name, false);
314 if (state == NULL || !state->initialized) {
315 return ESP_ERR_INVALID_STATE;
316 }
317
318 uint8_t raw_data[DHT22_DATA_BYTES];
319 esp_err_t ret = read_raw_data(state, raw_data);
320 if (ret != ESP_OK) {
321 return ret;
322 }
323
324 /* Parse raw data */
325 data->raw_humidity = (raw_data[0] << 8) | raw_data[1];
326 data->raw_temperature = (raw_data[2] << 8) | raw_data[3];
327 data->checksum = raw_data[4];
328
329 /* Verify checksum */
330 uint8_t calc_checksum =
331 star_bus_dht22_calculate_checksum(data->raw_humidity, data->raw_temperature);
332 data->checksum_valid = (calc_checksum == data->checksum);
333
334 /* Convert to physical values */
335 data->humidity_percent = star_bus_dht22_raw_to_humidity(data->raw_humidity, state->config.model);
336 data->temperature_c = star_bus_dht22_raw_to_celsius(data->raw_temperature, state->config.model);
337 data->timestamp_us = esp_timer_get_time();
338
339 return ESP_OK;
340}

344esp_err_t star_bus_dht22_init(star_bus_manager_t* manager,
345 const char* bus_name,
346 const star_dht22_config_t* config)
347{
348 if (manager == NULL || bus_name == NULL || config == NULL) {
349 ESP_LOGE(s_TAG, "Invalid parameters");
350 return ESP_ERR_INVALID_ARG;
351 }
352
353 if (config->gpio_pin < 0) {
354 ESP_LOGE(s_TAG, "Invalid GPIO pin");
355 return ESP_ERR_INVALID_ARG;
356 }
357
358 dht22_state_t* state = get_dht22_state(bus_name, true);
359 if (state == NULL) {
360 ESP_LOGE(s_TAG, "Failed to create state for bus '%s'", bus_name);
361 return ESP_ERR_NO_MEM;
362 }
363
364 memcpy(&state->config, config, sizeof(star_dht22_config_t));
365 memset(&state->stats, 0, sizeof(star_dht22_stats_t));
366 memset(&state->last_data, 0, sizeof(star_dht22_data_t));
367 state->last_read_time_us = 0;
368
369 /* Configure GPIO as open-drain output with pull-up */
370 gpio_config_t io_conf = {
371 .pin_bit_mask = (1ULL << config->gpio_pin),
372 .mode = GPIO_MODE_INPUT_OUTPUT_OD,
373 .pull_up_en = GPIO_PULLUP_ENABLE,
374 .pull_down_en = GPIO_PULLDOWN_DISABLE,
375 .intr_type = GPIO_INTR_DISABLE,
376 };
377
378 esp_err_t result = gpio_config(&io_conf);
379 if (result != ESP_OK) {
380 ESP_LOGE(s_TAG, "Failed to configure GPIO: %s", esp_err_to_name(result));
381 return result;
382 }
383
384 /* Release bus (high) */
385 gpio_high(config->gpio_pin);
386
387 state->initialized = true;
388
389 ESP_LOGI(s_TAG,
390 "DHT22 bus '%s' initialized on GPIO %d (model: %d)",
391 bus_name,
392 config->gpio_pin,
393 config->model);
394
395 return ESP_OK;
396}

431esp_err_t
432star_bus_dht22_read(star_bus_manager_t* manager, const char* bus_name, star_dht22_data_t* data)
433{
434 if (manager == NULL || bus_name == NULL || data == NULL) {
435 return ESP_ERR_INVALID_ARG;
436 }
437
438 dht22_state_t* state = get_dht22_state(bus_name, false);
439 if (state == NULL || !state->initialized) {
440 return ESP_ERR_NOT_FOUND;
441 }
442
443 int64_t now = esp_timer_get_time();
444 int64_t elapsed = now - state->last_read_time_us;
445
446 /* Check if minimum interval has passed */
447 if (state->last_read_time_us > 0 && elapsed < (STAR_DHT22_MIN_INTERVAL_MS * 1000)) {
448 /* Return cached data */
449 memcpy(data, &state->last_data, sizeof(star_dht22_data_t));
450 ESP_LOGD(s_TAG, "Returning cached data (elapsed: %" PRId64 " us)", elapsed);
451 return ESP_OK;
452 }
453
454 state->stats.total_reads++;
455 uint32_t start_time = esp_timer_get_time();
456
457 uint8_t raw_data[DHT22_DATA_BYTES];
458 esp_err_t ret = read_raw_data(state, raw_data);
459
460 uint32_t read_time = (uint32_t)(esp_timer_get_time() - start_time);
461 state->stats.last_read_time_us = read_time;
462
463 if (ret != ESP_OK) {
464 state->stats.failed_reads++;
465 if (ret == ESP_ERR_TIMEOUT) {
466 state->stats.timeout_errors++;
467 } else {
468 state->stats.no_response_errors++;
469 }
470 ESP_LOGD(s_TAG, "Read failed: %s", esp_err_to_name(ret));
471 return ret;
472 }
473
474 /* Parse raw data */
475 data->raw_humidity = (raw_data[0] << 8) | raw_data[1];
476 data->raw_temperature = (raw_data[2] << 8) | raw_data[3];
477 data->checksum = raw_data[4];
478
479 /* Verify checksum */
480 uint8_t calc_checksum =
481 star_bus_dht22_calculate_checksum(data->raw_humidity, data->raw_temperature);
482 data->checksum_valid = (calc_checksum == data->checksum);
483
484 if (!data->checksum_valid) {
485 state->stats.checksum_errors++;
486 state->stats.failed_reads++;
487 ESP_LOGD(s_TAG,
488 "Checksum mismatch: calculated 0x%02X, received 0x%02X",
489 calc_checksum,
490 data->checksum);
491 return ESP_ERR_INVALID_CRC;
492 }
493
494 /* Convert to physical values */
495 data->humidity_percent = star_bus_dht22_raw_to_humidity(data->raw_humidity, state->config.model);
496 data->temperature_c = star_bus_dht22_raw_to_celsius(data->raw_temperature, state->config.model);
497 data->timestamp_us = esp_timer_get_time();
498
499 /* Update state */
500 state->stats.successful_reads++;
501 state->stats.last_humidity = data->humidity_percent;
502 state->stats.last_temperature = data->temperature_c;
503 state->stats.last_read_timestamp_us = data->timestamp_us;
504 state->last_read_time_us = now;
505 memcpy(&state->last_data, data, sizeof(star_dht22_data_t));
506
507 ESP_LOGD(s_TAG,
508 "Read successful: T=%.1fC, H=%.1f%%, time=%lu us",
509 data->temperature_c,
510 data->humidity_percent,
511 (unsigned long)read_time);
512
513 return ESP_OK;
514}

675float star_bus_dht22_raw_to_celsius(uint16_t raw_value, star_dht22_model_t model)
676{
677 float temperature;
678
679 if (model == k_star_dht22_model_dht11) {
680 /* DHT11: integer values only, high byte is whole degrees */
681 temperature = (float)(raw_value >> 8);
682 } else {
683 /* DHT22/AM2302: 0.1 degree resolution, bit 15 is sign */
684 bool is_negative = (raw_value & 0x8000) != 0;
685 temperature = (float)(raw_value & 0x7FFF) / 10.0f;
686 if (is_negative) {
687 temperature = -temperature;
688 }
689 }
690
691 return temperature;
692}

694float star_bus_dht22_raw_to_humidity(uint16_t raw_value, star_dht22_model_t model)
695{
696 float humidity;
697
698 if (model == k_star_dht22_model_dht11) {
699 /* DHT11: integer values only, high byte is whole percent */
700 humidity = (float)(raw_value >> 8);
701 } else {
702 /* DHT22/AM2302: 0.1% resolution */
703 humidity = (float)raw_value / 10.0f;
704 }
705
706 /* Clamp to valid range */
707 if (humidity > 100.0f) {
708 humidity = 100.0f;
709 }
710 if (humidity < 0.0f) {
711 humidity = 0.0f;
712 }
713
714 return humidity;

717uint8_t star_bus_dht22_calculate_checksum(uint16_t raw_humidity, uint16_t raw_temperature)
718{

723float star_bus_dht22_calculate_heat_index(float temperature, float humidity)
724{
725 /* Convert to Fahrenheit for calculation */
726 float t = star_bus_dht22_celsius_to_fahrenheit(temperature);
727 float h = humidity;
728
729 /* Simple formula for temperatures below 80F */
730 if (t < 80.0f) {
731 float hi = 0.5f * (t + 61.0f + ((t - 68.0f) * 1.2f) + (h * 0.094f));
732 return (hi - 32.0f) * 5.0f / 9.0f; /* Convert back to Celsius */
733 }
734
735 /* Rothfusz regression equation */
736 float hi = -42.379f + 2.04901523f * t + 10.14333127f * h - 0.22475541f * t * h -
737 0.00683783f * t * t - 0.05481717f * h * h + 0.00122874f * t * t * h +
738 0.00085282f * t * h * h - 0.00000199f * t * t * h * h;
739
740 /* Adjustments for low/high humidity */
741 if (h < 13.0f && t >= 80.0f && t <= 112.0f) {
742 hi -= ((13.0f - h) / 4.0f) * sqrtf((17.0f - fabsf(t - 95.0f)) / 17.0f);
743 } else if (h > 85.0f && t >= 80.0f && t <= 87.0f) {
744 hi += ((h - 85.0f) / 10.0f) * ((87.0f - t) / 5.0f);
745 }
746
747 return (hi - 32.0f) * 5.0f / 9.0f; /* Convert back to Celsius */
748}

750float star_bus_dht22_calculate_dew_point(float temperature, float humidity)
751{
752 /* Magnus formula constants for temperature range -40 to +50C */
753 const float a = 17.27f;
754 const float b = 237.7f;
755
756 float gamma = (a * temperature) / (b + temperature) + logf(humidity / 100.0f);
757 float dew_point = (b * gamma) / (a - gamma);
758
759 return dew_point;
760}
