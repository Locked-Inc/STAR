32static watchdog_health_level_t internal_assess_sensor_health(void)
33{
34 shared_context_t* shared_ctx = shared_data_get_context();
35 if (shared_ctx == NULL) {
36 return k_watchdog_health_fault;
37 }
38
39 uint32_t hcsr04_failures = 0;
40 uint32_t hcsr04_reads = 0;
41 uint32_t temp_failures = 0;
42 uint32_t temp_reads = 0;
43
44 if (xSemaphoreTake(shared_ctx->health_mutex,
45 pdMS_TO_TICKS(STAR_SYSTEM_CONFIG.freertos.mutex_timeout_ms)) == pdTRUE) {
46 for (uint8_t i = 0; i < STAR_SYSTEM_CONFIG.num_hcsr04_sensors; i++) {
47 hcsr04_failures += shared_ctx->health_data.sensor_read_failures[i];
48 }
49 temp_failures = shared_ctx->health_data.temperature_read_failures;
50 hcsr04_reads = shared_ctx->health_data.total_sensor_reads;
51 temp_reads = shared_ctx->health_data.total_temperature_reads;
52
53 xSemaphoreGive(shared_ctx->health_mutex);
54 } else {
55 ESP_LOGE(s_TAG, "Failed to acquire health mutex for sensor assessment");
56 return k_watchdog_health_fault;
57 }
58
59 /* Assess HC-SR04 sensors (critical for operation) */
60 watchdog_health_level_t hcsr04_health = k_watchdog_health_good;
61 if (hcsr04_reads > 0) {
62 const float hcsr04_failure_rate = (float)hcsr04_failures / hcsr04_reads;
63
64 if (hcsr04_failure_rate > STAR_SYSTEM_CONFIG.health.failure_rate_critical) {
65 ESP_LOGE(s_TAG,
66 "Critical HC-SR04 failure rate: %.1f%%",
67 hcsr04_failure_rate * s_watchdog_percent_multiplier);
68 hcsr04_health = k_watchdog_health_critical;
69 } else if (hcsr04_failure_rate > STAR_SYSTEM_CONFIG.health.failure_rate_degraded) {
70 ESP_LOGW(s_TAG,
71 "Elevated HC-SR04 failure rate: %.1f%%",
72 hcsr04_failure_rate * s_watchdog_percent_multiplier);
73 hcsr04_health = k_watchdog_health_degraded;
74 } else {
75 ESP_LOGI(s_TAG,
76 "HC-SR04 health good: %.1f%% failure rate",
77 hcsr04_failure_rate * s_watchdog_percent_multiplier);
78 }
79 } else {
80 ESP_LOGW(s_TAG, "No HC-SR04 readings available for health assessment");
81 hcsr04_health = k_watchdog_health_degraded;
82 }
83
84 // Assess DHT22 sensor (optional - more tolerant)
85 watchdog_health_level_t temp_health = k_watchdog_health_good;
86 if (temp_reads > 0) {
87 const float temp_failure_rate = (float)temp_failures / temp_reads;
88
89 /* DHT22 is optional - only flag as critical if it was working then failed */
90 if (temp_failure_rate == s_watchdog_temp_complete_failure_rate) {
91 ESP_LOGD(s_TAG,
92 "DHT22 completely failed (likely not connected): %.1f%%",
93 temp_failure_rate * s_watchdog_percent_multiplier);
94 /* Don't degrade system for completely failed optional sensor */
95 } else if (temp_failure_rate > s_watchdog_temp_mostly_failing_rate) {
96 ESP_LOGW(s_TAG,
97 "DHT22 mostly failing: %.1f%%",
98 temp_failure_rate * s_watchdog_percent_multiplier);
99 temp_health = k_watchdog_health_degraded;
100 } else {
101 ESP_LOGI(s_TAG,
102 "DHT22 health good: %.1f%% failure rate",
103 temp_failure_rate * s_watchdog_percent_multiplier);
104 }
105 } else {
106 ESP_LOGD(s_TAG, "No DHT22 readings (sensor likely not connected)");
107 }
108
109 /* Overall health is the worst of the critical sensors (HC-SR04) */
110 /* DHT22 can degrade but shouldn't cause critical state if not connected */
111 if (hcsr04_health == k_watchdog_health_critical) {
112 return k_watchdog_health_critical;
113 } else if (hcsr04_health == k_watchdog_health_degraded ||
114 temp_health == k_watchdog_health_degraded) {
115 return k_watchdog_health_degraded;
116 }
117
118 return k_watchdog_health_good;
119}

121static watchdog_health_level_t internal_assess_task_health(void)
122{
123 bool critical_tasks_running = true;
124 uint32_t optional_task_failures = 0;
125
126 /* Critical tasks that must be running for system operation */
127 if (!sensor_task_is_running()) {
128 ESP_LOGE(s_TAG, "Sensor task not running - CRITICAL");
129 critical_tasks_running = false;
130 }
131
132 if (!led_task_is_running()) {
133 ESP_LOGE(s_TAG, "LED task not running - CRITICAL");
134 critical_tasks_running = false;
135 }
136
137 /* Optional tasks (DHT22 can fail if sensor not connected) */
138 if (!dht22_task_is_running()) {
139 ESP_LOGD(s_TAG, "DHT22 task not running (sensor may not be connected)");
140 optional_task_failures++;
141 }
142
143 if (!critical_tasks_running) {
144 return k_watchdog_health_critical;
145 }
146
147 /* Optional task failures don't degrade system health if sensor isn't connected */
148 /* The DHT22 task should handle sensor failures gracefully and keep running */
149
150 return k_watchdog_health_good;
151}

153static watchdog_health_level_t internal_assess_system_state(void)
154{
155 const system_state_t current_state = shared_data_get_system_state();
156
157 switch (current_state) {
158 case k_system_state_running:
159 return k_watchdog_health_good;
160 case k_system_state_error_recovery:
161 return k_watchdog_health_degraded;
162 case k_system_state_initializing:
163 return k_watchdog_health_degraded;
164 case k_system_state_fault:
165 default:
166 return k_watchdog_health_fault;
167 }
168}

170static watchdog_health_level_t internal_assess_overall_health(void)
171{
172 const watchdog_health_level_t sensor_health = internal_assess_sensor_health();
173 const watchdog_health_level_t task_health = internal_assess_task_health();
174 const watchdog_health_level_t state_health = internal_assess_system_state();
175
176 watchdog_health_level_t worst_health = k_watchdog_health_good;
177
178 if (sensor_health > worst_health)
179 worst_health = sensor_health;
180 if (task_health > worst_health)
181 worst_health = task_health;
182 if (state_health > worst_health)
183 worst_health = state_health;
184
185 return worst_health;
186}

188static esp_err_t internal_take_recovery_action(const watchdog_health_level_t health_level)
189{
190 esp_err_t ret = ESP_OK;
191 s_task_context.recovery_actions_taken++;
192
193 switch (health_level) {
194 case k_watchdog_health_degraded:
195 ESP_LOGW(s_TAG, "System degraded - logging detailed health status");
196 shared_data_set_system_state(k_system_state_error_recovery);
197 break;
198
199 case k_watchdog_health_critical:
200 ESP_LOGE(s_TAG, "System critical - attempting emergency LED shutdown");
201 shared_data_set_system_state(k_system_state_error_recovery);
202
203 if (led_task_is_running()) {
204 led_task_emergency_off();
205 }
206 break;
207
208 case k_watchdog_health_fault:
209 ESP_LOGE(s_TAG, "System fault - initiating system restart");
210 shared_data_set_system_state(k_system_state_fault);
211
212 ESP_LOGE(s_TAG, "System restart in %d ms...", STAR_SYSTEM_CONFIG.health.restart_delay_ms);
213 vTaskDelay(pdMS_TO_TICKS(STAR_SYSTEM_CONFIG.health.restart_delay_ms));
214 esp_restart();
215 break;
216
217 case k_watchdog_health_good:
218 if (shared_data_get_system_state() == k_system_state_error_recovery) {
219 ESP_LOGI(s_TAG, "System recovered - returning to normal operation");
220 shared_data_set_system_state(k_system_state_running);
221 }
222 break;
223
224 default:
225 ret = ESP_ERR_INVALID_ARG;
226 break;
227 }
228
229 return ret;
230}

232static void internal_watchdog_task_loop(void* const parameter)
233{
234 watchdog_task_context_t* const ctx = (watchdog_task_context_t*)parameter;
235
236 ESP_LOGI(s_TAG,
237 "System watchdog task started (interval: %dms)",
238 STAR_SYSTEM_CONFIG.tasks.watchdog.interval_ms);
239
240 TickType_t last_wake_time = xTaskGetTickCount();
241
242 while (ctx->task_running) {
243 const watchdog_health_level_t current_health = internal_assess_overall_health();
244 ctx->last_health_level = current_health;
245 ctx->health_checks_performed++;
246
247 if (current_health != k_watchdog_health_good) {
248 ESP_LOGW(s_TAG, "Health check %lu: Level %d", ctx->health_checks_performed, current_health);
249 internal_take_recovery_action(current_health);
250 } else {
251 ESP_LOGI(s_TAG, "Health check %lu: System healthy", ctx->health_checks_performed);
252 }
253
254 if (ctx->health_checks_performed % STAR_SYSTEM_CONFIG.health.health_check_log_interval == 0) {
255 const uint32_t uptime = shared_data_get_uptime_ms();
256 ESP_LOGI(s_TAG,
257 "Watchdog status - uptime: %lu ms, checks: %lu, recoveries: %lu",
258 uptime,
259 ctx->health_checks_performed,
260 ctx->recovery_actions_taken);
261 }
262
263 vTaskDelayUntil(&last_wake_time, pdMS_TO_TICKS(STAR_SYSTEM_CONFIG.tasks.watchdog.interval_ms));
264 }
265
266 ESP_LOGI(s_TAG, "System watchdog task ending");
267 ctx->task_handle = NULL;
268 vTaskDelete(NULL);
269}

273esp_err_t watchdog_task_start(void)
274{
275 if (s_g_context_initialized && s_task_context.task_running) {
276 ESP_LOGI(s_TAG, "Watchdog task already running");
277 return ESP_OK;
278 }
279
280 memset(&s_task_context, 0, sizeof(watchdog_task_context_t));
281
282 s_task_context.task_running = true;
283 s_task_context.health_checks_performed = 0;
284 s_task_context.recovery_actions_taken = 0;
285 s_task_context.last_health_level = k_watchdog_health_good;
286
287 const BaseType_t task_ret = xTaskCreate(internal_watchdog_task_loop,
288 STAR_SYSTEM_CONFIG.tasks.watchdog.name,
289 STAR_SYSTEM_CONFIG.tasks.watchdog.stack_size,
290 &s_task_context,
291 STAR_SYSTEM_CONFIG.tasks.watchdog.priority,
292 &s_task_context.task_handle);
293
294 if (task_ret != pdPASS) {
295 ESP_LOGE(s_TAG, "Failed to create watchdog task");
296 s_task_context.task_running = false;
297 return ESP_ERR_NO_MEM;
298 }
299
300 s_g_context_initialized = true;
301
302 shared_context_t* shared_ctx = shared_data_get_context();
303 if (shared_ctx != NULL) {
304 shared_ctx->watchdog_task_handle = s_task_context.task_handle;
305 }
306
307 ESP_LOGI(s_TAG, "System watchdog task started successfully");
308 return ESP_OK;
309}

311esp_err_t watchdog_task_stop(void)
312{
313 if (!s_g_context_initialized || !s_task_context.task_running) {
314 ESP_LOGI(s_TAG, "Watchdog task not running");
315 return ESP_OK;
316 }
317
318 ESP_LOGI(s_TAG, "Stopping watchdog task...");
319
320 s_task_context.task_running = false;
321
322 if (s_task_context.task_handle != NULL) {
323 vTaskDelay(pdMS_TO_TICKS(s_watchdog_recovery_delay_ms));
324 }
325
326 shared_context_t* shared_ctx = shared_data_get_context();
327 if (shared_ctx != NULL) {
328 shared_ctx->watchdog_task_handle = NULL;
329 }
330
331 s_g_context_initialized = false;
332
333 ESP_LOGI(s_TAG,
334 "Watchdog task stopped (checks: %lu, recoveries: %lu)",
335 s_task_context.health_checks_performed,
336 s_task_context.recovery_actions_taken);
337
338 memset(&s_task_context, 0, sizeof(watchdog_task_context_t));
339
340 return ESP_OK;
341}
