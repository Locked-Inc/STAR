98static void internal_init_global_state(void)
99{
100 /* Quick check without lock for common case */
101 if (g_global_mutex != NULL) {
102 return;
103 }
104
105 /* Use spinlock for thread-safe lazy initialization */
106 portENTER_CRITICAL(&g_init_spinlock);
107
108 /* Double-check after acquiring spinlock */
109 if (g_global_mutex == NULL) {
110 g_global_mutex = xSemaphoreCreateMutex();
111 }
112
113 portEXIT_CRITICAL(&g_init_spinlock);
114}

119static async_state_t* internal_get_async_state(const char* bus_name)
120{
121 internal_init_global_state();
122
123 if (g_global_mutex == NULL) {
124 ESP_LOGE(s_TAG, "Global mutex not initialized");
125 return NULL;
126 }
127
128 xSemaphoreTake(g_global_mutex, portMAX_DELAY);
129
130 /* Search for existing state by bus name */
131 for (uint8_t i = 0; i < g_num_async_states; i++) {
132 if (strcmp(g_async_states[i].bus_name, bus_name) == 0) {
133 xSemaphoreGive(g_global_mutex);
134 return &g_async_states[i];
135 }
136 }
137
138 /* Create new state if needed */
139 if (g_num_async_states < MAX_BUSES) {
140 async_state_t* state = &g_async_states[g_num_async_states];
141 memset(state, 0, sizeof(async_state_t));
142 strncpy(state->bus_name, bus_name, sizeof(state->bus_name) - 1);
143 state->bus_name[sizeof(state->bus_name) - 1] = '\0';
144 state->queue_mutex = xSemaphoreCreateMutex();
145 if (state->queue_mutex == NULL) {
146 ESP_LOGE(s_TAG, "Failed to create queue mutex for bus '%s'", bus_name);
147 xSemaphoreGive(g_global_mutex);
148 return NULL;
149 }
150 g_num_async_states++;
151 xSemaphoreGive(g_global_mutex);
152 return state;
153 }
154
155 ESP_LOGE(s_TAG, "Maximum async states reached (%d)", MAX_BUSES);
156 xSemaphoreGive(g_global_mutex);
157 return NULL;
158}

163static esp_err_t internal_enqueue_operation(async_state_t* state, async_operation_t* op)
164{
165 if (state == NULL || op == NULL) {
166 return ESP_ERR_INVALID_ARG;
167 }
168
169 xSemaphoreTake(state->queue_mutex, portMAX_DELAY);
170
171 if (state->pending_count >= (uint32_t)STAR_ASYNC_MAX_PENDING) {
172 xSemaphoreGive(state->queue_mutex);
173 ESP_LOGE(s_TAG, "Async queue full");
174 return ESP_ERR_NO_MEM;
175 }
176
177 /* Add to tail of queue */
178 op->next = NULL;
179 if (state->pending_tail != NULL) {
180 state->pending_tail->next = op;
181 }
182 state->pending_tail = op;
183
184 if (state->pending_head == NULL) {
185 state->pending_head = op;
186 }
187
188 state->pending_count++;
189
190 xSemaphoreGive(state->queue_mutex);
191
192 return ESP_OK;
193}

198static void internal_execute_operation(async_operation_t* op)
199{
200 if (op == NULL) {
201 return;
202 }
203
204 op->status = STAR_ASYNC_STATUS_RUNNING;
205 esp_err_t result = ESP_FAIL;
206
207 switch (op->type) {
208 case STAR_ASYNC_OP_I2C_WRITE:
209 result = star_bus_i2c_write(op->manager,
210 op->bus_name,
211 op->params.i2c.data,
212 op->params.i2c.length,
213 op->params.i2c.command,
214 NULL);
215 break;
216
217 case STAR_ASYNC_OP_I2C_READ:
218 result = star_bus_i2c_read(op->manager,
219 op->bus_name,
220 op->params.i2c.data,
221 op->params.i2c.length,
222 op->params.i2c.command,
223 NULL);
224 break;
225
226 case STAR_ASYNC_OP_SPI_TRANSMIT:
227 result = star_bus_spi_transmit(op->manager,
228 op->bus_name,
229 op->params.spi.tx_data,
230 op->params.spi.length,
231 0);
232 break;
233
234 case STAR_ASYNC_OP_SPI_RECEIVE:
235 result = star_bus_spi_receive(op->manager,
236 op->bus_name,
237 op->params.spi.rx_data,
238 op->params.spi.length,
239 0);
240 break;
241
242 case STAR_ASYNC_OP_SPI_TRANSCEIVE:
243 result = star_bus_spi_transfer(op->manager,
244 op->bus_name,
245 op->params.spi.tx_data,
246 op->params.spi.rx_data,
247 op->params.spi.length,
248 0);
249 break;
250
251 case STAR_ASYNC_OP_SMBUS_READ:
252 result = star_smbus_read_byte(op->manager,
253 op->bus_name,
254 op->params.smbus.addr,
255 op->params.smbus.command,
256 op->params.smbus.data);
257 break;
258
259 case STAR_ASYNC_OP_SMBUS_WRITE:
260 result = star_smbus_write_byte(op->manager,
261 op->bus_name,
262 op->params.smbus.addr,
263 op->params.smbus.command,
264 *op->params.smbus.data);
265 break;
266
267 default:
268 ESP_LOGE(s_TAG, "Unknown operation type: %d", op->type);
269 result = ESP_ERR_NOT_SUPPORTED;
270 break;
271 }
272
273 op->result = result;
274 op->status = (result == ESP_OK) ? STAR_ASYNC_STATUS_COMPLETE : STAR_ASYNC_STATUS_ERROR;
275
276 /* Set event bits if configured */
277 if (op->event_group != NULL) {
278 if (result == ESP_OK && op->complete_bit != 0) {
279 xEventGroupSetBits(op->event_group, op->complete_bit);
280 } else if (result != ESP_OK && op->error_bit != 0) {
281 xEventGroupSetBits(op->event_group, op->error_bit);
282 }
283 }
284
285 /* Signal wait semaphore if waiting */
286 if (op->wait_semaphore != NULL) {
287 xSemaphoreGive(op->wait_semaphore);
288 }
289
290 /* Invoke callback */
291 if (op->callback != NULL) {
292 op->callback((star_async_handle_t)op, op->status, result, op->user_context);
293 }
294}

299static void internal_async_worker_task(void* param)
300{
301 async_state_t* state = (async_state_t*)param;
302
303 ESP_LOGI(s_TAG, "Async worker task started");
304
305 while (state->worker_running) {
306 async_operation_t* op = NULL;
307
308 /* Dequeue next operation */
309 xSemaphoreTake(state->queue_mutex, portMAX_DELAY);
310
311 if (state->pending_head != NULL) {
312 op = state->pending_head;
313 state->pending_head = op->next;
314
315 if (state->pending_head == NULL) {
316 state->pending_tail = NULL;
317 }
318
319 state->pending_count--;
320 }
321
322 xSemaphoreGive(state->queue_mutex);
323
324 if (op != NULL) {
325 /* Check if operation was cancelled while in queue */
326 if (op->status == STAR_ASYNC_STATUS_CANCELLED) {
327 state->cancelled_ops++;
328 /* Don't execute, callback was already invoked during cancel */
329 } else {
330 /* Execute the operation */
331 internal_execute_operation(op);
332
333 /* Update statistics */
334 if (op->status == STAR_ASYNC_STATUS_COMPLETE) {
335 state->completed_ops++;
336 } else if (op->status == STAR_ASYNC_STATUS_ERROR) {
337 state->failed_ops++;
338 } else if (op->status == STAR_ASYNC_STATUS_CANCELLED) {
339 state->cancelled_ops++;
340 }
341 }
342 } else {
343 /* No operations pending, sleep briefly */
344 vTaskDelay(pdMS_TO_TICKS(10));
345 }
346 }
347
348 vTaskDelete(NULL);
349}

354static esp_err_t internal_ensure_worker_running(async_state_t* state)
355{
356 if (state == NULL) {
357 return ESP_ERR_INVALID_ARG;
358 }
359
360 /* Quick check without lock for common case */
361 if (state->worker_running) {
362 return ESP_OK;
363 }
364
365 /* Use queue_mutex to prevent race condition */
366 if (xSemaphoreTake(state->queue_mutex, portMAX_DELAY) != pdTRUE) {
367 ESP_LOGE(s_TAG, "Failed to take queue mutex for worker check");
368 return ESP_ERR_TIMEOUT;
369 }
370
371 esp_err_t result = ESP_OK;
372
373 /* Double-check after acquiring mutex */
374 if (!state->worker_running) {
375 state->worker_running = true;
376
377 BaseType_t created =
378 xTaskCreate(internal_async_worker_task, "async_worker", 4096, state, 5, &state->worker_task);
379
380 if (created != pdPASS) {
381 state->worker_running = false;
382 ESP_LOGE(s_TAG, "Failed to create worker task");
383 result = ESP_ERR_NO_MEM;
384 }
385 }
386
387 xSemaphoreGive(state->queue_mutex);
388 return result;
389}

518esp_err_t star_async_wait(star_async_handle_t handle, uint32_t timeout_ms)
519{
520 if (handle == NULL) {
521 return ESP_ERR_INVALID_ARG;
522 }
523
524 async_operation_t* op = (async_operation_t*)handle;
525
526 /* Create semaphore if not already created (thread-safe) */
527 if (op->wait_semaphore == NULL) {
528 portENTER_CRITICAL(&g_init_spinlock);
529 if (op->wait_semaphore == NULL) {
530 op->wait_semaphore = xSemaphoreCreateBinary();
531 }
532 portEXIT_CRITICAL(&g_init_spinlock);
533
534 if (op->wait_semaphore == NULL) {
535 return ESP_ERR_NO_MEM;
536 }
537 }
538
539 /* Wait for completion */
540 TickType_t ticks = (timeout_ms == 0) ? portMAX_DELAY : pdMS_TO_TICKS(timeout_ms);
541
542 if (xSemaphoreTake(op->wait_semaphore, ticks) == pdTRUE) {
543 return op->result;
544 }
545
546 return ESP_ERR_TIMEOUT;
547}

549esp_err_t star_async_cancel(star_async_handle_t handle)
550{
551 if (handle == NULL) {
552 return ESP_ERR_INVALID_ARG;
553 }
554
555 async_operation_t* op = (async_operation_t*)handle;
556
557 if (op->status == STAR_ASYNC_STATUS_COMPLETE || op->status == STAR_ASYNC_STATUS_ERROR) {
558 return ESP_ERR_INVALID_STATE;
559 }
560
561 op->status = STAR_ASYNC_STATUS_CANCELLED;
562 op->result = ESP_ERR_INVALID_STATE;
563
564 /* Invoke callback */
565 if (op->callback != NULL) {
566 op->callback(handle, op->status, op->result, op->user_context);
567 }
568
569 return ESP_OK;
570}

584void star_async_free_handle(star_async_handle_t handle)
585{
586 if (handle != NULL) {
587 async_operation_t* op = (async_operation_t*)handle;
588
589 if (op->wait_semaphore != NULL) {
590 vSemaphoreDelete(op->wait_semaphore);
591 }
592
593 /* Free SMBus write data if allocated */
594 if (op->type == STAR_ASYNC_OP_SMBUS_WRITE && op->params.smbus.data != NULL) {
595 free(op->params.smbus.data);
596 }
597
598 free(op);
